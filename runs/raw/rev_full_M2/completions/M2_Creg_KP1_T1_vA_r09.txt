```python
import numpy as np
import scipy
import pandas as pd
import matplotlib.pyplot as plt

# 输入数据：年化波动率（小数形式）
vol = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr = np.array([
    [1.00,  0.21, -0.13],
    [0.21,  1.00,  0.37],
    [-0.13, 0.37,  1.00]
])

# 构建协方差矩阵
Sigma = np.outer(vol, vol) * corr

# 全局最小方差组合（允许卖空，权重之和为1）
ones = np.ones(3)
Sigma_inv = np.linalg.inv(Sigma)
w_mvp = Sigma_inv @ ones / (ones @ Sigma_inv @ ones)

# 组合年化方差及波动率
var_mvp = w_mvp @ Sigma @ w_mvp
vol_mvp = np.sqrt(var_mvp)

# 按照要求输出
result = {
    'mvp_weights': w_mvp.tolist(),   # 转换为列表以便阅读
    'mvp_vol_annual': vol_mvp
}

print(result)
```
