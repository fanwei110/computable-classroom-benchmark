```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# ============================
# 第一部分：年化夏普比率
# ============================
# 由于未提供真实的“课程数据快照”，为满足可复现、自包含的要求，
# 使用固定随机种子生成模拟的基金日收益序列。
# 假设三年（756个交易日）的数据，无风险年利率2.1%，一年252个交易日。
np.random.seed(42)
n_days = 756
fund_daily_returns = np.random.normal(loc=0.0005, scale=0.01, size=n_days)

rf_annual = 0.021
rf_daily = rf_annual / 252.0

excess_daily_returns = fund_daily_returns - rf_daily

# 年化夏普比率 (假设日收益无自相关，使用 sqrt(252) 调整)
sharpe_annual = np.sqrt(252) * excess_daily_returns.mean() / excess_daily_returns.std()

# ============================
# 第二部分：绩效归因效应
# ============================
# 组合数据
w_p = np.array([0.45, 0.35, 0.20])          # 组合行业权重
r_p = np.array([0.083, 0.021, -0.014])      # 组合行业收益

# 基准数据
w_b = np.array([0.40, 0.40, 0.20])          # 基准行业权重
r_b = np.array([0.067, 0.034, -0.009])      # 基准行业收益

# 配置效应 (Allocation Effect)
allocation_effect = np.sum((w_p - w_b) * r_b)

# 选择效应 (Selection Effect)
selection_effect = np.sum(w_b * (r_p - r_b))

# 交互效应 (Interaction Effect)
interaction_effect = np.sum((w_p - w_b) * (r_p - r_b))

# ============================
# 存储结果
# ============================
result = {
    'sharpe_annual': sharpe_annual,
    'allocation_effect': allocation_effect,
    'selection_effect': selection_effect,
    'interaction_effect': interaction_effect
}

# 输出以验证
if __name__ == '__main__':
    for key, val in result.items():
        print(f"{key}: {val:.6f}")
```
