```python
import numpy as np
import matplotlib.pyplot as plt

# ----------------------------
# 1. 定义资产参数
# ----------------------------
mu = np.array([0.071, 0.124])          # 期望年收益
sigma = np.array([0.163, 0.289])       # 年化波动率
rhos = [0.15, 0.45, 0.75]             # 相关系数

# 目标收益
target_return = 0.10

# 权重扫描范围（涵盖卖空情况，使曲线完整）
w1_range = np.linspace(-0.5, 1.5, 600)

# 准备绘图
fig, ax = plt.subplots(figsize=(8, 5))
colors = ['tab:blue', 'tab:orange', 'tab:green']  # 三条曲线颜色

# 用于存储结果
result = {}

# ----------------------------
# 2. 对每个相关系数计算前沿并绘图
# ----------------------------
for i, rho in enumerate(rhos):
    # 协方差矩阵元素
    cov_11 = sigma[0] ** 2
    cov_22 = sigma[1] ** 2
    cov_12 = rho * sigma[0] * sigma[1]

    # 扫描组合权重，计算收益与波动率
    w2_range = 1.0 - w1_range
    port_mu = w1_range * mu[0] + w2_range * mu[1]
    port_var = (w1_range ** 2 * cov_11 +
                w2_range ** 2 * cov_22 +
                2 * w1_range * w2_range * cov_12)
    port_vol = np.sqrt(port_var)

    # 绘制前沿曲线
    line, = ax.plot(port_vol * 100, port_mu * 100,
                    label=f'ρ = {rho}', color=colors[i], linewidth=1.5)

    # 全局最小方差组合 (MVP) —— 允许权重和为1即可，无卖空限制
    w1_mvp = (cov_22 - cov_12) / (cov_11 + cov_22 - 2 * cov_12)
    w2_mvp = 1.0 - w1_mvp
    mu_mvp = w1_mvp * mu[0] + w2_mvp * mu[1]
    vol_mvp = np.sqrt(w1_mvp ** 2 * cov_11 +
                      w2_mvp ** 2 * cov_22 +
                      2 * w1_mvp * w2_mvp * cov_12)

    # 在曲线上标注MVP
    ax.scatter(vol_mvp * 100, mu_mvp * 100,
               color=line.get_color(), marker='o', s=50,
               zorder=5, edgecolors='k', linewidth=0.5)

    # 如果 rho == 0.45，保存所需结果
    if rho == 0.45:
        result['mvp_vol_at_rho45'] = vol_mvp

        # 给定目标收益，计算唯一组合的权重与波动率
        # 因为只有两个资产，给定收益即可确定权重
        w1_target = (target_return - mu[1]) / (mu[0] - mu[1])
        w2_target = 1.0 - w1_target
        vol_target = np.sqrt(w1_target ** 2 * cov_11 +
                             w2_target ** 2 * cov_22 +
                             2 * w1_target * w2_target * cov_12)
        result['frontier_vol_at_target'] = vol_target

# ----------------------------
# 3. 图形修饰
# ----------------------------
ax.set_xlabel('Annualized Volatility (%)', fontsize=12)
ax.set_ylabel('Expected Annual Return (%)', fontsize=12)
ax.set_title('Efficient Frontiers for Different Correlations', fontsize=14)
ax.legend(frameon=True, fontsize=10)
ax.grid(True, linestyle='--', alpha=0.6)

# 保存图形
fig_path = 'efficient_frontier.png'
fig.savefig(fig_path, dpi=200, bbox_inches='tight')
plt.close(fig)

result['figure_path'] = fig_path

# ----------------------------
# 4. 输出结果（教师可直接查看）
# ----------------------------
print("===== 计算结果 =====")
print(f"ρ=0.45 最小方差组合的波动率: {result['mvp_vol_at_rho45']:.4f} ({result['mvp_vol_at_rho45']*100:.2f}%)")
print(f"ρ=0.45 目标收益 10% 时的最小波动率: {result['frontier_vol_at_target']:.4f} ({result['frontier_vol_at_target']*100:.2f}%)")
print(f"图形已保存至: {result['figure_path']}")
```
