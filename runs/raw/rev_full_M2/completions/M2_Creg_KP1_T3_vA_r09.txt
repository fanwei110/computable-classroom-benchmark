```python
import numpy as np

def portfolio_vol(weights, sigmas, rho):
    """计算两资产组合的年化波动率"""
    w_a, w_b = weights
    sigma_a, sigma_b = sigmas
    variance = (w_a**2 * sigma_a**2 + 
                w_b**2 * sigma_b**2 + 
                2 * w_a * w_b * rho * sigma_a * sigma_b)
    return np.sqrt(variance)

# 给定数据
w = np.array([0.6, 0.4])                 # 权重 60/40
sigmas = np.array([0.184, 0.297])        # 年化波动率

rho_before = 0.3
rho_after = 0.8

# 计算组合波动率
vol_before = portfolio_vol(w, sigmas, rho_before)
vol_after = portfolio_vol(w, sigmas, rho_after)

# 按要求构建输出字典
result = {
    'vol_before_annual': vol_before,
    'vol_after_annual': vol_after
}

if __name__ == "__main__":
    print(result)
```
