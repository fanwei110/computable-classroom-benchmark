```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import numpy as np
from scipy.stats import norm
import pandas as pd
import matplotlib.pyplot as plt

# ---------- 输入参数 ----------
position_value = 1_850_000      # 头寸价值（元）
annual_vol = 0.218              # 年化收益波动率
trading_days = 252              # 年交易日数
confidence_95 = 0.95            # 95% 置信水平
confidence_99 = 0.99            # 99% 置信水平
horizon_10d = 10                # 十日 horizon

# ---------- 波动率换算 ----------
daily_vol = annual_vol / np.sqrt(trading_days)                # 一日波动率
ten_day_vol = daily_vol * np.sqrt(horizon_10d)                # 十日波动率（平方根法则）

# ---------- 正态分位数 ----------
z_95 = norm.ppf(confidence_95)   # 单尾 95% 分位数 ≈ 1.64485
z_99 = norm.ppf(confidence_99)   # 单尾 99% 分位数 ≈ 2.32635

# ---------- delta-normal VaR 计算（损失正数）----------
var_95_1d = position_value * daily_vol * z_95
var_99_10d = position_value * ten_day_vol * z_99

# ---------- 按契约存入字典 ----------
result = {
    'var_95_1d': var_95_1d,
    'var_99_10d': var_99_10d
}

# ---------- 查看结果（可复现） ----------
if __name__ == "__main__":
    print("Delta-Normal VaR 计算结果：")
    for k, v in result.items():
        print(f"  {k}: {v:,.2f} 元")
```
