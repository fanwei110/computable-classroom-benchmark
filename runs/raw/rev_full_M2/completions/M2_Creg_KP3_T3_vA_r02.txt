```python
import numpy as np

def main():
    # 债券参数
    face_value = 100.0
    coupon_rate = 0.046
    yield_initial = 0.053
    delta_yield = 0.008  # 80 个基点
    yield_new = yield_initial + delta_yield
    maturity = 7
    
    # 假设每年付息一次
    coupon = face_value * coupon_rate
    
    # 构建现金流数组：前6年为票息，第7年为票息+本金
    cashflows = np.full(maturity, coupon)
    cashflows[-1] += face_value
    
    # 计算折现因子
    periods = np.arange(1, maturity + 1)
    df_old = (1 + yield_initial) ** (-periods)
    df_new = (1 + yield_new) ** (-periods)
    
    # 计算价格
    price_old = np.sum(cashflows * df_old)
    price_new = np.sum(cashflows * df_new)
    
    # 价格变动百分比（负数表示下跌）
    price_drop_pct = (price_new - price_old) / price_old * 100
    
    # 输出结果
    result = {'price_drop_pct': price_drop_pct}
    return result

if __name__ == "__main__":
    result = main()
    print(result)
```
