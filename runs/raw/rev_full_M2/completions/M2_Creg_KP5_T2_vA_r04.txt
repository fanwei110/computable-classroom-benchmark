```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# ----------------------------------------------------------------------
# 可调参数
# ----------------------------------------------------------------------
CAPITAL = 1_000_000          # 头寸（元）
CONF_LEVEL = 0.95            # VaR 置信水平，可调整
DATA_FILE = "fund_data.csv"  # 课程数据快照文件，如不存在则使用可复现的示例数据
RANDOM_SEED = 42             # 生成示例数据时的随机种子，保证可复现
FIGURE_PATH = "var_histogram.png"  # 输出图片路径

# ----------------------------------------------------------------------
# 1. 读取 fund 列的日收益序列
# ----------------------------------------------------------------------
try:
    df = pd.read_csv(DATA_FILE)
    returns = df["fund"].values
    data_source = "file"
except FileNotFoundError:
    # 如果文件不存在，生成确定性示例数据，便于复现（符合“自包含”要求）
    np.random.seed(RANDOM_SEED)
    n_days = 500
    # 模拟一个基金的日对数收益：均值约为 0.02%，波动率约为 1.5%
    returns = np.random.normal(loc=0.0002, scale=0.015, size=n_days)
    data_source = "synthetic"
    print(f"未找到 {DATA_FILE}，使用确定性合成数据（{n_days} 个交易日，seed={RANDOM_SEED}）")
else:
    # 确保是一维数值数组
    returns = returns.flatten()

# ----------------------------------------------------------------------
# 2. 计算日损益分布
# ----------------------------------------------------------------------
pnl = CAPITAL * returns   # 100 万元头寸的每日损益（元）

# ----------------------------------------------------------------------
# 3. 计算历史 VaR
#    定义：在置信水平 CONF_LEVEL 下，日损失不会超过 VaR（正数）
#    损益分布的左尾分位数：alpha = 1 - CONF_LEVEL
#    VaR = - 损益分布的下 alpha 分位数
# ----------------------------------------------------------------------
alpha = 1.0 - CONF_LEVEL
var_cutoff_pnl = np.quantile(pnl, alpha)        # 损益分位点（通常为负值）
hist_var = -var_cutoff_pnl                       # 历史 VaR（正数）

# ----------------------------------------------------------------------
# 4. 绘图：日损益直方图 + VaR 竖线
# ----------------------------------------------------------------------
fig, ax = plt.subplots(figsize=(10, 6))
ax.hist(pnl, bins=80, alpha=0.7, color='steelblue', edgecolor='white',
        label='每日损益分布')

# 标出 95% VaR 对应的损益值
ax.axvline(var_cutoff_pnl, color='red', linestyle='--', linewidth=2,
           label=f'{CONF_LEVEL*100:.0f}% 1‑Day Historical VaR = ¥{hist_var:,.0f}')
ax.legend(fontsize=11)
ax.set_xlabel('日损益（元）', fontsize=12)
ax.set_ylabel('频次', fontsize=12)
ax.set_title(f'头寸 ¥{CAPITAL:,} 的日损益分布\n'
             f'{CONF_LEVEL*100:.0f}% 历史 VaR: ¥{hist_var:,.0f}', fontsize=13)
ax.grid(True, alpha=0.3)

# 在 VaR 竖线附近添加文字标注
y_lim = ax.get_ylim()
ax.text(var_cutoff_pnl * 1.15, y_lim[1] * 0.9,
        f'VaR = ¥{hist_var:,.0f}', color='red', fontsize=11,
        bbox=dict(facecolor='white', alpha=0.8, edgecolor='red'))

plt.tight_layout()
plt.savefig(FIGURE_PATH, dpi=150)
plt.close()

# ----------------------------------------------------------------------
# 5. 构建输出字典
# ----------------------------------------------------------------------
result = {
    'hist_var_95_1d': hist_var,
    'figure_path': FIGURE_PATH,
}

# 打印结果以便查看
print(f"数据来源: {data_source}")
print(f"头寸: ¥{CAPITAL:,}")
print(f"{CONF_LEVEL*100:.0f}% 一日历史 VaR: ¥{hist_var:,.2f}")
print(f"图片已保存至: {FIGURE_PATH}")
print(f"result = {result}")
```
