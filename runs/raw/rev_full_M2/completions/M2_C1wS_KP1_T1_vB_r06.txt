```python
import numpy as np
from scipy.optimize import minimize

# ========================
# 1. 输入参数
# ========================
# 年化波动率
vol = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵的下三角部分
# rho_12 = 0.21, rho_13 = -0.13, rho_23 = 0.37
corr = np.array([
    [1.00,  0.21, -0.13],
    [0.21,  1.00,  0.37],
    [-0.13, 0.37,  1.00]
])

# ========================
# 2. 构造协方差矩阵
# ========================
# 协方差矩阵 sigma_ij = rho_ij * sigma_i * sigma_j
sigma = np.diag(vol)    # 波动率对角矩阵
cov_mat = sigma @ corr @ sigma

# ========================
# 3. 求最小方差组合权重（允许做空，权重和=1）
# ========================
# 方法一：解析解（闭式解）
# 对于最小方差组合（无目标收益约束），权重 w* ∝ Σ^{-1} 1
# 即 w* = (Σ^{-1} 1) / (1' Σ^{-1} 1)

n = len(vol)                    # 资产数量
ones = np.ones(n)               # 全1向量
cov_inv = np.linalg.inv(cov_mat)  # 协方差矩阵的逆

# 未归一化权重
w_unnormed = cov_inv @ ones
# 归一化使权重和为1
w_mvp = w_unnormed / (ones @ w_unnormed)

# 方法二：数值优化（验证用）
# def portfolio_variance(w):
#     return w @ cov_mat @ w
# 
# constraints = ({'type': 'eq', 'fun': lambda w: np.sum(w) - 1})
# res = minimize(portfolio_variance, x0=np.ones(n)/n, constraints=constraints)
# w_mvp = res.x

# ========================
# 4. 计算组合波动率
# ========================
# 组合方差
mvp_var = w_mvp @ cov_mat @ w_mvp
# 年化组合波动率
mvp_vol = np.sqrt(mvp_var)

# ========================
# 5. 存储结果
# ========================
result = {
    'mvp_weights': w_mvp,
    'mvp_vol_annual': mvp_vol
}

# ========================
# 输出展示
# ========================
print("=== 最小方差组合（允许做空，全额投资） ===")
print(f"协方差矩阵:\n{cov_mat}\n")
print("最优权重:")
for i, w in enumerate(w_mvp, start=1):
    print(f"  资产 {i}: {w:.4f} ({w*100:.2f}%)")
print(f"\n组合年化波动率: {mvp_vol:.4f} ({mvp_vol*100:.2f}%)")

print("\n=== 结果字典 ===")
print(result)
```
