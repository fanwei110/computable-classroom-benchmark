```python
import numpy as np
from scipy.stats import norm

# ============================
# 输入数据
# ============================
position_value = 1_850_000          # 头寸价值（元）
annual_volatility = 0.218           # 年化收益波动率 21.8%

# ============================
# 波动率转换
# ============================
# 假设一年有 252 个交易日
trading_days_per_year = 252
daily_volatility = annual_volatility / np.sqrt(trading_days_per_year)

# ============================
# 分位数
# ============================
# 95% 置信度对应的 z 值
z_95 = norm.ppf(0.05)
# 99% 置信度对应的 z 值
z_99 = norm.ppf(0.01)

# ============================
# VaR 计算（delta-normal 参数法）
# ============================
# (i) 95% 一日 VaR
var_95_1d = abs(z_95) * daily_volatility * position_value

# (ii) 99% 十日 VaR
# 假设收益独立同分布，十日波动率 = 单日波动率 × sqrt(10)
ten_day_volatility = daily_volatility * np.sqrt(10)
var_99_10d = abs(z_99) * ten_day_volatility * position_value

# ============================
# 输出结果
# ============================
result = {
    'var_95_1d': var_95_1d,
    'var_99_10d': var_99_10d
}

print("计算结果：")
print(f"95% 一日 VaR: {var_95_1d:,.2f} 元")
print(f"99% 十日 VaR: {var_99_10d:,.2f} 元")
```
