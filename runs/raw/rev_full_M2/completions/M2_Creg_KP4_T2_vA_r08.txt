```python
import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt
import os

# =========================
# 可调参数
# =========================
K = 97.5          # 行权价
r = 0.043         # 无风险利率 (年化)
T = 0.58          # 剩余期限 (年)
S_target = 110    # 目标标的价格
sigma_target = 0.276  # 目标波动率 27.6%

# 可调的波动率列表 (三条曲线)
sigmas = [0.15, 0.276, 0.40]

# =========================
# 函数定义
# =========================
def delta_call(S, K, r, T, sigma):
    """计算不分红欧式看涨期权的 Delta (BSM 模型)"""
    if T <= 0:
        # 到期时 Delta 为 1 (实值) 或 0 (虚值)
        return np.where(S > K, 1.0, 0.0)
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    return norm.cdf(d1)

# =========================
# 计算 Delta@S=110, σ=27.6%
# =========================
delta_at_s110 = delta_call(S_target, K, r, T, sigma_target)

# =========================
# 绘制 Delta 曲线
# =========================
S_range = np.linspace(70, 140, 500)   # 标的价格从 70 到 140

plt.figure(figsize=(8, 5))
for sigma in sigmas:
    delta_vals = delta_call(S_range, K, r, T, sigma)
    plt.plot(S_range, delta_vals, label=f'$\sigma$ = {sigma*100:.1f}%')

plt.xlabel('标的价格 S')
plt.ylabel('Delta')
plt.title('欧式看涨期权 Delta 曲线 (K=97.5, T=0.58年)')
plt.legend()
plt.grid(True, alpha=0.3)

# 保存图像，获取绝对路径
figure_path = os.path.abspath('delta_curve.png')
plt.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close()

# =========================
# 组装输出字典
# =========================
result = {
    'delta_at_s110': delta_at_s110,
    'figure_path': figure_path
}

# 输出结果，便于验证
print(result)
```
