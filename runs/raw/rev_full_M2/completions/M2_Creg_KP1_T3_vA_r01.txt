```python
import numpy as np

def main() -> dict:
    """
    计算 A/B 60/40 组合在相关系数变化前后的年化波动率。
    
    资产 A 年化波动率: 18.4%
    资产 B 年化波动率: 29.7%
    组合权重: A=60%, B=40%
    相关系数变化: 0.3 → 0.8
    """
    # ---------- 已知参数 ----------
    sigma_A = 0.184        # 资产 A 年化波动率
    sigma_B = 0.297        # 资产 B 年化波动率
    w_A = 0.6              # 组合中 A 的权重
    w_B = 0.4              # 组合中 B 的权重
    rho_before = 0.3       # 变化前的相关系数
    rho_after = 0.8        # 变化后的相关系数

    # ---------- 组合方差与波动率 ----------
    # 公共部分：w_A^2 * σ_A^2 + w_B^2 * σ_B^2
    var_common = w_A**2 * sigma_A**2 + w_B**2 * sigma_B**2
    # 协方差项系数：2 * w_A * w_B * σ_A * σ_B
    cov_coeff = 2.0 * w_A * w_B * sigma_A * sigma_B

    # 变化前
    var_before = var_common + cov_coeff * rho_before
    vol_before_annual = np.sqrt(var_before)

    # 变化后
    var_after = var_common + cov_coeff * rho_after
    vol_after_annual = np.sqrt(var_after)

    # ---------- 结果封装 ----------
    result = {
        'vol_before_annual': vol_before_annual,
        'vol_after_annual': vol_after_annual
    }

    return result


if __name__ == "__main__":
    result = main()
    # 输出结果，确保可复现
    print(f"相关系数 0.3 时组合年化波动率: {result['vol_before_annual']:.4f} ({result['vol_before_annual']*100:.2f}%)")
    print(f"相关系数 0.8 时组合年化波动率: {result['vol_after_annual']:.4f} ({result['vol_after_annual']*100:.2f}%)")
    print(f"波动率变化 (后 - 前): {result['vol_after_annual'] - result['vol_before_annual']:+.4f}")
```
