```python
import numpy as np

# 债券参数
face_value = 100.0
coupon_rate = 0.046
years_to_maturity = 7
ytm = 0.053
coupon = face_value * coupon_rate

# 现金流时间点（年）
t = np.arange(1, years_to_maturity + 1)

# 现金流：最后一期包含面值
cashflows = np.full(years_to_maturity, coupon)
cashflows[-1] += face_value

# 贴现因子
discount = (1 + ytm) ** (-t)

# 现金流现值
pv_cf = cashflows * discount

# 价格
price = np.sum(pv_cf)

# 权重 = 每笔现金流现值 / 价格
weights = pv_cf / price

# 麦考利久期（年）
macaulay_duration_years = np.sum(t * weights)

# 修正久期（年）
modified_duration_years = macaulay_duration_years / (1 + ytm)

# 凸性计算：∑ [ t*(t+1)*PV(CF) ] / [ price * (1+ytm)^2 ]
convexity_numerator = np.sum(t * (t + 1) * pv_cf)
convexity = convexity_numerator / (price * (1 + ytm)**2)

# 结果存储
result = {
    'price': price,
    'macaulay_duration_years': macaulay_duration_years,
    'modified_duration_years': modified_duration_years,
    'convexity': convexity
}

# 输出结果
for key, value in result.items():
    print(f"{key}: {value:.6f}")
```
