```python
import numpy as np
import scipy
import pandas
import matplotlib

# 给定数据：年化波动率 (小数形式)
vols = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr = np.array([
    [ 1.00,  0.21, -0.13],
    [ 0.21,  1.00,  0.37],
    [-0.13,  0.37,  1.00]
])

# 构造协方差矩阵: Σ_ij = ρ_ij * σ_i * σ_j
cov = np.outer(vols, vols) * corr

# 全局最小方差组合 (允许卖空，sum(w)=1)
inv_cov = np.linalg.inv(cov)
ones = np.ones(len(vols))
w_mvp = (inv_cov @ ones) / (ones @ inv_cov @ ones)

# 年化组合方差与波动率
var_mvp = 1.0 / (ones @ inv_cov @ ones)
vol_mvp = np.sqrt(var_mvp)

# 按要求存入 result 字典
result = {
    'mvp_weights': w_mvp.tolist(),
    'mvp_vol_annual': float(vol_mvp)
}

if __name__ == "__main__":
    print("全局最小方差组合结果：")
    print(f"权重: {result['mvp_weights']}")
    print(f"年化波动率: {result['mvp_vol_annual']:.4%}")
```
