```python
import numpy as np
from scipy.stats import norm

# ------------------------------
# 给定的输入参数
# ------------------------------
position   = 1_850_000        # 仓位（人民币元）
annual_vol = 0.218            # 年化波动率
confidence_95 = 0.95          # 95% 置信水平
confidence_99 = 0.99          # 99% 置信水平
trading_days  = 252           # 假设一年中的交易日为 252 天
ten_days      = 10            # 10 天期限

# ------------------------------
# 1. 将年化波动率转换为日波动率
# ------------------------------
daily_vol = annual_vol / np.sqrt(trading_days)

# ------------------------------
# 2. 获取标准正态分布的分位数（单尾）
#    VaR 取正数表示损失金额，因此使用 (1-α) 分位数
#    norm.ppf(0.95) ≈ 1.64485，norm.ppf(0.99) ≈ 2.32635
# ------------------------------
z_95 = norm.ppf(confidence_95)
z_99 = norm.ppf(confidence_99)

# ------------------------------
# 3. 计算 VaR
#    - 一天 95% VaR
#    - 通过平方根法则将日波动率缩放到 10 天，计算 10 天 99% VaR
# ------------------------------
var_95_1d  = position * daily_vol * z_95
var_99_10d = position * daily_vol * np.sqrt(ten_days) * z_99

# ------------------------------
# 4. 结果字典（以人民币报告，保留两位小数）
# ------------------------------
result = {
    'var_95_1d':  round(var_95_1d, 2),
    'var_99_10d': round(var_99_10d, 2)
}

# 可选输出，方便课堂上查看
print(result)
```
