```python
import numpy as np
import pandas as pd

# ============================
# 1. 读取快照 CSV 数据
# ============================
# 根据题目，文件名假设为 "快照.csv"，且其中包含 "fund" 列（基金收益率序列）。
# 若实际文件名不同，请在此处修改。
DATA_FILE = "快照.csv"

# 读取数据
df = pd.read_csv(DATA_FILE)

# 提取 fund 列，剔除缺失值，确保为浮点数
fund_returns = df["fund"].dropna().astype(float)

# ============================
# 2. 参数设定
# ============================
rf_annual = 0.021               # 年化无风险利率 2.1%

# 频率假设：由于题目未指明收益率频率，此处假设为日频数据，一年按252个交易日计算。
# 若数据为其他频率（如月频12），只需修改 PERIODS_PER_YEAR 即可。
PERIODS_PER_YEAR = 252

# 将年化无风险利率转换为与数据同频的利率（简单比例法）
rf_period = rf_annual / PERIODS_PER_YEAR

# ============================
# 3. 计算全样本年化夏普比率
# ============================
# 超额收益率（每期）
excess_returns = fund_returns - rf_period

# 年化超额收益率
annualized_excess_return = excess_returns.mean() * PERIODS_PER_YEAR

# 年化波动率（标准差）
annualized_volatility = excess_returns.std() * np.sqrt(PERIODS_PER_YEAR)

# 年化夏普比率
sharpe_annual = annualized_excess_return / annualized_volatility

# ============================
# 4. 输出结果存入字典 result
# ============================
result = {"sharpe_annual": sharpe_annual}

# 在课堂投屏时可直接查看结果
print(result)
```
