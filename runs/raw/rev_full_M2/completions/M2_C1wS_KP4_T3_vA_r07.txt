```python
import numpy as np
from scipy.stats import norm

def bs_call_price(S, K, T, r, sigma):
    """
    计算欧式看涨期权的 Black-Scholes 价格。
    """
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
    return price

# 给定参数
S = 103.7        # 标的资产价格
K = 97.5         # 行权价
T = 0.58         # 剩余期限（年）
r = 0.043        # 无风险利率
sigma0 = 0.276   # 初始波动率
sigma1 = 0.286   # 上涨一个百分点后的波动率（+0.01）

# 计算波动率变化前后的期权价格
price0 = bs_call_price(S, K, T, r, sigma0)
price1 = bs_call_price(S, K, T, r, sigma1)

# 价格变化（隐含波动率上涨一个百分点）
price_change = price1 - price0

# 按要求存入字典
result = {'price_change': price_change}

if __name__ == "__main__":
    print(result)
```
