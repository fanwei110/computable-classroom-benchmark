```python
import numpy as np

# 债券参数
face_value = 100.0
coupon_rate = 0.046
years_to_maturity = 7
ytm = 0.053

# 现金流时间点 (1 到 7 年，最后一年包含本金)
times = np.arange(1, years_to_maturity + 1)

# 票息金额
coupon = face_value * coupon_rate

# 现金流：前6年只有票息，第7年票息+本金
cash_flows = np.full(years_to_maturity, coupon)
cash_flows[-1] += face_value

# 折现因子
discount_factors = 1 / (1 + ytm) ** times

# 现金流现值
present_values = cash_flows * discount_factors

# 价格 = 所有现金流现值之和
price = np.sum(present_values)

# 时间权重的现值
time_weighted_pv = times * present_values

# 麦考利久期 = sum(t * PV_t) / price
macaulay_duration_years = np.sum(time_weighted_pv) / price

# 修正久期 = 麦考利久期 / (1 + ytm)
modified_duration_years = macaulay_duration_years / (1 + ytm)

# 凸性 = sum(t * (t+1) * PV_t) / (price * (1+ytm)^2)
time_sq_plus_t = times * (times + 1)
convexity = np.sum(time_sq_plus_t * present_values) / (price * (1 + ytm) ** 2)

# 存入结果字典
result = {
    'price': price,
    'macaulay_duration_years': macaulay_duration_years,
    'modified_duration_years': modified_duration_years,
    'convexity': convexity
}

# 打印结果
for key, value in result.items():
    print(f"{key}: {value:.8f}")
```
