```python
import numpy as np
import pandas as pd

# ============================================================
# 题目条件
# ============================================================
beta = 1.42                # 股票的 beta
market_return_monthly = -0.058  # 上个月大盘收益（跌5.8%）
rf_annual = 0.047          # 年化无风险利率 4.7%

# ============================================================
# 步骤1：年化无风险利率折算为月利率
# 采用复利折算： (1 + r_annual)^(1/12) - 1
# ============================================================
rf_monthly = (1 + rf_annual) ** (1/12) - 1

# ============================================================
# 步骤2：CAPM —— 期望收益（月度层面）
# E[R_i] = R_f + beta * (E[R_m] - R_f)
# ============================================================
capm_return_monthly = rf_monthly + beta * (market_return_monthly - rf_monthly)

# ============================================================
# 步骤3：填充结果字典
# ============================================================
result = {
    'capm_return_monthly': capm_return_monthly
}

# ============================================================
# 输出结果（便于课堂投屏展示）
# ============================================================
print(f"年化无风险利率: {rf_annual*100:.2f}%")
print(f"折算月无风险利率: {rf_monthly*100:.4f}%")
print(f"市场月收益: {market_return_monthly*100:.2f}%")
print(f"股票 beta: {beta}")
print(f"CAPM 月度期望收益: {capm_return_monthly*100:.4f}%")
print("\n结果字典:")
print(result)
```
