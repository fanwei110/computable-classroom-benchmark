```python
import numpy as np

# 给定的参数
beta = 1.42
market_return_monthly = -0.058  # 市场下跌5.8%
risk_free_annual = 0.047        # 无风险利率年化4.7%

# 将年化无风险利率转换为月利率（简单比例法）
risk_free_monthly = risk_free_annual / 12

# 根据CAPM计算股票的月度预期收益
# CAPM: E(R_i) = R_f + beta * (E(R_m) - R_f)
capm_return_monthly = risk_free_monthly + beta * (market_return_monthly - risk_free_monthly)

# 按要求将结果存入字典
result = {
    'capm_return_monthly': capm_return_monthly
}

# 打印结果以便验证
print(result)
```
