```python
import numpy as np

# 给定数据：年化波动率（标准差）
sigma_A = 0.184   # A 的波动率
sigma_B = 0.297   # B 的波动率

# 组合权重：A 60%, B 40%
w_A = 0.6
w_B = 0.4

# 相关系数
rho_before = 0.3
rho_after  = 0.8

# 计算两个协方差矩阵中的协方差值
cov_before = rho_before * sigma_A * sigma_B
cov_after  = rho_after  * sigma_A * sigma_B

# 组合方差公式：w'Σw = w_A^2 σ_A^2 + w_B^2 σ_B^2 + 2 w_A w_B Cov(A,B)
var_before = (w_A**2 * sigma_A**2) + (w_B**2 * sigma_B**2) + (2 * w_A * w_B * cov_before)
var_after  = (w_A**2 * sigma_A**2) + (w_B**2 * sigma_B**2) + (2 * w_A * w_B * cov_after)

# 组合波动率（标准差）
vol_before = np.sqrt(var_before)
vol_after  = np.sqrt(var_after)

# 将所有要求的输出存入字典
result = {
    'vol_before_annual': vol_before,
    'vol_after_annual':  vol_after
}

# 打印结果以便教师投屏查看
print("组合波动率计算结果：")
print(f"相关系数 0.3 时组合年化波动率: {vol_before:.4f} ({vol_before*100:.2f}%)")
print(f"相关系数 0.8 时组合年化波动率: {vol_after:.4f} ({vol_after*100:.2f}%)")
print("\n要求输出的字典 result:", result)
```
