```python
import numpy as np
import matplotlib.pyplot as plt

# ============================================================
# 参数设置（可根据需要修改变动幅度）
# ============================================================
FACE_VALUE = 100.0
COUPON_RATE = 0.046          # 票息率 4.6%
MATURITY = 7                 # 剩余期限（年）
YTM_INITIAL = 0.053          # 初始到期收益率 5.3%
Y_START = 0.02               # 收益率下限
Y_END = 0.09                 # 收益率上限
DY_UP = 0.01                 # +100bp 变动幅度（可调）

# ============================================================
# 现金流生成（假设每年付息一次）
# ============================================================
t = np.arange(1, MATURITY + 1)               # 各期时间
coupon_cash = FACE_VALUE * COUPON_RATE
cf = np.full(MATURITY, coupon_cash)          # 各期票息
cf[-1] += FACE_VALUE                         # 最后一期加上本金

# ============================================================
# 定价与风险指标计算
# ============================================================

def price(y):
    """计算债券的精确价格，支持标量或数组输入"""
    y = np.asarray(y)
    # 向量化计算现金流现值总和
    return np.sum(cf / (1 + y[..., np.newaxis]) ** t, axis=-1)

# 初始价格与久期
P0 = price(YTM_INITIAL).item()
pv = cf / (1 + YTM_INITIAL)**t                     # 各期现金流现值
macaulay_duration = np.sum(t * pv) / P0            # 麦考利久期
modified_duration = macaulay_duration / (1 + YTM_INITIAL)  # 修正久期

# 上升100bp后的精确价格
y_up = YTM_INITIAL + DY_UP
P_up = price(y_up).item()

# 久期法估计的相对变化（ΔP/P ≈ -ModDur * Δy）
approx_relative_change = -modified_duration * DY_UP

# ============================================================
# 绘制价格-收益率曲线与久期近似
# ============================================================
y_grid = np.linspace(Y_START, Y_END, 300)          # 收益率网格
exact_prices = price(y_grid)                       # 精确价格
# 基于久期的切线近似：P(y) ≈ P0 * [1 - ModDur * (y - y0)]
approx_prices = P0 * (1 - modified_duration * (y_grid - YTM_INITIAL))

fig, ax = plt.subplots(figsize=(9, 6))
ax.plot(y_grid * 100, exact_prices, label='Exact Price',
        color='steelblue', linewidth=2)
ax.plot(y_grid * 100, approx_prices, label='Duration Approximation (tangent at 5.3%)',
        color='tomato', linestyle='--', linewidth=1.5)

# 标注关键点
ax.scatter([YTM_INITIAL * 100, y_up * 100], [P0, P_up],
           color='black', zorder=5, s=40)
ax.axvline(YTM_INITIAL * 100, color='gray', linestyle=':', alpha=0.6)
ax.axvline(y_up * 100, color='gray', linestyle=':', alpha=0.6)

ax.set_xlabel('Yield to Maturity (%)', fontsize=12)
ax.set_ylabel('Bond Price', fontsize=12)
ax.set_title('Bond Price vs. Yield\n(Face Value=100, Coupon=4.6%, Maturity=7y, YTM=5.3%)',
             fontsize=13, fontweight='bold')
ax.legend(fontsize=10)
ax.grid(True, alpha=0.3)
plt.tight_layout()

figure_path = 'bond_price_curve.png'
fig.savefig(figure_path, dpi=150)
plt.close(fig)   # 避免在无GUI环境下弹出窗口

# ============================================================
# 组装输出结果
# ============================================================
result = {
    'price_at_up100bp': P_up,                    # +100bp后的精确价格
    'dur_approx_change_up100bp': approx_relative_change,  # 久期法估计的相对变化
    'figure_path': figure_path                   # 图形文件路径
}

# 打印以供课堂验证
if __name__ == '__main__':
    print(f"初始 YTM = {YTM_INITIAL*100:.2f}%")
    print(f"初始精确价格 = {P0:.4f}")
    print(f"麦考利久期 = {macaulay_duration:.4f} 年")
    print(f"修正久期 = {modified_duration:.4f}")
    print(f"收益率上升 100bp 后精确价格 = {P_up:.4f}")
    print(f"久期法估计的相对变化 = {approx_relative_change:.6f} ({approx_relative_change*100:.4f}%)")
    print(f"图形已保存至: {figure_path}")
    print("\n输出字典 result:")
    for k, v in result.items():
        print(f"  {k}: {v}")
```
