```python
import numpy as np
import pandas as pd

# ===== 给定参数 =====
rf = 0.023          # 无风险利率 2.3%
E_Rm = 0.094        # 市场期望收益 9.4%
market_premium = E_Rm - rf   # 市场风险溢价

# 三只股票的 beta
beta = {
    'X': 0.62,
    'Y': 1.18,
    'Z': 1.51
}

# Y 的实际年收益
actual_return_Y = 0.131   # 13.1%

# ===== 1. CAPM 期望收益计算 =====
# E[Ri] = rf + beta_i * (E[Rm] - rf)
er_x = rf + beta['X'] * market_premium
er_y = rf + beta['Y'] * market_premium
er_z = rf + beta['Z'] * market_premium

# ===== 2. 计算 Y 的 alpha =====
alpha_y = actual_return_Y - er_y

# ===== 3. 填充结果字典 =====
result = {
    'er_x': er_x,
    'er_y': er_y,
    'er_z': er_z,
    'alpha_y': alpha_y
}

# ===== 输出结果（供课堂投屏查看） =====
print("=== CAPM 计算结果 ===")
print(f"无风险利率 (rf): {rf:.4f} ({rf*100:.2f}%)")
print(f"市场期望收益 (E[Rm]): {E_Rm:.4f} ({E_Rm*100:.2f}%)")
print(f"市场风险溢价: {market_premium:.4f} ({market_premium*100:.2f}%)")
print("-" * 30)
print(f"期望收益 X (er_x): {er_x:.6f} ({er_x*100:.4f}%)")
print(f"期望收益 Y (er_y): {er_y:.6f} ({er_y*100:.4f}%)")
print(f"期望收益 Z (er_z): {er_z:.6f} ({er_z*100:.4f}%)")
print(f"Y 实际收益: {actual_return_Y:.4f} ({actual_return_Y*100:.2f}%)")
print(f"Alpha of Y: {alpha_y:.6f} ({alpha_y*100:.4f}%)")
print("-" * 30)
print("输出字典 result:")
print(result)

# 可选：用 DataFrame 美化显示
df = pd.DataFrame({
    '股票': ['X', 'Y', 'Z'],
    'Beta': [beta['X'], beta['Y'], beta['Z']],
    'CAPM 期望收益': [er_x, er_y, er_z],
    '实际收益': [np.nan, actual_return_Y, np.nan],
    'Alpha': [np.nan, alpha_y, np.nan]
})
print("\n汇总表:")
print(df.to_string(index=False))
```
