```python
import numpy as np

# 给定数据
vols = np.array([0.187, 0.243, 0.312])             # 年化波动率
corr = np.array([
    [ 1.00,  0.21, -0.13],
    [ 0.21,  1.00,  0.37],
    [-0.13,  0.37,  1.00]
])                                                  # 相关系数矩阵

# 协方差矩阵: Σ = diag(vols) @ corr @ diag(vols)
cov_matrix = np.diag(vols) @ corr @ np.diag(vols)

# 全局最小方差组合 (允许卖空，权重和为1)
ones = np.ones(3)
inv_cov = np.linalg.inv(cov_matrix)
w_mvp = inv_cov @ ones / (ones @ inv_cov @ ones)

# 组合年化波动率
mvp_var = w_mvp @ cov_matrix @ w_mvp
mvp_vol = np.sqrt(mvp_var)

# 结果字典
result = {
    'mvp_weights': w_mvp,
    'mvp_vol_annual': mvp_vol
}

# 打印验证
if __name__ == "__main__":
    np.set_printoptions(precision=6, suppress=True)
    print("Global Minimum Variance Portfolio (GMVP)")
    print(f"Weights: {w_mvp}")
    print(f"Sum of weights: {w_mvp.sum():.6f}")
    print(f"Annualized Volatility: {mvp_vol:.6f}")
    print("\nResult dictionary:")
    print(result)
```
