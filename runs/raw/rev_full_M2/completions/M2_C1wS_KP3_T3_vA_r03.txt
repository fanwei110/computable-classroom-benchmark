```python
import numpy as np
from scipy.optimize import brentq

# ============================================================
# 债券参数设定
# ============================================================
face_value = 100.0          # 面值
coupon_rate = 0.046         # 年票息率 4.6%
ytm = 0.053                 # 当前到期收益率 5.3%
maturity = 7                # 剩余期限（年）
coupon_freq = 1             # 年付息次数（默认按年付息）
dy = 0.0080                 # 收益率变动 +80 bps

# ============================================================
# 现金流时间序列
# ============================================================
times = np.arange(1, maturity + 1)           # [1,2,...,7]
cash_flows = np.full(maturity, face_value * coupon_rate)  # 每年票息
cash_flows[-1] += face_value                 # 最后一年加本金

# ============================================================
# 贴现函数：给定收益率 y，计算全价（非净价）
# ============================================================
def bond_price(y):
    """按年复利贴现计算债券现值"""
    discount_factors = (1 + y) ** (-times)
    return np.sum(cash_flows * discount_factors)

# ============================================================
# 当前价格
# ============================================================
P0 = bond_price(ytm)

# ============================================================
# 麦考利久期、修正久期、凸性（解析计算）
# ============================================================
# 权重 = PV(CF) / P0
pv_cf = cash_flows * (1 + ytm) ** (-times)
weights = pv_cf / P0

# 麦考利久期（年）
macaulay_duration = np.sum(weights * times)

# 修正久期（年）
modified_duration = macaulay_duration / (1 + ytm)

# 凸性（年²）
convexity = np.sum(weights * times * (times + 1)) / (1 + ytm)**2

# ============================================================
# 使用二阶泰勒近似估算收益率 +80bps 时的价格变动百分比
# ============================================================
# ΔP/P ≈ -MD * Δy + 0.5 * Conv * (Δy)^2
delta_y = dy
price_change_approx = -modified_duration * delta_y + 0.5 * convexity * (delta_y ** 2)

# 转换为百分比跌幅（负号表示价格下跌，我们存正值）
price_drop_pct_approx = -price_change_approx * 100.0

# ============================================================
# 为更准确，用实际全价重算验证（非必需，但可确保泰勒近似合理）
# ============================================================
P1_actual = bond_price(ytm + dy)
price_drop_pct_actual = (P0 - P1_actual) / P0 * 100.0

# 采用实际精确计算的结果作为最终输出
result = {
    'price_drop_pct': round(price_drop_pct_actual, 6)
}

# ============================================================
# 打印结果，便于课堂展示
# ============================================================
print("===== 7年期债券：面值100、票息4.6%、收益率5.3% =====")
print(f"当前全价: {P0:.4f}")
print(f"麦考利久期: {macaulay_duration:.4f} 年")
print(f"修正久期:   {modified_duration:.4f} 年")
print(f"凸性:       {convexity:.4f} 年²")
print(f"收益率上升 80 bps 后全价: {P1_actual:.4f}")
print(f"泰勒近似价格跌幅: {price_drop_pct_approx:.4f}%")
print(f"实际价格跌幅:     {price_drop_pct_actual:.4f}%")
print("\nresult =", result)
```
