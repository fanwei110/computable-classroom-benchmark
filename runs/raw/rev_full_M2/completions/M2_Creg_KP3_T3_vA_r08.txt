import numpy as np

def bond_price(face_value, coupon_rate, yield_to_maturity, maturity, freq=2):
    """
    计算债券价格（假设半年付息，freq=2）。
    参数：
        face_value: 面值
        coupon_rate: 年票息率（小数）
        yield_to_maturity: 年化到期收益率（小数）
        maturity: 剩余年限
        freq: 年付息次数，默认 2
    返回：
        债券价格
    """
    periods = int(maturity * freq)
    coupon = face_value * coupon_rate / freq
    ytm_per = yield_to_maturity / freq
    # 现金流时间点 1..periods
    t = np.arange(1, periods + 1)
    # 现金流现值
    pv_coupons = np.sum(coupon / (1 + ytm_per) ** t)
    pv_face = face_value / (1 + ytm_per) ** periods
    return pv_coupons + pv_face

def main():
    # 债券参数
    face_value = 100.0
    coupon_rate = 0.046
    yield_initial = 0.053
    yield_shift = 0.0080  # 80 个基点
    maturity = 7.0
    freq = 2  # 半年付息，符合常见债券惯例

    # 计算初始价格
    price_initial = bond_price(face_value, coupon_rate, yield_initial, maturity, freq)

    # 收益率上升后的新价格
    yield_new = yield_initial + yield_shift
    price_new = bond_price(face_value, coupon_rate, yield_new, maturity, freq)

    # 价格跌幅百分比（正数表示下跌）
    price_drop_pct = (price_initial - price_new) / price_initial * 100.0

    # 输出契约
    result = {'price_drop_pct': price_drop_pct}
    print(result)

if __name__ == '__main__':
    main()
