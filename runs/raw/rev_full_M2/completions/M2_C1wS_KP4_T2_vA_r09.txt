```python
import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt

# ============================================================
# 可调参数
# ============================================================
K = 97.5            # 行权价
r = 0.043           # 无风险利率（连续复利）
T = 0.58            # 剩余到期时间（年）
S_min, S_max = 70, 140   # 标的价格范围
S_points = 500      # 网格点数

# 波动率列表（可自由增删或修改数值）
volatilities = [0.15, 0.276, 0.40]

# 指定报告点
S_report = 110.0
sigma_report = 0.276

# ============================================================
# 函数定义：欧式看涨期权的 delta
# ============================================================
def bs_call_delta(S, K, r, T, sigma):
    """Black-Scholes 欧式看涨期权 delta: N(d1)"""
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    return norm.cdf(d1)

# ============================================================
# 1. 计算网格上的 delta
# ============================================================
S_grid = np.linspace(S_min, S_max, S_points)

# ============================================================
# 2. 画图
# ============================================================
fig, ax = plt.subplots(figsize=(10, 6))

for sigma in volatilities:
    delta_vals = bs_call_delta(S_grid, K, r, T, sigma)
    ax.plot(S_grid, delta_vals, lw=2, label=f'σ = {sigma*100:.1f}%')

# 标注行权价
ax.axvline(x=K, color='gray', linestyle='--', alpha=0.7, label=f'Strike = {K}')

ax.set_xlabel('Spot Price S', fontsize=12)
ax.set_ylabel('Delta', fontsize=12)
ax.set_title('Black-Scholes Call Delta vs Spot Price', fontsize=14)
ax.legend(fontsize=11)
ax.grid(True, alpha=0.3)

figure_path = 'delta_vs_spot.png'
fig.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close(fig)

# ============================================================
# 3. 报告单点 delta：S=110, σ=27.6%
# ============================================================
delta_at_s110 = bs_call_delta(S_report, K, r, T, sigma_report)

# ============================================================
# 输出字典
# ============================================================
result = {
    'delta_at_s110': delta_at_s110,
    'figure_path': figure_path
}

# 课堂展示时可直接打印结果
if __name__ == '__main__':
    print(f"Delta at S={S_report}, σ={sigma_report*100:.1f}%: {delta_at_s110:.6f}")
    print(f"Figure saved to: {figure_path}")
```
