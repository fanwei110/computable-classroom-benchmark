import numpy as np
from scipy.stats import norm

def black_scholes_price(S, K, T, r, sigma, option_type='call'):
    """
    Calculate the Black-Scholes price of a European option.
    """
    d1 = (np.log(S / K) + (r + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    if option_type == 'call':
        price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
    else:
        price = K * np.exp(-r * T) * norm.cdf(-d2) - S * norm.cdf(-d1)
    return price

# Given market data
S = 103.7        # underlying price
K = 97.5         # strike price
T = 0.58         # time to maturity (years)
r = 0.043        # risk-free rate
sigma_current = 0.276  # current implied volatility (27.6%)
sigma_new = sigma_current + 0.01  # IV increases by 1 percentage point

# Price the option under current and new volatility
price_before = black_scholes_price(S, K, T, r, sigma_current)
price_after = black_scholes_price(S, K, T, r, sigma_new)

# Price change due to 1% increase in IV
price_change = price_after - price_before

# Store result in expected format
result = {'price_change': price_change}

# Output for verification
print(result)
