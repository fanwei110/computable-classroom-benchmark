```python
import numpy as np
from scipy.stats import norm

# =========================
# 输入参数
# =========================
position_value = 1_850_000.0   # 头寸价值（人民币）
annual_vol = 0.218             # 年化波动率
trading_days_per_year = 252    # 年交易日数（市场惯例）

# =========================
# 波动率换算
# =========================
daily_vol = annual_vol / np.sqrt(trading_days_per_year)            # 日波动率
vol_10d = daily_vol * np.sqrt(10)                                  # 10日波动率

# =========================
# 正态分位数（置信水平对应的标准正态分位点）
# =========================
z_95 = norm.ppf(0.95)   # 95% 置信水平单侧分位数 ≈ 1.6449
z_99 = norm.ppf(0.99)   # 99% 置信水平单侧分位数 ≈ 2.3263

# =========================
# 计算 VaR（参数法，均值假设为 0）
# =========================
var_95_1d = position_value * daily_vol * z_95   # 1日 95% VaR
var_99_10d = position_value * vol_10d * z_99    # 10日 99% VaR

# =========================
# 输出结果（保留两位小数）
# =========================
result = {
    'var_95_1d': round(var_95_1d, 2),
    'var_99_10d': round(var_99_10d, 2)
}

# 打印以便投屏查看
print(result)
```
