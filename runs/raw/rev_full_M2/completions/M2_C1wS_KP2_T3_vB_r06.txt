```python
import numpy as np

# 给定数据
beta = 1.42
market_return_monthly = -0.058  # 大盘跌5.8%
rf_annual = 0.047               # 年化无风险利率

# 1. 年化无风险利率折算为月利率
# 采用单利折算（金融教学中常见处理），即直接除以12
rf_monthly = rf_annual / 12

# 2. CAPM 公式： E(R) = rf + beta * (Rm - rf)
# 注意：所有收益率均为月度值
capm_return_monthly = rf_monthly + beta * (market_return_monthly - rf_monthly)

# 3. 填写结果字典
result = {
    'capm_return_monthly': capm_return_monthly
}

# 输出结果，供课堂投屏展示
print("CAPM 计算过程与结果")
print(f"Beta: {beta}")
print(f"上月市场收益率（Rm）: {market_return_monthly:.4f} ({market_return_monthly*100:.2f}%)")
print(f"年化无风险利率: {rf_annual*100:.2f}%")
print(f"折算月化无风险利率（单利/12）: {rf_monthly:.6f} ({rf_monthly*100:.4f}%)")
print(f"根据 CAPM 计算的上月期望收益: {capm_return_monthly:.6f} ({capm_return_monthly*100:.4f}%)")
print()
print("结果字典：")
print(result)
```
