```python
import numpy as np
import scipy  # noqa: F401 (虽然未直接使用，但按要求导入)
import pandas as pd  # noqa: F401
import matplotlib.pyplot as plt  # noqa: F401

def main():
    # 给定参数
    beta = 1.42
    market_return_monthly = -0.058  # 上个月市场跌了5.8%
    risk_free_annual = 0.047        # 无风险利率年化4.7%

    # 将年化无风险利率转换为月利率（采用简单比例法，也可使用复利，但通常按比例）
    risk_free_monthly = risk_free_annual / 12.0

    # 根据CAPM计算预期月收益: E(R) = Rf + beta * (Rm - Rf)
    expected_return = risk_free_monthly + beta * (market_return_monthly - risk_free_monthly)

    # 存储结果到字典
    result = {
        'capm_return_monthly': expected_return
    }

    # 输出结果
    print("计算结果：")
    print(f"月无风险利率: {risk_free_monthly:.6f} ({risk_free_monthly*100:.4f}%)")
    print(f"市场月收益: {market_return_monthly:.6f} ({market_return_monthly*100:.2f}%)")
    print(f"CAPM预期月收益: {expected_return:.6f} ({expected_return*100:.4f}%)")
    print("\n结果字典:")
    print(result)

if __name__ == "__main__":
    main()
```
