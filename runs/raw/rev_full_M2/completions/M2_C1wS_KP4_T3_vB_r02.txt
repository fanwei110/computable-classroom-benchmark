```python
import numpy as np
from scipy.stats import norm

# ---------- 期权参数 ----------
S = 103.7          # 标的资产价格
K = 97.5           # 执行价格
r = 0.043          # 无风险利率（连续复利）
T = 0.58           # 剩余期限（年）
sigma0 = 0.276     # 当前隐含波动率 (27.6%)
sigma1 = 0.286     # 上涨1个百分点后的波动率 (28.6%)

# ---------- Black-Scholes 看涨期权定价 ----------
def bs_call_price(S, K, r, sigma, T):
    d1 = (np.log(S / K) + (r + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    return S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)

price0 = bs_call_price(S, K, r, sigma0, T)
price1 = bs_call_price(S, K, r, sigma1, T)

# 期权价格变化（Vega 效应，vega 对看涨和看跌相同）
price_change = price1 - price0

# ---------- 输出结果 ----------
result = {'price_change': price_change}
print(result)
```
