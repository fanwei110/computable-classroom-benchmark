import numpy as np

def bond_price(face, coupon_rate, ytm, maturity, freq=1):
    """
    计算债券价格（年复利，freq=1表示每年付息一次）。
    可扩展freq，但本题明确未给频率，默认1。
    """
    periods = int(maturity * freq)
    coupon = face * coupon_rate / freq
    t = np.arange(1, periods + 1)
    cf = np.full(periods, coupon)
    cf[-1] += face  # 最后一期还本
    pv_factors = (1 + ytm / freq) ** (-t / freq)  # 通用但freq=1时即 (1+ytm)^-t
    price = np.sum(cf * pv_factors)
    return price

def bond_duration_convexity(face, coupon_rate, ytm, maturity, freq=1):
    """
    计算麦考利久期、修正久期、凸性。
    采用年复利，freq=1。
    """
    periods = int(maturity * freq)
    coupon = face * coupon_rate / freq
    t = np.arange(1, periods + 1)
    cf = np.full(periods, coupon)
    cf[-1] += face
    y = ytm / freq
    pv = cf / (1 + y) ** (t / freq)
    price = np.sum(pv)
    weights = pv / price
    mac_dur = np.sum(weights * t / freq)
    mod_dur = mac_dur / (1 + y)
    convexity = np.sum(weights * (t / freq) * (t / freq + 1)) / (1 + y) ** 2
    return mac_dur, mod_dur, convexity

# ---------- 参数 ----------
face = 100.0
coupon_rate = 0.046   # 4.6%
ytm = 0.053          # 5.3%
maturity = 7          # 7年
delta_y = 0.0080      # 80个基点

# ---------- 1. 利率敏感性 ----------
price0 = bond_price(face, coupon_rate, ytm, maturity)
mac_dur, mod_dur, conv = bond_duration_convexity(face, coupon_rate, ytm, maturity)

print(f"当前价格: {price0:.4f}")
print(f"麦考利久期: {mac_dur:.4f} 年")
print(f"修正久期: {mod_dur:.4f}")
print(f"凸性: {conv:.4f}")

# ---------- 2. 收益率上升80bp的价格影响 ----------
# 精确重定价
price1 = bond_price(face, coupon_rate, ytm + delta_y, maturity)
exact_drop = (price0 - price1) / price0 * 100   # 跌幅，正数表示下跌百分比

# 久期+凸性近似
approx_change = -mod_dur * delta_y + 0.5 * conv * (delta_y ** 2)  # ΔP/P
approx_drop = -approx_change * 100   # 转为跌幅百分比（正数）

print(f"收益率上升80bp后价格: {price1:.4f}")
print(f"精确价格跌幅: {exact_drop:.4f}%")
print(f"久期凸性近似跌幅: {approx_drop:.4f}%")

# ---------- 3. 存入 result ----------
result = {
    'price_drop_pct': round(exact_drop, 4)   # 精确计算的价格跌幅百分比
}

print("\n=== 输出字典 ===")
print(result)
