```python
import numpy as np
import matplotlib.pyplot as plt

# ========================
# 资产参数
# ========================
r1 = 0.071
r2 = 0.124
sigma1 = 0.163
sigma2 = 0.289
target_return = 0.10  # 目标期望收益

# 不同相关系数
rhos = [0.15, 0.45, 0.75]

# ========================
# 辅助函数
# ========================
def portfolio_stats(w, r1, r2, sigma1, sigma2, rho):
    """给定权重w（资产1的比例），返回组合期望收益和标准差"""
    rp = w * r1 + (1 - w) * r2
    var = w**2 * sigma1**2 + (1 - w)**2 * sigma2**2 + 2 * w * (1 - w) * rho * sigma1 * sigma2
    sd = np.sqrt(np.maximum(var, 0))  # 数值安全
    return rp, sd

def mvp_weight(r1, r2, sigma1, sigma2, rho):
    """全局最小方差组合中资产1的权重（允许卖空，满仓）"""
    cov = rho * sigma1 * sigma2
    num = sigma2**2 - cov
    den = sigma1**2 + sigma2**2 - 2 * cov
    return num / den

# ========================
# 计算所需结果
# ========================
# 相关系数0.45下的最小方差组合波动率
rho_target = 0.45
w_mvp_045 = mvp_weight(r1, r2, sigma1, sigma2, rho_target)
_, mvp_vol_045 = portfolio_stats(w_mvp_045, r1, r2, sigma1, sigma2, rho_target)

# 目标收益10%下的组合权重和波动率（唯一确定）
w_target = (target_return - r2) / (r1 - r2)
_, frontier_vol_target = portfolio_stats(w_target, r1, r2, sigma1, sigma2, rho_target)

# ========================
# 绘制均值-方差前沿
# ========================
fig, ax = plt.subplots(figsize=(8, 6))

# 为每条相关系数曲线生成数据
w_range = np.linspace(-2.5, 3.5, 500)  # 足够覆盖卖空区域

colors = ['blue', 'red', 'green']
for rho, color in zip(rhos, colors):
    # 计算整个前沿
    rp_vals, sd_vals = portfolio_stats(w_range, r1, r2, sigma1, sigma2, rho)
    ax.plot(sd_vals, rp_vals, color=color, label=f'ρ = {rho}')
    
    # 找出并标注最小方差组合
    w_mvp = mvp_weight(r1, r2, sigma1, sigma2, rho)
    rp_mvp, sd_mvp = portfolio_stats(w_mvp, r1, r2, sigma1, sigma2, rho)
    ax.scatter(sd_mvp, rp_mvp, color=color, marker='o', s=60, zorder=5)
    ax.annotate(f'MVP\n({sd_mvp:.3f}, {rp_mvp:.3f})',
                (sd_mvp, rp_mvp),
                textcoords="offset points",
                xytext=(10, 10),
                fontsize=8,
                color=color)

# 单独标记目标收益点（仅rho=0.45的情况）
ax.scatter(frontier_vol_target, target_return, color='red', marker='s', s=80, zorder=6)
ax.annotate(f'Target 10%\n({frontier_vol_target:.3f}, {target_return:.2f})',
            (frontier_vol_target, target_return),
            textcoords="offset points",
            xytext=(10, -15),
            fontsize=9,
            color='darkred',
            fontweight='bold')

# 图形格式
ax.set_xlabel('Annualized Volatility (Standard Deviation)')
ax.set_ylabel('Expected Return')
ax.set_title('Mean-Variance Frontier for Two Risky Assets')
ax.legend()
ax.grid(True, linestyle='--', alpha=0.6)

# 保存图像
figure_path = 'mean_variance_frontier.png'
fig.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close(fig)

# ========================
# 存储结果
# ========================
result = {
    'mvp_vol_at_rho45': mvp_vol_045,
    'frontier_vol_at_target': frontier_vol_target,
    'figure_path': figure_path
}

# 可选输出
print(result)
```
