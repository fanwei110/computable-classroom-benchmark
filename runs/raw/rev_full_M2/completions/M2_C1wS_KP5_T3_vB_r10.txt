```python
import numpy as np
from scipy.stats import norm

# ---------- 参数设置 ----------
# 头寸价值（元）
position = 2_700_000.0
# 年化波动率
sigma_annual = 0.24
# 一年交易日数（行业惯例）
trading_days = 252
# 置信水平
confidence = 0.95

# ---------- 1. 年化波动率换算为一日波动率 ----------
sigma_daily = sigma_annual / np.sqrt(trading_days)

# ---------- 2. 95% 正态分位数 ----------
# 对于 95% 置信水平，正态分布的分位数为 z_{0.95}
z_score = norm.ppf(confidence)

# ---------- 3. 计算一日 VaR ----------
# 假设期望收益为 0，VaR = 头寸 * 一日波动率 * 正态分位数
var_95_1d = position * sigma_daily * z_score

# ---------- 输出结果 ----------
result = {
    'var_95_1d': var_95_1d
}

print(result)
```
