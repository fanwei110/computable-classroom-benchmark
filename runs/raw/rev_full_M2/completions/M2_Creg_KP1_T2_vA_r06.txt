```python
import numpy as np
import matplotlib.pyplot as plt

# ------------------ 参数 ------------------
mu = np.array([0.071, 0.124])          # 期望年收益
sigma = np.array([0.163, 0.289])       # 年化波动率
rhos = [0.15, 0.45, 0.75]             # 需要绘图的相关系数
rho_target = 0.45                      # 用于报告的相关系数
target_return = 0.10                   # 目标期望收益

# 组合统计量计算函数
def portfolio_stats(w: float, rho: float):
    """给定资产1权重 w，计算组合期望收益与波动率"""
    w2 = 1.0 - w
    mu_p = w * mu[0] + w2 * mu[1]
    var_p = (w**2 * sigma[0]**2 +
             w2**2 * sigma[1]**2 +
             2 * w * w2 * rho * sigma[0] * sigma[1])
    return mu_p, np.sqrt(var_p)

def min_variance_weights(rho: float):
    """给定相关系数，返回最小方差组合的资产1权重"""
    cov = rho * sigma[0] * sigma[1]
    denom = sigma[0]**2 + sigma[1]**2 - 2 * cov
    w1 = (sigma[1]**2 - cov) / denom
    return w1

# ------------------ 计算所需输出 ------------------
# 相关系数0.45时的最小方差组合波动率
w_mvp_45 = min_variance_weights(rho_target)
_, mvp_vol_45 = portfolio_stats(w_mvp_45, rho_target)   # 小数

# 目标收益10%下的波动率 (唯一权重)
w_target = (target_return - mu[1]) / (mu[0] - mu[1])
_, front_vol_target = portfolio_stats(w_target, rho_target)

# ------------------ 绘图 ------------------
fig, ax = plt.subplots(figsize=(8, 6))

# 生成权重范围，保证曲线覆盖足够的收益区间
w_range = np.linspace(-1.5, 2.5, 600)  # 足够宽以展示卖空区域

for rho in rhos:
    mu_curve = []
    sigma_curve = []
    for w in w_range:
        mu_p, sig_p = portfolio_stats(w, rho)
        mu_curve.append(mu_p * 100)      # 转换为百分比
        sigma_curve.append(sig_p * 100)
    ax.plot(sigma_curve, mu_curve, label=f'ρ = {rho}')

    # 计算并标出最小方差组合
    w_mvp = min_variance_weights(rho)
    mu_mvp, sig_mvp = portfolio_stats(w_mvp, rho)
    ax.scatter(sig_mvp * 100, mu_mvp * 100, s=50,
               zorder=5, edgecolors='k', linewidth=0.5)
    # 添加简要文字标注
    ax.annotate(f'MVP\n({sig_mvp*100:.1f}%, {mu_mvp*100:.1f}%)',
                xy=(sig_mvp*100, mu_mvp*100),
                xytext=(sig_mvp*100 + 2, mu_mvp*100 - 0.3),
                fontsize=8,
                arrowprops=dict(arrowstyle='->', lw=0.8),
                )

ax.set_xlabel('Annualized Volatility (%)')
ax.set_ylabel('Expected Annual Return (%)')
ax.set_title('Mean–Variance Frontier for Two Risky Assets')
ax.legend()
ax.grid(True, linestyle='--', alpha=0.6)

figure_path = 'mean_variance_frontier.png'
fig.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close(fig)

# ------------------ 输出结果 ------------------
result = {
    'mvp_vol_at_rho45': mvp_vol_45,
    'frontier_vol_at_target': front_vol_target,
    'figure_path': figure_path
}

# 打印结果以验证
print(result)
```
