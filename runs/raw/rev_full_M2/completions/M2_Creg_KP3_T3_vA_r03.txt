```python
import numpy as np

# 债券参数
face_value = 100.0        # 面值
coupon_rate = 0.046       # 票息率
yield_initial = 0.053     # 初始收益率
nper = 7                  # 期限（年）
coupon = face_value * coupon_rate  # 年付息额

# 收益率上升 80 个基点 (1 bp = 0.01%)
yield_change = 80 / 10000  # 0.0080
yield_new = yield_initial + yield_change

# 计算初始价格（使用 numpy.pv，因为是现金流出，取负号得到正价格）
price_initial = -np.pv(yield_initial, nper, coupon, face_value)
# 计算新价格
price_new = -np.pv(yield_new, nper, coupon, face_value)

# 价格下跌百分比
price_drop_pct = (price_initial - price_new) / price_initial * 100

# 按照输出契约，存入字典
result = {
    'price_drop_pct': price_drop_pct
}
```
