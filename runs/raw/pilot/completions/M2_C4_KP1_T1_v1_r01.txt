```python
import numpy as np

# 1. 构造协方差矩阵
sigma = np.array([0.187, 0.243, 0.312])  # 年化波动率（小数）

# 相关系数矩阵
corr = np.array([
    [1.00,  0.21, -0.13],
    [0.21,  1.00,  0.37],
    [-0.13, 0.37,  1.00]
])

# 计算协方差矩阵: Σ_ij = σ_i * σ_j * ρ_ij
Sigma = np.outer(sigma, sigma) * corr

# 2. 最小方差组合权重（允许卖空，满仓约束 w_sum = 1）
# 闭式解: w = (Σ^{-1} * 1) / (1^T * Σ^{-1} * 1)
ones = np.ones(3)
inv_Sigma = np.linalg.inv(Sigma)
w = inv_Sigma @ ones / (ones @ inv_Sigma @ ones)  # 权重向量，自动满足和为1

# 3. 组合年化波动率
mvp_var = w @ Sigma @ w
mvp_vol = np.sqrt(mvp_var)  # 因为是年化数据，结果即为年化波动率

# 4. 输出结果
result = {
    'mvp_weights': w.tolist(),   # 以列表形式存储，方便查看
    'mvp_vol_annual': mvp_vol     # 小数表示的年化波动率
}

print(result)
```
