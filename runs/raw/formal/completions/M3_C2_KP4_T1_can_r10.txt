```python
import numpy as np
from scipy.stats import norm

# ==================== 参数设定 ====================
S = 103.7       # 标的资产现价
K = 97.5        # 行权价
sigma = 0.276   # 隐含波动率（年化）
r = 0.043       # 无风险利率（年化）
T = 0.58        # 剩余期限（年）

# ==================== 步骤 1: 计算 d1 与 d2 ====================
d1 = (np.log(S / K) + (r + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
d2 = d1 - sigma * np.sqrt(T)

# ==================== 步骤 2: 计算期权价格与 Delta ====================
# 欧式看涨期权价格公式
call_price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)

# 欧式看涨期权 Delta
call_delta = norm.cdf(d1)

# ==================== 步骤 3: 计算 Vega ====================
# Vega = S * N'(d1) * sqrt(T)，表示波动率变动1单位(即100%)时，期权价格的变化量
# N'(d1) 为标准正态分布的概率密度函数
call_vega = S * norm.pdf(d1) * np.sqrt(T)

# ==================== 步骤 4: 填充 result ====================
result = {
    'call_price': call_price,
    'call_delta': call_delta,
    'call_vega': call_vega
}

# 打印结果以便课堂展示
print(f"d1 = {d1:.6f}")
print(f"d2 = {d2:.6f}")
print(f"Call Price = {call_price:.4f}")
print(f"Call Delta = {call_delta:.4f}")
print(f"Call Vega  = {call_vega:.4f}")
```
