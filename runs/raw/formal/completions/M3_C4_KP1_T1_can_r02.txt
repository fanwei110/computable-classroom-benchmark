```python
import numpy as np

# ===== 1. 由波动率与相关系数构造协方差矩阵 =====
vols = np.array([0.187, 0.243, 0.312])
corr = np.array([
    [1.00,  0.21, -0.13],
    [0.21,  1.00,  0.37],
    [-0.13, 0.37,  1.00]
])

# 协方差矩阵 Σ: Σ_ij = ρ_ij * σ_i * σ_j
cov_matrix = np.outer(vols, vols) * corr

# ===== 2. 求最小方差权重，归一化使其和为 1 =====
ones = np.ones(len(vols))

# 闭式解：w_raw = Σ^{-1} * 1
cov_inv = np.linalg.inv(cov_matrix)
w_raw = cov_inv @ ones

# 归一化：w_mvp = w_raw / sum(w_raw)，使资金全部投出 (∑w = 1)
w_mvp = w_raw / np.sum(w_raw)

# ===== 3. 组合波动率取 w'Σw 的平方根，用小数表示 =====
# 组合方差
var_mvp = w_mvp @ cov_matrix @ w_mvp
# 年化波动率
vol_mvp_annual = np.sqrt(var_mvp)

# ===== 4. 按要求键名填充 result =====
result = {
    'mvp_weights': w_mvp.tolist(),
    'mvp_vol_annual': float(vol_mvp_annual)
}

# 控制台输出供教师投屏检验
print(f"全局最小方差组合权重: {result['mvp_weights']}")
print(f"全局最小方差组合年化波动率: {result['mvp_vol_annual']:.5f}")
```
