```python
import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt
import os

# Black-Scholes 欧式看涨期权 delta 计算函数
def black_scholes_call_delta(S, K, T, r, sigma):
    """
    计算欧式看涨期权的 delta (N(d1))

    参数:
    S : 标的资产价格
    K : 行权价
    T : 剩余期限（年）
    r : 无风险利率（连续复利）
    sigma : 波动率（小数）

    返回:
    delta : 期权的 delta 值
    """
    d1 = (np.log(S / K) + (r + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    delta = norm.cdf(d1)
    return delta

# 参数设置
K = 97.5          # 行权价
r = 0.043         # 无风险利率（4.3%）
T = 0.58          # 剩余期限（0.58 年）
S_range = np.linspace(70, 140, 100)  # 标的价格范围

# 波动率列表
sigmas = [0.15, 0.276, 0.40]

# 计算不同波动率下的 delta 曲线
plt.figure(figsize=(10, 6))
for sigma in sigmas:
    deltas = black_scholes_call_delta(S_range, K, T, r, sigma)
    plt.plot(S_range, deltas, label=f'σ = {sigma * 100:.1f}%')

# 图形设置
plt.title('欧式看涨期权 Delta 曲线 (K=97.5, r=4.3%, T=0.58年)')
plt.xlabel('标的资产价格')
plt.ylabel('Delta')
plt.legend()
plt.grid(True)

# 保存图形
figure_path = 'bs_delta_curves.png'
plt.savefig(figure_path)
plt.close()

# 计算特定条件下的 delta
S_target = 110
sigma_target = 0.276
delta_at_s110 = black_scholes_call_delta(S_target, K, T, r, sigma_target)

# 构建结果字典
result = {
    'delta_at_s110': delta_at_s110,
    'figure_path': os.path.abspath(figure_path)
}

# 输出结果（供课堂展示）
print(result)
```
