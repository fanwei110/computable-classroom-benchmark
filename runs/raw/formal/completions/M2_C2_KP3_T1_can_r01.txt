```python
import numpy as np

def main():
    # 债券参数
    face_value = 100.0
    coupon_rate = 0.046       # 票息率 4.6%
    years = 7                 # 期限
    ytm = 0.053              # 到期收益率 5.3%
    coupon = face_value * coupon_rate  # 年票息

    # 时间点：1, 2, ..., 7
    t = np.arange(1, years + 1)

    # 现金流：每年票息 + 最后一年还本
    cash_flows = np.full(years, coupon)
    cash_flows[-1] += face_value

    # 贴现因子及现值
    discount_factors = (1 + ytm) ** (-t)
    pv_flows = cash_flows * discount_factors

    # 1. 价格
    price = np.sum(pv_flows)

    # 2. 麦考利久期 = sum(t * PV) / price
    macaulay_duration = np.sum(t * pv_flows) / price

    # 3. 修正久期 = 麦考利久期 / (1 + ytm)
    modified_duration = macaulay_duration / (1 + ytm)

    # 4. 凸性 = sum( t*(t+1) * pv_flows ) / ( price * (1+ytm)^2 )
    convexity = np.sum(t * (t + 1) * pv_flows) / (price * (1 + ytm) ** 2)

    # 输出字典
    result = {
        'price': round(price, 6),
        'macaulay_duration_years': round(macaulay_duration, 6),
        'modified_duration_years': round(modified_duration, 6),
        'convexity': round(convexity, 6)
    }
    return result

if __name__ == "__main__":
    result = main()
    print(result)
```
