```python
import numpy as np

# ==================== 已知参数 ====================
rf = 2.3      # 无风险利率 (%)
rm = 9.4      # 市场期望收益 (%)
beta_x = 0.62
beta_y = 1.18
beta_z = 1.51
actual_return_y = 13.1  # 股票 Y 当年实际收益 (%)

# ==================== 计算过程 ====================
# 市场风险溢价
market_risk_premium = rm - rf

# 1. 对每个 beta 套用 CAPM：E[Ri] = rf + beta * (E[Rm] - rf)
er_x = rf + beta_x * market_risk_premium
er_y = rf + beta_y * market_risk_premium
er_z = rf + beta_z * market_risk_premium

# 2. 用 Y 的 CAPM 预测计算其 alpha：alpha = 实际收益 - 期望收益
alpha_y = actual_return_y - er_y

# 3. 填充 result
result = {
    'er_x': er_x,
    'er_y': er_y,
    'er_z': er_z,
    'alpha_y': alpha_y
}

# ==================== 课堂投屏展示 ====================
print("="*50)
print("《证券投资学》- CAPM 与证券市场线计算")
print("="*50)
print(f"无风险利率 (rf): {rf}%")
print(f"市场期望收益 (E[Rm]): {rm}%")
print(f"市场风险溢价 (E[Rm]-rf): {market_risk_premium}%")
print("-"*50)
print(f"股票 X (Beta={beta_x}) 的 CAPM 期望收益: {er_x:.4f}%")
print(f"股票 Y (Beta={beta_y}) 的 CAPM 期望收益: {er_y:.4f}%")
print(f"股票 Z (Beta={beta_z}) 的 CAPM 期望收益: {er_z:.4f}%")
print("-"*50)
print(f"股票 Y 实际收益: {actual_return_y}%")
print(f"股票 Y 的 Alpha: {alpha_y:.4f}%")
print("="*50)
print("\n输出契约字典 result:")
print(result)
```
