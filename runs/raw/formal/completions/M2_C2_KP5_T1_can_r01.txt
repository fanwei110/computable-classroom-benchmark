```python
import numpy as np
from scipy.stats import norm

# ---------- 给定参数 ----------
position_value = 1_850_000           # 头寸价值（元）
annual_vol = 0.218                   # 年化收益波动率
trading_days = 252                   # 一年中的交易日数（行业惯例）

# ---------- 1. 波动率期限转换 ----------
daily_vol = annual_vol / np.sqrt(trading_days)                # 日波动率
ten_day_vol = annual_vol * np.sqrt(10 / trading_days)         # 十日波动率

# ---------- 2. 正态分位数 ----------
z_95 = norm.ppf(0.95)   # 95% 置信水平对应的标准正态分位数（右尾，取正值）
z_99 = norm.ppf(0.99)   # 99% 置信水平对应的标准正态分位数

# ---------- 3. VaR 计算 ----------
var_95_1d = position_value * daily_vol * z_95     # 95% 一日 VaR
var_99_10d = position_value * ten_day_vol * z_99  # 99% 十日 VaR

# ---------- 4. 填充结果 ----------
result = {
    'var_95_1d': var_95_1d,
    'var_99_10d': var_99_10d
}

# 输出结果供教师查看
if __name__ == "__main__":
    print(result)
    print(f"95% 一日 VaR: {var_95_1d:,.2f} 元")
    print(f"99% 十日 VaR: {var_99_10d:,.2f} 元")
```
