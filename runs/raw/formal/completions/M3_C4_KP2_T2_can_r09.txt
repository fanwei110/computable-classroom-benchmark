```python
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.ticker import PercentFormatter

# ==================== 1. 参数化设定 ====================
# 课程约定：所有利率与收益率均使用小数表示
rf = 0.023  # 无风险利率
rm = 0.094  # 市场期望收益

# ==================== 2. 计算SML斜率与特定Beta期望收益 ====================
sml_slope = rm - rf
beta_target = 1.27
er_at_beta_127 = rf + beta_target * sml_slope

# ==================== 3. 绘制SML与股票点 ====================
# 生成 beta 从 0 到 2 的序列
betas = np.linspace(0, 2, 100)
sml_er = rf + betas * sml_slope

# 定义三只股票的数据
stocks = {
    'X': {'beta': 0.62, 'return': 0.081},
    'Y': {'beta': 1.18, 'return': 0.131},
    'Z': {'beta': 1.51, 'return': 0.099}
}

# 初始化图形
fig, ax = plt.subplots(figsize=(10, 6))

# 绘制证券市场线 (SML)
ax.plot(betas, sml_er, label=f'SML (Slope = {sml_slope:.3f})', color='royalblue', linewidth=2.5)

# 标出无风险利率与市场组合点
ax.scatter(0, rf, color='black', zorder=5)
ax.annotate(f'$r_f$ = {rf:.1%}', xy=(0, rf), xytext=(15, -15), textcoords='offset points', fontsize=11)
ax.scatter(1, rm, color='black', zorder=5)
ax.annotate(f'$E[R_m]$ = {rm:.1%}', xy=(1, rm), xytext=(15, -15), textcoords='offset points', fontsize=11)

# 绘制三只股票并带标注（设置不同偏移量避免重叠）
offsets = {'X': (15, 15), 'Y': (15, 15), 'Z': (15, -20)}
for name, data in stocks.items():
    beta_i = data['beta']
    ret_i = data['return']
    ax.scatter(beta_i, ret_i, color='red', s=70, zorder=5)
    ax.annotate(f'{name} ({ret_i:.1%})', 
                xy=(beta_i, ret_i), 
                xytext=offsets[name], 
                textcoords='offset points',
                fontsize=12, fontweight='bold',
                arrowprops=dict(arrowstyle='->', color='darkred', lw=1.5))

# 设置图表格式
ax.set_xlabel('Beta ($\\beta$)', fontsize=14)
ax.set_ylabel('Expected Return $E[R_i]$', fontsize=14)
ax.set_title('Security Market Line (SML) & Stock Alphas', fontsize=16)
ax.legend(fontsize=12, loc='upper left')
ax.grid(True, linestyle='--', alpha=0.6)
ax.set_xlim(0, 2)
ax.set_ylim(0, 0.18)
ax.yaxis.set_major_formatter(PercentFormatter(xmax=1.0, decimals=1))

# ==================== 4. 保存图形并填充结果 ====================
figure_path = 'sml_plot.png'
fig.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close()

# 将结果存入字典
result = {
    'sml_slope': sml_slope,
    'er_at_beta_127': er_at_beta_127,
    'figure_path': figure_path
}

# 供课堂投屏检验
if __name__ == '__main__':
    print(f"SML 斜率: {result['sml_slope']:.5f}")
    print(f"Beta=1.27 处 CAPM 期望收益: {result['er_at_beta_127']:.5f}")
    print(f"图形已保存至: {result['figure_path']}")
    print("Result Dictionary:", result)
```
