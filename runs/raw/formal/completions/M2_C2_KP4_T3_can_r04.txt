```python
import numpy as np
from scipy.stats import norm

# 输入参数
S = 103.7        # 标的现价
K = 97.5         # 行权价
sigma = 0.276    # 波动率（小数形式）
r = 0.043        # 无风险利率
T = 0.58         # 剩余年限
dsigma = 0.01    # 隐含波动率上升 1 个百分点（即 0.01）

# 计算 d1, d2
d1 = (np.log(S / K) + (r + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
d2 = d1 - sigma * np.sqrt(T)

# Vega：期权价格对波动率（小数形式）的偏导数
vega = S * norm.pdf(d1) * np.sqrt(T)

# 波动率上升一个百分点对应的价格变化
price_change = vega * dsigma

# 存储结果
result = {'price_change': price_change}

# 可选输出，方便查看
if __name__ == "__main__":
    print(result)
```
