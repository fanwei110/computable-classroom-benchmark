```python
import numpy as np
from scipy.stats import norm

# 输入参数
annual_vol = 0.24          # 年化波动率
position = 2_700_000       # 头寸金额
trading_days = 252         # 一年交易日数
alpha = 0.05               # 置信水平 95%

# 1. 年化波动率换算为一日波动率
daily_vol = annual_vol / np.sqrt(trading_days)

# 2. 正态分布 95% 分位数 (5% 左尾)
z_alpha = norm.ppf(alpha)   # 负值，例如 -1.64485

# 3. 一日 VaR（损失金额，正值）
var_95_1d = -position * daily_vol * z_alpha

# 存入结果字典
result = {'var_95_1d': var_95_1d}

# 输出结果（供课堂投屏查看）
print(result)
```
