```python
import numpy as np
import pandas as pd

def main():
    # -----------------------------
    # 第一部分：年化夏普比率
    # -----------------------------
    # 读取数据
    df = pd.read_csv("data/market_snapshot_v1.csv")
    # 确保存在 fund 列
    fund_daily = df["fund"].dropna()  # 日收益率（小数）

    # 参数
    rf_annual = 0.021               # 年化无风险利率
    rf_daily = rf_annual / 252      # 日无风险利率
    trading_days = 252              # 每年交易日

    # 日超额收益
    excess_daily = fund_daily - rf_daily

    # 样本标准差（ddof=1）
    daily_std = excess_daily.std(ddof=1)
    # 年化波动率
    annual_std = daily_std * np.sqrt(trading_days)

    # 年化平均超额收益
    annual_excess_return = excess_daily.mean() * trading_days

    # 年化夏普比率
    sharpe_annual = annual_excess_return / annual_std

    # -----------------------------
    # 第二部分：Brinson-Hood-Beebower 业绩归因
    # -----------------------------
    # 组合权重与行业收益
    w_p = np.array([0.45, 0.35, 0.20])
    r_p = np.array([0.083, 0.021, -0.014])

    # 基准权重与行业收益
    w_b = np.array([0.40, 0.40, 0.20])
    r_b = np.array([0.067, 0.034, -0.009])

    # 配置效应 = Σ (w_p - w_b) * r_b
    allocation_effect = np.sum((w_p - w_b) * r_b)

    # 选择效应 = Σ w_b * (r_p - r_b)
    selection_effect = np.sum(w_b * (r_p - r_b))

    # 交互效应 = Σ (w_p - w_b) * (r_p - r_b)
    interaction_effect = np.sum((w_p - w_b) * (r_p - r_b))

    # -----------------------------
    # 输出结果
    # -----------------------------
    result = {
        "sharpe_annual": sharpe_annual,
        "allocation_effect": allocation_effect,
        "selection_effect": selection_effect,
        "interaction_effect": interaction_effect,
    }

    return result

if __name__ == "__main__":
    res = main()
    print(res)
```
