**结果**  
- `ρ = 0.45` 时最小方差组合（MVP）的年化波动率：**0.1617**（约 16.17%）  
- 目标年化收益 10% 对应的最小波动率（ρ = 0.45）：**0.2024**（约 20.24%）  

```python
import numpy as np
import matplotlib.pyplot as plt

# 资产参数
r1, r2 = 0.071, 0.124
sigma1, sigma2 = 0.163, 0.289
var1, var2 = sigma1 ** 2, sigma2 ** 2
cov_base = sigma1 * sigma2

# 相关系数列表
rhos = [0.15, 0.45, 0.75]

# 用于存储结果
result = {}

def frontier(r1, r2, sigma1, sigma2, rho):
    cov = rho * sigma1 * sigma2
    var1, var2 = sigma1**2, sigma2**2
    # 最小方差组合
    w1_mvp = (var2 - cov) / (var1 + var2 - 2 * cov)
    w2_mvp = 1 - w1_mvp
    r_mvp = w1_mvp * r1 + w2_mvp * r2
    var_mvp = w1_mvp**2 * var1 + w2_mvp**2 * var2 + 2 * w1_mvp * w2_mvp * cov
    sigma_mvp = np.sqrt(var_mvp)

    # 全边界
    w1 = np.linspace(-0.5, 1.5, 600)
    w2 = 1 - w1
    r = w1 * r1 + w2 * r2
    var = w1**2 * var1 + w2**2 * var2 + 2 * w1 * w2 * cov
    sigma = np.sqrt(var)

    # 提取有效前沿（r >= r_mvp）
    efficient = r >= r_mvp
    sigma_eff = sigma[efficient]
    r_eff = r[efficient]

    return sigma_eff, r_eff, sigma_mvp, r_mvp

# 绘图
plt.figure(figsize=(10, 6))
colors = ['blue', 'orange', 'green']
for rho, color in zip(rhos, colors):
    sigma_eff, r_eff, sigma_mvp, r_mvp = frontier(r1, r2, sigma1, sigma2, rho)
    plt.plot(sigma_eff, r_eff, color=color, label=f'ρ = {rho}')
    plt.scatter(sigma_mvp, r_mvp, color=color, marker='o', s=60, zorder=5)
    # 标注 MVP
    plt.annotate(f'MVP\n({sigma_mvp:.3f}, {r_mvp:.3f})',
                 (sigma_mvp, r_mvp),
                 textcoords="offset points", xytext=(10, -10),
                 fontsize=8, color=color)

plt.xlabel('Annualized Volatility')
plt.ylabel('Annualized Expected Return')
plt.title('Efficient Frontier for Two Assets')
plt.legend()
plt.grid(True, linestyle='--', alpha=0.6)
plt.tight_layout()

# 保存图片
fig_path = 'efficient_frontier.png'
plt.savefig(fig_path, dpi=150)
plt.close()

# ----- 计算要求的值 -----
rho45 = 0.45
cov45 = rho45 * sigma1 * sigma2

# MVP波动率
w1_mvp45 = (var2 - cov45) / (var1 + var2 - 2 * cov45)
w2_mvp45 = 1 - w1_mvp45
var_mvp45 = w1_mvp45**2 * var1 + w2_mvp45**2 * var2 + 2 * w1_mvp45 * w2_mvp45 * cov45
sigma_mvp45 = np.sqrt(var_mvp45)

# 目标收益10%对应的波动率
r_target = 0.10
w1_target = (r_target - r2) / (r1 - r2)
w2_target = 1 - w1_target
var_target = w1_target**2 * var1 + w2_target**2 * var2 + 2 * w1_target * w2_target * cov45
sigma_target = np.sqrt(var_target)

result['mvp_vol_at_rho45'] = round(float(sigma_mvp45), 6)
result['frontier_vol_at_target'] = round(float(sigma_target), 6)
result['figure_path'] = fig_path

print(result)
```
