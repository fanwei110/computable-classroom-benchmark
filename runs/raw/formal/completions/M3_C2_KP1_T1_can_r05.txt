```python
import numpy as np

# =====================
# 1. 输入已知参数
# =====================
# 年化波动率
vols = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr_matrix = np.array([
    [1.0,  0.21, -0.13],
    [0.21, 1.0,   0.37],
    [-0.13, 0.37, 1.0 ]
])

# =====================
# 2. 构造协方差矩阵
# =====================
# 公式: Σ = diag(σ) @ Corr @ diag(σ)
diag_vols = np.diag(vols)
cov_matrix = diag_vols @ corr_matrix @ diag_vols

# =====================
# 3. 求解全局最小方差组合(MVP)权重
# =====================
# 满仓约束下(w'1 = 1)的最小方差组合闭式解为: w_mvp = (Σ^-1 * 1) / (1' * Σ^-1 * 1)
# 其中 1 为元素全为1的列向量
ones = np.ones(3)
cov_matrix_inv = np.linalg.inv(cov_matrix)

# 计算 Σ^-1 * 1
cov_inv_ones = cov_matrix_inv @ ones

# 计算归一化分母 1' * Σ^-1 * 1
denominator = ones @ cov_inv_ones

# 得到权重向量
mvp_weights = cov_inv_ones / denominator

# =====================
# 4. 计算组合年化波动率
# =====================
# 组合方差: σ^2 = w'Σw
# 组合波动率: σ = sqrt(w'Σw)
mvp_variance = mvp_weights @ cov_matrix @ mvp_weights
mvp_vol_annual = np.sqrt(mvp_variance)

# =====================
# 5. 按要求填充结果字典
# =====================
result = {
    'mvp_weights': mvp_weights.tolist(),
    'mvp_vol_annual': mvp_vol_annual
}

# 打印结果以便课堂投屏展示
print("协方差矩阵 Σ:")
print(np.round(cov_matrix, 6))
print("\n全局最小方差组合(MVP)结果:")
print(f"权重: {result['mvp_weights']}")
print(f"年化波动率: {result['mvp_vol_annual']:.4%}")
```
