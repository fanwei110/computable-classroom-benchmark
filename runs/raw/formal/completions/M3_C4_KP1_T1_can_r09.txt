```python
import numpy as np

# 1. 由波动率与相关系数构造协方差矩阵
# 年化波动率（小数表示）
vols = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr_matrix = np.array([
    [1.0,  0.21, -0.13],
    [0.21, 1.0,  0.37],
    [-0.13, 0.37, 1.0]
])

# 协方差矩阵 Σ = diag(σ) @ P @ diag(σ)
cov_matrix = np.diag(vols) @ corr_matrix @ np.diag(vols)

# 2. 求最小方差权重（闭式解：w_mvp = (Σ^-1 * 1) / (1' * Σ^-1 * 1)）
ones = np.ones(3)

# 使用求解器代替直接求逆以提升数值稳定性，求解 Σ * x = 1
x = np.linalg.solve(cov_matrix, ones)

# 归一化使其和为 1（满仓约束）
mvp_weights = x / np.sum(x)

# 3. 计算组合年化波动率：σ_p = sqrt(w' Σ w)
port_var = mvp_weights @ cov_matrix @ mvp_weights
mvp_vol_annual = np.sqrt(port_var)

# 4. 按要求键名填充 result
result = {
    'mvp_weights': mvp_weights.tolist(),
    'mvp_vol_annual': mvp_vol_annual
}

# 打印结果供课堂投屏查看
print(result)
```
