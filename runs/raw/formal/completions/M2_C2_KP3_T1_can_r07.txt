import numpy as np

def bond_price_duration_convexity(face_value, coupon_rate, maturity_years, ytm, freq=1):
    """
    计算债券价格、麦考利久期、修正久期和凸性。

    参数:
        face_value : float, 面值
        coupon_rate : float, 年票息率（小数形式）
        maturity_years : int, 到期年数
        ytm : float, 到期收益率（年化，小数形式）
        freq : int, 每年付息次数（默认1，即按年付息）

    返回:
        dict: 包含 'price', 'macaulay_duration_years', 
              'modified_duration_years', 'convexity' 的字典
    """
    periods = maturity_years * freq
    coupon = face_value * coupon_rate / freq   # 每期票息
    discount_rate = ytm / freq                # 每期贴现率

    # 现金流时间点（从1开始）
    t = np.arange(1, periods + 1)
    # 各期现金流（最后一期加本金）
    cash_flows = np.full(periods, coupon)
    cash_flows[-1] += face_value

    # 各期现金流的贴现因子
    discount_factors = (1 + discount_rate) ** t
    pv = cash_flows / discount_factors

    # 价格
    price = np.sum(pv)

    # 麦考利久期（以期为单位）
    macaulay_periods = np.sum(t * pv) / price
    # 转换为年
    macaulay_years = macaulay_periods / freq

    # 修正久期（年）
    modified_duration_years = macaulay_years / (1 + ytm / freq)

    # 凸性（常用公式：Convexity = sum [ t*(t+1/freq?) * PV ] / (P * (1+y/freq)^2) ）
    # 具体推导：Convexity = (1/P) * sum [ CF_t/(1+y/freq)^t * t * (t+1/freq)? ] / (1+y/freq)^2
    # 这里我们采用标准定义：
    # Convexity (periods) = sum_{t} [ CF_t / (1+y/freq)^t * t * (t+1) ] / (P * (1+y/freq)^2)
    # 然后年化凸性 = Convexity (periods) / freq^2
    # 注意：有些教材直接按年化计算。为严谨，这里按期间计算再年化。
    convexity_periods = np.sum(pv * t * (t + 1)) / (price * (1 + discount_rate)**2)
    convexity = convexity_periods / (freq ** 2)

    return {
        'price': round(float(price), 6),
        'macaulay_duration_years': round(float(macaulay_years), 6),
        'modified_duration_years': round(float(modified_duration_years), 6),
        'convexity': round(float(convexity), 6)
    }

# 给定参数
face_value = 100.0
coupon_rate = 0.046      # 4.6%
maturity_years = 7
ytm = 0.053              # 5.3%

# 按题意假设每年付息一次
result = bond_price_duration_convexity(face_value, coupon_rate, maturity_years, ytm, freq=1)

# 输出结果供查看（教师投屏需要清晰）
print("债券定价与风险指标计算结果（年付息一次）：")
for key, value in result.items():
    print(f"{key}: {value}")
