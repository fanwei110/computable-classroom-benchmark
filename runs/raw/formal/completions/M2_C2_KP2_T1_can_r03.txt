```python
import numpy as np

def main():
    # 给定参数
    rf = 0.023          # 无风险利率 2.3%
    er_m = 0.094        # 市场期望收益 9.4%
    
    # 各股票的 beta
    beta_x = 0.62
    beta_y = 1.18
    beta_z = 1.51
    
    # Y 的实际收益
    actual_y = 0.131    # 13.1%
    
    # 市场风险溢价
    market_premium = er_m - rf  # E[Rm] - rf
    
    # 1. 用 CAPM 计算各股票的期望收益
    # E[Ri] = rf + beta_i * (E[Rm] - rf)
    er_x = rf + beta_x * market_premium
    er_y = rf + beta_y * market_premium
    er_z = rf + beta_z * market_premium
    
    # 2. 计算股票 Y 的 alpha
    # alpha = 实际收益 - CAPM 预测收益
    alpha_y = actual_y - er_y
    
    # 3. 填充结果字典
    result = {
        'er_x': er_x,
        'er_y': er_y,
        'er_z': er_z,
        'alpha_y': alpha_y
    }
    
    # 打印结果供课堂查看
    print("CAPM 计算结果")
    print("=" * 40)
    print(f"无风险利率 (rf):        {rf*100:.2f}%")
    print(f"市场期望收益 (E[Rm]):   {er_m*100:.2f}%")
    print(f"市场风险溢价:           {market_premium*100:.2f}%")
    print("-" * 40)
    print(f"股票 X | beta = {beta_x:.2f} | CAPM 期望收益 = {er_x*100:.4f}%")
    print(f"股票 Y | beta = {beta_y:.2f} | CAPM 期望收益 = {er_y*100:.4f}%")
    print(f"股票 Z | beta = {beta_z:.2f} | CAPM 期望收益 = {er_z*100:.4f}%")
    print("-" * 40)
    print(f"股票 Y 实际收益:        {actual_y*100:.2f}%")
    print(f"股票 Y 的 alpha:        {alpha_y*100:.4f}%")
    print("=" * 40)
    
    return result

# 确保脚本直接运行时执行
if __name__ == "__main__":
    result = main()
```
