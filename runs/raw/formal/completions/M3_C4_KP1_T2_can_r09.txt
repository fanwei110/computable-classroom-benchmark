```python
import numpy as np
import matplotlib.pyplot as plt

# ==================== 数据与参数 ====================
mu = np.array([0.071, 0.124])       # 期望收益
sigma = np.array([0.163, 0.289])    # 波动率
rhos = [0.15, 0.45, 0.75]          # 相关系数
target_return = 0.10                # 目标收益

# ==================== 扫描权重并计算前沿 ====================
# 满仓约束: w1 + w2 = 1，设 w1 = w，则 w2 = 1 - w
w_range = np.linspace(-0.5, 1.5, 1000)
port_ret = w_range * mu[0] + (1 - w_range) * mu[1]

plt.figure(figsize=(10, 7))
colors = ['blue', 'green', 'red']

# 用来保存各个相关系数下的关键点
mvp_vol_at_rho45 = None
frontier_vol_at_target = None

for i, rho in enumerate(rhos):
    # 构造协方差矩阵
    cov_mat = np.array([
        [sigma[0]**2, rho * sigma[0] * sigma[1]],
        [rho * sigma[0] * sigma[1], sigma[1]**2]
    ])
    
    # 计算组合波动率
    port_var = (w_range**2) * cov_mat[0, 0] + \
               ((1 - w_range)**2) * cov_mat[1, 1] + \
               2 * w_range * (1 - w_range) * cov_mat[0, 1]
    port_vol = np.sqrt(port_var)
    
    # 解析求解最小方差组合权重
    # 对 Var(w) 求导令其等于0：
    # w_mvp = (sigma2^2 - rho*sigma1*sigma2) / (sigma1^2 + sigma2^2 - 2*rho*sigma1*sigma2)
    denom = sigma[0]**2 + sigma[1]**2 - 2 * rho * sigma[0] * sigma[1]
    w1_mvp = (sigma[1]**2 - rho * sigma[0] * sigma[1]) / denom
    w2_mvp = 1 - w1_mvp
    w_mvp = np.array([w1_mvp, w2_mvp])
    
    # 最小方差组合的收益与波动率
    mvp_ret = w_mvp @ mu
    mvp_var = w_mvp @ cov_mat @ w_mvp
    mvp_vol = np.sqrt(mvp_var)
    
    # 针对 rho = 0.45 记录要求的数据
    if rho == 0.45:
        mvp_vol_at_rho45 = mvp_vol
        
        # 目标收益下的最小方差组合权重（解析解）
        # w1_target = (target_ret - mu2) / (mu1 - mu2)
        w1_target = (target_return - mu[1]) / (mu[0] - mu[1])
        w2_target = 1 - w1_target
        w_target = np.array([w1_target, w2_target])
        
        target_var = w_target @ cov_mat @ w_target
        frontier_vol_at_target = np.sqrt(target_var)
    
    # 画前沿曲线
    plt.plot(port_vol, port_ret, color=colors[i], 
             label=f'ρ = {rho:.2f}')
    
    # 标出最小方差组合点
    plt.scatter(mvp_vol, mvp_ret, color=colors[i], zorder=5)
    plt.annotate(f'MVP (ρ={rho:.2f})\nσ={mvp_vol:.2%}, μ={mvp_ret:.2%}',
                 xy=(mvp_vol, mvp_ret), 
                 xytext=(mvp_vol + 0.015, mvp_ret - 0.015),
                 arrowprops=dict(facecolor='black', shrink=0.05, width=1, headwidth=5),
                 fontsize=9)

# ==================== 图表美化 ====================
plt.title('Mean-Variance Frontiers for Different Correlations', fontsize=14)
plt.xlabel('Volatility (σ)', fontsize=12)
plt.ylabel('Expected Return (μ)', fontsize=12)
plt.legend(fontsize=11)
plt.grid(True, linestyle='--', alpha=0.7)
plt.xlim(0, 0.45)
plt.ylim(0.02, 0.18)

figure_path = 'markowitz_frontiers.png'
plt.savefig(figure_path, dpi=300, bbox_inches='tight')
plt.close()

# ==================== 填充结果 ====================
result = {
    'mvp_vol_at_rho45': mvp_vol_at_rho45,
    'frontier_vol_at_target': frontier_vol_at_target,
    'figure_path': figure_path
}

# 打印验证（可选，便于课堂展示）
print(f"相关系数 0.45 时的最小方差组合年化波动率: {result['mvp_vol_at_rho45']:.6f}")
print(f"目标期望收益 10% 下可达到的最小年化波动率: {result['frontier_vol_at_target']:.6f}")
```
