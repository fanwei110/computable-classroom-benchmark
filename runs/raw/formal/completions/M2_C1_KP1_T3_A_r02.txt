```python
import math

# 给定的参数
sigma_A = 0.184  # A的波动率
sigma_B = 0.297  # B的波动率
w_A = 0.6        # A的权重
w_B = 0.4        # B的权重
rho_before = 0.3 # 原相关系数
rho_after = 0.8  # 新相关系数

# 计算组合方差
var_before = (w_A**2 * sigma_A**2 +
              w_B**2 * sigma_B**2 +
              2 * w_A * w_B * rho_before * sigma_A * sigma_B)
var_after = (w_A**2 * sigma_A**2 +
             w_B**2 * sigma_B**2 +
             2 * w_A * w_B * rho_after * sigma_A * sigma_B)

# 年化波动率
vol_before_annual = math.sqrt(var_before)
vol_after_annual = math.sqrt(var_after)

result = {
    'vol_before_annual': round(vol_before_annual, 6),
    'vol_after_annual': round(vol_after_annual, 6)
}
result
```
