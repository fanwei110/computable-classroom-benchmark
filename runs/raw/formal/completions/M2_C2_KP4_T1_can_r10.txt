```python
import numpy as np
from scipy.stats import norm

# ============================================================
# 参数设定
# ============================================================
S0 = 103.7        # 标的现价
K = 97.5          # 行权价
sigma = 0.276     # 年化隐含波动率 (27.6%)
r = 0.043         # 年化无风险利率 (4.3%)
T = 0.58          # 剩余期限 (年)

# ============================================================
# 1. 计算 d1 与 d2
# ============================================================
d1 = (np.log(S0 / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
d2 = d1 - sigma * np.sqrt(T)

# ============================================================
# 2. 计算欧式看涨期权价格与 Delta
# ============================================================
call_price = S0 * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
call_delta = norm.cdf(d1)   # 看涨期权的 Delta

# ============================================================
# 3. 计算 Vega (∂C/∂σ，波动率每变动1单位的期权价格变化)
# ============================================================
call_vega = S0 * np.sqrt(T) * norm.pdf(d1)

# ============================================================
# 4. 填充结果字典
# ============================================================
result = {
    'call_price': call_price,
    'call_delta': call_delta,
    'call_vega': call_vega
}

# 打印结果以供投屏展示
print(result)
```
