import numpy as np

# 资产参数
sigma_A = 0.184  # 年化波动率 A
sigma_B = 0.297  # 年化波动率 B
w = np.array([0.6, 0.4])  # A 占 60%, B 占 40%

# 协方差矩阵函数
def cov_matrix(rho):
    return np.array([[sigma_A**2, rho * sigma_A * sigma_B],
                     [rho * sigma_A * sigma_B, sigma_B**2]])

# 计算组合波动率
def portfolio_vol(Sigma, w):
    return np.sqrt(w.T @ Sigma @ w)
