```python
import math

# 给定参数
annual_vol = 0.24               # 年化波动率
position = 2_700_000            # 头寸价值
confidence = 0.95               # 置信水平
trading_days = 252              # 年交易日数

# 计算日波动率
daily_vol = annual_vol / math.sqrt(trading_days)

# 95% 置信水平对应的正态分布分位数（单尾）
z_score = abs(-1.6448536269514722)  # scipy.stats.norm.ppf(0.05) 的绝对值

# VaR 计算
var_95_1d = position * daily_vol * z_score

# 按约定存入字典
result = {'var_95_1d': round(var_95_1d, 2)}
```
