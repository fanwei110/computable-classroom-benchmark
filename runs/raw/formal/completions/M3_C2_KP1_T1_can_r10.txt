```python
import numpy as np

# =====================
# 1. 构造协方差矩阵
# =====================
# 年化波动率
volatilities = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr_matrix = np.array([
    [1.0,  0.21, -0.13],
    [0.21, 1.0,   0.37],
    [-0.13, 0.37, 1.0 ]
])

# 协方差矩阵：Σ = diag(σ) @ Corr @ diag(σ)
cov_matrix = np.outer(volatilities, volatilities) * corr_matrix

# =====================
# 2. 求全局最小方差组合权重
# =====================
# 满仓约束下，全局最小方差组合(MVP)权重的闭式解为：w_mvp = (Σ^{-1} * 1) / (1' * Σ^{-1} * 1)
# 其中 1 为全1列向量

inv_cov_matrix = np.linalg.inv(cov_matrix)
ones = np.ones(3)

# 计算 Σ^{-1} * 1
inv_cov_ones = inv_cov_matrix @ ones

# 归一化，使权重和为1
mvp_weights = inv_cov_ones / np.sum(inv_cov_ones)

# =====================
# 3. 计算组合年化波动率
# =====================
# 组合方差：σ² = w'Σw
# 组合波动率：σ = sqrt(w'Σw)
mvp_var = mvp_weights @ cov_matrix @ mvp_weights
mvp_vol_annual = np.sqrt(mvp_var)

# =====================
# 4. 封装结果
# =====================
result = {
    'mvp_weights': mvp_weights.tolist(),
    'mvp_vol_annual': float(mvp_vol_annual)
}

# 打印结果供课堂投屏展示
print("全局最小方差组合(MVP)计算结果：")
print(f"权重: {result['mvp_weights']}")
print(f"年化波动率: {result['mvp_vol_annual']:.4f} ({result['mvp_vol_annual']*100:.2f}%)")
```
