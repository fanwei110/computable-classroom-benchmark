```python
import numpy as np
from scipy.stats import norm

# 课程计算约定参数
TRADING_DAYS_PER_YEAR = 252
CONFIDENCE_LEVEL = 0.95
ANNUALIZED_VOLATILITY = 0.24
POSITION_VALUE = 2_700_000

# 1. 去年化波动率到一日
daily_volatility = ANNUALIZED_VOLATILITY / np.sqrt(TRADING_DAYS_PER_YEAR)

# 2. 单尾 95% 正态分位数（均值=0）
z_score = norm.ppf(1 - CONFIDENCE_LEVEL)

# 3. 计算 VaR（正的损失金额）
var_95_1d = POSITION_VALUE * z_score * daily_volatility

# 输出契约
result = {
    'var_95_1d': round(var_95_1d, 2)  # 保留两位小数，符合金额习惯
}
```
