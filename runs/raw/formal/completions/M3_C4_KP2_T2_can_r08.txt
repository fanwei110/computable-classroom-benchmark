```python
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.ticker as mticker

# ==========================================
# 1. 参数化无风险利率与市场期望收益
# ==========================================
rf = 0.023   # 无风险利率 2.3%
rm = 0.094   # 市场期望收益 9.4%

# ==========================================
# 2. 计算 SML 斜率与指定 Beta 处的期望收益
# ==========================================
sml_slope = rm - rf
beta_target = 1.27
er_at_beta_127 = rf + beta_target * sml_slope

# ==========================================
# 3. 准备绘图数据
# ==========================================
# Beta 范围 0 到 2
betas = np.linspace(0, 2, 100)
# SML 上的期望收益
sml_returns = rf + betas * sml_slope

# 股票数据
stocks = {
    'X': {'beta': 0.62, 'return': 0.081},
    'Y': {'beta': 1.18, 'return': 0.131},
    'Z': {'beta': 1.51, 'return': 0.099}
}

# ==========================================
# 4. 绘制证券市场线 (SML) 及股票点
# ==========================================
# 设置中文字体与负号支持（若系统无SimHei会自动降级，但不影响程序运行）
plt.rcParams['font.sans-serif'] = ['SimHei', 'Arial Unicode MS', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

fig, ax = plt.subplots(figsize=(10, 6))

# 绘制 SML
ax.plot(betas, sml_returns, label='证券市场线 (SML)', color='blue', linewidth=2)

# 标出无风险利率与市场组合点
ax.scatter(0, rf, color='black', s=80, zorder=5, label=f'无风险利率 (r_f={rf:.1%})')
ax.scatter(1, rm, color='green', s=80, zorder=5, label=f'市场组合 (r_m={rm:.1%})')

# 绘制股票点、标注及 Alpha 指示线
for name, data in stocks.items():
    beta_s = data['beta']
    ret_s = data['return']
    
    # 计算该 Beta 下的 CAPM 期望收益及 Alpha
    ret_sml = rf + beta_s * sml_slope
    alpha = ret_s - ret_sml
    
    # 画出股票点
    ax.scatter(beta_s, ret_s, color='red', s=60, zorder=5)
    
    # 画出偏离 SML 的垂直虚线 (即 Alpha)
    ax.vlines(beta_s, min(ret_s, ret_sml), max(ret_s, ret_sml), 
              colors='gray', linestyles='dashed', linewidth=1.2)
    
    # 标注股票信息
    # 根据Alpha正负微调标注位置，防止遮挡
    y_offset = 0.008 if alpha >= 0 else -0.015
    ax.annotate(f'{name}\nβ={beta_s}, r={ret_s:.1%}\nα={alpha:+.1%}', 
                xy=(beta_s, ret_s), 
                xytext=(beta_s + 0.06, ret_s + y_offset),
                fontsize=10,
                bbox=dict(boxstyle="round,pad=0.3", fc="white", ec="gray", alpha=0.8),
                arrowprops=dict(arrowstyle='->', color='black', lw=1.2))

# 图表格式美化
ax.set_title('CAPM 与证券市场线 (SML)', fontsize=16)
ax.set_xlabel('Beta (β)', fontsize=13)
ax.set_ylabel('期望收益率', fontsize=13)

# Y轴以百分比格式显示
ax.yaxis.set_major_formatter(mticker.PercentFormatter(xmax=1, decimals=1))

ax.set_xlim(0, 2)
ax.set_ylim(0, 0.20)
ax.axhline(0, color='black', linewidth=0.5)
ax.grid(True, linestyle='--', alpha=0.6)
ax.legend(fontsize=11, loc='upper left')

# ==========================================
# 5. 保存图形
# ==========================================
figure_path = 'sml_plot.png'
fig.savefig(figure_path, dpi=300, bbox_inches='tight')
plt.close(fig)

# ==========================================
# 6. 填充结果字典
# ==========================================
result = {
    'sml_slope': sml_slope,
    'er_at_beta_127': er_at_beta_127,
    'figure_path': figure_path
}

# 输出结果供查验
print(f"SML 斜率: {result['sml_slope']:.5f}")
print(f"Beta=1.27 处的 CAPM 期望收益: {result['er_at_beta_127']:.5f}")
print(f"图形已保存至: {result['figure_path']}")
print("\nresult 字典内容:")
print(result)
```
