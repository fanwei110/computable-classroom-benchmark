```python
import math

# 已知参数
annual_vol = 0.24
position = 2700000
z_95 = 1.645  # 95%置信水平下的Z值
trading_days = 252  # 假设一年有252个交易日

# 计算日波动率
daily_vol = annual_vol / math.sqrt(trading_days)

# 计算95%一日VaR
var_95_1d = position * z_95 * daily_vol

# 按照输出契约要求存入字典
result = {'var_95_1d': var_95_1d}
```
