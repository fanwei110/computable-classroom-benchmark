```python
import numpy as np
from scipy.stats import norm

def black_scholes_call(S, K, T, r, sigma):
    """
    欧式看涨期权 Black-Scholes 定价（连续复利）
    S: 标的现价
    K: 行权价
    T: 剩余期限（年）
    r: 无风险利率（小数）
    sigma: 波动率（小数）
    返回: 期权价格
    """
    if T <= 0:
        return max(S - K, 0.0)
    if sigma <= 0:
        return max(S * np.exp(-r * T) - K * np.exp(-r * T), 0.0)  # 实际上 discount 差异，直接用 payoff
        
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    
    call_price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
    return call_price

# ---------- 参数设定 ----------
S0 = 103.7      # 现价
K = 97.5        # 行权价
T = 0.58        # 剩余期限（年）
r = 0.043       # 无风险利率（连续复利，小数形式）
sigma_base = 0.276   # 当前隐含波动率 27.6%
sigma_up = 0.286     # 上升一个百分点后的波动率 28.6%

# ---------- 精确重定价 ----------
price_base = black_scholes_call(S0, K, T, r, sigma_base)
price_up = black_scholes_call(S0, K, T, r, sigma_up)

# 精确价差（波动率上升一个百分点导致的期权价格变化）
price_change = price_up - price_base

# ---------- 输出结果 ----------
result = {
    'price_change': price_change
}

# 打印结果以便课堂投屏展示
print("=== Black-Scholes 精确重定价 ===")
print(f"标的现价 S = {S0}")
print(f"行权价 K = {K}")
print(f"剩余期限 T = {T} 年")
print(f"无风险利率 r = {r*100:.2f}%")
print(f"波动率 σ_base = {sigma_base*100:.2f}%")
print(f"波动率 σ_up   = {sigma_up*100:.2f}%")
print("-" * 40)
print(f"σ = {sigma_base*100:.2f}% 时期权价格: {price_base:.6f}")
print(f"σ = {sigma_up*100:.2f}%  时期权价格: {price_up:.6f}")
print(f"价格变化 (精确价差): {price_change:.6f}")
print("=" * 40)
print("输出字典 result:")
print(result)
```
