```python
import numpy as np

# ==================
# 1. 参数设定 (小数表示)
# ==================
beta = 1.42
market_return_monthly = -0.058  # 上个月市场跌幅 5.8%
risk_free_rate_annual = 0.047   # 无风险年利率 4.7%

# ==================
# 2. 计算过程
# ==================
# 步骤1：把年无风险利率按简单除法折算为月利率（年利率/12）
risk_free_rate_monthly = risk_free_rate_annual / 12

# 步骤2：在月度层面套用 CAPM
# CAPM公式: E(R_i) = R_f + beta * (E(R_m) - R_f)
capm_return_monthly = risk_free_rate_monthly + beta * (market_return_monthly - risk_free_rate_monthly)

# ==================
# 3. 填充 result
# ==================
result = {
    'capm_return_monthly': capm_return_monthly
}

# 便于课堂投屏展示计算结果
print(f"年无风险利率折算为月利率: {risk_free_rate_monthly:.6f}")
print(f"按CAPM计算的上月期望收益率: {capm_return_monthly:.6f}")
print(f"即该股票上个月本应收益: {capm_return_monthly*100:.4f}%")
```
