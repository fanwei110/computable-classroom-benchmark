```python
import numpy as np

def main():
    # 债券参数
    face = 100.0         # 面值
    coupon_rate = 0.046  # 票息率 4.6%
    coupon = face * coupon_rate  # 每年票息 = 4.6
    y = 0.053            # 年收益率 5.3%
    T = 7                # 期限 7 年
    dy = 0.008           # 收益率上升 80 个基点

    # 现金流时间点 1,2,...,T
    t = np.arange(1, T + 1, dtype=float)
    # 各年现金流：前 T-1 年为票息，最后一年为票息+面值
    cf = np.full(T, coupon)
    cf[-1] += face

    # 贴现因子与现值
    disc = (1 + y) ** t
    pv = cf / disc
    price = np.sum(pv)

    # 麦考利久期
    weights = pv / price
    mac_duration = np.sum(t * weights)

    # 修正久期 (年复利)
    mod_duration = mac_duration / (1 + y)

    # 一阶近似跌幅 dP/P = - D_mod * dy，取正的小数表示跌幅
    price_drop_pct = mod_duration * dy

    # 输出结果字典
    result = {'price_drop_pct': price_drop_pct}
    print(result)

if __name__ == "__main__":
    main()
```
