```python
import numpy as np

# ==================== 债券参数设置 ====================
F = 100.0          # 债券面值
c = 0.046          # 票息率 (4.6%)
y = 0.053          # 当前收益率 (5.3%)
T = 7              # 期限 (7年)
dy = 0.008         # 收益率变动 (上升80个基点)

# 假设：按年付息，期末收回本金
t = np.arange(1, T + 1)
cfs = np.full(T, c * F)
cfs[-1] += F       # 最后一期现金流包含本金

# ==================== 1. 计算债券当前价格与利率敏感性 ====================
# 计算贴现因子
discount_factors = (1 + y) ** -t

# 债券当前价格 (现金流贴现之和)
P = np.sum(cfs * discount_factors)

# 麦考利久期 (Macaulay Duration)
mac_duration = np.sum(t * cfs * discount_factors) / P

# 修正久期 (Modified Duration) - 衡量价格对收益率的一阶敏感性
mod_duration = mac_duration / (1 + y)

# 凸性 (Convexity) - 衡量价格对收益率的二阶敏感性
# 严格公式: Convexity = Sum[ t*(t+1) * CF_t / (1+y)^(t+2) ] / P
convexity = np.sum(t * (t + 1) * cfs * discount_factors) / (P * (1 + y) ** 2)

# ==================== 2. 估算收益率上升80个基点的价格影响 ====================
# 使用泰勒展开式（久期+凸性）估算价格变动百分比：
# ΔP/P ≈ -D_mod * Δy + (1/2) * Convexity * (Δy)^2
delta_P_pct = -mod_duration * dy + 0.5 * convexity * (dy ** 2)

# ==================== 3. 存入结果 ====================
# 题目要求“跌幅”，因此取变动率的绝对值（以正数表示下跌幅度）
price_drop_pct = -delta_P_pct

result = {
    'price_drop_pct': price_drop_pct
}

# 课堂投屏辅助打印（便于验证逻辑）
print(f"债券当前价格: {P:.4f}")
print(f"修正久期: {mod_duration:.4f}")
print(f"凸性: {convexity:.4f}")
print(f"收益率上升80bp后，价格变动率: {delta_P_pct:.4%}")
print(f"预计跌幅: {price_drop_pct:.4%}")
print(f"\n最终结果 result: {result}")
```
