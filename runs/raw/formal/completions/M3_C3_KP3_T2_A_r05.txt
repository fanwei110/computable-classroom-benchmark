```python
import numpy as np
import matplotlib.pyplot as plt

# --- 债券参数 ---
face_value = 100
coupon_rate = 0.046
maturity = 7
y0 = 0.053

# --- 可调参数：收益率变动幅度（绘图范围） ---
# 收益率从 y_min 到 y_max
y_min = 0.02
y_max = 0.09

# --- 债券定价与久期计算函数 ---
def bond_price(y, FV, c, N):
    """计算债券精确价格（年复利）"""
    coupon = FV * c
    if y == 0:
        return FV + coupon * N
    pv_coupons = coupon * (1 - (1 + y)**-N) / y
    pv_face = FV * (1 + y)**-N
    return pv_coupons + pv_face

def modified_duration(y, FV, c, N):
    """计算修正久期"""
    coupon = FV * c
    P = bond_price(y, FV, c, N)
    # 计算麦考利久期
    weighted_sum = 0
    for t in range(1, N + 1):
        weighted_sum += t * coupon / (1 + y)**t
    weighted_sum += N * FV / (1 + y)**N
    mac_dur = weighted_sum / P
    # 修正久期 = 麦考利久期 / (1 + y)
    return mac_dur / (1 + y)

# --- 计算初始状态 ---
P0 = bond_price(y0, face_value, coupon_rate, maturity)
D_mod = modified_duration(y0, face_value, coupon_rate, maturity)

# --- 生成绘图数据 ---
yields = np.linspace(y_min, y_max, 500)
exact_prices = [bond_price(y, face_value, coupon_rate, maturity) for y in yields]
# 久期近似：P_approx = P0 * (1 - D_mod * (y - y0))
approx_prices = [P0 * (1 - D_mod * (y - y0)) for y in yields]

# --- 绘制图表 ---
plt.figure(figsize=(10, 6))
plt.plot(yields * 100, exact_prices, label='精确价格', color='blue', linewidth=2)
plt.plot(yields * 100, approx_prices, label='久期近似', color='red', linestyle='--', linewidth=2)
plt.axvline(x=y0 * 100, color='grey', linestyle=':', label=f'初始收益率 ({y0*100:.1f}%)')
plt.scatter([y0 * 100], [P0], color='black', zorder=5)

plt.title('债券价格随收益率变化曲线')
plt.xlabel('收益率 (%)')
plt.ylabel('债券价格')
plt.legend()
plt.grid(True, linestyle='--', alpha=0.7)

# 保存图表
figure_path = 'bond_price_yield_curve.png'
plt.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close()

# --- 计算要求的指标 ---
dy = 0.01  # 100个基点 = 1% = 0.01
y_up100bp = y0 + dy

# 收益率上升100个基点后的精确价格
price_at_up100bp = bond_price(y_up100bp, face_value, coupon_rate, maturity)

# 用久期估计的相对价格变化：dP/P = -D_mod * dy
dur_approx_change_up100bp = -D_mod * dy

# --- 封装输出 ---
result = {
    'price_at_up100bp': price_at_up100bp,
    'dur_approx_change_up100bp': dur_approx_change_up100bp,
    'figure_path': figure_path
}

# 打印结果供查看
print(f"初始价格: {P0:.4f}")
print(f"修正久期: {D_mod:.4f}")
print(f"收益率上升100bp后精确价格: {result['price_at_up100bp']:.4f}")
print(f"久期近似相对价格变化: {result['dur_approx_change_up100bp']:.4f} (即 {result['dur_approx_change_up100bp']*100:.4f}%)")
print(f"图表已保存至: {result['figure_path']}")
```
