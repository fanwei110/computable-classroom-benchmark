```python
import numpy as np
from scipy.stats import norm

def black_scholes_call(S, K, T, r, sigma):
    """
    欧式看涨期权 Black-Scholes 定价及希腊字母
    S     : 标的现价
    K     : 行权价
    T     : 剩余期限（年）
    r     : 连续复利无风险利率（小数）
    sigma : 年化波动率（小数）
    返回  : (价格, delta, vega)
    """
    # 计算 d1, d2
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)

    # 期权价格
    call_price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)

    # delta
    call_delta = norm.cdf(d1)

    # vega: 价格对波动率的偏导数（每单位波动率，σ 用小数表示）
    call_vega = S * norm.pdf(d1) * np.sqrt(T)

    return call_price, call_delta, call_vega

# ---------- 给定参数 ----------
S     = 103.7      # 股票现价
K     = 97.5       # 行权价
T     = 0.58       # 剩余期限（年）
r     = 0.043      # 无风险利率（4.3%）
sigma = 0.276      # 隐含波动率（27.6%）

# 计算
call_price, call_delta, call_vega = black_scholes_call(S, K, T, r, sigma)

# 结果字典（键名严格按要求）
result = {
    'call_price': call_price,
    'call_delta': call_delta,
    'call_vega': call_vega
}

# 输出结果（可直接投屏检查）
print(result)
```
