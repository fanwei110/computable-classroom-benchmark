```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import norm

# ========================
# 可调参数（波动率列表）
# ========================
VOLATILITIES = [0.15, 0.276, 0.40]   # 对应 15%, 27.6%, 40%
S_MIN, S_MAX, N_POINTS = 70, 140, 600  # 标的网格

# 期权参数
K = 97.5         # 行权价
r = 0.043        # 无风险利率（年化）
T = 0.58         # 剩余期限（年）
S0 = 110.0       # 待报告的标的价格
SIGMA0 = 0.276   # 待报告的波动率

def bs_call_delta(S, K, r, T, sigma):
    """Black-Scholes 欧式看涨期权 delta"""
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    return norm.cdf(d1)

# ---------- 1. 计算三个波动率下的 delta 曲线 ----------
S_grid = np.linspace(S_MIN, S_MAX, N_POINTS)
delta_curves = {}
for sig in VOLATILITIES:
    delta_curves[sig] = bs_call_delta(S_grid, K, r, T, sig)

# ---------- 2. 画图 ----------
plt.figure(figsize=(10, 6))
for sig in VOLATILITIES:
    plt.plot(S_grid, delta_curves[sig], lw=2,
             label=f'$\\sigma$ = {sig*100:.1f}%')
plt.xlabel('Underlying Spot Price')
plt.ylabel('Delta')
plt.title('European Call Option Delta (Black-Scholes)')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()

figure_path = 'delta_curve.png'
plt.savefig(figure_path, dpi=150)
plt.close()

# ---------- 3. 报告特定点 delta ----------
delta_s110 = bs_call_delta(S0, K, r, T, SIGMA0)

# ---------- 4. 保存到 result 字典 ----------
result = {
    'delta_at_s110': delta_s110,
    'figure_path': figure_path
}

print(result)   # 老师运行时会直接看到输出
```
