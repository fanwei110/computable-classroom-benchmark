```python
import numpy as np
from scipy.stats import norm

# ==========================================
# 参数与假设设置
# ==========================================
position_value = 1_850_000  # 头寸价值（人民币元）
annual_vol = 0.218         # 年化收益波动率

# 假设：一年有252个交易日，采用平方根法则进行波动率的期限缩放
trading_days_per_year = 252

# ==========================================
# 步骤 1: 将年化波动率换算到一日期限
# ==========================================
# 日波动率 = 年化波动率 / sqrt(一年交易日数)
daily_vol = annual_vol / np.sqrt(trading_days_per_year)

# ==========================================
# 步骤 2: 对两个置信水平套用正态分位数
# ==========================================
# 在 delta-normal 模型下，假设损益服从正态分布
# VaR = 头寸价值 × 日波动率 × |正态分布下对应分位数|
# 95%置信水平对应左尾5%的分位数，99%置信水平对应左尾1%的分位数
z_95 = abs(norm.ppf(0.05))  # 等价于 norm.ppf(0.95)
z_99 = abs(norm.ppf(0.01))  # 等价于 norm.ppf(0.99)

# ==========================================
# 步骤 3: 计算 1日 VaR 及缩放至 10日 VaR
# ==========================================
# (i) 95% 一日 VaR
var_95_1d = position_value * daily_vol * z_95

# (ii) 99% 十日 VaR
# 期限缩放法则：VaR_T = VaR_1 * sqrt(T)
scaling_factor_10d = np.sqrt(10)
var_99_10d = position_value * daily_vol * z_99 * scaling_factor_10d

# ==========================================
# 步骤 4: 填充 result
# ==========================================
result = {
    'var_95_1d': round(var_95_1d, 2),
    'var_99_10d': round(var_99_10d, 2)
}

# ==========================================
# 课堂投屏展示辅助信息
# ==========================================
print("="*50)
print("《证券投资学》课堂实时计算 - Delta-Normal VaR")
print("="*50)
print(f"头寸价值: \t\t\t{position_value:,.2f} 元")
print(f"年化波动率: \t\t\t{annual_vol:.2%}")
print(f"假设年交易日数: \t\t{trading_days_per_year} 天")
print("-"*50)
print(f"日波动率: \t\t\t{daily_vol:.6%}")
print(f"正态分位数 (95%): \t\t{z_95:.4f}")
print(f"正态分位数 (99%): \t\t{z_99:.4f}")
print("-"*50)
print(f"(i) 95% 一日 VaR: \t\t{result['var_95_1d']:,.2f} 元")
print(f"(ii) 99% 十日 VaR: \t\t{result['var_99_10d']:,.2f} 元")
print("="*50)
```
