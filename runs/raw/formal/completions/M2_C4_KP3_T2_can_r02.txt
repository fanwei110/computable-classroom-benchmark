```python
import numpy as np
import matplotlib.pyplot as plt

# ---------- 债券参数 ----------
face_value = 100.0          # 面值
coupon_rate = 0.046         # 票息率 (4.6%)
maturity = 7                # 期限 (年)
y0 = 0.053                  # 当前到期收益率 (5.3%)
dy_shock = 0.01             # 收益率上升 100 bp

# ---------- 现金流 ----------
# 每年支付一次票息，最后一年票息+面值
times = np.arange(1, maturity + 1)              # t = 1,2,...,7
cash_flows = np.full(maturity, face_value * coupon_rate)
cash_flows[-1] += face_value                    # 最后一期加上面值

# ---------- 定价函数 ----------
def bond_price(y):
    """给定年复利收益率 y，返回债券价格"""
    disc = (1 + y) ** times
    return np.sum(cash_flows / disc)

P0 = bond_price(y0)

# ---------- 麦考利久期 & 修正久期 ----------
pv = cash_flows / (1 + y0) ** times
weights = pv / P0
D_mac = np.sum(times * weights)                 # 麦考利久期 (年)
D_mod = D_mac / (1 + y0)                        # 修正久期

# ---------- 凸性 (按照课程公式) ----------
# Conv = Σ [ t(t+1) * CF_t / (1+y)^(t+2) ] / P
conv_numerator = np.sum(times * (times + 1) * cash_flows / (1 + y0) ** (times + 2))
Conv = conv_numerator / P0                      # 凸性 (年的平方)

# ---------- 精确价格-收益率曲线 ----------
y_grid = np.linspace(0.02, 0.09, 300)           # 2% 到 9%
P_exact = np.array([bond_price(y) for y in y_grid])

# 一阶久期近似：P(y) ≈ P0 * (1 - D_mod * (y - y0))
P_approx1 = P0 * (1 - D_mod * (y_grid - y0))

# 二阶久期+凸性近似：P(y) ≈ P0 * (1 - D_mod * Δy + 0.5 * Conv * Δy^2)
dy = y_grid - y0
P_approx2 = P0 * (1 - D_mod * dy + 0.5 * Conv * dy ** 2)

# ---------- 收益率上升100bp的结果 ----------
y_up = y0 + dy_shock
price_up100bp = bond_price(y_up)                # 精确价格
dur_approx_change = -D_mod * dy_shock           # 一阶相对变化 (小数)

# ---------- 画图 ----------
fig, ax = plt.subplots(figsize=(10, 6))
ax.plot(y_grid * 100, P_exact, 'b-', linewidth=2, label='Exact Price')
ax.plot(y_grid * 100, P_approx1, 'r--', linewidth=1.8, label='Duration Approximation (Linear)')
ax.plot(y_grid * 100, P_approx2, 'g--', linewidth=1.8, label='Duration + Convexity Approximation')
ax.axvline(y0 * 100, color='gray', linestyle=':', alpha=0.7, label=f'Current Yield = {y0*100:.1f}%')
ax.axvline(y_up * 100, color='orange', linestyle=':', alpha=0.7, label=f'Up 100bp = {y_up*100:.1f}%')

ax.set_xlabel('Yield to Maturity (%)')
ax.set_ylabel('Bond Price')
ax.set_title('Bond Price–Yield Curve (7yr, 4.6% Coupon, Annual)')
ax.legend()
ax.grid(True, alpha=0.3)

# 保存图片
figure_path = 'bond_price_yield_curve.png'
fig.savefig(figure_path, dpi=200, bbox_inches='tight')
plt.close(fig)   # 避免在 notebook 中重复显示

# ---------- 输出字典 ----------
result = {
    'price_at_up100bp': price_up100bp,
    'dur_approx_change_up100bp': dur_approx_change,
    'figure_path': figure_path
}

# 可选：打印结果以便当堂查看
print("=== 计算结果 ===")
print(f"当前收益率 {y0*100:.2f}% 的精确价格: {P0:.4f}")
print(f"收益率上升100bp后的精确价格: {price_up100bp:.4f}")
print(f"久期法一阶近似的相对价格变化: {dur_approx_change:.6f} ({dur_approx_change*100:.4f}%)")
print(f"\n麦考利久期: {D_mac:.4f} 年")
print(f"修正久期:   {D_mod:.4f}")
print(f"凸性:       {Conv:.6f} 年的平方")
print(f"\n图像已保存至: {figure_path}")
print(f"\nresult 字典: {result}")
```
