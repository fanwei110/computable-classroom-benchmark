```python
import numpy as np
import matplotlib.pyplot as plt
import os

# 课程计算约定参数
risk_free_rate = 0.023  # 无风险利率 2.3%
market_return = 0.094   # 市场期望收益 9.4%

# 证券市场线 (SML) 函数
def calculate_sml_er(beta, risk_free_rate, market_return):
    return risk_free_rate + beta * (market_return - risk_free_rate)

# 生成 beta 从 0 到 2 的 SML 数据
betas = np.linspace(0, 2, 100)
sml_returns = calculate_sml_er(betas, risk_free_rate, market_return)

# 股票数据
stocks = {
    'X': {'beta': 0.62, 'return': 0.081},
    'Y': {'beta': 1.18, 'return': 0.131},
    'Z': {'beta': 1.51, 'return': 0.099}
}

# 创建图形
plt.figure(figsize=(10, 6))
plt.plot(betas, sml_returns, label='Security Market Line (SML)', color='blue')

# 绘制股票点
for stock, data in stocks.items():
    plt.scatter(data['beta'], data['return'], label=f'Stock {stock}', s=100)
    plt.text(data['beta'] + 0.02, data['return'] - 0.005, stock,
             fontsize=10, verticalalignment='top')

# 图形设置
plt.title('Security Market Line (SML) and Stock Positions')
plt.xlabel('Beta')
plt.ylabel('Expected Return')
plt.xlim(0, 2)
plt.ylim(0, 0.15)
plt.grid(True)
plt.legend()

# 保存图形
figure_path = 'sml_plot.png'
plt.savefig(figure_path)
plt.close()

# 计算 SML 斜率
sml_slope = market_return - risk_free_rate

# 计算 beta=1.27 处的期望收益
er_at_beta_127 = calculate_sml_er(1.27, risk_free_rate, market_return)

# 结果字典
result = {
    'sml_slope': sml_slope,
    'er_at_beta_127': er_at_beta_127,
    'figure_path': os.path.abspath(figure_path)
}

# 输出结果（供课堂展示）
print(result)
```
