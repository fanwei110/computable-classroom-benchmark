```python
import numpy as np

# 1. 基础数据
vols = np.array([0.187, 0.243, 0.312])
corr12, corr13, corr23 = 0.21, -0.13, 0.37

# 2. 构造相关系数矩阵与协方差矩阵
corr_mat = np.array([
    [1.0,  corr12,  corr13],
    [corr12, 1.0,  corr23],
    [corr13, corr23, 1.0 ]
])

# 协方差矩阵 Σ = diag(σ) @ Corr @ diag(σ)
cov_mat = np.diag(vols) @ corr_mat @ np.diag(vols)

# 3. 求全局最小方差组合 (MVP) 权重
# 闭式解: w_mvp = (Σ^-1 @ 1) / (1' @ Σ^-1 @ 1)
ones = np.ones(len(vols))
cov_inv = np.linalg.inv(cov_mat)
w_mvp = (cov_inv @ ones) / (ones @ cov_inv @ ones)

# 4. 计算组合波动率
# 组合方差 = w'Σw，年化波动率 = sqrt(w'Σw)
var_mvp = w_mvp @ cov_mat @ w_mvp
vol_mvp = np.sqrt(var_mvp)

# 5. 按输出契约填充结果
result = {
    'mvp_weights': w_mvp.tolist(),
    'mvp_vol_annual': vol_mvp
}

# 课堂展示辅助打印（可选，便于投屏观察）
if __name__ == '__main__':
    print("协方差矩阵 Σ:")
    print(cov_mat)
    print("\n全局最小方差组合权重:")
    print(f"  资产1: {result['mvp_weights'][0]:.6f}")
    print(f"  资产2: {result['mvp_weights'][1]:.6f}")
    print(f"  资产3: {result['mvp_weights'][2]:.6f}")
    print(f"  权重和: {sum(result['mvp_weights']):.6f}")
    print(f"\n全局最小方差组合年化波动率: {result['mvp_vol_annual']:.6f}")
```
