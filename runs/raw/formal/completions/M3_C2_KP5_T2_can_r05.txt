```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

def main():
    # ==================== 参数设置 ====================
    position = 1_000_000          # 头寸（人民币/元）
    confidence_level = 0.95       # 置信水平（可调参数）
    alpha = 1 - confidence_level  # 显著性水平
    data_path = 'data/market_snapshot_v1.csv'
    figure_path = 'daily_pnl_hist_var.png'
    
    # 设置中文字体和负号显示，以防图表中出现乱码
    plt.rcParams['font.sans-serif'] = ['SimHei', 'Arial Unicode MS', 'DejaVu Sans']
    plt.rcParams['axes.unicode_minus'] = False

    # ==================== 1. 读取数据，构造日损益 ====================
    # 读取课程数据快照
    df = pd.read_csv(data_path)
    
    # 提取 'fund' 列的日收益率序列，并剔除缺失值
    # 假设：'fund'列为日收益率（小数形式，如0.01表示1%）
    daily_returns = df['fund'].dropna()
    
    # 构造头寸的日损益 (P&L = 头寸金额 * 日收益率)
    pnl = position * daily_returns

    # ==================== 2. 计算历史 VaR ====================
    # 历史 VaR：损益分布的下 alpha 分位数
    # np.percentile 计算经验分位数，alpha * 100 对应 5% 分位数
    var_pnl_quantile = np.percentile(pnl, alpha * 100)
    
    # VaR 惯例以正数报告损失金额
    hist_var_95_1d = -var_pnl_quantile

    # ==================== 3. 画直方图并加带标注的 VaR 线 ====================
    fig, ax = plt.subplots(figsize=(10, 6))

    # 绘制日损益分布直方图
    ax.hist(pnl, bins=50, color='steelblue', edgecolor='black', alpha=0.75)

    # 绘制 VaR 竖线（位于损益的分位点位置，即负数区域）
    ax.axvline(
        x=var_pnl_quantile, 
        color='red', 
        linestyle='--', 
        linewidth=2,
        label=f'{confidence_level*100:.0f}% 1日历史VaR: {hist_var_95_1d:,.2f} 元'
    )

    # 添加图表标签和标题
    ax.set_title(f'头寸日损益分布与 {confidence_level*100:.0f}% 历史 VaR', fontsize=14)
    ax.set_xlabel('日损益（人民币/元）', fontsize=12)
    ax.set_ylabel('频数', fontsize=12)
    ax.legend(fontsize=12)
    ax.grid(True, linestyle=':', alpha=0.6)

    # ==================== 4. 保存图形并填充 result ====================
    fig.savefig(figure_path, dpi=150, bbox_inches='tight')
    plt.close(fig)

    # 按输出契约构造结果字典
    result = {
        'hist_var_95_1d': hist_var_95_1d,
        'figure_path': figure_path
    }

    # 控制台输出以供教师投屏参考
    print(f"计算完成：{confidence_level*100:.0f}% 一日历史 VaR 为 {hist_var_95_1d:,.2f} 元")
    print(f"图形已保存至: {figure_path}")

    return result

if __name__ == '__main__':
    result = main()
```
