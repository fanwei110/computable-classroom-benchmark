import numpy as np

def main():
    # 给定的年化波动率（小数）
    sigma = np.array([0.187, 0.243, 0.312])
    # 相关系数矩阵的上三角（不含对角线）
    corr = np.array([
        [1.0,  0.21, -0.13],
        [0.21, 1.0,   0.37],
        [-0.13,0.37,  1.0]
    ])
    # 1. 构造协方差矩阵
    # Σ[i,j] = ρ[i,j] * σ[i] * σ[j]
    cov_matrix = np.zeros((3, 3))
    for i in range(3):
        for j in range(3):
            cov_matrix[i, j] = corr[i, j] * sigma[i] * sigma[j]
    
    # 2. 全局最小方差组合权重（允许卖空，权重和为1）
    # 闭式解：w = (Σ^{-1} 1) / (1^T Σ^{-1} 1)
    inv_cov = np.linalg.inv(cov_matrix)
    ones = np.ones(3)
    numerator = inv_cov @ ones   # Σ^{-1} 1
    denominator = ones @ numerator  # 1^T Σ^{-1} 1
    mvp_weights = numerator / denominator
    
    # 3. 组合年化波动率
    var_portfolio = mvp_weights @ cov_matrix @ mvp_weights
    mvp_vol_annual = np.sqrt(var_portfolio)
    
    # 存储结果
    result = {
        'mvp_weights': mvp_weights.tolist(),  # 转为列表便于阅读
        'mvp_vol_annual': float(mvp_vol_annual)
    }
    
    # 输出到控制台（供投屏查看）
    print("全局最小方差组合结果：")
    print(f"权重 (资产1, 资产2, 资产3): {result['mvp_weights']}")
    print(f"年化波动率: {result['mvp_vol_annual']:.4f} ({result['mvp_vol_annual']*100:.2f}%)")
    return result

if __name__ == "__main__":
    result = main()
