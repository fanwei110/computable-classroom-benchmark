```python
import numpy as np

# ==========================================
# 1. 由波动率与相关系数构造协方差矩阵
# ==========================================
# 给定的年化波动率
vols = np.array([0.187, 0.243, 0.312])

# 给定的相关系数矩阵
corr = np.array([
    [1.00,  0.21, -0.13],
    [0.21,  1.00,  0.37],
    [-0.13, 0.37,  1.00]
])

# 协方差矩阵 Σ = diag(σ) @ Corr @ diag(σ)
cov_matrix = np.diag(vols) @ corr @ np.diag(vols)

# ==========================================
# 2. 求最小方差权重 (闭式解)
# ==========================================
# 满仓约束(w之和为1)下的全局最小方差组合权重公式为：w = Σ^{-1} * 1 / (1' * Σ^{-1} * 1)
ones = np.ones(3)
cov_inv = np.linalg.inv(cov_matrix)

w_raw = cov_inv @ ones
mvp_weights = w_raw / np.sum(w_raw)  # 归一化使其和为 1

# ==========================================
# 3. 计算组合波动率
# ==========================================
# 组合方差为 w'Σw，波动率取平方根
mvp_var = mvp_weights.T @ cov_matrix @ mvp_weights
mvp_vol = np.sqrt(mvp_var)

# ==========================================
# 4. 按要求键名填充 result
# ==========================================
result = {
    'mvp_weights': mvp_weights.tolist(),
    'mvp_vol_annual': float(mvp_vol)
}

# 当作脚本运行时输出结果，便于课堂投屏展示
if __name__ == '__main__':
    print("全局最小方差组合结果：")
    print(f"权重: {result['mvp_weights']}")
    print(f"年化波动率: {result['mvp_vol_annual']:.4%}")
```
