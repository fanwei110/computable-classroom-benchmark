```python
import numpy as np
import matplotlib.pyplot as plt
import os

# --- 1. 定义资产参数 ---
E_R1 = 0.071
E_R2 = 0.124
vol1 = 0.163
vol2 = 0.289

var1 = vol1 ** 2
var2 = vol2 ** 2

# --- 2. 计算投资组合预期收益和波动率的函数 ---
def portfolio_return(w1, E_R1, E_R2):
    return w1 * E_R1 + (1 - w1) * E_R2

def portfolio_vol(w1, vol1, vol2, rho):
    var = w1**2 * vol1**2 + (1 - w1)**2 * vol2**2 + 2 * w1 * (1 - w1) * rho * vol1 * vol2
    return np.sqrt(var)

# --- 3. 计算最小方差组合(MVP)权重的函数 ---
def mvp_weight(vol1, vol2, rho):
    var1 = vol1 ** 2
    var2 = vol2 ** 2
    cov12 = rho * vol1 * vol2
    w1 = (var2 - cov12) / (var1 + var2 - 2 * cov12)
    return w1

# --- 4. 计算特定相关系数下的要求指标 ---
rho_target = 0.45

# 计算 rho=0.45 时的 MVP 权重及波动率
w1_mvp_45 = mvp_weight(vol1, vol2, rho_target)
vol_mvp_45 = portfolio_vol(w1_mvp_45, vol1, vol2, rho_target)

# 计算 rho=0.45 时，目标收益 10% 的最小波动率
# 对于两资产，权重与收益一一对应
target_ret = 0.10
w1_target = (target_ret - E_R2) / (E_R1 - E_R2)
vol_target_45 = portfolio_vol(w1_target, vol1, vol2, rho_target)

# --- 5. 绘图 ---
rhos = [0.15, 0.45, 0.75]
w1_array = np.linspace(-0.5, 1.5, 500) # 包含卖空情况以画出完整双曲线

plt.figure(figsize=(10, 7))

for rho in rhos:
    # 计算前沿曲线
    rets = portfolio_return(w1_array, E_R1, E_R2)
    vols = portfolio_vol(w1_array, vol1, vol2, rho)
    
    # 计算 MVP 点
    w1_mvp = mvp_weight(vol1, vol2, rho)
    ret_mvp = portfolio_return(w1_mvp, E_R1, E_R2)
    vol_mvp = portfolio_vol(w1_mvp, vol1, vol2, rho)
    
    # 画前沿曲线
    plt.plot(vols, rets, label=f'ρ = {rho}')
    
    # 画 MVP 点
    plt.scatter(vol_mvp, ret_mvp, marker='o', s=60, zorder=5)
    plt.annotate(f'MVP (ρ={rho})\nVol: {vol_mvp:.2%}', 
                 xy=(vol_mvp, ret_mvp), 
                 xytext=(vol_mvp + 0.015, ret_mvp - 0.005),
                 arrowprops=dict(facecolor='black', shrink=0.05, width=1, headwidth=5),
                 fontsize=9)

# 在 rho=0.45 曲线上标出目标收益10%的点
ret_10 = 0.10
vol_10 = vol_target_45
plt.scatter(vol_10, ret_10, color='red', marker='*', s=150, zorder=5, label='Target 10% Return (ρ=0.45)')

plt.title('Efficient Frontier for Two Assets with Different Correlations', fontsize=14)
plt.xlabel('Annualized Volatility', fontsize=12)
plt.ylabel('Expected Annual Return', fontsize=12)
plt.legend()
plt.grid(True, linestyle='--', alpha=0.6)

# 保存图片
fig_path = 'efficient_frontier.png'
plt.savefig(fig_path, dpi=150, bbox_inches='tight')
plt.close()

# --- 6. 构造输出字典 ---
result = {
    'mvp_vol_at_rho45': round(vol_mvp_45, 6),
    'frontier_vol_at_target': round(vol_target_45, 6),
    'figure_path': fig_path
}

print(result)
```
