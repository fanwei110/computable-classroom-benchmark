```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from pathlib import Path

# 固定随机种子和其他设置，保证可复现
np.random.seed(42)

# ========== 参数 ==========
DATA_PATH = Path("data/market_snapshot_v1.csv")
RISK_FREE_RATE_ANNUAL = 0.021        # 年化无风险利率
TRADING_DAYS = 252                   # 一年交易日数
WINDOW = 60                          # 滚动窗口长度（可调）
FIGURE_FILENAME = "rolling_sharpe.png"

# ========== 1. 读取数据 ==========
df = pd.read_csv(DATA_PATH)

# 自动识别日期列（假设存在 date 或 Date 列）
date_col = None
for col in df.columns:
    if col.lower() == 'date':
        date_col = col
        break
if date_col is None:
    raise ValueError("CSV 中未找到日期列 (date)")

df[date_col] = pd.to_datetime(df[date_col])
df = df.set_index(date_col)
fund_returns = df['fund'].astype(float)   # 日收益序列，小数形式

# 去除缺失值
fund_returns = fund_returns.dropna()

# ========== 2. 计算无风险日利率和超额收益 ==========
rf_daily = RISK_FREE_RATE_ANNUAL / TRADING_DAYS
excess_returns = fund_returns - rf_daily

# ========== 3. 计算60日滚动年化夏普比率 ==========
# 年化夏普 = sqrt(252) * (超额日收益均值 / 超额日收益标准差)
rolling_mean = excess_returns.rolling(window=WINDOW).mean()
rolling_std  = excess_returns.rolling(window=WINDOW).std()
rolling_sharpe = np.sqrt(TRADING_DAYS) * (rolling_mean / rolling_std)

# 移除因窗口不足产生的 NaN
rolling_sharpe = rolling_sharpe.dropna()

# ========== 4. 报告最后一个窗口的夏普值 ==========
rolling_sharpe_last = rolling_sharpe.iloc[-1]

# ========== 5. 画图并保存 ==========
fig, ax = plt.subplots(figsize=(12, 6))
ax.plot(rolling_sharpe.index, rolling_sharpe.values, color='steelblue', linewidth=1.2)
ax.set_title(f"{WINDOW}-Day Rolling Annualized Sharpe Ratio (Fund)", fontsize=14)
ax.set_xlabel("Date")
ax.set_ylabel("Annualized Sharpe Ratio")
ax.grid(True, alpha=0.3)
fig.tight_layout()
figure_path = FIGURE_FILENAME
fig.savefig(figure_path, dpi=150)
plt.close(fig)

# ========== 6. 整理结果字典 ==========
result = {
    'rolling_sharpe_last': round(rolling_sharpe_last, 6),
    'figure_path': figure_path
}

# 输出结果供教学检查
print(result)
```
