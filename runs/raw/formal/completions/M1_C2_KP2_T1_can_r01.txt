```python
import numpy as np

# 给定参数
risk_free_rate = 0.023  # 无风险利率 2.3%
market_return = 0.094   # 市场期望收益 9.4%
betas = {
    'X': 0.62,
    'Y': 1.18,
    'Z': 1.51
}
actual_return_y = 0.131  # 股票 Y 实际收益 13.1%

# 步骤 1: 计算每个股票的 CAPM 期望收益
er_x = risk_free_rate + betas['X'] * (market_return - risk_free_rate)
er_y = risk_free_rate + betas['Y'] * (market_return - risk_free_rate)
er_z = risk_free_rate + betas['Z'] * (market_return - risk_free_rate)

# 步骤 2: 计算股票 Y 的 alpha
alpha_y = actual_return_y - er_y

# 步骤 3: 填充 result 字典
result = {
    'er_x': er_x,
    'er_y': er_y,
    'er_z': er_z,
    'alpha_y': alpha_y
}

# 输出 result 以供验证（可选，实际使用时可注释掉）
print(result)
```
