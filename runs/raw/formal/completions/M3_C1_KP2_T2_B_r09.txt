    new_ret_line = rf + beta_line * (rm_init - rf)
    step = dict(
        method="update", 
        label=f"{rf:.1%}", 
        args=[{"y": [new_ret_line, rets_assets, [rf, rm_init]]}]
    )
    rf_steps.append(step)

rm_steps = []
for rm in rm_range:
    new_ret_line = rf_init + beta_line * (rm - rf_init)
    step = dict(
        method="update", 
        label=f"{rm:.1%}", 
        args=[{"y": [new_ret_line, rets_assets, [rf_init, rm]]}]
    )
    rm_steps.append(step)

fig.update_layout(sliders=[
    dict(active=4, steps=rf_steps, currentvalue=dict(prefix="无风险利率 rf: ")),
    dict(active=7, steps=rm_steps, currentvalue=dict(prefix="市场收益 rm: "))
])
