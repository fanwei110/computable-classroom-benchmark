```python
import numpy as np

# ==================== 输入参数 (小数表示) ====================
beta = 1.42                     # 股票的 Beta 值
market_return_monthly = -0.058  # 上个月市场收益率 (-5.8%)
risk_free_rate_annual = 0.047   # 年化无风险利率 (4.7%)

# ==================== 步骤 1：折算月度无风险利率 ====================
# 按课程约定：年利率转月利率使用简单除法 (年利率 / 12)
risk_free_rate_monthly = risk_free_rate_annual / 12

# ==================== 步骤 2：在月度层面套用 CAPM ====================
# CAPM 公式: E(R_i) = R_f + β_i * (E(R_m) - R_f)
# 此处 R_f 和 E(R_m) 均已转换为月度数据
capm_return_monthly = risk_free_rate_monthly + beta * (market_return_monthly - risk_free_rate_monthly)

# ==================== 步骤 3：填充 result ====================
result = {
    'capm_return_monthly': capm_return_monthly
}

# 课堂展示辅助打印 (可选，便于投屏观察计算过程与结果)
print(f"月度无风险利率: {risk_free_rate_monthly:.6f} ({risk_free_rate_monthly*100:.4f}%)")
print(f"CAPM 月度期望收益: {capm_return_monthly:.6f} ({capm_return_monthly*100:.4f}%)")
print(f"结果字典 result: {result}")
```
