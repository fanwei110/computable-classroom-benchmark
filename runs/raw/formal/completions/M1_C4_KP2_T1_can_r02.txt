```python
import numpy as np

# 课程计算约定
rf = 0.023          # 无风险利率 (2.3%)
er_market = 0.094   # 市场期望收益 (9.4%)

# 股票 beta 值
beta_x = 0.62
beta_y = 1.18
beta_z = 1.51

# 1. 计算各股票的 CAPM 期望年收益
er_x = rf + beta_x * (er_market - rf)
er_y = rf + beta_y * (er_market - rf)
er_z = rf + beta_z * (er_market - rf)

# 2. 计算股票 Y 的 alpha
actual_return_y = 0.131  # 股票 Y 当年实际收益 (13.1%)
alpha_y = actual_return_y - er_y

# 3. 填充 result 字典
result = {
    'er_x': er_x,
    'er_y': er_y,
    'er_z': er_z,
    'alpha_y': alpha_y
}

# 输出 result 以便课堂展示
print(result)
```
