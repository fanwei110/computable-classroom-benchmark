```python
import numpy as np
import pandas as pd

# 给定数据（小数表示）
rf = 0.023          # 无风险利率 2.3%
rm = 0.094          # 市场期望收益 9.4%
beta_x = 0.62
beta_y = 1.18
beta_z = 1.51

# 股票 Y 的实际年收益
actual_return_y = 0.131  # 13.1%

# 市场风险溢价
market_premium = rm - rf  # E[Rm] - rf

# CAPM 期望收益计算函数
def capm_expected_return(beta):
    return rf + beta * market_premium

# 计算各股票的 CAPM 期望收益
er_x = capm_expected_return(beta_x)
er_y = capm_expected_return(beta_y)
er_z = capm_expected_return(beta_z)

# 计算股票 Y 的 alpha
alpha_y = actual_return_y - er_y

# 汇总结果至字典 result
result = {
    'er_x': er_x,
    'er_y': er_y,
    'er_z': er_z,
    'alpha_y': alpha_y
}

# 输出结果（保留足够小数，便于课堂查看）
print("CAPM 计算结果：")
for key, value in result.items():
    print(f"{key}: {value:.6f}  ({value*100:.4f}%)")
```
