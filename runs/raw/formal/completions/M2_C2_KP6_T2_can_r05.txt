```python
import os
import numpy as np
import pandas as pd
from matplotlib import pyplot as plt

# ========================
# 用户可调参数
# ========================
WINDOW = 60                 # 滚动窗口长度（交易日）
RF_ANNUAL = 0.021           # 年化无风险利率
TRADING_DAYS_PER_YEAR = 252 # 年交易天数

# ========================
# 1. 读取 CSV 并准备数据
# ========================
csv_path = "data/market_snapshot_v1.csv"

# 读取时将第一列作为日期索引，自动解析日期
df = pd.read_csv(csv_path, index_col=0, parse_dates=True)

# 查找含有 "fund" 的列名（不区分大小写），作为基金收益率
fund_col = None
for col in df.columns:
    if "fund" in col.lower():
        fund_col = col
        break
if fund_col is None:
    raise ValueError("CSV 中找不到 'fund' 列，请检查列名。")

# 假设收益率已是小数形式（如 0.01 表示 1%）。若数据为百分数，可在此处处理。
fund_returns = df[fund_col].dropna()

# 计算日化无风险利率（简单比例法，保持内部一致）
rf_daily = RF_ANNUAL / TRADING_DAYS_PER_YEAR

# 超额日收益率
excess_returns = fund_returns - rf_daily

# ========================
# 2. 计算滚动年化夏普比率
# ========================
rolling_mean = excess_returns.rolling(window=WINDOW, min_periods=WINDOW).mean()
rolling_std  = excess_returns.rolling(window=WINDOW, min_periods=WINDOW).std()

# 年化夏普比率 = (日超额均值 / 日超额标准差) * sqrt(年交易天数)
annual_factor = np.sqrt(TRADING_DAYS_PER_YEAR)
rolling_sharpe = (rolling_mean / rolling_std) * annual_factor

# ========================
# 3. 最近一个窗口的夏普值
# ========================
rolling_sharpe_last = rolling_sharpe.dropna().iloc[-1]

# ========================
# 4. 画图并保存
# ========================
fig, ax = plt.subplots(figsize=(12, 5))
ax.plot(rolling_sharpe.index, rolling_sharpe, linewidth=1.2, color='#1f77b4')
ax.set_title(f"{WINDOW}-Day Rolling Annualized Sharpe Ratio of Fund", fontsize=14)
ax.set_xlabel("Date")
ax.set_ylabel("Annualized Sharpe Ratio")
ax.axhline(0, color='gray', linestyle='--', linewidth=0.8)
ax.grid(True, alpha=0.3)
plt.tight_layout()

figure_filename = "rolling_sharpe.png"
plt.savefig(figure_filename, dpi=150, bbox_inches='tight')
plt.close()

# ========================
# 5. 填充结果字典
# ========================
result = {
    "rolling_sharpe_last": round(float(rolling_sharpe_last), 6),
    "figure_path": os.path.abspath(figure_filename)
}

# 脚本运行完毕后，变量 result 即符合输出契约。
# 教师可直接检查 result 字典的内容。
if __name__ == "__main__":
    print("滚动年化夏普比率计算完成。")
    print("result =", result)
```
