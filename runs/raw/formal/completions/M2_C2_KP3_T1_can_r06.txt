```python
import numpy as np

# 债券参数
face_value = 100.0          # 面值
coupon_rate = 0.046         # 票息率
years = 7                   # 剩余期限（年）
ytm = 0.053                 # 到期收益率

# 构造现金流：每年一次付息，到期还本
t = np.arange(1, years + 1, dtype=float)
coupon = face_value * coupon_rate
cash_flows = np.full(years, coupon)
cash_flows[-1] += face_value   # 最后一期加入本金

# 按年复利贴现
discount_factors = (1 + ytm) ** (-t)
pv = cash_flows * discount_factors
price = np.sum(pv)

# 麦考利久期
weighted_t = t * pv
macaulay_duration = np.sum(weighted_t) / price

# 修正久期
modified_duration = macaulay_duration / (1 + ytm)

# 凸性（标准定义：现金流时间效应 + 二阶导/价格）
convexity = np.sum(t * (t + 1) * pv) / (price * (1 + ytm)**2)

# 结果输出
result = {
    'price': price,
    'macaulay_duration_years': macaulay_duration,
    'modified_duration_years': modified_duration,
    'convexity': convexity
}

if __name__ == "__main__":
    # 打印结果供课堂投屏查看
    for key, value in result.items():
        print(f"{key}: {value:.6f}")
```
