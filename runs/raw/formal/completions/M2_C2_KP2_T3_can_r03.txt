```python

import json


def main():
    # ---------- 给定数据 ----------
    beta = 1.42                     # 股票的系统性风险
    market_return_monthly = -0.058  # 上月市场收益率（下跌5.8%）
    rf_annual = 0.047               # 年化无风险利率（4.7%）

    # ---------- 步骤1：年无风险利率折算为月利率 ----------
    # 假设年无风险利率为有效年利率，按复利方式转换为月有效利率
    # 月利率 = (1 + 年利率)^(1/12) - 1
    rf_monthly = (1.0 + rf_annual) ** (1.0 / 12.0) - 1.0

    # ---------- 步骤2：在月度层面套用CAPM ----------
    # CAPM公式：E(R_i) = R_f + beta * (R_m - R_f)
    capm_return = rf_monthly + beta * (market_return_monthly - rf_monthly)

    # ---------- 步骤3：按要求构造输出字典 ----------
    result = {
        'capm_return_monthly': capm_return
    }

    # 输出结果（供课堂投屏查看）
    print(json.dumps(result, indent=4))
    return result


if __name__ == "__main__":
    main()
```
