```python
import numpy as np

# ==================== 1. 基础参数设置 ====================
# 年化波动率（小数表示）
vols = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr_matrix = np.array([
    [1.0,  0.21, -0.13],
    [0.21, 1.0,   0.37],
    [-0.13, 0.37, 1.0 ]
])

# ==================== 2. 构造协方差矩阵 ====================
# Σ_ij = ρ_ij * σ_i * σ_j
cov_matrix = np.outer(vols, vols) * corr_matrix

# ==================== 3. 求全局最小方差组合权重 ====================
# 允许卖空且满仓约束(w'1=1)下，最小方差组合闭式解为：
# w_mvp = Σ^{-1} * 1 / (1' * Σ^{-1} * 1)
inv_cov_matrix = np.linalg.inv(cov_matrix)
ones = np.ones(3)

# 计算权重并自动归一化
w_mvp = (inv_cov_matrix @ ones) / (ones.T @ inv_cov_matrix @ ones)

# ==================== 4. 计算组合年化波动率 ====================
# 组合方差 σ² = w'Σw，波动率 σ = sqrt(w'Σw)
var_mvp = w_mvp.T @ cov_matrix @ w_mvp
vol_mvp = np.sqrt(var_mvp)

# ==================== 5. 按契约填充 result ====================
result = {
    'mvp_weights': w_mvp.tolist(),
    'mvp_vol_annual': vol_mvp
}

# 课堂投屏展示输出
print("全局最小方差组合权重:", result['mvp_weights'])
print("全局最小方差组合年化波动率:", result['mvp_vol_annual'])
```
