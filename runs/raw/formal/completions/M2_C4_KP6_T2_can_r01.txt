```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# =========================
# 1. 读取数据与参数设置
# =========================
# 从课程的 CSV 快照中读取基金日收益率
data = pd.read_csv("data/market_snapshot_v1.csv", parse_dates=True)
fund_returns = data["fund"]  # 假设列名为 fund，存储小数形式的日收益率

# 无风险利率参数（年化 2.1%）
rf_annual = 0.021
rf_daily = rf_annual / 252.0  # 简单除法转换为日无风险利率

# 滚动窗口长度（可根据需要调整）
window = 60

# =========================
# 2. 计算滚动年化夏普比率
# =========================
# 超额收益 = 基金日收益 - 无风险日收益
excess_returns = fund_returns - rf_daily

# 滚动均值与滚动标准差（样本标准差，ddof=1）
rolling_mean = excess_returns.rolling(window=window).mean()
rolling_std = excess_returns.rolling(window=window).std(ddof=1)

# 年化夏普比率 = sqrt(252) * (年化超额收益 / 年化波动率)
# 由于我们使用的是日度数据，直接对日超额收益的均值和标准差进行年化
rolling_sharpe = np.sqrt(252) * (rolling_mean / rolling_std)

# 去除前 window-1 个 NaN 值，以便画图和取最后一个有效值
rolling_sharpe_clean = rolling_sharpe.dropna()

# 最后一个窗口的夏普值（小数）
last_sharpe = rolling_sharpe_clean.iloc[-1]

# =========================
# 3. 保存图形
# =========================
fig, ax = plt.subplots(figsize=(10, 5))
ax.plot(rolling_sharpe_clean.index, rolling_sharpe_clean.values,
        color='steelblue', linewidth=1.2, label=f'{window}-Day Rolling Sharpe')
ax.axhline(0, color='gray', linestyle='--', linewidth=0.8)
ax.set_title('Rolling Annualised Sharpe Ratio')
ax.set_xlabel('Date')
ax.set_ylabel('Sharpe Ratio')
ax.legend()
ax.grid(True, alpha=0.3)
fig.tight_layout()

figure_path = "rolling_sharpe.png"
fig.savefig(figure_path, dpi=150)
plt.close(fig)

# =========================
# 4. 组装输出结果
# =========================
result = {
    'rolling_sharpe_last': last_sharpe,
    'figure_path': figure_path
}

# 如果直接运行脚本，打印结果供检查
if __name__ == "__main__":
    print(f"最后一个 {window} 日滚动年化夏普比率: {last_sharpe:.6f}")
    print(f"图形已保存至: {figure_path}")
```
