```python
import numpy as np

# 1. 确定权重对应方式：w_A = 0.6, w_B = 0.4
weights = np.array([0.6, 0.4])

# 2. 资产年化波动率（标准差）
vol_A_annual = 0.184
vol_B_annual = 0.297

# 相关系数
rho_before = 0.3
rho_after = 0.8

# 构造协方差矩阵
cov_before = np.array([
    [vol_A_annual**2, rho_before * vol_A_annual * vol_B_annual],
    [rho_before * vol_A_annual * vol_B_annual, vol_B_annual**2]
])

cov_after = np.array([
    [vol_A_annual**2, rho_after * vol_A_annual * vol_B_annual],
    [rho_after * vol_A_annual * vol_B_annual, vol_B_annual**2]
])

# 3. 计算组合波动率（年化）
portfolio_vol_before = np.sqrt(weights @ cov_before @ weights)
portfolio_vol_after = np.sqrt(weights @ cov_after @ weights)

# 4. 填充 result 字典
result = {
    'vol_before_annual': portfolio_vol_before,
    'vol_after_annual': portfolio_vol_after
}

# 打印结果以便课堂展示
print("60/40 组合波动率变化：")
print(f"相关系数 0.3 时的年化波动率: {portfolio_vol_before:.4f}")
print(f"相关系数 0.8 时的年化波动率: {portfolio_vol_after:.4f}")
```
