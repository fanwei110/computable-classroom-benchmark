```python
import numpy as np
from scipy.optimize import newton

# 债券参数
face_value = 100.0          # 面值
annual_coupon_rate = 0.046  # 年票息率 4.6%
years_to_maturity = 7       # 剩余期限 7 年
ytm_current = 0.053         # 当前收益率 5.3%
yield_change_bps = 80       # 收益率变动 80 个基点
yield_change = yield_change_bps / 10000  # 转换为小数

# 计算每期现金流（半年付息）
coupon_payment = face_value * annual_coupon_rate / 2
periods = years_to_maturity * 2
cash_flows = np.full(periods, coupon_payment)
cash_flows[-1] += face_value  # 最后一期加上面值

# 计算当前价格
def bond_price(yield_per_period, cash_flows):
    periods = len(cash_flows)
    discount_factors = (1 + yield_per_period) ** np.arange(1, periods + 1)
    return np.sum(cash_flows / discount_factors)

yield_per_period_current = ytm_current / 2
price_current = bond_price(yield_per_period_current, cash_flows)

# 计算久期（Macaulay Duration）
def macaulay_duration(yield_per_period, cash_flows):
    periods = len(cash_flows)
    discount_factors = (1 + yield_per_period) ** np.arange(1, periods + 1)
    discounted_cash_flows = cash_flows / discount_factors
    weights = discounted_cash_flows / np.sum(discounted_cash_flows)
    return np.sum(weights * np.arange(1, periods + 1))

mac_dur = macaulay_duration(yield_per_period_current, cash_flows)
mod_dur = mac_dur / (1 + yield_per_period_current)  # 修正久期

# 计算凸性
def convexity(yield_per_period, cash_flows):
    periods = len(cash_flows)
    discount_factors = (1 + yield_per_period) ** np.arange(1, periods + 1)
    discounted_cash_flows = cash_flows / discount_factors
    weights = discounted_cash_flows / np.sum(discounted_cash_flows)
    return np.sum(weights * np.arange(1, periods + 1) * (np.arange(1, periods + 1) + 1)) / (1 + yield_per_period)**2

conv = convexity(yield_per_period_current, cash_flows)

# 估算价格变动（久期 + 凸性近似）
price_drop_pct = -mod_dur * yield_change + 0.5 * conv * (yield_change ** 2)
price_drop_pct *= 100  # 转换为百分比

# 精确计算新价格（验证）
yield_per_period_new = (ytm_current + yield_change) / 2
price_new = bond_price(yield_per_period_new, cash_flows)
price_drop_pct_exact = (price_new - price_current) / price_current * 100

# 存储结果
result = {
    'price_drop_pct': price_drop_pct  # 使用久期+凸性近似值
}

# 输出结果（可选，便于调试）
print(f"当前价格: {price_current:.4f}")
print(f"久期近似跌幅: {price_drop_pct:.4f}%")
print(f"精确跌幅: {price_drop_pct_exact:.4f}%")
```
