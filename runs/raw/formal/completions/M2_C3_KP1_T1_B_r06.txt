```python
import numpy as np

# 输入数据
sigma = np.array([0.187, 0.243, 0.312])
rho = np.array([
    [1.0, 0.21, -0.13],
    [0.21, 1.0, 0.37],
    [-0.13, 0.37, 1.0]
])

# 构建协方差矩阵
cov_matrix = np.diag(sigma) @ rho @ np.diag(sigma)

# 计算最小方差组合权重（允许做空，资金全部投入，权重和为1）
ones = np.ones(3)
inv_cov = np.linalg.inv(cov_matrix)
w_mvp = inv_cov @ ones / (ones @ inv_cov @ ones)

# 组合年化波动率
vol_mvp = np.sqrt(w_mvp @ cov_matrix @ w_mvp)

result = {
    'mvp_weights': w_mvp.tolist(),
    'mvp_vol_annual': vol_mvp
}

result
```
