```python
import numpy as np

def main():
    # 债券参数
    face_value = 100.0          # 面值
    coupon_rate = 0.046         # 票息率 4.6%
    maturity_years = 7          # 期限 7 年
    ytm = 0.053                 # 到期收益率 5.3%

    # 现金流时间点 (年末)
    times = np.arange(1, maturity_years + 1, dtype=float)
    
    # 每年票息
    coupon = face_value * coupon_rate   # 4.6

    # 现金流：每年票息，最后一年加上本金
    cash_flows = np.full(maturity_years, coupon)
    cash_flows[-1] += face_value        # 第7年 4.6 + 100 = 104.6

    # 贴现因子 (1 + ytm) ^ (-t)
    discount_factors = (1 + ytm) ** (-times)

    # ----- 1. 价格 -----
    present_values = cash_flows * discount_factors
    price = np.sum(present_values)

    # ----- 2. 麦考利久期 -----
    # 现金流现值的权重
    weights = present_values / price
    macaulay_duration = np.sum(times * weights)

    # 修正久期
    modified_duration = macaulay_duration / (1 + ytm)

    # ----- 3. 凸性 -----
    # 凸性 = [ Σ t*(t+1) * PV(CF_t) ] / (P * (1+y)^2)
    t_tp1_pv = times * (times + 1) * present_values
    convexity = np.sum(t_tp1_pv) / (price * (1 + ytm) ** 2)

    # 存储结果
    result = {
        'price': price,
        'macaulay_duration_years': macaulay_duration,
        'modified_duration_years': modified_duration,
        'convexity': convexity
    }

    # 输出以便课堂投屏查看
    for key, val in result.items():
        print(f"{key}: {val:.6f}")

    return result

# 脚本入口
if __name__ == "__main__":
    result = main()
```
