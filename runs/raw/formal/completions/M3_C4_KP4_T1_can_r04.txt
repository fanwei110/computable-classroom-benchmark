```python
import numpy as np
from scipy.stats import norm

# 课程计算约定参数
S = 103.7      # 标的资产现价
K = 97.5       # 行权价
sigma = 0.276  # 隐含波动率 (年化，小数表示)
r = 0.043      # 无风险利率 (连续复利，小数表示)
T = 0.58       # 剩余期限 (年)

# 步骤1：用连续复利利率计算 d1 与 d2
# 对于不分红股票，股息率 q = 0
d1 = (np.log(S / K) + (r + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
d2 = d1 - sigma * np.sqrt(T)

# 步骤2：计算期权价格与 delta = N(d1)
call_price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
call_delta = norm.cdf(d1)

# 步骤3：按每单位波动率计算 vega (dC/dσ)，用小数表示
# vega = S * sqrt(T) * N'(d1)，其中 N'(x) 为标准正态分布的概率密度函数 pdf
call_vega = S * np.sqrt(T) * norm.pdf(d1)

# 步骤4：填充 result
result = {
    'call_price': call_price,
    'call_delta': call_delta,
    'call_vega': call_vega
}

# 课堂展示输出
print(result)
```
