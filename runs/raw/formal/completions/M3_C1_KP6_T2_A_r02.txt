import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.widgets import Slider
import os

# ===== 1. 加载数据 =====
# 尝试所有常见格式
data_path = None
for fname in os.listdir('/mnt/data'):
    fpath = os.path.join('/mnt/data', fname)
    if os.path.isfile(fpath):
        data_path = fpath
        break

if data_path.endswith('.csv'):
    df = pd.read_csv(data_path)
elif data_path.endswith(('.xlsx', '.xls')):
    df = pd.read_excel(data_path)
elif data_path.endswith('.parquet'):
    df = pd.read_parquet(data_path)

print("数据形状:", df.shape)
print("列名:", df.columns.tolist())
print(df.head(10))

# ===== 2. 识别日期列和fund列 =====
# 找到日期列
date_col = None
for col in df.columns:
    if any(kw in str(col).lower() for kw in ['date', '日期', 'time', 'time']):
        date_col = col
        break

    # 尝试第一列
    date_col = df.columns[0]

df[date_col] = pd.to_datetime(df[date_col])
df = df.sort_values(date_col).reset_index(drop=True)

# 找到fund列
fund_col = None
for col in df.columns:
    if 'fund' in str(col).lower():
        fund_col = col
        break
if fund_col is None:
    # 取第一个数值列
    for col in df.columns:
        if df[col].dtype in ['float64', 'int64', 'float32', 'int32']:
            fund_col = col
            break

print(f"日期列: {date_col}, fund列: {fund_col}")

# ===== 3. 计算滚动年化夏普比率 =====
rf_annual = 0.021  # 年化无风险利率 2.1%
rf_daily = rf_annual / 252  # 日化无风险利率

# 计算日收益率
returns = df[fund_col].pct_change().dropna()

def calc_rolling_sharpe(returns_series, window=60, rf_annual=0.021):
    """计算滚动年化夏普比率"""
    rf_daily = rf_annual / 252
    excess_returns = returns_series - rf_daily
    rolling_mean = excess_returns.rolling(window=window).mean()
    rolling_std = returns_series.rolling(window=window).std()
    # 年化: mean * 252 / (std * sqrt(252)) = (mean/std) * sqrt(252)
    rolling_sharpe = (rolling_mean / rolling_std) * np.sqrt(252)
    return rolling_sharpe

window = 60
rolling_sharpe = calc_rolling_sharpe(returns, window=window, rf_annual=rf_annual)

# 最后一个窗口的数值
rolling_sharpe_last = rolling_sharpe.dropna().iloc[-1]
print(f"\n{window}日滚动年化夏普比率 - 最后一个窗口的值: {rolling_sharpe_last:.6f}")

fig, ax = plt.subplots(figsize=(14, 7))
plt.subplots_adjust(bottom=0.2)

dates = df[date_col].iloc[1:len(returns)+1]  # 对齐日期

# 初始曲线
line, = ax.plot(dates.iloc[window-1:], rolling_sharpe.dropna(),

ax.axhline(y=0, color='gray', linestyle='--', alpha=0.5)

slider.on_changed(update)

plt.close(fig)

print(f"\n图片已保存至: {fig_path}")
print(f"最后一个窗口的滚动年化夏普: {rolling_sharpe_last}")

# ===== 5. 输出结果 =====
result = {

}

# 3. 计算滚动年化夏普比率

rf_annual = 0.021  # 2.1%

def calc_rolling_sharpe(prices, window=60, rf_annual=0.021):
    """
    从价格序列计算滚动年化夏普比率
    夏普比率 = (超额收益均值 / 收益标准差) * sqrt(252)
    """
    rf_daily = rf_annual / 252
    prices_s = pd.Series(prices).astype(float)
    returns = prices_s.pct_change()
    excess_returns = returns - rf_daily
    rolling_mean = excess_returns.rolling(window=window).mean()
    rolling_std = returns.rolling(window=window).std()
    rolling_sharpe = (rolling_mean / rolling_std) * np.sqrt(252)
    return rolling_sharpe

这个完整的解决方案：

1. **自动加载数据**：支持 csv/xlsx/xls/parquet 格式
2. **自动识别列**：智能识别日期列和fund列
3. **计算滚动年化夏普**：公式为 `(超额收益均值 / 收益标准差) × √252`，其中日无风险利率 = 2.1% / 252
4. **交互滑块**：窗口天数可在10~252天之间调节
5. **输出结果**：`result` 字典包含 `rolling_sharpe_last` 和 `figure_path`

**关键计算说明：**
- 滚动夏普 = `[E(R - Rf) / σ(R)] × √252`
- R：日收益率，Rf：日无风险利率 = 2.1%/252
- σ(R)：日收益率标准差（不是超额收益的标准差）
- 年化因子：√252
