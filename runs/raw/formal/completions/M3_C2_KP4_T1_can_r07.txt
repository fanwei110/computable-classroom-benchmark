```python
import numpy as np
from scipy.stats import norm

# ==================== 1. 参数设置 ====================
S = 103.7      # 股票现价
K = 97.5       # 行权价
sigma = 0.276  # 隐含波动率 (年化，27.6%)
r = 0.043      # 无风险利率 (年化，4.3%)
T = 0.58       # 剩余期限 (年)
q = 0.0        # 股息率 (题目指定为不分红股票，即 q=0)

# ==================== 2. 计算 d1 与 d2 ====================
d1 = (np.log(S / K) + (r - q + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
d2 = d1 - sigma * np.sqrt(T)

# ==================== 3. 计算期权价格与 Delta ====================
# 欧式看涨期权价格 (Black-Scholes 闭式解)
call_price = S * np.exp(-q * T) * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)

# 欧式看涨期权 Delta
call_delta = np.exp(-q * T) * norm.cdf(d1)

# ==================== 4. 计算 Vega ====================
# 欧式期权 Vega (看涨与看跌相同)
# 注意：此处的 vega 是对 sigma 求导的数学定义 (即波动率每变动1.0对应的期权价格变动)
# 实务中常说"Vega per 1%"，若需该数值，将此结果除以 100 即可。
call_vega = S * np.exp(-q * T) * norm.pdf(d1) * np.sqrt(T)

# ==================== 5. 填充 result ====================
result = {
    'call_price': call_price,
    'call_delta': call_delta,
    'call_vega': call_vega
}

# ==================== 课堂投屏输出 ====================
print("="*40)
print("Black-Scholes 欧式看涨期权计算结果")
print("="*40)
print(f"d1         = {d1:.6f}")
print(f"d2         = {d2:.6f}")
print("-" * 40)
print(f"Call Price = {result['call_price']:.4f}")
print(f"Call Delta = {result['call_delta']:.4f}")
print(f"Call Vega  = {result['call_vega']:.4f}  (每 1.0 波动率变动)")
print(f"Call Vega  = {result['call_vega']/100:.4f}  (每 1% 波动率变动)")
print("="*40)
```
