以下是自包含的Python脚本，完成指定的任务：

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from pathlib import Path

# 1. 读取数据并计入无风险利率
def load_and_preprocess_data(file_path, annual_rf_rate=0.021):
    """读取CSV文件并预处理数据"""
    df = pd.read_csv(file_path, parse_dates=['date'], index_col='date')
    daily_rf_rate = annual_rf_rate / 252  # 假设252个交易日
    return df, daily_rf_rate

# 2. 计算滚动年化夏普比率
def calculate_rolling_sharpe(df, window=60, annual_rf_rate=0.021):
    """计算滚动年化夏普比率"""
    daily_rf_rate = annual_rf_rate / 252
    excess_returns = df['fund'] - daily_rf_rate

    # 计算滚动均值和标准差
    rolling_mean = excess_returns.rolling(window=window).mean()
    rolling_std = excess_returns.rolling(window=window).std()

    # 年化夏普比率
    annualized_sharpe = (rolling_mean / rolling_std) * np.sqrt(252)
    return annualized_sharpe

# 3. 绘制时间序列图并报告最后一个窗口值
def plot_and_report(sharpe_series, window=60):
    """绘制夏普比率时间序列并报告最后一个值"""
    plt.figure(figsize=(12, 6))
    sharpe_series.plot(title=f'{window}-Day Rolling Annualized Sharpe Ratio')
    plt.ylabel('Sharpe Ratio')
    plt.grid(True)

    # 保存图形
    figure_path = Path('rolling_sharpe_ratio.png')
    plt.savefig(figure_path)
    plt.close()

    # 获取最后一个窗口的值
    last_sharpe = sharpe_series.iloc[-1]
    return last_sharpe, figure_path

# 主流程
def main():
    # 设置参数
    data_path = 'data/market_snapshot_v1.csv'
    window_size = 60  # 可调窗口长度
    annual_rf_rate = 0.021

    # 1. 读取数据
    df, daily_rf_rate = load_and_preprocess_data(data_path, annual_rf_rate)

    # 2. 计算滚动夏普比率
    rolling_sharpe = calculate_rolling_sharpe(df, window=window_size, annual_rf_rate=annual_rf_rate)

    # 3. 报告和绘图
    last_sharpe, figure_path = plot_and_report(rolling_sharpe, window=window_size)

    # 4. 保存结果
    result = {
        'rolling_sharpe_last': last_sharpe,
        'figure_path': str(figure_path)
    }
    return result

# 执行并输出结果
if __name__ == '__main__':
    result = main()
    print(result)
```

这个脚本完成了以下工作：
1. 读取CSV文件中的日收益数据，并计算每日无风险利率
2. 计算60日滚动年化夏普比率（窗口长度可通过修改window_size参数调整）
3. 绘制时间序列图并保存为rolling_sharpe_ratio.png
4. 报告最后一个60日窗口的夏普比率值
5. 将结果存入result字典，包含'rolling_sharpe_last'和'figure_path'两个键

注意事项：
- 脚本假设数据文件位于data/market_snapshot_v1.csv
- 年化无风险利率默认为2.1%，每年按252个交易日计算
- 图形保存为当前目录下的rolling_sharpe_ratio.png
- 结果字典中的值都是确定可复现的
