```python
import numpy as np
import matplotlib.pyplot as plt

# ==========================================
# 1. 基础参数设置
# ==========================================
mu1, mu2 = 0.071, 0.124       # 期望年收益
sig1, sig2 = 0.163, 0.289     # 年化波动率
rhos = [0.15, 0.45, 0.75]     # 给定的三种相关系数

# 扫描权重，允许卖空，故范围扩展至[-1, 2]以充分展示双曲线
w1_scan = np.linspace(-0.5, 1.5, 1000)

# ==========================================
# 2. 绘图与计算
# ==========================================
plt.figure(figsize=(10, 7))
# 设置中文字体和负号显示
plt.rcParams['font.sans-serif'] = ['SimHei', 'Arial Unicode MS', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

mvp_vol_at_rho45 = None
frontier_vol_at_target = None
target_return = 0.10  # 目标期望收益 10%

for rho in rhos:
    # 构造协方差矩阵元素
    cov12 = rho * sig1 * sig2
    
    # --- 步骤1：在组合权重上扫描画出前沿 ---
    # 组合期望收益
    mu_p = w1_scan * mu1 + (1 - w1_scan) * mu2
    # 组合方差
    sig_p_sq = (w1_scan**2) * (sig1**2) + \
               ((1 - w1_scan)**2) * (sig2**2) + \
               2 * w1_scan * (1 - w1_scan) * cov12
    # 组合波动率
    sig_p = np.sqrt(sig_p_sq)
    
    # 绘制前沿曲线
    plt.plot(sig_p, mu_p, label=f'相关系数 ρ = {rho}', lw=2)
    
    # --- 步骤2：标出最小方差组合 (MVP) ---
    # 利用解析解求最小方差组合权重
    w1_mvp = (sig2**2 - cov12) / (sig1**2 + sig2**2 - 2 * cov12)
    w2_mvp = 1 - w1_mvp
    
    mu_mvp = w1_mvp * mu1 + w2_mvp * mu2
    sig_mvp_sq = (w1_mvp**2) * (sig1**2) + (w2_mvp**2) * (sig2**2) + 2 * w1_mvp * w2_mvp * cov12
    sig_mvp = np.sqrt(sig_mvp_sq)
    
    # 在图上标记MVP点
    plt.scatter(sig_mvp, mu_mvp, marker='*', s=200, zorder=5)
    plt.annotate(f'MVP (ρ={rho})\nσ={sig_mvp:.2%}', 
                 xy=(sig_mvp, mu_mvp), 
                 xytext=(10, 10), 
                 textcoords='offset points',
                 fontsize=9,
                 arrowprops=dict(arrowstyle='->', color='black', lw=1.2))
    
    # --- 步骤3：针对相关系数 0.45 的特定计算 ---
    if rho == 0.45:
        mvp_vol_at_rho45 = sig_mvp
        
        # 计算目标期望收益 10% 下的最优权重与最小波动率
        w1_target = (target_return - mu2) / (mu1 - mu2)
        w2_target = 1 - w1_target
        
        sig_target_sq = (w1_target**2) * (sig1**2) + \
                        (w2_target**2) * (sig2**2) + \
                        2 * w1_target * w2_target * cov12
        frontier_vol_at_target = np.sqrt(sig_target_sq)

# 图表装饰
plt.title('马科维茨均值-方差前沿 (允许卖空、满仓)', fontsize=14)
plt.xlabel('年化波动率 (σ)', fontsize=12)
plt.ylabel('期望年收益 (μ)', fontsize=12)
plt.legend(fontsize=11)
plt.grid(True, linestyle='--', alpha=0.7)

# 标出两项资产自身的位置
plt.scatter([sig1, sig2], [mu1, mu2], color='red', marker='o', s=60, zorder=5, label='风险资产')
plt.text(sig1, mu1, '  资产1', verticalalignment='center', fontsize=10, color='red')
plt.text(sig2, mu2, '  资产2', verticalalignment='center', fontsize=10, color='red')

# ==========================================
# 4. 保存图形并填充结果字典
# ==========================================
figure_path = 'markowitz_frontier.png'
plt.savefig(figure_path, dpi=300, bbox_inches='tight')
plt.close()

# 将输出结果存入字典，确保为标准 Python float
result = {
    'mvp_vol_at_rho45': float(mvp_vol_at_rho45),
    'frontier_vol_at_target': float(frontier_vol_at_target),
    'figure_path': figure_path
}

# 打印结果以便教师核验
print(f"相关系数0.45时最小方差组合年化波动率: {result['mvp_vol_at_rho45']:.6%}")
print(f"相关系数0.45时目标收益10%的最小年化波动率: {result['frontier_vol_at_target']:.6%}")
print(f"图形已保存至: {result['figure_path']}")
```
