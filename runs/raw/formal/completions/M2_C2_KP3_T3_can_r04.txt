import numpy as np

def bond_price(face, coupon_rate, ytm, maturity, freq=1):
    """计算债券价格（面值100，每年付息freq次）"""
    periods = int(maturity * freq)
    coupon = face * coupon_rate / freq
    ytm_per = ytm / freq
    t = np.arange(1, periods + 1)
    pv_coupons = coupon / (1 + ytm_per) ** t
    pv_face = face / (1 + ytm_per) ** periods
    return np.sum(pv_coupons) + pv_face

def macaulay_duration(face, coupon_rate, ytm, maturity, freq=1):
    """麦考利久期（年）"""
    periods = int(maturity * freq)
    coupon = face * coupon_rate / freq
    ytm_per = ytm / freq
    t = np.arange(1, periods + 1)
    pv_cf = coupon / (1 + ytm_per) ** t
    pv_face = face / (1 + ytm_per) ** periods
    price = np.sum(pv_cf) + pv_face
    weighted_time = np.sum(t * pv_cf) + periods * pv_face
    mac_dur = weighted_time / price / freq  # 年化
    return mac_dur

def modified_duration(face, coupon_rate, ytm, maturity, freq=1):
    """修正久期"""
    mac_dur = macaulay_duration(face, coupon_rate, ytm, maturity, freq)
    ytm_per = ytm / freq
    mod_dur = mac_dur / (1 + ytm_per)
    return mod_dur

def convexity(face, coupon_rate, ytm, maturity, freq=1):
    """凸性（年化）"""
    periods = int(maturity * freq)
    coupon = face * coupon_rate / freq
    ytm_per = ytm / freq
    t = np.arange(1, periods + 1)
    pv_cf = coupon / (1 + ytm_per) ** t
    pv_face = face / (1 + ytm_per) ** periods
    price = np.sum(pv_cf) + pv_face
    # 修正凸性分母包含 (1+y)^2，这里使用通常的凸性公式
    convex = (np.sum(t * (t + 1) * pv_cf) + periods * (periods + 1) * pv_face) / \
             (price * (1 + ytm_per) ** 2) / (freq ** 2)  # 年化
    return convex

# 参数
face_value = 100.0
coupon_rate = 0.046
ytm = 0.053
maturity = 7.0
freq = 1  # 按年付息
delta_y = 0.008  # 80个基点

# 计算当前价格
price_current = bond_price(face_value, coupon_rate, ytm, maturity, freq)

# 敏感性指标
mod_dur = modified_duration(face_value, coupon_rate, ytm, maturity, freq)
conv = convexity(face_value, coupon_rate, ytm, maturity, freq)

# 估计价格变化百分比（久期+凸性调整）
price_change_pct = -mod_dur * delta_y + 0.5 * conv * (delta_y ** 2)  # 小数形式

# 跌幅（用负的百分比变化表示，若跌则为负）
price_drop_pct = price_change_pct * 100.0  # 转为百分比点数

# 输出字典
result = {'price_drop_pct': price_drop_pct}

# 打印结果（用于课堂展示）
print("债券定价与久期凸性分析")
print(f"面值: {face_value}, 票息率: {coupon_rate*100}%, 收益率: {ytm*100}%, 期限: {maturity}年")
print(f"当前价格: {price_current:.4f}")
print(f"修正久期: {mod_dur:.4f} 年")
print(f"凸性: {conv:.4f}")
print(f"收益率变动: +{delta_y*100:.0f} 个基点")
print(f"估计价格变动百分比: {price_drop_pct:.4f}%")
print(f"\n结果字典: {result}")
