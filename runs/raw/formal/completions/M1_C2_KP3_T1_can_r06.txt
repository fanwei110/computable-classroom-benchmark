以下是自包含的 Python 脚本，计算债券的价格、麦考利久期、修正久期和凸性：

```python
import numpy as np
from scipy.optimize import newton

# 债券参数
face_value = 100.0
coupon_rate = 0.046
years_to_maturity = 7
ytm = 0.053  # 到期收益率
coupon_payment = face_value * coupon_rate
periods_per_year = 1  # 假设每年付息一次

# 1. 计算债券价格（现金流贴现之和）
def calculate_price(ytm, face_value, coupon_payment, years_to_maturity, periods_per_year=1):
    total_periods = years_to_maturity * periods_per_year
    discount_factors = np.array([(1 + ytm / periods_per_year) ** (-t) for t in range(1, total_periods + 1)])
    cash_flows = np.full(total_periods, coupon_payment)
    cash_flows[-1] += face_value  # 最后一期加上面值
    price = np.sum(cash_flows * discount_factors)
    return price

price = calculate_price(ytm, face_value, coupon_payment, years_to_maturity, periods_per_year)

# 2. 计算麦考利久期和修正久期
def calculate_duration(ytm, face_value, coupon_payment, years_to_maturity, periods_per_year=1):
    total_periods = years_to_maturity * periods_per_year
    discount_factors = np.array([(1 + ytm / periods_per_year) ** (-t) for t in range(1, total_periods + 1)])
    cash_flows = np.full(total_periods, coupon_payment)
    cash_flows[-1] += face_value

    # 麦考利久期（加权平均时间）
    weighted_cash_flows = cash_flows * discount_factors * np.arange(1, total_periods + 1)
    macaulay_duration_periods = np.sum(weighted_cash_flows) / price
    macaulay_duration_years = macaulay_duration_periods / periods_per_year

    # 修正久期
    modified_duration_years = macaulay_duration_years / (1 + ytm / periods_per_year)

    return macaulay_duration_years, modified_duration_years

macaulay_duration_years, modified_duration_years = calculate_duration(
    ytm, face_value, coupon_payment, years_to_maturity, periods_per_year
)

# 3. 计算凸性
def calculate_convexity(ytm, face_value, coupon_payment, years_to_maturity, periods_per_year=1):
    total_periods = years_to_maturity * periods_per_year
    discount_factors = np.array([(1 + ytm / periods_per_year) ** (-t) for t in range(1, total_periods + 1)])
    cash_flows = np.full(total_periods, coupon_payment)
    cash_flows[-1] += face_value

    # 凸性（二阶导数）
    weighted_cash_flows = cash_flows * discount_factors * np.arange(1, total_periods + 1) * np.arange(2, total_periods + 2)
    convexity_periods = np.sum(weighted_cash_flows) / price
    convexity = convexity_periods / (periods_per_year ** 2 * (1 + ytm / periods_per_year) ** 2)

    return convexity

convexity = calculate_convexity(ytm, face_value, coupon_payment, years_to_maturity, periods_per_year)

# 4. 填充结果字典
result = {
    'price': round(price, 4),
    'macaulay_duration_years': round(macaulay_duration_years, 4),
    'modified_duration_years': round(modified_duration_years, 4),
    'convexity': round(convexity, 4)
}

print(result)
```

输出结果将存储在 `result` 字典中，包含以下键值对：
- `price`: 债券价格（保留4位小数）
- `macaulay_duration_years`: 麦考利久期（年）
- `modified_duration_years`: 修正久期（年）
- `convexity`: 凸性（保留4位小数）
