```python
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.ticker import PercentFormatter

# ==========================================
# 1. 参数化设定
# ==========================================
rf = 0.023  # 无风险利率 2.3%
rm = 0.094  # 市场期望收益 9.4%

# 股票数据
stocks = {
    'X': {'beta': 0.62, 'return': 0.081},
    'Y': {'beta': 1.18, 'return': 0.131},
    'Z': {'beta': 1.51, 'return': 0.099}
}

# ==========================================
# 2. 核心计算
# ==========================================
# SML斜率即为市场风险溢价
sml_slope = rm - rf

# 计算.beta=1.27处的CAPM期望收益
beta_target = 1.27
er_at_beta_127 = rf + beta_target * sml_slope

# ==========================================
# 3. 绘图
# ==========================================
# 生成 beta 数据
betas = np.linspace(0, 2, 200)
# 计算 SML 对应的期望收益
er_sml = rf + betas * sml_slope

# 创建画布
fig, ax = plt.subplots(figsize=(10, 7))

# 绘制证券市场线 (SML)
ax.plot(betas, er_sml, label='Security Market Line (SML)', color='royalblue', linewidth=2.5)

# 绘制股票点并标注（包含 alpha 信息以体现偏离）
for name, data in stocks.items():
    beta_val = data['beta']
    ret_val = data['return']
    
    # 计算 CAPM 期望收益与 Alpha
    er_capm = rf + beta_val * sml_slope
    alpha_val = ret_val - er_capm
    
    # 画点
    ax.scatter(beta_val, ret_val, color='crimson', s=70, zorder=5)
    
    # 标注设置：为了避免重叠，对不同股票做微小偏移
    offset_map = {
        'X': (12, 12),
        'Y': (12, 12),
        'Z': (12, -22)
    }
    
    # 标注内容
    annotation_text = f"Stock {name}\nβ={beta_val}, R={ret_val:.1%}\nα={alpha_val:.2%}"
    
    ax.annotate(annotation_text,
                xy=(beta_val, ret_val),
                xytext=offset_map[name],
                textcoords='offset points',
                arrowprops=dict(arrowstyle='->', color='black', lw=1.2),
                bbox=dict(boxstyle="round,pad=0.3", fc="lightyellow", ec="gray", alpha=0.9),
                fontsize=9)

# 格式化图表
ax.set_title('CAPM & Security Market Line (SML)', fontsize=16, fontweight='bold')
ax.set_xlabel('Beta (β)', fontsize=13)
ax.set_ylabel('Expected Return', fontsize=13)
ax.yaxis.set_major_formatter(PercentFormatter(1.0, decimals=1))
ax.xaxis.set_major_locator(plt.MultipleLocator(0.2))
ax.set_xlim(0, 2)
ax.set_ylim(0, 0.18)
ax.grid(True, linestyle='--', alpha=0.6)
ax.legend(fontsize=12, loc='upper left')

# 保存图形
figure_path = 'sml_plot.png'
fig.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close(fig)

# ==========================================
# 4. 构建 result 字典
# ==========================================
result = {
    'sml_slope': sml_slope,
    'er_at_beta_127': er_at_beta_127,
    'figure_path': figure_path
}

# (可选) 打印验证
if __name__ == '__main__':
    print("计算结果与输出路径：")
    for k, v in result.items():
        if isinstance(v, float):
            print(f"{k}: {v:.4f} ({v*100:.2f}%)")
        else:
            print(f"{k}: {v}")
```
