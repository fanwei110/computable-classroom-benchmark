```python
import numpy as np

# ========================
# 债券参数设定（假设每年付息一次）
# ========================
face_value = 100.0       # 面值
coupon_rate = 0.046      # 票面利率（年化）
ytm = 0.053              # 到期收益率（年化）
maturity = 7             # 剩余期限（年）

# ========================
# 生成现金流
# ========================
coupon = face_value * coupon_rate           # 每年票息 = 4.6
t = np.arange(1, maturity + 1)              # 现金流发生年份 1~7
cash_flows = np.full(maturity, coupon)      # 前6年：只派息
cash_flows[-1] += face_value                # 最后一年：加上本金

# ========================
# 1. 计算当前价格
# ========================
discount_factors = (1 + ytm) ** t
present_values = cash_flows / discount_factors
price = np.sum(present_values)

# ========================
# 2. 利率敏感性：久期与凸性
# ========================
# 麦考利久期
macaulay_duration = np.sum(t * present_values) / price

# 修正久期
modified_duration = macaulay_duration / (1 + ytm)

# 凸性（公式：1/(P*(1+y)^2) * Σ [t*(t+1)*CF/(1+y)^t]）
convexity = np.sum(t * (t + 1) * present_values) / (price * (1 + ytm) ** 2)

# ========================
# 3. 估算收益率上升 80 个基点的影响
# ========================
delta_y = 0.0080   # 80 bp = 0.80%

# 近似价格变动比例（使用久期+凸性）
approx_price_change = -modified_duration * delta_y + 0.5 * convexity * (delta_y ** 2)

# 精确价格变动比例（用于对比验证）
new_ytm = ytm + delta_y
new_pv = cash_flows / (1 + new_ytm) ** t
new_price = np.sum(new_pv)
exact_price_change = (new_price - price) / price

# ========================
# 4. 跌幅（百分比形式，正数表示下跌了多少%）
# ========================
price_drop_pct = -approx_price_change * 100   # 转换为百分比，取正表示为跌幅

# ========================
# 输出结果至字典
# ========================
result = {
    'price_drop_pct': price_drop_pct
}

# 在屏幕上输出关键信息（便于课堂投屏观察）
print(f"当前债券价格: {price:.4f}")
print(f"麦考利久期:   {macaulay_duration:.4f} 年")
print(f"修正久期:     {modified_duration:.4f}")
print(f"凸性:         {convexity:.4f}")
print(f"收益率变动:   +{delta_y*100:.0f} 个基点")
print(f"近似价格变动比例: {approx_price_change:.6f} ({approx_price_change*100:.4f}%)")
print(f"精确价格变动比例: {exact_price_change:.6f} ({exact_price_change*100:.4f}%)")
print(f"预计价格跌幅: {price_drop_pct:.4f}%")
print("\n最终结果字典:")
print(result)
```
