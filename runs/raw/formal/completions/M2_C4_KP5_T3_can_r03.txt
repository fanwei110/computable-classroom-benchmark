```python
import numpy as np
from scipy.stats import norm

# ---- 输入参数 ----
annual_vol = 0.24          # 年化波动率（小数）
position_value = 2_700_000 # 头寸价值（元）
confidence = 0.95          # 置信水平
trading_days = 252         # 年交易日数

# ---- 1. 去年化：日波动率 ----
daily_vol = annual_vol / np.sqrt(trading_days)

# ---- 2. 单尾 95% 正态分位数（均值取零） ----
# 单尾 5% 分位数对应于收益率分布的下分位数
z_score = norm.ppf(1 - confidence)   # 得到负值，如 -1.64485
var_dollar = position_value * daily_vol * abs(z_score)  # VaR 为正损失金额

# ---- 3. 存储结果 ----
result = {
    'var_95_1d': var_dollar
}

# 输出以便投屏展示
if __name__ == "__main__":
    print(f"年化波动率: {annual_vol*100:.2f}%")
    print(f"日波动率: {daily_vol*100:.4f}%")
    print(f"单尾 95% 正态分位数 (z): {z_score:.5f}")
    print(f"95% 一日 VaR (人民币): {var_dollar:,.2f} 元")
    print(f"\n结果字典: {result}")
```
