```python
# 计算债券各项指标
face_value = 100
coupon_rate = 0.046
coupon = face_value * coupon_rate  # 每年票息 4.6
ytm = 0.053
years = 7

# 1. 价格 (Price)
# P = C * [1 - (1+y)^(-T)] / y + F * (1+y)^(-T)
discount = (1 + ytm) ** years
pv_factor = 1 / discount
annuity_factor = (1 - pv_factor) / ytm
price = coupon * annuity_factor + face_value * pv_factor

# 2. 麦考利久期 (Macaulay Duration)
# 计算各年现金流现值及其时间加权
pv_coupons = [coupon / (1 + ytm)**t for t in range(1, years + 1)]
pv_face = face_value / (1 + ytm)**years
total_pv = sum(pv_coupons) + pv_face
weighted_sum = sum(t * pv for t, pv in enumerate(pv_coupons, start=1)) + years * pv_face
macaulay_duration = weighted_sum / total_pv

# 3. 修正久期 (Modified Duration)
modified_duration = macaulay_duration / (1 + ytm)

# 4. 凸性 (Convexity)
# C = [1 / (P * (1+y)^2)] * sum[ t*(t+1) * PV(CF_t) ]
# 或者直接用各年现金流
convexity_sum = sum(t * (t + 1) * pv for t, pv in enumerate(pv_coupons, start=1)) + years * (years + 1) * pv_face
convexity = convexity_sum / (total_pv * (1 + ytm)**2)

# 结果保留6位小数，可按需调整
result = {
    'price': round(price, 6),
    'macaulay_duration_years': round(macaulay_duration, 6),
    'modified_duration_years': round(modified_duration, 6),
    'convexity': round(convexity, 6)
}
```
