```python
import numpy as np
import matplotlib.pyplot as plt

# 资产参数
mu = np.array([0.071, 0.124])
sigma = np.array([0.163, 0.289])
var = sigma ** 2
cov_base = np.outer(sigma, sigma)  # σ_i * σ_j 矩阵

# 相关系数列表
rhos = [0.15, 0.45, 0.75]

# 生成有效前沿的函数
def frontier_points(mu, sigma, rho, n_points=200):
    """返回到期收益对应的最小标准差"""
    cov = np.array([[sigma[0]**2, rho*sigma[0]*sigma[1]],
                    [rho*sigma[0]*sigma[1], sigma[1]**2]])
    inv_cov = np.linalg.inv(cov)
    ones = np.ones(2)
    A = ones @ inv_cov @ ones
    B = ones @ inv_cov @ mu
    C = mu @ inv_cov @ mu
    D = A*C - B**2
    # 最小方差组合
    w_mvp = inv_cov @ ones / A
    mu_mvp = w_mvp @ mu
    sigma_mvp = np.sqrt(1 / A)
    # 有效前沿：mu 从 mu_mvp 到 max(mu)
    mu_range = np.linspace(mu_mvp, max(mu)*1.05, n_points)
    # 根据公式 σ² = (A*μ² - 2*B*μ + C) / D
    sigma_range = np.sqrt((A * mu_range**2 - 2 * B * mu_range + C) / D)
    return mu_range, sigma_range, w_mvp, mu_mvp, sigma_mvp

# 绘图
plt.figure(figsize=(10, 6))
for rho in rhos:
    mu_range, sigma_range, w_mvp, mu_mvp, sigma_mvp = frontier_points(mu, sigma, rho)
    plt.plot(sigma_range * 100, mu_range * 100, label=f'ρ = {rho}')
    plt.scatter(sigma_mvp * 100, mu_mvp * 100, marker='o', s=60)  # 标注最小方差组合

plt.xlabel('年化波动率 (%)')
plt.ylabel('期望年收益 (%)')
plt.title('两资产有效前沿')
plt.legend()
plt.grid(True)
plt.tight_layout()
fig_path = 'effective_frontier.png'
plt.savefig(fig_path)
```
