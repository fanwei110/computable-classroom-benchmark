```python
import numpy as np
from scipy.stats import norm

def main():
    # 给定参数
    position_value = 1_850_000  # 头寸价值（元）
    annual_vol = 0.218          # 年化收益波动率（小数形式）
    trading_days = 252          # 一年交易日数

    # 1. 年化波动率去年化到一日
    daily_vol = annual_vol / np.sqrt(trading_days)

    # 2. 单尾正态分位数（对于 VaR 我们使用左侧分位数对应的正值）
    # 95% 置信水平：左尾概率 5%，对应 z 分位数为负值，损失取绝对值即为 norm.ppf(0.95)
    z_95 = norm.ppf(0.95)
    # 99% 置信水平
    z_99 = norm.ppf(0.99)

    # 3. 计算一日 VaR（均值取零）
    var_95_1d = position_value * daily_vol * z_95

    # 4. 十日 VaR：一日波动率按 sqrt(10) 缩放
    var_99_10d = position_value * daily_vol * np.sqrt(10) * z_99

    # 5. 创建结果字典
    result = {
        'var_95_1d': var_95_1d,
        'var_99_10d': var_99_10d
    }

    # 打印结果以供课堂投屏展示
    print("=== 在险价值 (VaR) 计算结果 ===")
    print(f"头寸价值: {position_value:,.0f} 元")
    print(f"年化波动率: {annual_vol*100:.1f}%")
    print(f"日波动率: {daily_vol*100:.4f}%")
    print(f"95% 一日 VaR: {var_95_1d:,.2f} 元")
    print(f"99% 十日 VaR: {var_99_10d:,.2f} 元")
    print("\n结果字典:", result)

    return result

if __name__ == "__main__":
    result = main()
```
