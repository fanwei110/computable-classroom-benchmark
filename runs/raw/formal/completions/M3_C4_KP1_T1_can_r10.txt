```python
import numpy as np

# ==================== 1. 参数设定 ====================
# 年化波动率 (小数表示)
vols = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr = np.array([
    [1.00,  0.21, -0.13],
    [0.21,  1.00,  0.37],
    [-0.13, 0.37,  1.00]
])

# ==================== 2. 构造协方差矩阵 ====================
# 协方差矩阵 Σ = diag(σ) @ Corr @ diag(σ)
cov_matrix = np.diag(vols) @ corr @ np.diag(vols)

# ==================== 3. 求解全局最小方差组合(MVP)权重 ====================
# 满仓约束(∑w=1)下，MVP闭式解为: w_mvp = Σ^{-1} * 1 / (1' * Σ^{-1} * 1)
# 此处使用 np.linalg.solve 替代直接求逆，以获得更好的数值稳定性
ones = np.ones(len(vols))
w_raw = np.linalg.solve(cov_matrix, ones)
w_mvp = w_raw / np.sum(w_raw)

# ==================== 4. 计算组合年化波动率 ====================
# 组合方差 σ_p^2 = w' Σ w
mvp_var = w_mvp.T @ cov_matrix @ w_mvp
mvp_vol = np.sqrt(mvp_var)

# ==================== 5. 按契约要求存入字典 ====================
result = {
    'mvp_weights': w_mvp.tolist(),
    'mvp_vol_annual': float(mvp_vol)
}

# ==================== 课堂投屏打印 ====================
print("====== 马科维茨均值-方差组合理论 ======")
print("协方差矩阵 Σ:\n", np.round(cov_matrix, 6))
print("-" * 40)
print(f"全局最小方差组合权重: {result['mvp_weights']}")
print(f"全局最小方差组合年化波动率: {result['mvp_vol_annual']:.4%}")
```
