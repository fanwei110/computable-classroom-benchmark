```python
import numpy as np

# 给定参数
beta = 1.42
market_return_monthly = -0.058      # 上个月市场收益率（下跌 5.8%）
annual_risk_free = 0.047            # 年化无风险利率 4.7%

# 1. 将年无风险利率折算为月利率（假设采用单利：年利率/12）
risk_free_monthly = annual_risk_free / 12

# 2. 在月度层面套用 CAPM 公式
#    E(R_i) = R_f + beta * (R_m - R_f)
#    其中 R_m 使用上个月实际市场收益率，计算事后应有的预期收益
capm_return_monthly = risk_free_monthly + beta * (market_return_monthly - risk_free_monthly)

# 3. 按要求存入字典
result = {
    'capm_return_monthly': capm_return_monthly
}

# 输出结果（供投屏查看）
print(result)
```
