```python
import numpy as np

# 债券参数
face_value = 100.0
coupon_rate = 0.046
years = 7
yield_to_maturity = 0.053

# 现金流时间点
t = np.arange(1, years + 1)

# 每年现金流：票息，最后一年加上面值
cash_flows = np.full(years, face_value * coupon_rate)
cash_flows[-1] += face_value  # 到期还本

# 折现因子
discount_factors = (1 + yield_to_maturity) ** t

# 各期现金流现值
present_values = cash_flows / discount_factors

# 价格
price = np.sum(present_values)

# 麦考利久期（年）
weighted_times = t * present_values
macaulay_duration_years = np.sum(weighted_times) / price

# 修正久期（年）
modified_duration_years = macaulay_duration_years / (1 + yield_to_maturity)

# 凸性（年的平方） Σ[t(t+1)CF/(1+y)^(t+2)] / P
convexity_numerator = np.sum(t * (t + 1) * cash_flows / (1 + yield_to_maturity) ** (t + 2))
convexity = convexity_numerator / price

# 结果字典
result = {
    'price': price,
    'macaulay_duration_years': macaulay_duration_years,
    'modified_duration_years': modified_duration_years,
    'convexity': convexity
}

# 输出，方便课堂展示
for k, v in result.items():
    print(f"{k}: {v:.6f}")
```
