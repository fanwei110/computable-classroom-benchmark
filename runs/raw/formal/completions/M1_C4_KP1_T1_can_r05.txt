```python
import numpy as np
from scipy.optimize import minimize

# 给定数据
vols = np.array([0.187, 0.243, 0.312])  # 年化波动率
corr_matrix = np.array([
    [1.00, 0.21, -0.13],
    [0.21, 1.00,  0.37],
    [-0.13, 0.37, 1.00]
])

# 步骤1：构造协方差矩阵
cov_matrix = np.outer(vols, vols) * corr_matrix

# 步骤2：求最小方差组合权重（满仓约束）
n_assets = len(vols)
# 目标函数：组合方差 w'Σw
def portfolio_variance(w):
    return w @ cov_matrix @ w

# 约束：权重之和为1
constraints = ({'type': 'eq', 'fun': lambda w: np.sum(w) - 1})
# 初始猜测：等权重
initial_guess = np.ones(n_assets) / n_assets
# 优化
result_opt = minimize(portfolio_variance, initial_guess, constraints=constraints)
mvp_weights = result_opt.x

# 步骤3：计算组合波动率
mvp_vol_annual = np.sqrt(portfolio_variance(mvp_weights))

# 步骤4：填充结果字典
result = {
    'mvp_weights': mvp_weights.tolist(),
    'mvp_vol_annual': float(mvp_vol_annual)
}

print(result)
```
