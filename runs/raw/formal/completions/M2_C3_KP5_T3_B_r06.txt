```python
import math

# 给定参数
annual_vol = 0.24
position = 2_700_000
confidence = 0.95
trading_days = 252

# 正态分布95%分位数（单尾）
z_score = 1.6448536269514722  # scipy.stats.norm.ppf(0.95)

# 日波动率
daily_vol = annual_vol / math.sqrt(trading_days)

# 一日VaR（金额，正损失）
var_1d = position * daily_vol * z_score

result = {'var_95_1d': var_1d}
print(result)
```
