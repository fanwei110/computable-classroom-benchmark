```python
import numpy as np
import matplotlib.pyplot as plt

# 资产参数
r = np.array([0.071, 0.124])          # 期望年收益
sigma = np.array([0.163, 0.289])     # 年化波动率
rhos = [0.15, 0.45, 0.75]            # 相关系数

# 生成扫描权重（满仓，允许卖空）
w = np.linspace(-3, 4, 2000)         # 足够宽的范围以覆盖前沿
weights = np.column_stack((w, 1 - w))

# 绘图
plt.figure(figsize=(8, 6))
colors = ['blue', 'red', 'green']    # 三条曲线颜色

# 用于存储关键字结果
mvp_vol_rho45 = None
target_vol_rho45 = None

for idx, rho in enumerate(rhos):
    # 协方差矩阵元素
    cov12 = rho * sigma[0] * sigma[1]
    var1 = sigma[0]**2
    var2 = sigma[1]**2
    
    # 组合方差与期望收益（向量化计算）
    port_var = (weights[:, 0]**2 * var1) + (weights[:, 1]**2 * var2) + \
               (2 * weights[:, 0] * weights[:, 1] * cov12)
    port_std = np.sqrt(port_var)
    port_ret = weights[:, 0] * r[0] + weights[:, 1] * r[1]
    
    # 绘制均值-方差前沿曲线
    plt.plot(port_std, port_ret, color=colors[idx], label=rf'$\rho$ = {rho}')
    
    # 计算最小方差组合 (MVP)
    # 解析公式：w_mvp = (sigma2^2 - rho * sigma1 * sigma2) / (sigma1^2 + sigma2^2 - 2*rho*sigma1*sigma2)
    w_mvp = (var2 - cov12) / (var1 + var2 - 2 * cov12)
    mu_mvp = w_mvp * r[0] + (1 - w_mvp) * r[1]
    # 组合方差
    var_mvp = (w_mvp**2 * var1) + ((1 - w_mvp)**2 * var2) + (2 * w_mvp * (1 - w_mvp) * cov12)
    vol_mvp = np.sqrt(var_mvp)
    
    # 在曲线上标出 MVP
    plt.scatter(vol_mvp, mu_mvp, color=colors[idx], marker='*', s=100,
                edgecolor='black', zorder=5, label=f'MVP ($\\rho$={rho})')
    
    # 对 rho=0.45 记录要求的数据
    if rho == 0.45:
        mvp_vol_rho45 = vol_mvp
        
        # 目标收益 10% 下的组合
        # 权重由 w_target = (target - r2) / (r1 - r2)   (满仓)
        target = 0.10
        w_target = (target - r[1]) / (r[0] - r[1])
        # 对应波动率
        var_target = (w_target**2 * var1) + ((1 - w_target)**2 * var2) + \
                     (2 * w_target * (1 - w_target) * cov12)
        target_vol_rho45 = np.sqrt(var_target)

plt.xlabel('Annualized Volatility')
plt.ylabel('Annualized Expected Return')
plt.title('Mean-Variance Frontier under Different Correlations\n(Full Investment, Shorting Allowed)')
plt.legend()
plt.grid(True, linestyle='--', alpha=0.7)
plt.tight_layout()

# 保存图片
fig_path = 'mean_variance_frontier.png'
plt.savefig(fig_path, dpi=150)
plt.close()

# 构造输出字典
result = {
    'mvp_vol_at_rho45': mvp_vol_rho45,
    'frontier_vol_at_target': target_vol_rho45,
    'figure_path': fig_path
}

# 打印结果以便课堂查看
print("========== 结果输出 ==========")
print(f"相关系数 0.45 时最小方差组合年化波动率: {mvp_vol_rho45:.6f} ({mvp_vol_rho45*100:.4f}%)")
print(f"目标期望收益 10% 下的最小年化波动率:   {target_vol_rho45:.6f} ({target_vol_rho45*100:.4f}%)")
print(f"图形已保存至: {fig_path}")
print("result 字典内容:", result)

# 脚本结束时 result 变量可用于后续调用
```
