```python
import numpy as np
from scipy.stats import norm

# ================= 市场参数设定 =================
S = 103.7       # 标的现价
K = 97.5        # 行权价
r = 0.043       # 无风险利率 (连续复利, 4.3%)
T = 0.58        # 剩余到期时间 (年)
sigma_1 = 0.276 # 初始波动率 (27.6%)
sigma_2 = 0.286 # 上升1个百分点后的波动率 (28.6%)

# ================= BS 欧式看涨期权定价公式 =================
def bs_call_price(S, K, r, T, sigma):
    """
    利用 Black-Scholes 闭式解计算欧式看涨期权价格
    """
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    
    call_price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
    return call_price

# ================= 分别在两个波动率下精确重定价 =================
price_sigma_1 = bs_call_price(S, K, r, T, sigma_1)
price_sigma_2 = bs_call_price(S, K, r, T, sigma_2)

# ================= 计算精确价差 =================
price_change = price_sigma_2 - price_sigma_1

# ================= 按照输出契约存入字典 =================
result = {
    'price_change': price_change
}

# （可选：课堂投屏展示中间过程，方便讲解）
if __name__ == '__main__':
    print(f"波动率 27.6% 下的看涨期权精确价格: {price_sigma_1:.4f}")
    print(f"波动率 28.6% 下的看涨期权精确价格: {price_sigma_2:.4f}")
    print(f"隐含波动率上升1个百分点，期权价格变动: {price_change:.4f}")
```
