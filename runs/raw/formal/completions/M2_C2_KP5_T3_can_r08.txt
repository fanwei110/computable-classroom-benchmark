```python
import numpy as np
from scipy.stats import norm

# 给定参数
position = 2_700_000        # 头寸金额（元）
annual_vol = 0.24           # 年化波动率
confidence = 0.95           # 置信水平

# 假设：一年有 252 个交易日
trading_days = 252

# 1. 将年化波动率换算为一日波动率
daily_vol = annual_vol / np.sqrt(trading_days)

# 2. 95% 置信水平对应的标准正态分位数（单尾）
z_score = norm.ppf(confidence)   # 约 1.64485

# 3. 计算 95% 一日 VaR（金额，正值表示损失）
var_95_1d = position * daily_vol * z_score

# 输出契约
result = {'var_95_1d': var_95_1d}

print(result)
```
