```python
import numpy as np

# 债券参数
face_value = 100.0          # 面值
coupon_rate = 0.046         # 年票息率
yield_to_maturity = 0.053   # 年收益率（年复利）
years_to_maturity = 7       # 剩余期限（年）
dy = 0.008                  # 收益率变动（80个基点）

# 1. 计算每期现金流
periods = years_to_maturity
coupon_payment = face_value * coupon_rate
cash_flows = np.full(periods, coupon_payment)
cash_flows[-1] += face_value  # 最后一期加上面值

# 2. 计算麦考利久期
discount_factors = np.array([(1 + yield_to_maturity) ** (-t) for t in range(1, periods + 1)])
present_values = cash_flows * discount_factors
bond_price = np.sum(present_values)

# 麦考利久期
macaulay_duration = np.sum([t * pv for t, pv in zip(range(1, periods + 1), present_values)]) / bond_price

# 修正久期
modified_duration = macaulay_duration / (1 + yield_to_maturity)

# 3. 估算价格跌幅
price_drop_pct = modified_duration * dy  # dP/P = -D_mod * dy，跌幅为正数

# 输出结果
result = {
    'price_drop_pct': price_drop_pct
}
```
