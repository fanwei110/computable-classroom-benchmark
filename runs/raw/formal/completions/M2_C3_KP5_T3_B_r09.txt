```python
import numpy as np
from scipy.stats import norm

# 给定参数
annual_vol = 0.24
position = 2_700_000
confidence = 0.95
trading_days = 252

# 计算日波动率
daily_vol = annual_vol / np.sqrt(trading_days)

# 95%置信水平下的z值（单尾）
z_score = norm.ppf(1 - confidence)  # 左尾分位数，为负值
var_95_1d = abs(position * daily_vol * z_score)  # VaR报告为正的损失金额

result = {'var_95_1d': var_95_1d}
print(result)
```
