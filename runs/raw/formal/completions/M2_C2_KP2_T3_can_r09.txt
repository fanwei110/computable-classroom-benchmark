```python
import numpy as np

def main():
    # 给定参数
    beta = 1.42
    market_return_monthly = -0.058   # 市场下跌 5.8%
    rf_annual = 0.047                # 年化无风险利率 4.7%
    
    # 将年化无风险利率折算为月利率（复利方式）
    # 假设：年利率为有效年利率，按月复利折算
    rf_monthly = (1 + rf_annual) ** (1 / 12) - 1
    
    # 在月度频率上应用 CAPM 公式：E(R) = Rf + beta * (Rm - Rf)
    capm_return_monthly = rf_monthly + beta * (market_return_monthly - rf_monthly)
    
    # 按要求构造输出字典
    result = {
        'capm_return_monthly': capm_return_monthly
    }
    
    # 输出结果（教师可直接投屏查看）
    print("CAPM 预期月度收益（小数形式）：", capm_return_monthly)
    print(f"即 {capm_return_monthly * 100:.2f}%")
    print("\nresult =", result)

if __name__ == "__main__":
    main()
```
