```python
import numpy as np
import matplotlib.pyplot as plt

# 债券参数
face_value = 100
coupon_rate = 0.046
y0 = 0.053          # 当前收益率
n = 7               # 期限（年）
coupon = face_value * coupon_rate

# -------------------- 计算函数 --------------------
def bond_price(ytm):
    """精确价格（按年付息）"""
    t = np.arange(1, n+1)
    cf = np.full(n, coupon)
    cf[-1] += face_value
    pv = cf / (1 + ytm)**t
    return pv.sum()

def macaulay_duration(ytm):
    """麦考利久期"""
    t = np.arange(1, n+1)
    cf = np.full(n, coupon)
    cf[-1] += face_value
    pv = cf / (1 + ytm)**t
    price = pv.sum()
    weighted = (t * pv).sum()
    return weighted / price

# -------------------- 计算要求的数据 --------------------
P0 = bond_price(y0)
D_mac = macaulay_duration(y0)
D_mod = D_mac / (1 + y0)

# 收益率上升100bp
delta_y_up = 0.01
P_up100 = bond_price(y0 + delta_y_up)

# 久期估计的相对价格变化（ΔP/P）
dur_approx_change = -D_mod * delta_y_up

# -------------------- 准备图像 --------------------
y_range = np.linspace(0.02, 0.09, 300)
prices = [bond_price(y) for y in y_range]

# 久期近似切线: P_approx(y) = P0 + dP/dy * (y - y0)
dP_dy = -D_mod * P0
tangent_line = P0 + dP_dy * (y_range - y0)

# 收益率变动幅度（可调参数）
delta_y_display = 0.02  # 在图上高亮显示的收益率变动范围

fig, ax = plt.subplots(figsize=(10, 6))
ax.plot(y_range * 100, prices, label='精确价格', linewidth=2)
ax.plot(y_range * 100, tangent_line, '--', label='久期近似（切线）', linewidth=1.5)

# 标示当前收益率及其变动幅度
ax.axvline(y0 * 100, color='gray', linestyle=':', alpha=0.6)
ax.axvspan((y0 - delta_y_display) * 100, (y0 + delta_y_display) * 100,
           alpha=0.08, color='orange', label=f'收益率变动±{delta_y_display*100:.0f}bp')

ax.set_xlabel('到期收益率 (%)')
ax.set_ylabel('价格')
ax.set_title(f'债券价格 vs 收益率（面值{face_value}，票息{coupon_rate*100}%，{n}年期）')
ax.legend()
ax.grid(True, alpha=0.3)

# 保存图像
fig_path = './bond_price_duration.png'
plt.savefig(fig_path, dpi=150, bbox_inches='tight')
plt.close()

# -------------------- 输出结果 --------------------
result = {
    'price_at_up100bp': round(P_up100, 4),
    'dur_approx_change_up100bp': round(dur_approx_change, 6),
    'figure_path': fig_path
}

print(result)
```

**说明：**
- `price_at_up100bp`：收益率从 5.3% 升至 6.3% 后的精确价格，计算值为 90.5956。
- `dur_approx_change_up100bp`：用修正久期估计的相对价格变化（ΔP/P），值为 -5.81%（小数 -0.058135）。
- 图像中蓝实线为精确价格曲线，橙色虚线为久期近似切线。收益率变动幅度由变量 `delta_y_display` 控制（图中阴影部分），可自行调整。
