```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# 设置中文字体和负号显示
plt.rcParams['font.sans-serif'] = ['SimHei', 'Arial Unicode MS']
plt.rcParams['axes.unicode_minus'] = False

# 1. 读取数据
df = pd.read_csv('data/market_snapshot_v1.csv')

# 2. 计算日收益率
# 假设fund列为基金净值，通过pct_change计算日收益率；若fund列已是日收益率，则直接使用
# 为兼容两种情况，若数据均值大于1（通常为净值），则计算pct_change
if df['fund'].mean() > 1.5:
    daily_returns = df['fund'].pct_change().dropna()
else:
    daily_returns = df['fund']

# 3. 参数设置
position = 1000000.0  # 头寸100万
confidence_level = 0.95  # 置信度可调
alpha = 1 - confidence_level

# 4. 计算损益 (P&L)
pnl = position * daily_returns

# 5. 计算历史法VaR
# 取损益序列的alpha分位数，VaR为正的损失金额
var_pnl = np.percentile(pnl, alpha * 100)
hist_var_95_1d = max(0.0, -var_pnl)  # 确保VaR报告为正的损失金额

# 6. 绘制损益直方图及VaR标根线
plt.figure(figsize=(10, 6))
plt.hist(pnl, bins=50, alpha=0.75, edgecolor='black', color='steelblue')
plt.axvline(x=var_pnl, color='red', linestyle='--', linewidth=2, 
            label=f'{confidence_level*100:.0f}% 1-Day VaR: {hist_var_95_1d:,.2f}')
plt.title('损益直方图与历史法VaR', fontsize=14)
plt.xlabel('损益金额 (P&L)', fontsize=12)
plt.ylabel('频数', fontsize=12)
plt.legend(fontsize=12)
plt.grid(axis='y', alpha=0.3)

# 7. 保存图表
figure_path = 'pnl_hist_var.png'
plt.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close()

# 8. 构建输出字典
result = {
    'hist_var_95_1d': hist_var_95_1d,
    'figure_path': figure_path
}

print(f"历史法95%一日VaR: {result['hist_var_95_1d']:,.2f}")
print(f"图表已保存至: {result['figure_path']}")
```
