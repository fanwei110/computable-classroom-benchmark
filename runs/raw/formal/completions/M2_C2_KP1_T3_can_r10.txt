```python
import numpy as np

# 资产参数：年化波动率
sigma_A = 0.184   # 资产A
sigma_B = 0.297   # 资产B

# 60/40组合权重：A占60%，B占40%
w = np.array([0.6, 0.4])

# 相关系数情景
rho_before = 0.3
rho_after  = 0.8

def make_cov(s1, s2, rho):
    """由波动率和相关系数构造协方差矩阵"""
    cov12 = rho * s1 * s2
    return np.array([[s1**2, cov12],
                     [cov12, s2**2]])

cov_before = make_cov(sigma_A, sigma_B, rho_before)
cov_after  = make_cov(sigma_A, sigma_B, rho_after)

# 组合方差与波动率（年化）
var_before = w @ cov_before @ w
vol_before = np.sqrt(var_before)

var_after = w @ cov_after @ w
vol_after = np.sqrt(var_after)

# 输出字典
result = {
    'vol_before_annual': vol_before,
    'vol_after_annual': vol_after
}

print(result)
```
