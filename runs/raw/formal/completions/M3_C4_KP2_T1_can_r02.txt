```python
import numpy as np

# 1. 设定已知参数（按约定使用小数表示）
rf = 0.023       # 无风险利率 2.3%
erm = 0.094      # 市场期望收益 9.4%
beta_x = 0.62
beta_y = 1.18
beta_z = 1.51
actual_return_y = 0.131  # 股票 Y 的实际收益 13.1%

# 计算市场风险溢价
market_risk_premium = erm - rf

# 2. 对每个 beta 套用 CAPM 公式：E[Ri] = rf + beta * (E[Rm] - rf)
er_x = rf + beta_x * market_risk_premium
er_y = rf + beta_y * market_risk_premium
er_z = rf + beta_z * market_risk_premium

# 3. 计算 Y 的 alpha：实际收益减去 CAPM 期望收益
alpha_y = actual_return_y - er_y

# 4. 填充 result 字典
result = {
    'er_x': er_x,
    'er_y': er_y,
    'er_z': er_z,
    'alpha_y': alpha_y
}

# 打印结果以便课堂投屏展示
print(f"股票 X 的 CAPM 期望收益: {er_x:.5f} ({er_x*100:.3f}%)")
print(f"股票 Y 的 CAPM 期望收益: {er_y:.5f} ({er_y*100:.3f}%)")
print(f"股票 Z 的 CAPM 期望收益: {er_z:.5f} ({er_z*100:.3f}%)")
print(f"股票 Y 的 Alpha: {alpha_y:.5f} ({alpha_y*100:.3f}%)")
print("\n最终输出字典 result:")
print(result)
```
