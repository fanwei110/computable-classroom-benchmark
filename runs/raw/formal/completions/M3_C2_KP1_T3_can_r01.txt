```python
import numpy as np

# ==========================================
# 《证券投资学》课堂演示：马科维茨组合方差与相关系数
# ==========================================

# 1. 自行确定 60/40 权重与两只资产的对应方式
# 假设：资产A占60%，资产B占40%
w_A = 0.6
w_B = 0.4
w = np.array([w_A, w_B])

# 资产波动率（年化）
vol_A = 0.184
vol_B = 0.297

# 资产方差
var_A = vol_A ** 2
var_B = vol_B ** 2

# 2. 构造相关系数 0.3 与 0.8 两个协方差矩阵
rho_before = 0.3
rho_after = 0.8

# 协方差 = 相关系数 * 资产A标准差 * 资产B标准差
cov_before = rho_before * vol_A * vol_B
cov_after = rho_after * vol_A * vol_B

Sigma_before = np.array([
    [var_A, cov_before],
    [cov_before, var_B]
])

Sigma_after = np.array([
    [var_A, cov_after],
    [cov_after, var_B]
])

# 3. 计算两个组合波动率
# 组合方差公式：w'Σw
port_var_before = w.T @ Sigma_before @ w
port_var_after = w.T @ Sigma_after @ w

# 组合波动率（标准差）
vol_before = float(np.sqrt(port_var_before))
vol_after = float(np.sqrt(port_var_after))

# 4. 填充 result
result = {
    'vol_before_annual': vol_before,
    'vol_after_annual': vol_after
}

# 课堂投屏辅助打印（验证可复现性）
if __name__ == "__main__":
    print(f"假设资产A权重为60%，资产B权重为40%：")
    print(f"当相关系数 = {rho_before} 时，组合年化波动率 = {vol_before:.4%}")
    print(f"当相关系数 = {rho_after} 时，组合年化波动率 = {vol_after:.4%}")
    print(f"波动率变化 = {(vol_after - vol_before):.4%} (相关系数提升导致分散化效应减弱，波动率上升)")
    print("\n结果字典 result：")
    print(result)
```
