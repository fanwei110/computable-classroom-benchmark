```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# ==================== 可调参数 ====================
confidence_level = 0.95          # 置信水平，可在此修改
position_value = 1_000_000.0     # 头寸金额（人民币元）
data_path = 'data/market_snapshot_v1.csv'
figure_filename = 'var_hist.png'
# =================================================

# ---------- 1. 读取数据并构造日损益序列 ----------
df = pd.read_csv(data_path)
# 假定 "fund" 列为日收益率（小数形式，例如 0.01 代表 1%）
# 实际教学中请确认列名；若数据格式有差异，可在此微调
daily_returns = df['fund'].dropna()   # 去除缺失值
# 损益 = 头寸 * 日收益率
pnl = position_value * daily_returns

# ---------- 2. 计算历史 VaR ----------
# 经验分位数（默认 linear 插值）
left_tail_q = 100 * (1 - confidence_level)
var_cutoff = np.percentile(pnl, left_tail_q)   # 损益分布的左侧分位数（通常为负）
hist_var = -var_cutoff                         # VaR 报告为正的损失金额

# ---------- 3. 绘图 ----------
plt.figure(figsize=(10, 6))
plt.hist(pnl, bins=50, edgecolor='k', alpha=0.7, color='steelblue')
plt.axvline(x=var_cutoff, color='red', linestyle='--', linewidth=2,
            label=f'{int(confidence_level*100)}% 历史VaR: ¥{hist_var:,.2f}')
plt.xlabel('日损益（人民币元）')
plt.ylabel('频数')
plt.title(f'头寸 ¥{position_value:,.0f} 的日损益分布')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()

# 保存图片
plt.savefig(figure_filename)
plt.close()

# ---------- 4. 组装结果 ----------
result = {
    'hist_var_95_1d': hist_var,
    'figure_path': figure_filename
}

# 可选：将结果输出到控制台，方便教师查看
print(result)
```
