```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import norm

# 参数设置
K = 97.5
r_annual = 0.043  # 年复利报价
T = 0.58
vol_list = [0.15, 0.276, 0.40]
S_range = np.linspace(70, 140, 500)

# 无风险利率转换为连续复利
r_cont = np.log(1 + r_annual)

# 计算 delta 函数（看涨期权）
def delta_call(S, K, r, T, sigma):
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    return norm.cdf(d1)

# 画图
plt.figure(figsize=(10, 6))
for sigma in vol_list:
    delta = delta_call(S_range, K, r_cont, T, sigma)
    plt.plot(S_range, delta, label=f'σ = {sigma*100:.1f}%')

plt.xlabel('标的资产价格')
plt.ylabel('Delta')
plt.title('看涨期权 Delta 曲线')
plt.legend()
plt.grid(True)
plt.tight_layout()

# 保存图形
fig_path = 'delta_curve.png'
plt.savefig(fig_path)
plt.close()

# 特定点计算：S=110, sigma=27.6%
S_target = 110
sigma_target = 0.276
delta_at_s110 = delta_call(S_target, K, r_cont, T, sigma_target)

# 结果汇总
result = {
    'delta_at_s110': round(delta_at_s110, 6),
    'figure_path': fig_path
}

result
```
