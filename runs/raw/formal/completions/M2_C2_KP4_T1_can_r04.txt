import numpy as np
from scipy.stats import norm

def black_scholes_call(S, K, T, r, sigma):
    """
    计算欧式看涨期权的价格、delta 和 vega (无股息)
    
    参数:
        S: 标的资产现价
        K: 行权价
        T: 剩余期限（年）
        r: 无风险利率（年化，连续复利）
        sigma: 波动率（年化）
    
    返回:
        price: 期权理论价格
        delta: 期权 delta
        vega:  期权 vega (对波动率的偏导数)
    """
    # 计算 d1, d2
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    
    # 标准正态分布的累积分布函数和概率密度函数
    N_d1 = norm.cdf(d1)
    N_d2 = norm.cdf(d2)
    n_d1 = norm.pdf(d1)
    
    # 看涨期权价格
    call_price = S * N_d1 - K * np.exp(-r * T) * N_d2
    
    # delta
    call_delta = N_d1
    
    # vega (对波动率的偏导数) 注意：通常vega定义为价格对sigma的导数
    call_vega = S * np.sqrt(T) * n_d1
    
    return call_price, call_delta, call_vega

if __name__ == "__main__":
    # 已知参数
    S = 103.7
    K = 97.5
    sigma = 0.276
    r = 0.043
    T = 0.58

    # 计算
    price, delta, vega = black_scholes_call(S, K, T, r, sigma)

    # 存入结果字典
    result = {
        'call_price': price,
        'call_delta': delta,
        'call_vega': vega
    }

    # 输出结果（用于检查）
    print(result)
