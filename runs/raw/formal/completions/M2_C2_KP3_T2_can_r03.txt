```python
import numpy as np
import matplotlib.pyplot as plt

# ================== åæ°è®¾ç½® ==================
FACE_VALUE = 100
COUPON_RATE = 0.046        # ç¥¨æ¯ç 4.6%
MATURITY = 7               # å¹´æ 7 å¹´
Y0 = 0.053                 # å½åæ¶çç 5.3%

# æ¶ççåå¨å¹ï¼åºç¹ï¼ï¼å¯è°
SHIFT_BP = 100             # ä¸å 100 ä¸ªåºç¹
DY = SHIFT_BP / 10_000     # Îy = 0.01

# ================== å®ä»·å½æ°ï¼å¹´ä»æ¯ï¼ ==================
def bond_price(ytm, face=FACE_VALUE, coupon_rate=COUPON_RATE, maturity=MATURITY):
    """
    è®¡ç®åºå¸ä»·æ ¼ï¼å¹´ä»æ¯ï¼å¹´å¤å©ï¼
    æ¯ææ éææ°ç»æ¶çç
    """
    ytm = np.asarray(ytm, dtype=float)
    scalar_input = ytm.ndim == 0
    ytm = np.atleast_1d(ytm)
    t = np.arange(1, maturity + 1)
    # ç°éæµï¼å°æååä¸ºç¥¨æ¯ï¼å°æå¹´åº¦ä¸ºç¥¨æ¯+é¢å¼
    cf = np.full(maturity, coupon_rate * face)
    cf[-1] += face
    # æç°å å­ (1+y)^{-t}ï¼å½¢ç¶ (len(ytm), maturity)
    discount = (1 + ytm[:, np.newaxis]) ** -t
    price = np.sum(cf * discount, axis=1)
    if scalar_input:
        return price.item()
    return price

# ================== å½åæ¶ççä¸çä»·æ ¼ & ä¹æ ==================
P0 = bond_price(Y0)

# éº¦èå©ä¹æ

# ä¿®æ­£ä¹æ

# ================== ç»å¶ç²¾ç¡®æ²çº¿ & ä¹æè¿ä¼¼ ==================

         linewidth=2)

            label=f'Current yield ({Y0*100:.2f}%)')

            label=f'Up {SHIFT_BP}bp ({(Y0 + DY)*100:.2f}%)')

plt.ylabel('Bond Price')
plt.title('PriceâYield Curve: 7-Year, 4.6% Coupon')
```
