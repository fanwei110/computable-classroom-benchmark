```python
import numpy as np

# ================= 根据题目给定参数 =================
# 年化波动率
vols = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr = np.array([
    [1.00,  0.21, -0.13],
    [0.21,  1.00,  0.37],
    [-0.13, 0.37,  1.00]
])

# ================= 步骤1：构造协方差矩阵 =================
# Σ = diag(σ) @ C @ diag(σ) 或等价地使用外积 element-wise 乘相关系数
cov_matrix = np.outer(vols, vols) * corr

# ================= 步骤2：求全局最小方差组合权重 =================
# 允许卖空且满仓约束下，MVP闭式解为: w = Σ^(-1) * 1 / (1' * Σ^(-1) * 1)
ones = np.ones(3)
cov_inv = np.linalg.inv(cov_matrix)

# 计算未归一化的权重
w_unnorm = cov_inv @ ones

# 归一化使其和为1
w_mvp = w_unnorm / np.sum(w_unnorm)

# ================= 步骤3：计算组合年化波动率 =================
# 组合方差 = w' Σ w，组合波动率取其平方根
var_mvp = w_mvp.T @ cov_matrix @ w_mvp
vol_mvp = np.sqrt(var_mvp)

# ================= 步骤4：按要求填充结果 =================
result = {
    'mvp_weights': w_mvp.tolist(),
    'mvp_vol_annual': float(vol_mvp)
}

# 课堂投屏展示结果
if __name__ == "__main__":
    print("全局最小方差组合(MVP)计算结果：")
    print(f"权重: {result['mvp_weights']}")
    print(f"年化波动率: {result['mvp_vol_annual']:.4%}")
```
