```python
import numpy as np

def main():
    # 债券参数
    face_value = 100.0
    coupon_rate = 0.046
    ytm = 0.053
    maturity = 7
    dy = 0.008  # 收益率上升80个基点

    # 现金流发生的时间（年）
    t = np.arange(1, maturity + 1)
    # 各期现金流：最后一期包含本金
    cash_flows = np.full(maturity, face_value * coupon_rate)
    cash_flows[-1] += face_value

    # 各期现金流现值
    discount_factors = (1 + ytm) ** t
    present_values = cash_flows / discount_factors
    price = np.sum(present_values)

    # 麦考利久期
    macaulay_duration = np.sum(t * present_values) / price

    # 修正久期（年复利）
    modified_duration = macaulay_duration / (1 + ytm)

    # 根据一阶近似：dP/P = -D_mod * dy
    # 价格跌幅（正小数）
    price_drop_pct = modified_duration * dy

    # 输出结果
    result = {
        'price_drop_pct': price_drop_pct
    }
    print(result)

if __name__ == "__main__":
    main()
```
