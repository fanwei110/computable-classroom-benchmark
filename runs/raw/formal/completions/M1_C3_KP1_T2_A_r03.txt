\[
\mu_p = w \mu_1 + (1 - w) \mu_2

\[

\]

\[
w_{mvp} = \frac{\sigma_2^2 - \sigma_1 \sigma_2 \rho}{\sigma_1^2 + \sigma_2^2 - 2 \sigma_1 \sigma_2 \rho}
\]
