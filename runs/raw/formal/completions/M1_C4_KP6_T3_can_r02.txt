```python
import numpy as np
import pandas as pd

# 1. 读取数据
data_path = 'data/market_snapshot_v1.csv'
df = pd.read_csv(data_path)

# 提取 fund 列的日收益率
daily_returns = df['fund'].dropna().values  # 确保无缺失值

# 2. 计算日无风险利率
annual_risk_free_rate = 0.021
daily_risk_free_rate = annual_risk_free_rate / 252

# 计算超额日收益率
excess_daily_returns = daily_returns - daily_risk_free_rate

# 3. 计算样本标准差（ddof=1）
daily_volatility = np.std(excess_daily_returns, ddof=1)

# 年化夏普比率
annualized_sharpe = np.mean(excess_daily_returns) / daily_volatility * np.sqrt(252)

# 存储结果
result = {
    'sharpe_annual': annualized_sharpe
}
```
