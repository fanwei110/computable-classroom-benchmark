```python
import numpy as np

# ==================== 已知参数 ====================
beta = 1.42                     # 股票的 Beta 值
market_return_monthly = -0.058  # 上个月市场收益率 (-5.8%)
risk_free_rate_annual = 0.047   # 年化无风险利率 (4.7%)

# ==================== 步骤 1：把年无风险利率折算为月利率 ====================
# 假设：采用简单算术平均折算（即年利率除以12），这是金融学中计算月度CAPM最常用的标准做法，
# 尤其在对应月度市场收益和回归估计时，算术折算保证了内部一致性。
risk_free_rate_monthly = risk_free_rate_annual / 12

# ==================== 步骤 2：在月度层面套用 CAPM ====================
# CAPM 公式: E(R_i) = R_f + beta * (E(R_m) - R_f)
# 其中：
# E(R_i) 为股票预期收益率
# R_f 为无风险利率
# E(R_m) 为市场收益率
# (E(R_m) - R_f) 为市场风险溢价

market_risk_premium_monthly = market_return_monthly - risk_free_rate_monthly
capm_return_monthly = risk_free_rate_monthly + beta * market_risk_premium_monthly

# ==================== 步骤 3：填充 result ====================
result = {
    'capm_return_monthly': capm_return_monthly
}

# ==================== 控制台输出（便于投屏展示验证） ====================
print(f"月度无风险利率: {risk_free_rate_monthly:.6f} ({risk_free_rate_monthly*100:.4f}%)")
print(f"月度市场风险溢价: {market_risk_premium_monthly:.6f} ({market_risk_premium_monthly*100:.4f}%)")
print(f"按CAPM计算的该股票上月预期收益: {capm_return_monthly:.6f} ({capm_return_monthly*100:.4f}%)")
print("-" * 40)
print(f"result字典: {result}")
```
