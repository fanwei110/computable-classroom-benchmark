```python
import numpy as np

# 给定数据：年化波动率（小数）
sigma = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵（对称）
rho = np.array([
    [1.00, 0.21, -0.13],
    [0.21, 1.00, 0.37],
    [-0.13, 0.37, 1.00]
])

# 1. 构造协方差矩阵 Sigma: Σ_ij = σ_i * σ_j * ρ_ij
Sigma = np.outer(sigma, sigma) * rho

# 2. 计算全局最小方差组合权重（闭式解）
# 最小化 w' Σ w，满足 sum(w) = 1
# 闭式解: w = (Σ^{-1} 1) / (1' Σ^{-1} 1)
ones = np.ones(len(sigma))
Sigma_inv = np.linalg.inv(Sigma)
w_mvp = Sigma_inv @ ones / (ones @ Sigma_inv @ ones)

# 3. 组合年化波动率：sqrt(w' Σ w)
var_mvp = w_mvp @ Sigma @ w_mvp
vol_mvp = np.sqrt(var_mvp)

# 输出结果
result = {
    'mvp_weights': w_mvp.tolist(),
    'mvp_vol_annual': vol_mvp
}

print("全局最小方差组合 (允许卖空)")
print(f"权重: {result['mvp_weights']}")
print(f"年化波动率: {result['mvp_vol_annual']:.6f}")
```
