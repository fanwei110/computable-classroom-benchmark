```python
import numpy as np
from scipy.stats import norm

# ==================== 输入参数 ====================
position_value = 1_850_000    # 头寸价值（元）
annual_vol = 0.218            # 年化收益波动率
trading_days = 252            # 年交易日数
confidence_95 = 0.95          # 95% 置信水平
confidence_99 = 0.99          # 99% 置信水平
horizon_1d = 1                # 1 日
horizon_10d = 10              # 10 日

# ==================== 计算 ====================
# 1. 去年化：日波动率
daily_vol = annual_vol / np.sqrt(trading_days)

# 2. 单尾正态分位数（均值为零）
z_95 = norm.ppf(confidence_95)   # 95% 分位数
z_99 = norm.ppf(confidence_99)   # 99% 分位数

# 3. 一日 VaR（95%）
var_95_1d = position_value * daily_vol * z_95   # 正数表示损失

# 4. 十日 VaR（99%）：用 sqrt(10) 缩放
var_99_10d = position_value * daily_vol * np.sqrt(horizon_10d) * z_99

# ==================== 输出 ====================
result = {
    'var_95_1d': round(var_95_1d, 2),   # 保留两位小数（分）
    'var_99_10d': round(var_99_10d, 2)
}

print(result)
```
