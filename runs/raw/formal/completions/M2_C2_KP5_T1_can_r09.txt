```python
import numpy as np
from scipy.stats import norm

# 头寸与市场参数
position = 1_850_000         # 头寸价值（元）
annual_vol = 0.218           # 年化收益波动率
trading_days = 252           # 一年交易日数（学界/业界常用假设）

# 1. 年化波动率换算为一日波动率
daily_vol = annual_vol / np.sqrt(trading_days)

# 2. 置信水平对应的标准正态分位数
z_95 = norm.ppf(0.95)    # 95% 单尾分位数 ≈ 1.6449
z_99 = norm.ppf(0.99)    # 99% 单尾分位数 ≈ 2.3263

# 3. 计算 VaR （均假设零期望收益，VaR 以正数表示损失额）
var_95_1d = position * daily_vol * z_95

# 十日 VaR：波动率按 √时间 缩放
var_99_10d = position * daily_vol * np.sqrt(10) * z_99

# 输出契约
result = {
    'var_95_1d': var_95_1d,
    'var_99_10d': var_99_10d
}

# 课堂演示：打印结果（保留两位小数供参考）
print("95% 一日 VaR:  ¥{:,.2f}".format(result['var_95_1d']))
print("99% 十日 VaR: ¥{:,.2f}".format(result['var_99_10d']))
```
