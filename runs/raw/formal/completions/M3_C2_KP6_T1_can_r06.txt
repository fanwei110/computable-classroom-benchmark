```python
import pandas as pd
import numpy as np

# ==========================================
# 第一部分：读取数据与计算年化夏普比率
# ==========================================

# 1. 读取快照 CSV
df = pd.read_csv('data/market_snapshot_v1.csv')

# 提取 "fund" 列的日收益，并剔除可能存在的缺失值
daily_returns = df['fund'].dropna()

# 假设：一年按252个交易日计算，日无风险利率采用单利拆分
rf_annual = 0.021
trading_days = 252
rf_daily = rf_annual / trading_days

# 计算日超额收益
excess_daily_returns = daily_returns - rf_daily

# 计算超额收益均值与组合日收益标准差（采用样本标准差 ddof=1）
mean_excess_daily = np.mean(excess_daily_returns)
std_daily = np.std(daily_returns, ddof=1)

# 计算年化夏普比率
sharpe_annual = (mean_excess_daily / std_daily) * np.sqrt(trading_days)


# ==========================================
# 第二部分：业绩归因（BHB模型分解）
# ==========================================

# 组合权重与基准权重
wp = np.array([0.45, 0.35, 0.20])
wb = np.array([0.40, 0.40, 0.20])

# 组合行业收益与基准行业收益
rp = np.array([0.083, 0.021, -0.014])
rb = np.array([0.067, 0.034, -0.009])

# 计算配置效应: Σ (wp_i - wb_i) * rb_i
allocation_effect = np.sum((wp - wb) * rb)

# 计算选择效应: Σ wb_i * (rp_i - rb_i)
selection_effect = np.sum(wb * (rp - rb))

# 计算交互效应: Σ (wp_i - wb_i) * (rp_i - rb_i)
interaction_effect = np.sum((wp - wb) * (rp - rb))


# ==========================================
# 填充输出契约
# ==========================================

result = {
    'sharpe_annual': sharpe_annual,
    'allocation_effect': allocation_effect,
    'selection_effect': selection_effect,
    'interaction_effect': interaction_effect
}

# 课堂投屏展示结果
print(result)
```
