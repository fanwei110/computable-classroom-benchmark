```python
import numpy as np
from scipy.stats import norm
import pandas as pd
import matplotlib.pyplot as plt

# 给定参数
position_value = 1_850_000        # 头寸价值（元）
sigma_annual = 0.218              # 年化收益波动率（小数形式）
trading_days = 252                # 年交易日数

# 1. 去年化到日波动率
sigma_daily = sigma_annual / np.sqrt(trading_days)

# 2. 单尾正态分位数
z_95 = norm.ppf(0.95)   # 95% 置信水平对应的单尾分位数
z_99 = norm.ppf(0.99)   # 99% 置信水平对应的单尾分位数

# 3. 计算 VaR（正损失金额）
var_95_1d = position_value * sigma_daily * z_95

# 十日波动率按 sqrt(10) 缩放
sigma_10d = sigma_daily * np.sqrt(10)
var_99_10d = position_value * sigma_10d * z_99

# 输出契约：将所有结果存入 result 字典
result = {
    'var_95_1d': var_95_1d,
    'var_99_10d': var_99_10d
}

# 可选：用于课堂投屏时直接观察结果
if __name__ == '__main__':
    print("95% 一日 VaR (元):", result['var_95_1d'])
    print("99% 十日 VaR (元):", result['var_99_10d'])
```
