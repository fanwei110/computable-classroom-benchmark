```python
import numpy as np

# 年化波动率
vol = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr = np.array([
    [ 1.00,  0.21, -0.13],
    [ 0.21,  1.00,  0.37],
    [-0.13,  0.37,  1.00]
])

# 1. 协方差矩阵
Sigma = np.outer(vol, vol) * corr

# 2. 最小方差组合权重（允许卖空，满仓）
ones = np.ones(len(vol))
inv_Sigma = np.linalg.inv(Sigma)
w_mvp = inv_Sigma @ ones / (ones @ inv_Sigma @ ones)

# 3. 组合年化波动率
mvp_vol = np.sqrt(w_mvp @ Sigma @ w_mvp)

# 4. 整理结果
result = {
    'mvp_weights': w_mvp.tolist(),   # 以列表形式给出
    'mvp_vol_annual': float(mvp_vol)
}

print(result)
```
