```python
import numpy as np
from scipy.stats import norm

# ==========================================
# 欧式看涨期权 Black-Scholes 定价与希腊字母计算
# ==========================================

# 1. 参数设置 (均按小数表示)
S = 103.7      # 标的资产现价
K = 97.5       # 行权价
sigma = 0.276  # 隐含波动率 (27.6%)
r = 0.043      # 无风险利率 (4.3%)
T = 0.58       # 剩余期限 (年)

# 2. 计算 d1 与 d2
d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
d2 = d1 - sigma * np.sqrt(T)

# 3. 计算期权价格与 delta
N_d1 = norm.cdf(d1)
N_d2 = norm.cdf(d2)

call_price = S * N_d1 - K * np.exp(-r * T) * N_d2
call_delta = N_d1  # 不分红股票的看涨期权 delta = N(d1)

# 4. 计算 vega (dC/dσ，按每单位波动率报价)
n_d1 = norm.pdf(d1)
call_vega = S * np.sqrt(T) * n_d1  # 不分红股票的 vega = S * sqrt(T) * N'(d1)

# 5. 填充 result 字典
result = {
    'call_price': call_price,
    'call_delta': call_delta,
    'call_vega': call_vega
}

# 输出结果，便于课堂投屏展示
print("="=40)
print("Black-Scholes 欧式看涨期权计算结果")
print("="=40)
print(f"d1         = {d1:.6f}")
print(f"d2         = {d2:.6f}")
print(f"Call Price = {result['call_price']:.4f}")
print(f"Call Delta = {result['call_delta']:.6f}")
print(f"Call Vega  = {result['call_vega']:.4f}")
print("="=40)
print("Result Dictionary:")
print(result)
```
