```python
import numpy as np
import matplotlib.pyplot as plt

# ==================== 数据与参数设置 ====================
mu1, mu2 = 0.071, 0.124         # 期望年收益
sig1, sig2 = 0.163, 0.289       # 年化波动率
rhos = [0.15, 0.45, 0.75]       # 相关系数列表
target_mu = 0.10                # 目标期望收益 (10%)

# ==================== 初始化结果字典 ====================
result = {
    'mvp_vol_at_rho45': None,
    'frontier_vol_at_target': None,
    'figure_path': 'markowitz_frontier.png'
}

# ==================== 绘图准备 ====================
plt.figure(figsize=(10, 7))
# 设定不同相关系数对应的颜色，以便图例与MVP标记匹配
colors = ['blue', 'green', 'red']

# 允许卖空，满仓约束下 w1 + w2 = 1，扫描 w1 即可
w1_range = np.linspace(-0.5, 1.5, 1000)
w2_range = 1 - w1_range

# ==================== 主计算循环 ====================
for i, rho in enumerate(rhos):
    # 构造协方差矩阵元素
    var1 = sig1 ** 2
    var2 = sig2 ** 2
    cov12 = rho * sig1 * sig2
    
    # 1. 扫描权重计算前沿组合的收益与波动率
    port_ret = w1_range * mu1 + w2_range * mu2
    port_var = w1_range**2 * var1 + w2_range**2 * var2 + 2 * w1_range * w2_range * cov12
    port_vol = np.sqrt(port_var)
    
    # 画出均值-方差前沿 (横轴波动率，纵轴期望收益)
    plt.plot(port_vol, port_ret, label=f'ρ = {rho:.2f}', color=colors[i], linewidth=2)
    
    # 2. 计算并标出最小方差组合 (MVP)
    # 利用解析解求满仓约束下最小方差组合的权重
    w1_mvp = (var2 - cov12) / (var1 + var2 - 2 * cov12)
    w2_mvp = 1 - w1_mvp
    
    mvp_ret = w1_mvp * mu1 + w2_mvp * mu2
    mvp_var = w1_mvp**2 * var1 + w2_mvp**2 * var2 + 2 * w1_mvp * w2_mvp * cov12
    mvp_vol = np.sqrt(mvp_var)
    
    # 在图上标出MVP点
    plt.scatter(mvp_vol, mvp_ret, color=colors[i], s=60, zorder=5, edgecolor='black', label=f'MVP (ρ={rho:.2f})')
    
    # 3. 针对相关系数 0.45 的特定计算
    if np.isclose(rho, 0.45):
        # 记录 0.45 下的最小方差组合年化波动率
        result['mvp_vol_at_rho45'] = mvp_vol
        
        # 计算目标期望收益 10% 下的权重及可达到的最小年化波动率
        w1_target = (target_mu - mu2) / (mu1 - mu2)
        w2_target = 1 - w1_target
        
        target_var = w1_target**2 * var1 + w2_target**2 * var2 + 2 * w1_target * w2_target * cov12
        target_vol = np.sqrt(target_var)
        
        result['frontier_vol_at_target'] = target_vol

# ==================== 图形美化与保存 ====================
plt.xlabel('Volatility (σ)', fontsize=12)
plt.ylabel('Expected Return (μ)', fontsize=12)
plt.title('Mean-Variance Frontier for Two Risky Assets', fontsize=14)
plt.legend(fontsize=10)
plt.grid(True, linestyle='--', alpha=0.6)

# 保存图形
plt.savefig(result['figure_path'], dpi=300, bbox_inches='tight')
plt.close()

# ==================== 输出结果验证 ====================
print("计算结果：")
print(f"相关系数0.45时，最小方差组合的年化波动率: {result['mvp_vol_at_rho45']:.6f}")
print(f"相关系数0.45时，目标期望收益10%下的最小年化波动率: {result['frontier_vol_at_target']:.6f}")
print(f"图形已保存至: {result['figure_path']}")
```
