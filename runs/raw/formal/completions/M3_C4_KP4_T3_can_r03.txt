```python
import numpy as np
from scipy.stats import norm

def black_scholes_call_price(S, K, r, T, sigma):
    """
    计算欧式看涨期权的 Black-Scholes 价格
    S: 标的资产现价
    K: 行权价
    r: 无风险利率（连续复利）
    T: 剩余到期时间（年）
    sigma: 波动率
    """
    d1 = (np.log(S / K) + (r + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
    return price

# 已知参数
S = 103.7       # 标的资产现价
K = 97.5        # 行权价
r = 0.043       # 无风险利率 4.3%
T = 0.58        # 剩余到期时间 0.58 年
sigma_old = 0.276  # 原波动率 27.6%
sigma_new = 0.286  # 上升一个百分点后的波动率 28.6%

# 1. 分别在波动率 28.6% 与 27.6% 下精确重定价
price_old = black_scholes_call_price(S, K, r, T, sigma_old)
price_new = black_scholes_call_price(S, K, r, T, sigma_new)

# 2. 把精确价差存入 result
price_change = price_new - price_old

result = {'price_change': price_change}

# 为了在课堂投屏时直观展示，可打印结果
if __name__ == '__main__':
    print(f"波动率 27.6% 下的期权价格: {price_old:.4f}")
    print(f"波动率 28.6% 下的期权价格: {price_new:.4f}")
    print(f"期权价格变化量: {price_change:.4f}")
    print(f"result字典: {result}")
```
