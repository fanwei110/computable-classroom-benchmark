```python
import numpy as np
import matplotlib.pyplot as plt

# ==========================================
# 1. 参数化无风险利率与市场期望收益
# ==========================================
rf = 0.023  # 无风险利率 2.3%
rm = 0.094  # 市场期望收益 9.4%

# ==========================================
# 2. 核心计算
# ==========================================
# SML斜率即为市场风险溢价
sml_slope = rm - rf

# 计算 beta = 1.27 处的 CAPM 期望收益
beta_target = 1.27
er_at_beta_127 = rf + beta_target * sml_slope

# ==========================================
# 3. 定义股票数据
# ==========================================
stocks = {
    'X': {'beta': 0.62, 'er': 0.081},
    'Y': {'beta': 1.18, 'er': 0.131},
    'Z': {'beta': 1.51, 'er': 0.099}
}

# ==========================================
# 4. 绘制证券市场线 (SML) 与股票点
# ==========================================
# 设置绘图风格与中文字体支持
plt.rcParams['font.sans-serif'] = ['SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

fig, ax = plt.subplots(figsize=(10, 7))

# 生成 beta 数据
betas = np.linspace(0, 2, 200)
# 计算 SML 对应的期望收益
sml_ers = rf + betas * sml_slope

# 绘制 SML
ax.plot(betas, sml_ers, color='blue', linewidth=2, label='Security Market Line (SML)')

# 标出无风险利率与市场收益点
ax.scatter(0, rf, color='green', s=80, zorder=5, label=f'Risk-Free Rate (β=0, E(r)={rf:.1%})')
ax.scatter(1, rm, color='orange', s=80, zorder=5, label=f'Market Portfolio (β=1, E(r)={rm:.1%})')

# 绘制股票 X, Y, Z 并添加标注与 Alpha 演示
for name, data in stocks.items():
    beta_i = data['beta']
    er_i = data['er']
    sml_er_i = rf + beta_i * sml_slope
    
    # 绘制股票点
    ax.scatter(beta_i, er_i, color='red', s=100, zorder=6, marker='D')
    
    # 绘制 Alpha 偏离线 (从 SML 到实际收益的垂直距离)
    ax.plot([beta_i, beta_i], [sml_er_i, er_i], color='gray', linestyle='--', linewidth=1.5)
    
    # 添加文本标注 (微调位置防止重叠)
    # X 在线上方，Y 在线上方，Z 在线下方
    if er_i >= sml_er_i:
        xytext_offset = (10, 10)
    else:
        xytext_offset = (10, -15)
        
    ax.annotate(f"Stock {name}\nβ={beta_i}, E(r)={er_i:.1%}\n(Alpha={er_i - sml_er_i:.2%})",
                xy=(beta_i, er_i),
                xytext=xytext_offset,
                textcoords='offset points',
                fontsize=10,
                bbox=dict(boxstyle="round,pad=0.3", fc="white", ec="gray", alpha=0.9))

# 图表格式美化
ax.set_title('CAPM & Security Market Line (SML)', fontsize=16, fontweight='bold')
ax.set_xlabel('Beta (β)', fontsize=13)
ax.set_ylabel('Expected Return E(r)', fontsize=13)
ax.set_xlim(0, 2)
ax.set_ylim(0, 0.20)
ax.yaxis.set_major_formatter(plt.FuncFormatter(lambda y, _: '{:.1%}'.format(y)))
ax.grid(True, linestyle='--', alpha=0.7)
ax.legend(loc='upper left', fontsize=11)

# 保存图形
figure_path = 'sml_plot.png'
plt.savefig(figure_path, dpi=300, bbox_inches='tight')
plt.close()

# ==========================================
# 5. 填充结果字典
# ==========================================
result = {
    'sml_slope': sml_slope,
    'er_at_beta_127': er_at_beta_127,
    'figure_path': figure_path
}

# (可选) 打印验证结果
if __name__ == '__main__':
    print(f"SML 斜率 (市场风险溢价): {result['sml_slope']:.4f}")
    print(f"Beta=1.27 处的 CAPM 期望收益: {result['er_at_beta_127']:.4f}")
    print(f"图形已保存至: {result['figure_path']}")
```
