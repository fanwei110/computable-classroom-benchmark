```python
import numpy as np
from scipy.stats import norm

# 期权与市场参数
S = 103.7          # 标的现价
K = 97.5           # 行权价
T = 0.58           # 剩余年限
r = 0.043          # 连续复利无风险利率
q = 0.0            # 连续股息率（未提及，默认0）

# 两个隐含波动率场景
sigma_1 = 0.276    # 当前波动率 27.6%
sigma_2 = 0.286    # 上升一个百分点后的波动率 28.6%

def bsm_call_price(S, K, T, r, sigma, q=0.0):
    """Black-Scholes 欧式看涨期权定价"""
    if T <= 1e-12:
        return max(S - K, 0.0)
    d1 = (np.log(S / K) + (r - q + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    price = S * np.exp(-q * T) * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
    return price

# 计算两个波动率下的期权价格
price_1 = bsm_call_price(S, K, T, r, sigma_1)
price_2 = bsm_call_price(S, K, T, r, sigma_2)

# 精准价差
price_change = price_2 - price_1

# 按要求存入字典
result = {'price_change': price_change}

# 输出以供课堂投屏查看
print("期权价格变动（σ从27.6% → 28.6%）:")
print(f"价格(σ=27.6%) = {price_1:.6f}")
print(f"价格(σ=28.6%) = {price_2:.6f}")
print(f"精确价差        = {price_change:.6f}")
print(f"\nresult = {result}")
```
