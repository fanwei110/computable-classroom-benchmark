```python
import numpy as np

# 给定数据
vol_A = 0.184
vol_B = 0.297
w_A = 0.6
w_B = 0.4
rho_before = 0.3
rho_after = 0.8

# 组合波动率公式: σ_p = sqrt((w_A*σ_A)^2 + (w_B*σ_B)^2 + 2*w_A*w_B*σ_A*σ_B*ρ)
vol_before = np.sqrt((w_A * vol_A)**2 + (w_B * vol_B)**2 + 2 * w_A * w_B * vol_A * vol_B * rho_before)
vol_after  = np.sqrt((w_A * vol_A)**2 + (w_B * vol_B)**2 + 2 * w_A * w_B * vol_A * vol_B * rho_after)

result = {
    'vol_before_annual': vol_before,
    'vol_after_annual': vol_after
}

print(result)
```
