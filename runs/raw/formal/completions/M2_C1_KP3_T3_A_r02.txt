```python
import numpy as np

# 债券参数
face_value = 100
coupon_rate = 0.046
ytm_initial = 0.053
ytm_change = 0.008  # 80 bps
years = 7

# 计算初始价格
coupon = face_value * coupon_rate
cash_flows = np.array([coupon] * years)
cash_flows[-1] += face_value
discount_factors = (1 + ytm_initial) ** np.arange(1, years + 1)
price_initial = np.sum(cash_flows / discount_factors)

# 计算新收益率下的价格
ytm_new = ytm_initial + ytm_change
discount_factors_new = (1 + ytm_new) ** np.arange(1, years + 1)
price_new = np.sum(cash_flows / discount_factors_new)

# 计算价格跌幅百分比
price_drop_pct = (price_initial - price_new) / price_initial * 100

result = {'price_drop_pct': round(price_drop_pct, 2)}
print(result)
```
