```python
import numpy as np

# 债券参数
face_value = 100
coupon_rate = 0.046
yield_initial = 0.053
yield_change = 0.008  # 80个基点
maturity = 7

# 假设每年付息一次
cash_flows = np.full(maturity, coupon_rate * face_value)
cash_flows[-1] += face_value
times = np.arange(1, maturity + 1)

# 初始价格
disc_factors_initial = (1 + yield_initial) ** (-times)
price_initial = np.sum(cash_flows * disc_factors_initial)

# 新收益率下的价格
yield_new = yield_initial + yield_change
disc_factors_new = (1 + yield_new) ** (-times)
price_new = np.sum(cash_flows * disc_factors_new)

# 价格跌幅百分比（用小数表示）
price_drop_pct = (price_initial - price_new) / price_initial

result = {'price_drop_pct': price_drop_pct}
```
