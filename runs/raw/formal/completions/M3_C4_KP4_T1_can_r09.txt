```python
import numpy as np
from scipy.stats import norm

# =====================
# 1. 参数设置
# =====================
S = 103.7       # 标的资产现价
K = 97.5        # 行权价
sigma = 0.276   # 隐含波动率（年化，小数表示）
r = 0.043       # 无风险利率（连续复利，小数表示）
T = 0.58        # 剩余期限（年）
q = 0.0         # 股息率（不分红股票）

# =====================
# 2. 计算 d1 与 d2
# =====================
# Black-Scholes 公式下的 d1 与 d2（使用连续复利利率）
d1 = (np.log(S / K) + (r - q + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
d2 = d1 - sigma * np.sqrt(T)

# =====================
# 3. 计算期权价格与 delta
# =====================
# 欧式看涨期权价格 C
call_price = S * np.exp(-q * T) * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)

# Delta = N(d1) * exp(-q*T)，因 q=0，故 delta = N(d1)
call_delta = np.exp(-q * T) * norm.cdf(d1)

# =====================
# 4. 计算 vega
# =====================
# Vega = dC/dσ，按每单位波动率报价（即波动率变化1.0时，期权价格的变化量）
# Vega = S * exp(-q*T) * sqrt(T) * N'(d1)，其中 N'(d1) 为标准正态分布的概率密度函数
call_vega = S * np.exp(-q * T) * np.sqrt(T) * norm.pdf(d1)

# =====================
# 5. 封装结果
# =====================
result = {
    'call_price': call_price,
    'call_delta': call_delta,
    'call_vega': call_vega
}

# 输出结果（教师在投屏时可以直接观察此字典）
print(result)
```
