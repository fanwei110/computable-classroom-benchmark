```python
import numpy as np

# 给定的年化波动率（小数）
sigma = np.array([0.187, 0.243, 0.312])

# 给定相关系数矩阵（对称）
corr = np.array([
    [1.0,  0.21, -0.13],
    [0.21, 1.0,   0.37],
    [-0.13, 0.37,  1.0]
])

# 1. 构造协方差矩阵
# 协方差公式：cov(i,j) = rho(i,j) * sigma_i * sigma_j
cov_matrix = np.outer(sigma, sigma) * corr

# 2. 计算全局最小方差组合权重（允许卖空，权重和=1）
# 闭式解：w = (Σ^{-1} * 1) / (1^T Σ^{-1} 1)
inv_cov = np.linalg.inv(cov_matrix)
ones = np.ones(3)
w_mvp = inv_cov @ ones / (ones @ inv_cov @ ones)

# 3. 计算组合年化波动率
var_mvp = w_mvp @ cov_matrix @ w_mvp
vol_mvp = np.sqrt(var_mvp)

# 4. 按要求的键名存入结果字典
result = {
    'mvp_weights': w_mvp.tolist(),   # 转为列表便于查看
    'mvp_vol_annual': float(vol_mvp)
}

# 可选打印，方便检查
if __name__ == "__main__":
    print("全局最小方差组合权重：", result['mvp_weights'])
    print("年化波动率：", result['mvp_vol_annual'])
```
