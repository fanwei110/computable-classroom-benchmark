```python
import numpy as np
import matplotlib.pyplot as plt
import os

# ============================
# 参数设定
# ============================
r1 = 7.1      # 资产1期望年收益（%）
r2 = 12.4     # 资产2期望年收益（%）
s1 = 16.3     # 资产1年化波动率（%）
s2 = 28.9     # 资产2年化波动率（%）
rhos = [0.15, 0.45, 0.75]          # 需要考察的相关系数
target_return = 10.0               # 目标期望收益（%）

# ============================
# 工具函数：计算最小方差组合（允许卖空，满仓）
# ============================
def compute_mvp(r1, r2, s1, s2, rho):
    """返回 (w1, 组合收益, 组合波动率)"""
    denom = s1**2 + s2**2 - 2 * rho * s1 * s2
    w1 = (s2**2 - rho * s1 * s2) / denom
    w2 = 1.0 - w1
    ret = w1 * r1 + w2 * r2
    var = w1**2 * s1**2 + w2**2 * s2**2 + 2 * w1 * w2 * rho * s1 * s2
    vol = np.sqrt(var)
    return w1, ret, vol

# ============================
# 针对相关系数 0.45 的计算
# ============================
# 1) 最小方差组合的年化波动率
_, mvp_ret_45, mvp_vol_45 = compute_mvp(r1, r2, s1, s2, 0.45)

# 2) 目标期望收益 10% 的最小年化波动率
w_target = (target_return - r2) / (r1 - r2)   # 满仓约束下唯一确定权重
var_target = (w_target**2 * s1**2
              + (1 - w_target)**2 * s2**2
              + 2 * w_target * (1 - w_target) * 0.45 * s1 * s2)
vol_target = np.sqrt(var_target)

# ============================
# 结果汇总
# ============================
result = {
    'mvp_vol_at_rho45': mvp_vol_45,
    'frontier_vol_at_target': vol_target,
    'figure_path': 'mean_variance_frontier.png'
}

# ============================
# 画图：均值-方差前沿
# ============================
w_range = np.linspace(-3, 4, 800)   # 权重扫描范围，覆盖足够卖空/杠杆区域

fig, ax = plt.subplots(figsize=(10, 6))
for rho in rhos:
    s12 = rho * s1 * s2
    # 扫描所有满仓组合
    rets = w_range * r1 + (1 - w_range) * r2
    vars_ = (w_range**2 * s1**2
             + (1 - w_range)**2 * s2**2
             + 2 * w_range * (1 - w_range) * s12)
    vols = np.sqrt(vars_)
    # 按波动率升序排列，保证曲线连续
    idx = np.argsort(vols)
    line, = ax.plot(vols[idx], rets[idx], label=f'ρ = {rho}')
    # 标记最小方差组合
    _, mvp_ret, mvp_vol = compute_mvp(r1, r2, s1, s2, rho)
    ax.scatter(mvp_vol, mvp_ret, color=line.get_color(),
               s=80, zorder=5, marker='o', edgecolors='white')

# 图形装饰
ax.set_xlabel('Annualized Volatility (%)')
ax.set_ylabel('Expected Return (%)')
ax.set_title('Mean-Variance Frontier (Two Risky Assets)')
ax.legend()
ax.grid(True)
plt.tight_layout()

# 保存图形
plt.savefig(result['figure_path'], dpi=150)
plt.show()

# 输出结果字典（课堂投屏可查看）
print(result)
```
