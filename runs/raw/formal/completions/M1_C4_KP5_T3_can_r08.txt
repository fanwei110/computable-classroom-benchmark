```python
import numpy as np
from scipy.stats import norm

# 课程计算约定参数
TRADING_DAYS_PER_YEAR = 252
CONFIDENCE_LEVEL = 0.95

# 输入数据
annualized_volatility = 0.24          # 年化波动率 24%
position_size = 2_700_000             # 头寸 2,700,000 元

# 1. 去年化一日波动率
daily_volatility = annualized_volatility / np.sqrt(TRADING_DAYS_PER_YEAR)

# 2. 单尾 95% 正态分位数（均值取零）
z_score = norm.ppf(CONFIDENCE_LEVEL)

# 3. 计算一日 VaR（正的损失金额）
var_95_1d = position_size * z_score * daily_volatility

# 输出契约
result = {
    'var_95_1d': round(var_95_1d, 2)   # 保留两位小数，符合金额习惯
}
```
