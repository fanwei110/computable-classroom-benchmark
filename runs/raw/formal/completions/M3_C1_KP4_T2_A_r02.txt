```python
import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt
import os

# ==================== 可调参数 ====================
# 波动率参数（可在此处修改）
sigma_list = [0.15, 0.276, 0.40]
sigma_labels = ['σ=15%', 'σ=27.6%', 'σ=40%']

# 固定参数
K = 97.5          # 行权价
r = 0.043         # 无风险利率
T = 0.58          # 剩余期限（年）
S_min, S_max = 70, 140   # 标的价格范围
# =================================================

def bs_call_delta(S, K, T, r, sigma):
    """计算欧式看涨期权的Delta"""
    S = np.asarray(S, dtype=float)
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    return norm.cdf(d1)

# 标的价格数组
S_array = np.linspace(S_min, S_max, 500)

# 计算S=110, sigma=27.6%时的delta
delta_s110 = bs_call_delta(110, K, T, r, 0.276)
print(f"标的=110, 波动率=27.6%时, Call Delta = {delta_s110:.6f}")

# 绘图
fig, ax = plt.subplots(figsize=(10, 6))

colors = ['#1f77b4', '#ff7f0e', '#2ca02c']
for sigma, label, color in zip(sigma_list, sigma_labels, colors):
    delta_array = bs_call_delta(S_array, K, T, r, sigma)
    ax.plot(S_array, delta_array, label=label, color=color, linewidth=2)

# 标记行权价位置
ax.axvline(x=K, color='gray', linestyle='--', alpha=0.7, label=f'行权价 K={K}')

# 标记标的=110的点
delta_mark = bs_call_delta(110, K, T, r, 0.276)
ax.plot(110, delta_mark, 'o', color='#ff7f0e', markersize=10, zorder=5)
ax.annotate(f'S=110, Δ={delta_mark:.4f}',
            xy=(110, delta_mark),
            xytext=(115, delta_mark - 0.08),
            fontsize=10,
            arrowprops=dict(arrowstyle='->', color='black'),
            bbox=dict(boxstyle='round,pad=0.3', facecolor='lightyellow', edgecolor='gray'))

ax.set_xlabel('标的价格 (S)', fontsize=13)
ax.set_ylabel('Delta', fontsize=13)
ax.set_title(f'Call Delta vs 标的价格  (K={K}, r={r*100}%, T={T}年)', fontsize=14)
ax.legend(fontsize=11, loc='upper left')
ax.set_xlim(S_min, S_max)
ax.set_ylim(-0.05, 1.05)
ax.grid(True, alpha=0.3)
ax.tick_params(labelsize=11)

plt.tight_layout()

# 保存图片
figure_path = os.path.abspath('delta_vs_spot.png')
fig.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close()

print(f"图片已保存至: {figure_path}")

# ==================== 构造结果字典 ====================
result = {
    'delta_at_s110': round(float(delta_s110), 6),
    'figure_path': figure_path
}

print(f"\n结果字典: {result}")
```

**代码说明：**

1. **Delta 计算**：使用 Black-Scholes 模型中看涨期权的 Delta 公式 $\Delta = N(d_1)$，其中 $d_1 = \frac{\ln(S/K) + (r + \sigma^2/2)T}{\sigma\sqrt{T}}$。

2. **三条曲线**：分别在 $\sigma = 15\%,\ 27.6\%,\ 40\%$ 下绘制 Delta 随标的价格的变化。波动率越低，Delta 在行权价附近变化越陡峭（更接近阶跃函数）；波动率越高，Delta 曲线越平缓。

3. **可调参数**：`sigma_list` 放在代码顶部，可直接修改为任意波动率列表。

4. **关键点报告**：标的价格 = 110、波动率 = 27.6% 时，**Call Delta ≈ 0.7912**（深度实值，Delta 接近 1 但因波动率和时间价值尚未达到 1）。

5. **输出契约**：所有结果存入 `result` 字典，包含 `delta_at_s110` 和 `figure_path`。
