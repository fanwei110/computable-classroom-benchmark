import numpy as np

def main():
    # 给定年化波动率
    vol1, vol2, vol3 = 0.187, 0.243, 0.312
    vols = np.array([vol1, vol2, vol3])

    # 相关系数矩阵
    corr = np.array([
        [1.00,  0.21, -0.13],
        [0.21,  1.00,  0.37],
        [-0.13, 0.37,  1.00]
    ])

    # 1. 构造协方差矩阵 (年化)
    Sigma = np.outer(vols, vols) * corr
    # 也可以使用: Sigma = np.diag(vols) @ corr @ np.diag(vols)

    # 2. 全局最小方差组合 (允许卖空, 全额投资: sum(w)=1)
    ones = np.ones(3)
    inv_Sigma = np.linalg.inv(Sigma)

    # 闭式解: w = (Σ^{-1} * 1) / (1^T * Σ^{-1} * 1)
    w_mvp = inv_Sigma @ ones / (ones @ inv_Sigma @ ones)

    # 3. 组合年化波动率
    mvp_var = w_mvp @ Sigma @ w_mvp
    mvp_vol = np.sqrt(mvp_var)

    # 4. 按要求填充结果字典
    result = {
        'mvp_weights': w_mvp.tolist(),   # 列表, 顺序对应资产1,2,3
        'mvp_vol_annual': float(mvp_vol)
    }

    # 打印以便课堂投屏观察
    print("全局最小方差组合 (允许卖空, 全额投资):")
    print(f"权重: {result['mvp_weights']}")
    print(f"年化波动率: {result['mvp_vol_annual']:.4%}")

    return result

if __name__ == "__main__":
    result = main()
