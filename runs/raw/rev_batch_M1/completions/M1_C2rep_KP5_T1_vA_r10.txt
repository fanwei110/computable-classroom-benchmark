```python
import numpy as np
from scipy.stats import norm

# 输入参数
position_value = 1_850_000          # 头寸价值（元）
annual_volatility = 0.218           # 年化波动率（21.8%）

# 1. 年化波动率换算到一日波动率
# 假设一年有252个交易日
daily_volatility = annual_volatility / np.sqrt(252)

# 2. 正态分位数（Z-score）
z_95 = norm.ppf(0.95)               # 95%置信水平对应的Z值
z_99 = norm.ppf(0.99)               # 99%置信水平对应的Z值

# 3. 计算VaR
# (i) 95% 一日VaR
var_95_1d = position_value * daily_volatility * z_95

# (ii) 99% 十日VaR
# 十日波动率 = daily_volatility * sqrt(10)
ten_day_volatility = daily_volatility * np.sqrt(10)
var_99_10d = position_value * ten_day_volatility * z_99

# 4. 填充result字典
result = {
    'var_95_1d': round(var_95_1d, 2),    # 保留两位小数，单位：元
    'var_99_10d': round(var_99_10d, 2)   # 保留两位小数，单位：元
}

# 输出结果（可选，便于调试）
print(result)
```
