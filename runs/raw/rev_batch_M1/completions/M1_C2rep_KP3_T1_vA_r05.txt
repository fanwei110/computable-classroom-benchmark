以下是自包含的 Python 脚本，计算债券的价格、麦考利久期、修正久期和凸性：

```python
import numpy as np
from scipy.optimize import newton

# 债券参数
face_value = 100.0
coupon_rate = 0.046
years_to_maturity = 7
ytm = 0.053  # 到期收益率
coupon_payment = face_value * coupon_rate
periods_per_year = 1  # 假设每年付息一次

# 1. 计算债券价格（现金流贴现之和）
def bond_price(yield_to_maturity, face_value, coupon_payment, years_to_maturity, periods_per_year=1):
    total_periods = years_to_maturity * periods_per_year
    cash_flows = np.array([coupon_payment] * (total_periods - 1) + [coupon_payment + face_value])
    discount_factors = np.array([(1 + yield_to_maturity / periods_per_year) ** (-t) for t in range(1, total_periods + 1)])
    price = np.sum(cash_flows * discount_factors)
    return price

price = bond_price(ytm, face_value, coupon_payment, years_to_maturity, periods_per_year)

# 2. 计算麦考利久期和修正久期
def macaulay_duration(yield_to_maturity, face_value, coupon_payment, years_to_maturity, periods_per_year=1):
    total_periods = years_to_maturity * periods_per_year
    cash_flows = np.array([coupon_payment] * (total_periods - 1) + [coupon_payment + face_value])
    discount_factors = np.array([(1 + yield_to_maturity / periods_per_year) ** (-t) for t in range(1, total_periods + 1)])
    present_values = cash_flows * discount_factors
    weighted_times = np.array([t for t in range(1, total_periods + 1)]) / periods_per_year
    mac_dur = np.sum(weighted_times * present_values) / np.sum(present_values)
    return mac_dur

def modified_duration(macaulay_duration, yield_to_maturity, periods_per_year=1):
    mod_dur = macaulay_duration / (1 + yield_to_maturity / periods_per_year)
    return mod_dur

macaulay_dur = macaulay_duration(ytm, face_value, coupon_payment, years_to_maturity, periods_per_year)
modified_dur = modified_duration(macaulay_dur, ytm, periods_per_year)

# 3. 计算凸性
def convexity(yield_to_maturity, face_value, coupon_payment, years_to_maturity, periods_per_year=1):
    total_periods = years_to_maturity * periods_per_year
    cash_flows = np.array([coupon_payment] * (total_periods - 1) + [coupon_payment + face_value])
    discount_factors = np.array([(1 + yield_to_maturity / periods_per_year) ** (-t) for t in range(1, total_periods + 1)])
    present_values = cash_flows * discount_factors
    weighted_times = np.array([t * (t + 1) for t in range(1, total_periods + 1)]) / (periods_per_year ** 2)
    conv = np.sum(weighted_times * present_values) / (np.sum(present_values) * (1 + yield_to_maturity / periods_per_year) ** 2)
    return conv

conv = convexity(ytm, face_value, coupon_payment, years_to_maturity, periods_per_year)

# 4. 填充 result 字典
result = {
    'price': price,
    'macaulay_duration_years': macaulay_dur,
    'modified_duration_years': modified_dur,
    'convexity': conv
}

# 输出结果（可选，便于调试）
print(result)
```

### 输出说明：
- `price`: 债券的现值（现金流贴现之和）。
- `macaulay_duration_years`: 麦考利久期（单位：年）。
- `modified_duration_years`: 修正久期（单位：年）。
- `convexity`: 凸性（无单位，但通常表示为百分比变化的平方）。

### 假设：
1. 每年付息一次（`periods_per_year = 1`）。
2. 现金流包括每年的票息和到期时的本金。
3. 久期和凸性的计算基于离散复利（与 YTM 的定义一致）。

运行脚本后，`result` 字典将包含所有要求的输出值。
