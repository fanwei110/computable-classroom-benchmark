```python
import numpy as np
from scipy.optimize import newton

# 债券参数
face_value = 100.0          # 面值
coupon_rate = 0.046         # 票息率 4.6%
ytm_initial = 0.053         # 初始收益率 5.3%
years_to_maturity = 7       # 到期年限
coupon_frequency = 2        # 每年付息次数（半年付息）
yield_change_bps = 80       # 收益率变动（基点）

# 计算每期票息
coupon_payment = face_value * coupon_rate / coupon_frequency

# 定义债券定价函数：现金流贴现之和
def bond_price(ytm, face_value, coupon_payment, years_to_maturity, coupon_frequency):
    periods = years_to_maturity * coupon_frequency
    cash_flows = np.full(periods, coupon_payment)
    cash_flows[-1] += face_value  # 最后一期包含面值
    discount_factors = (1 + ytm / coupon_frequency) ** np.arange(1, periods + 1)
    return np.sum(cash_flows / discount_factors)

# 计算初始价格
price_initial = bond_price(ytm_initial, face_value, coupon_payment, years_to_maturity, coupon_frequency)

# 定义久期和凸性计算函数
def macaulay_duration_convexity(ytm, face_value, coupon_payment, years_to_maturity, coupon_frequency):
    periods = years_to_maturity * coupon_frequency
    cash_flows = np.full(periods, coupon_payment)
    cash_flows[-1] += face_value
    discount_factors = (1 + ytm / coupon_frequency) ** np.arange(1, periods + 1)
    pv_cash_flows = cash_flows / discount_factors
    weights = pv_cash_flows / np.sum(pv_cash_flows)
    time_periods = np.arange(1, periods + 1) / coupon_frequency
    duration = np.sum(weights * time_periods)
    convexity = np.sum(weights * time_periods ** 2) / (1 + ytm / coupon_frequency) ** 2
    return duration, convexity

# 计算久期和凸性
duration, convexity = macaulay_duration_convexity(ytm_initial, face_value, coupon_payment, years_to_maturity, coupon_frequency)
modified_duration = duration / (1 + ytm_initial / coupon_frequency)

# 估算价格变动（使用久期和凸性）
yield_change = yield_change_bps / 10000  # 基点转换为小数
price_drop_pct_approx = -modified_duration * yield_change * 100 + 0.5 * convexity * (yield_change ** 2) * 100

# 精确计算新价格
ytm_new = ytm_initial + yield_change
price_new = bond_price(ytm_new, face_value, coupon_payment, years_to_maturity, coupon_frequency)
price_drop_pct_exact = (price_new - price_initial) / price_initial * 100

# 结果存储
result = {
    'price_drop_pct': price_drop_pct_exact  # 使用精确计算的跌幅
}

print(result)
```
