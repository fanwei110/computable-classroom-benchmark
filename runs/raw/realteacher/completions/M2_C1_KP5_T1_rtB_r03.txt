```python
import math

# 给定参数
position = 1850000          # 头寸
annual_vol = 0.218          # 年化波动率
trading_days = 250          # 假设一年250个交易日

# 正态分布分位数
z_95 = 1.6448536269514722   # 95% 单尾
z_99 = 2.3263478740408408   # 99% 单尾

# 日波动率
daily_vol = annual_vol / math.sqrt(trading_days)

# 95% 一日 VaR
var_95_1d = position * daily_vol * z_95

# 99% 十日 VaR (使用年化波动率直接计算)
var_99_10d = position * annual_vol * math.sqrt(10 / trading_days) * z_99

# 存入字典
result = {
    'var_95_1d': round(var_95_1d, 2),
    'var_99_10d': round(var_99_10d, 2)
}
```
