```python
import math
from scipy.stats import norm

# 给定参数
position = 1_850_000         # 头寸价值
sigma_annual = 0.218         # 年化波动率
trading_days = 252           # 每年交易日

# 日波动率
sigma_daily = sigma_annual / math.sqrt(trading_days)

# Z值
z_95 = norm.ppf(0.95)      # 95%置信水平
z_99 = norm.ppf(0.99)      # 99%置信水平

# 一日VaR 95%
var_95_1d = position * z_95 * sigma_daily

# 十日波动率
sigma_10d = sigma_annual * math.sqrt(10 / trading_days)

# 十日VaR 99%
var_99_10d = position * z_99 * sigma_10d

# 存储结果，保留两位小数
result = {
    'var_95_1d': round(var_95_1d, 2),
    'var_99_10d': round(var_99_10d, 2)
}
result
```
