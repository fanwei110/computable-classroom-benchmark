```python
# 给定数据
Rf = 2.3  # 无风险利率，%
Rm = 9.4  # 市场收益率，%
market_premium = Rm - Rf  # 市场风险溢价

# 三只股票的beta
beta_x = 0.62
beta_y = 1.18
beta_z = 1.51

# Y股票的实际收益
actual_return_y = 13.1  # %

# 按CAPM计算期望收益
er_x = Rf + beta_x * market_premium
er_y = Rf + beta_y * market_premium
er_z = Rf + beta_z * market_premium

# 计算Y的alpha
alpha_y = actual_return_y - er_y

# 结果字典
result = {
    'er_x': er_x,
    'er_y': er_y,
    'er_z': er_z,
    'alpha_y': alpha_y
}
```
