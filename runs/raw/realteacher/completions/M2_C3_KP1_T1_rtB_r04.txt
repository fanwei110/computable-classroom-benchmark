```python
import numpy as np

# 给定数据
sigma = np.array([0.187, 0.243, 0.312])
rho = np.array([
    [1.0,  0.21, -0.13],
    [0.21, 1.0,   0.37],
    [-0.13,0.37,  1.0 ]
])

# 计算协方差矩阵
Sigma = np.outer(sigma, sigma) * rho

# 全1向量
ones = np.ones(3)

# 最小方差组合权重（允许卖空、全额投资）
Sigma_inv = np.linalg.inv(Sigma)
x = Sigma_inv @ ones
mvp_weights = x / np.sum(x)

# 组合年化波动率
mvp_var = mvp_weights @ Sigma @ mvp_weights
mvp_vol_annual = np.sqrt(mvp_var)

# 存储结果
result = {
    'mvp_weights': mvp_weights.tolist(),
    'mvp_vol_annual': round(float(mvp_vol_annual), 6)
}

print(result)
```
