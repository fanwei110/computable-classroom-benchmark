```python
import math

S = 103.7
K = 97.5
sigma = 0.276
r = 0.043
T = 0.58

# 计算 d1
d1 = (math.log(S / K) + (r + sigma**2 / 2) * T) / (sigma * math.sqrt(T))
# 正态分布密度函数 N'(d1)
N_prime_d1 = math.exp(-d1**2 / 2) / math.sqrt(2 * math.pi)
# Vega（波动率每变动1个单位的价格变化）
vega = S * math.sqrt(T) * N_prime_d1
# 隐含波动率上涨1个百分点（0.01）对应的价格变化
price_change = vega * 0.01

result = {'price_change': round(price_change, 4)}
```
