```python
import math

# 已知参数
annual_vol = 0.24
position = 2700000
z_95 = 1.645  # 95%置信水平对应的正态分布分位数
trading_days = 252  # 默认一年252个交易日

# 计算日波动率
daily_vol = annual_vol / math.sqrt(trading_days)

# 计算95%一日VaR
var_95_1d = position * z_95 * daily_vol

# 按照输出契约存入字典
result = {'var_95_1d': round(var_95_1d, 2)}
```
