```python
# 给定数据
rf = 0.023       # 无风险利率
rm = 0.094       # 市场期望收益
beta_x = 0.62
beta_y = 1.18
beta_z = 1.51
actual_y = 0.131 # Y的实际收益

# 市场风险溢价
mrp = rm - rf

# CAPM 期望收益
er_x = rf + beta_x * mrp
er_y = rf + beta_y * mrp
er_z = rf + beta_z * mrp

# Y的alpha
alpha_y = actual_y - er_y

# 存入字典
result = {
    'er_x': er_x,
    'er_y': er_y,
    'er_z': er_z,
    'alpha_y': alpha_y
}
result
```
