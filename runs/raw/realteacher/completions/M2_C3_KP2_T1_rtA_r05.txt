```python
# 给定数据
Rf = 0.023      # 无风险利率
Rm = 0.094      # 市场收益率
beta_x = 0.62
beta_y = 1.18
beta_z = 1.51
actual_y = 0.131  # Y的实际收益

# 市场风险溢价
market_premium = Rm - Rf

# 按CAPM计算期望收益
er_x = Rf + beta_x * market_premium
er_y = Rf + beta_y * market_premium
er_z = Rf + beta_z * market_premium

# Y的alpha
alpha_y = actual_y - er_y

# 存储结果
result = {
    'er_x': er_x,
    'er_y': er_y,
    'er_z': er_z,
    'alpha_y': alpha_y
}
```
