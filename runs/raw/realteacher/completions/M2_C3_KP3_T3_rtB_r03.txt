```python
import numpy as np

# 债券参数
face_value = 100
coupon_rate = 0.046
ytm = 0.053
n_years = 7
delta_y = 0.008  # 80 bp

# 假设年付息，与年复利报价一致
coupon = face_value * coupon_rate
times = np.arange(1, n_years + 1)
cash_flows = np.full(n_years, coupon)
cash_flows[-1] += face_value

# 当前价格
discount_factors = (1 + ytm) ** times
pv_cf = cash_flows / discount_factors
price0 = pv_cf.sum()

# 收益率上行后价格
ytm_new = ytm + delta_y
discount_factors_new = (1 + ytm_new) ** times
price1 = (cash_flows / discount_factors_new).sum()

# 价格跌幅百分比（正数表示损失）
price_drop_pct = (price0 - price1) / price0 * 100

result = {'price_drop_pct': round(price_drop_pct, 4)}
result
```
