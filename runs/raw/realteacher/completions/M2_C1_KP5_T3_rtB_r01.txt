```python
import math

# 给定参数
annual_vol = 0.24
position = 2_700_000  # 270万
confidence = 0.95
trading_days = 252  # 常用年交易日数

# 单尾95%对应的z值
z_score = 1.6448536269514722  # scipy.stats.norm.ppf(0.95)

# 日波动率
daily_vol = annual_vol / math.sqrt(trading_days)

# VaR计算
var_95_1d = position * daily_vol * z_score

result = {"var_95_1d": round(var_95_1d, 2)}
print(result)
```
