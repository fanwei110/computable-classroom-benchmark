```python
import numpy as np

# 输入数据
volatilities = np.array([0.187, 0.243, 0.312])
correlations = np.array([
    [1.0, 0.21, -0.13],
    [0.21, 1.0, 0.37],
    [-0.13, 0.37, 1.0]
])

# 构建协方差矩阵
cov_matrix = np.outer(volatilities, volatilities) * correlations

# 求解最小方差组合
# 公式: w = (Σ^-1 * 1) / (1^T * Σ^-1 * 1)
# 组合方差: σ^2 = 1 / (1^T * Σ^-1 * 1)
inv_cov_matrix = np.linalg.inv(cov_matrix)
ones = np.ones(3)

w_mvp = inv_cov_matrix @ ones / (ones.T @ inv_cov_matrix @ ones)
var_mvp = 1 / (ones.T @ inv_cov_matrix @ ones)
vol_mvp = np.sqrt(var_mvp)

# 输出契约组装
result = {
    'mvp_weights': [round(w, 4) for w in w_mvp.tolist()],  # 保留4位小数
    'mvp_vol_annual': round(vol_mvp, 4)                     # 保留4位小数（若需百分比形式则为14.65%）
}

# 为了直观展示，此处打印结果（实际提交仅保留字典即可）
print(result)
```
