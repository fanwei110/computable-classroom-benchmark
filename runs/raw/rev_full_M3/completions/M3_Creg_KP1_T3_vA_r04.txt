```python
import numpy as np

# 给定参数
vol_A = 0.184  # 资产 A 的年化波动率
vol_B = 0.297  # 资产 B 的年化波动率
weight_A = 0.6 # 资产 A 的权重
weight_B = 0.4 # 资产 B 的权重
corr_before = 0.3 # 变化前的相关系数
corr_after = 0.8  # 变化后的相关系数

# 计算变化前的协方差
cov_before = corr_before * vol_A * vol_B

# 计算变化后的协方差
cov_after = corr_after * vol_A * vol_B

# 构建变化前的协方差矩阵
cov_matrix_before = np.array([
    [vol_A**2, cov_before],
    [cov_before, vol_B**2]
])

# 构建变化后的协方差矩阵
cov_matrix_after = np.array([
    [vol_A**2, cov_after],
    [cov_after, vol_B**2]
])

# 权重向量
weights = np.array([weight_A, weight_B])

# 计算组合方差: w^T * Σ * w
var_before = weights.T @ cov_matrix_before @ weights
var_after = weights.T @ cov_matrix_after @ weights

# 计算组合年化波动率 (标准差)
vol_before_annual = np.sqrt(var_before)
vol_after_annual = np.sqrt(var_after)

# 将结果存入字典
result = {
    'vol_before_annual': vol_before_annual,
    'vol_after_annual': vol_after_annual
}

print(result)
```
