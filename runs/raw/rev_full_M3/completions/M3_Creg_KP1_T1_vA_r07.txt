```python
import numpy as np

# 1. 定义输入数据
# 年化波动率
vols = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr = np.array([
    [1.00, 0.21, -0.13],
    [0.21, 1.00,  0.37],
    [-0.13, 0.37,  1.00]
])

# 2. 计算协方差矩阵
# 公式: Cov = diag(vols) @ Corr @ diag(vols) = vols.T * Corr * vols (逐元素相乘)
cov_matrix = np.outer(vols, vols) * corr

# 3. 计算全局最小方差组合 (MVP) 的权重
# 构造全1向量
ones = np.ones(len(vols))

# 为了数值稳定性，使用求解线性方程组代替显式矩阵求逆
# 解方程 Cov @ x = 1
x = np.linalg.solve(cov_matrix, ones)

# 权重公式: w = x / (1^T @ x)
S = ones @ x
mvp_weights = x / S

# 4. 计算全局最小方差组合的年化波动率
# 组合方差: w^T @ Cov @ w
mvp_var = mvp_weights @ cov_matrix @ mvp_weights
mvp_vol_annual = np.sqrt(mvp_var)

# 5. 封装结果
result = {
    'mvp_weights': mvp_weights,
    'mvp_vol_annual': mvp_vol_annual
}

# 打印结果以便验证
if __name__ == "__main__":
    print("全局最小方差组合权重:", result['mvp_weights'])
    print("权重之和:", np.sum(result['mvp_weights']))
    print("全局最小方差组合年化波动率:", result['mvp_vol_annual'])
```
