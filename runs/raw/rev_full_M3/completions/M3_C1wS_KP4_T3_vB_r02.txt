```python
import numpy as np
from scipy.stats import norm

# ================= 1. 参数设定 =================
S = 103.7      # 标的资产价格
K = 97.5       # 行权价
sigma = 0.276  # 初始波动率 (27.6%)
r = 0.043      # 无风险利率 (4.3%)
T = 0.58       # 到期时间 (年)

# 波动率变化：IV涨1个点(1个百分点)，即绝对值变化 0.01
delta_sigma = 0.01

# ================= 2. 计算 d1 与 Vega =================
# Black-Scholes 模型中的 d1
d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))

# 希腊字母 Vega 的闭式解 (对看涨与看跌期权相同)
# Vega = S * N'(d1) * sqrt(T)
vega = S * norm.pdf(d1) * np.sqrt(T)

# ================= 3. 推算期权价格变化 =================
# 概念：利用希腊字母 Vega 推算波动率微小变动引起的价格响应（一阶线性近似）
price_change = vega * delta_sigma

# 【教学补充验证】使用 BS 闭式解计算精确的价格差
# 注：根据平价关系，看涨与看跌期权对波动率的敏感度及价格变化量完全一致
d2 = d1 - sigma * np.sqrt(T)
price_old = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)

sigma_new = sigma + delta_sigma
d1_new = (np.log(S / K) + (r + 0.5 * sigma_new**2) * T) / (sigma_new * np.sqrt(T))
d2_new = d1_new - sigma_new * np.sqrt(T)
price_new = S * norm.cdf(d1_new) - K * np.exp(-r * T) * norm.cdf(d2_new)
exact_price_change = price_new - price_old

# ================= 4. 输出结果 =================
# 按照输出契约存入字典
result = {'price_change': price_change}

# 课堂投屏打印
print(f"=== Black-Scholes 期权定价与希腊字母推算 ===")
print(f"标的: {S}, 行权: {K}, 波动率: {sigma*100:.1f}%, 利率: {r*100:.1f}%, 期限: {T}年")
print(f"计算所得 d1: {d1:.4f}")
print(f"计算所得 Vega: {vega:.4f}")
print(f"推算的价格变化 (Vega × 0.01): {price_change:.4f}")
print(f"[验证] 精确的 BS 价格变化: {exact_price_change:.4f}")
print("-" * 45)
print(f"result = {result}")
```
