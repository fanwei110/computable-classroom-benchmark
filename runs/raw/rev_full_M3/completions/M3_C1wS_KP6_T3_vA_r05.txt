```python
import pandas as pd
import numpy as np

# ==================== 假设说明 ====================
# 1. 数据频率：假设 snapshot.csv 中的 fund 列为日频净值数据。
# 2. 交易日历：假设一年有 252 个交易日，用于日度收益和年化因子的转换。
# 3. 收益率计算：采用简单收益率，与夏普比率标准定义保持一致。
# 4. 无风险利率折算：年化 2.1% 除以 252 转换为日度无风险利率。
# ==================================================

# 1. 读取快照 CSV
df = pd.read_csv('snapshot.csv')

# 提取基金净值序列并剔除缺失值
fund_nav = df['fund'].dropna()

# 计算日度简单收益率
daily_returns = fund_nav.pct_change().dropna()

# 2. 计算全样本年化夏普比率
rf_annual = 0.021
trading_days = 252
rf_daily = rf_annual / trading_days

# 计算日度超额收益
daily_excess_returns = daily_returns - rf_daily

# 计算日度超额收益的均值与标准差（使用样本标准差，ddof=1，pandas默认即为1）
mean_excess_daily = daily_excess_returns.mean()
std_excess_daily = daily_excess_returns.std()

# 计算日度夏普比率并年化（乘以 sqrt(252)）
daily_sharpe = mean_excess_daily / std_excess_daily
sharpe_annual = daily_sharpe * np.sqrt(trading_days)

# 3. 把结果存入 result
result = {
    'sharpe_annual': sharpe_annual
}

# 课堂投屏打印展示，便于直观确认
print(f"全样本年化夏普比率: {sharpe_annual:.4f}")
```
