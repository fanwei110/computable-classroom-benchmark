```python
import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt

def bs_call_delta(S, K, T, r, sigma):
    """
    计算欧式看涨期权的 Black-Scholes Delta。
    假设：无分红，欧式看涨期权。
    """
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    return norm.cdf(d1)

# ---------------- 参数设定 ----------------
K = 97.5          # 行权价
r = 0.043         # 无风险利率 4.3%
T = 0.58          # 剩余期限 0.58 年
S_min = 70.0      # 标的网格下限
S_max = 140.0     # 标的网格上限
n_points = 500    # 网格点数

# 波动率参数化
volatilities = [0.15, 0.276, 0.40]

# ---------------- 步骤1：计算 Delta ----------------
S_grid = np.linspace(S_min, S_max, n_points)
delta_results = {}

for sigma in volatilities:
    delta_results[sigma] = bs_call_delta(S_grid, K, T, r, sigma)

# ---------------- 步骤2：画图 ----------------
plt.figure(figsize=(10, 6))

for sigma in volatilities:
    plt.plot(S_grid, delta_results[sigma], 
             label=f'σ = {sigma*100:.1f}%', 
             linewidth=2)

plt.xlabel('Underlying Price (S)', fontsize=12)
plt.ylabel('Delta (Call)', fontsize=12)
plt.title('European Call Option Delta vs. Underlying Price', fontsize=14)
plt.axvline(x=K, color='gray', linestyle='--', linewidth=1, label=f'Strike K = {K}')
plt.legend(fontsize=11)
plt.grid(True, linestyle=':', alpha=0.7)

# 保存图形
figure_path = 'delta_vs_spot.png'
plt.savefig(figure_path, dpi=300, bbox_inches='tight')
plt.close()

# ---------------- 步骤3：报告特定点 Delta ----------------
S_target = 110.0
sigma_target = 0.276
delta_at_s110 = bs_call_delta(S_target, K, T, r, sigma_target)

print(f"标的价 = {S_target}, 波动率 = {sigma_target*100:.1f}% 时的 Delta = {delta_at_s110:.6f}")

# ---------------- 步骤4：填充 result 字典 ----------------
result = {
    'delta_at_s110': delta_at_s110,
    'figure_path': figure_path
}
```
