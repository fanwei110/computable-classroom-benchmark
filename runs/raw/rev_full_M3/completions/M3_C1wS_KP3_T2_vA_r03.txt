```python
"""
《证券投资学》课堂演示：债券定价与久期近似
债券：面值100、票息4.6%（每年付一次）、7年、当前收益率5.3%
"""
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# ----------------------------------------------------------------------------
# 1. 债券基本参数与计算函数
# ----------------------------------------------------------------------------
FACE = 100.0
COUPON_RATE = 0.046
MATURITY = 7          # 年
Y0 = 0.053            # 基准收益率
COUPON = FACE * COUPON_RATE   # 每年票息 = 4.6

# 未指明处采用的假设：按年付息，复利贴现（市场惯例）

def bond_price(y, face=FACE, c=COUPON, n=MATURITY):
    """给定(一组)收益率，返回债券精确价格。现金流贴现求和。"""
    y = np.atleast_1d(np.asarray(y, dtype=float))
    t = np.arange(1, n + 1)                     # 1..n
    cf = np.full(n, c, dtype=float)
    cf[-1] += face                              # 最后一期本金
    disc = (1.0 + y)[:, None] ** t[None, :]     # (m, n)
    pv = cf[None, :] / disc
    return pv.sum(axis=1)

def bond_duration(y, face=FACE, c=COUPON, n=MATURITY):
    """返回 (price, macaulay_duration, modified_duration)。"""
    y = float(y)
    t = np.arange(1, n + 1)
    cf = np.full(n, c, dtype=float)
    cf[-1] += face
    pv = cf / (1.0 + y) ** t
    price = pv.sum()
    mac_dur = (t * pv).sum() / price
    mod_dur = mac_dur / (1.0 + y)
    return price, mac_dur, mod_dur

# ----------------------------------------------------------------------------
# 2. 精确价格曲线（收益率 2% ~ 9%）
# ----------------------------------------------------------------------------
y_grid = np.linspace(0.02, 0.09, 300)
P_exact = bond_price(y_grid)

# 基准点信息
P0, D_mac, D_mod = bond_duration(Y0)

# ----------------------------------------------------------------------------
# 3. 久期近似曲线（在 5.3% 附近，窗口可调）
#    ΔP/P ≈ -D_mod * Δy   =>   P(y) ≈ P0 * (1 - D_mod * (y - Y0))
# ----------------------------------------------------------------------------
DELTA_Y_WINDOW = 0.02          # 可调：在基准收益率 ±200bp 窗口内画近似曲线
y_approx = np.linspace(Y0 - DELTA_Y_WINDOW, Y0 + DELTA_Y_WINDOW, 200)
P_approx = P0 * (1.0 - D_mod * (y_approx - Y0))

# ----------------------------------------------------------------------------
# 4. +100bp 报告：精确价格 与 久期法估计的相对价格变化
# ----------------------------------------------------------------------------
DY = 0.01                      # +100 基点
Y_UP = Y0 + DY
P_up_exact = float(bond_price(Y_UP)[0])
dur_rel_change_up100 = float(-D_mod * DY)   # 相对价格变化 ΔP/P

# ----------------------------------------------------------------------------
# 5. 作图
# ----------------------------------------------------------------------------
fig, ax = plt.subplots(figsize=(9.5, 6))

ax.plot(y_grid * 100, P_exact, color='#1f4e79', lw=2.2,
        label='Exact price  $P(y)=\\sum_t CF_t/(1+y)^t$')
ax.plot(y_approx * 100, P_approx, color='#c0392b', lw=2.0, ls='--',
        label=f'Duration approx  $P\\approx P_0(1-D_{{mod}}\\Delta y)$  (±{DELTA_Y_WINDOW*100:.0f}bp)')

# 标注基准点与 +100bp 点
ax.scatter([Y0*100], [P0], color='black', zorder=5)
ax.annotate(f'Base\ny={Y0*100:.1f}%, P={P0:.4f}',
            xy=(Y0*100, P0), xytext=(Y0*100-1.2, P0+3.5),
            fontsize=9, arrowprops=dict(arrowstyle='->', lw=0.8))

ax.scatter([Y_UP*100], [P_up_exact], color='green', zorder=5)
ax.annotate(f'+100bp\ny={Y_UP*100:.2f}%, P={P_up_exact:.4f}',
            xy=(Y_UP*100, P_up_exact), xytext=(Y_UP*100+0.3, P_up_exact+3.0),
            fontsize=9, arrowprops=dict(arrowstyle='->', lw=0.8))

# 久期法在 +100bp 处的估计点
P_up_dur = P0 * (1.0 + dur_rel_change_up100)
ax.scatter([Y_UP*100], [P_up_dur], marker='x', color='#c0392b', s=90, zorder=5,
           label=f'Duration-est @+100bp: P≈{P_up_dur:.4f}')

ax.set_xlabel('Yield y (%)', fontsize=11)
ax.set_ylabel('Bond price', fontsize=11)
ax.set_title('Bond Price vs Yield: Exact Curve and Duration Approximation\n'
             f'(FV={FACE}, coupon={COUPON_RATE*100}%, {MATURITY}y, base y={Y0*100:.1f}%)',
             fontsize=11)
ax.grid(True, alpha=0.3)
ax.legend(loc='upper right', fontsize=9)
fig.tight_layout()

fig_path = 'bond_price_duration.png'
fig.savefig(fig_path, dpi=120)

# ----------------------------------------------------------------------------
# 6. 输出契约
# ----------------------------------------------------------------------------
result = {
    'price_at_up100bp': P_up_exact,                  # 精确价格：y=6.3%
    'dur_approx_change_up100bp': dur_rel_change_up100,  # 久期法估计的相对价格变化
    'figure_path': fig_path,
}

# 控制台简要报告（便于课堂投屏核对）
print('='*64)
print('债券参数:', f'FV={FACE}, coupon={COUPON_RATE*100}%, n={MATURITY}, y0={Y0*100:.1f}%')
print(f'基准价格 P0              = {P0:.6f}')
print(f'Macaulay duration        = {D_mac:.6f} 年')
print(f'Modified duration        = {D_mod:.6f}')
print('-'*64)
print(f'精确价格 @ y=6.30% (+100bp)   = {P_up_exact:.6f}')
print(f'久期法估计的相对价格变化       = {dur_rel_change_up100*100:.4f}%')
print(f'  (即 P_up 估计 = P0 * (1 + 该值) = {P0*(1+dur_rel_change_up100):.6f})')
print('-'*64)
print(f'图保存至: {fig_path}')
print('='*64)
print('result =', result)
```

### 关键假设与说明
- **付息频率**：题面未指明，按**年付息**处理；复利贴现。
- **久期类型**：用修正久期 `D_mod = D_mac / (1+y)`，对应一阶相对价格敏感度 `ΔP/P ≈ -D_mod · Δy`。
- **近似窗口**：`DELTA_Y_WINDOW = 0.02`（±200bp），可在脚本顶部直接调节，方便课堂演示线性近似"近端准、远端偏"的特征。
- **复现性**：所有数值由解析公式给出，无随机成分，无网络依赖。

### 期望输出（数值参考）
- 基准价格 P0 ≈ **96.0065**
- 修正久期 D_mod ≈ **5.8577**
- 精确价格 @ y=6.30%（+100bp）≈ **90.4674**
- 久期法估计的相对价格变化 ≈ **-5.8577%**（对应估计价 ≈ 90.3967）

可以看到久期法在小幅变动时误差很小（约 7 个 bp 量级），但随 |Δy| 增大，因忽略凸性而偏离精确曲线——这正是图中虚线与实线开口所要传达的概念。
