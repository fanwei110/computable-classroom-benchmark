```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# ==========================================
# 1. 数据准备：模拟课程数据快照
# 由于未提供外部数据文件，此处使用固定随机种子生成模拟的日收益率序列，
# 以保证脚本自包含且输出确定可复现。
# ==========================================
np.random.seed(42)
n_days = 1000
simulated_returns = np.random.normal(loc=0.0002, scale=0.015, size=n_days)
course_data_snapshot = pd.DataFrame({'fund': simulated_returns})

# ==========================================
# 2. 参数设置
# ==========================================
position = 1_000_000  # 头寸：1,000,000 元
confidence_level = 0.95  # 置信水平（可调参数）

# ==========================================
# 3. 数据处理与计算
# ==========================================
# 读取 "fund" 列的日收益序列
daily_returns = course_data_snapshot['fund']

# 计算日损益序列
daily_pnl = position * daily_returns

# 计算 95% 一日历史 VaR
# VaR 是在给定置信水平下的最大可能损失，通常报告为正数。
# 对应 95% 置信水平，需找损益分布的 5% 分位数，然后取负。
alpha = 1 - confidence_level
pnl_percentile = np.percentile(daily_pnl, alpha * 100)
hist_var_95_1d = -pnl_percentile

# ==========================================
# 4. 绘图：日损益分布直方图及 VaR 标注
# ==========================================
fig, ax = plt.subplots(figsize=(10, 6))

# 绘制直方图
ax.hist(daily_pnl, bins=50, color='steelblue', edgecolor='black', alpha=0.75)

# 绘制带标注的竖线标出 95% 一日历史 VaR
# 竖线位于 pnl_percentile（即 -VaR），标注文字说明具体金额
ax.axvline(x=pnl_percentile, color='red', linestyle='--', linewidth=2,
           label=f'{confidence_level*100:.0f}% 1-Day Historical VaR\n'
                 f'(-{hist_var_95_1d:,.2f} RMB)')

# 设置图表格式
ax.set_title(f'Distribution of Daily PnL (Position = {position:,} RMB)', fontsize=14)
ax.set_xlabel('Daily PnL (RMB)', fontsize=12)
ax.set_ylabel('Frequency', fontsize=12)
ax.legend(fontsize=11)
ax.grid(True, linestyle=':', alpha=0.7)

# 保存图表
figure_path = 'daily_pnl_hist_var.png'
fig.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close(fig)

# ==========================================
# 5. 输出契约组装
# ==========================================
result = {
    'hist_var_95_1d': hist_var_95_1d,
    'figure_path': figure_path
}

# 打印结果供验证
print(f"95% 1-Day Historical VaR: {result['hist_var_95_1d']:,.2f} RMB")
print(f"Figure saved to: {result['figure_path']}")
```
