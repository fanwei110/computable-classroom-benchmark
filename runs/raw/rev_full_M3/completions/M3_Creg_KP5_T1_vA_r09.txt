```python
import numpy as np
from scipy.stats import norm

# --- 输入参数 ---
portfolio_value = 1_850_000       # 头寸价值（元）
annual_volatility = 0.218         # 年化收益波动率
trading_days_per_year = 252       # 一年的交易天数（计算日波动率的标准假设）

# --- 计算波动率 ---
# 使用平方根法则将年化波动率转换为特定期限的波动率
daily_volatility = annual_volatility / np.sqrt(trading_days_per_year)
ten_day_volatility = annual_volatility * np.sqrt(10 / trading_days_per_year)

# --- 计算分位数 (Z-score) ---
# 95% 置信水平对应的标准正态分布下分位数
z_95 = norm.ppf(0.95)
# 99% 置信水平对应的标准正态分布下分位数
z_99 = norm.ppf(0.99)

# --- 计算 VaR (Delta-Normal 模型) ---
# VaR = 头寸价值 × Z-score × 期限波动率
var_95_1d = portfolio_value * z_95 * daily_volatility
var_99_10d = portfolio_value * z_99 * ten_day_volatility

# --- 输出结果 ---
result = {
    'var_95_1d': var_95_1d,
    'var_99_10d': var_99_10d
}

# 打印结果以供验证
print(f"95% 一日 VaR: {result['var_95_1d']:.2f} 元")
print(f"99% 十日 VaR: {result['var_99_10d']:.2f} 元")
```
