```python
import numpy as np

# 资产参数
vol_A = 0.184  # 资产 A 的年化波动率
vol_B = 0.297  # 资产 B 的年化波动率

# 组合权重
w_A = 0.6
w_B = 0.4

# 相关系数变化
rho_before = 0.3
rho_after = 0.8

# 组合方差公式：Var_p = w_A^2 * vol_A^2 + w_B^2 * vol_B^2 + 2 * w_A * w_B * vol_A * vol_B * rho
# 计算不依赖于相关系数的常数部分
var_const = w_A**2 * vol_A**2 + w_B**2 * vol_B**2

# 计算依赖于相关系数的协方差部分系数
cov_coeff = 2 * w_A * w_B * vol_A * vol_B

# 计算相关系数变化前后的组合年化波动率
vol_before_annual = np.sqrt(var_const + cov_coeff * rho_before)
vol_after_annual = np.sqrt(var_const + cov_coeff * rho_after)

# 按照输出契约存入字典
result = {
    'vol_before_annual': vol_before_annual,
    'vol_after_annual': vol_after_annual
}

# 打印结果以供查看
print(f"相关性从 0.3 升至 0.8 时，60/40 组合的年化波动率变化：")
print(f"变化前: {result['vol_before_annual']:.4f} ({result['vol_before_annual']*100:.2f}%)")
print(f"变化后: {result['vol_after_annual']:.4f} ({result['vol_after_annual']*100:.2f}%)")
print(f"波动率增加: {(result['vol_after_annual'] - result['vol_before_annual'])*100:.2f}%")
```
