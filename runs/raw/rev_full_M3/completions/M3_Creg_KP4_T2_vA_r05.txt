```python
import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt

# ==================== 参数设定 ====================
strike_price = 97.5          # 行权价 K
risk_free_rate = 0.043       # 无风险利率 r
time_to_maturity = 0.58      # 剩余期限 T (年)
volatilities = [0.15, 0.276, 0.40]  # 波动率可调参数列表

# 标的价格范围
S_min, S_max = 70, 140
S_array = np.linspace(S_min, S_max, 700)

# ==================== 核心计算函数 ====================
def calculate_european_call_delta(S, K, r, T, sigma):
    """
    计算不分红欧式看涨期权的 Delta
    公式: Delta = N(d1)
    其中: d1 = [ln(S/K) + (r + 0.5 * sigma^2) * T] / (sigma * sqrt(T))
    """
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    delta = norm.cdf(d1)
    return delta

# ==================== 计算特定值 ====================
# 报告标的 110、波动率 27.6% 时的 delta
specific_S = 110
specific_sigma = 0.276
delta_at_s110 = calculate_european_call_delta(
    S=specific_S, 
    K=strike_price, 
    r=risk_free_rate, 
    T=time_to_maturity, 
    sigma=specific_sigma
)

# ==================== 绘图 ====================
plt.figure(figsize=(10, 6))

# 遍历波动率，计算 Delta 并画线
for sigma in volatilities:
    delta_array = calculate_european_call_delta(
        S=S_array, 
        K=strike_price, 
        r=risk_free_rate, 
        T=time_to_maturity, 
        sigma=sigma
    )
    plt.plot(S_array, delta_array, label=f'Volatility = {sigma*100:.1f}%')

# 图表格式设置
plt.title('European Call Option Delta vs Underlying Price', fontsize=14)
plt.xlabel('Underlying Price (S)', fontsize=12)
plt.ylabel('Delta', fontsize=12)
plt.legend(fontsize=11)
plt.grid(True, linestyle='--', alpha=0.7)

# 保存图片
figure_path = 'delta_curve.png'
plt.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close()

# ==================== 输出契约 ====================
result = {
    'delta_at_s110': delta_at_s110,
    'figure_path': figure_path
}

# 打印结果以便验证
print(f"Delta at S=110, sigma=27.6%: {result['delta_at_s110']:.6f}")
print(f"Figure saved to: {result['figure_path']}")
```
