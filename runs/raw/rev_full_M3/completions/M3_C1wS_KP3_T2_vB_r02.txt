```python
import numpy as np
import matplotlib.pyplot as plt

# ==========================================
# 参数设置 (变动幅度在此可调)
# ==========================================
F = 100                  # 面值
c_rate = 0.046           # 票息率 4.6%
C = F * c_rate           # 每期票息
T = 7                    # 期限 7 年
y0 = 0.053               # 初始到期收益率 5.3%
delta_y = 0.01           # 收益率变动幅度 100bp (0.01)，可在此处调整

# ==========================================
# 核心计算函数 (向量化)
# ==========================================
# 时间数组
t_arr = np.arange(1, T + 1)

# 现金流数组
cf_arr = np.full(T, C)
cf_arr[-1] += F

# 定义精确价格计算函数
def calc_exact_price(y, cf_arr, t_arr):
    """计算给定收益率下的债券精确价格"""
    return np.sum(cf_arr / (1 + y)**t_arr)

# 定义麦考利久期计算函数
def calc_mac_duration(y, cf_arr, t_arr):
    """计算麦考利久期"""
    P = calc_exact_price(y, cf_arr, t_arr)
    return np.sum(t_arr * cf_arr / (1 + y)**t_arr) / P

# ==========================================
# 1. 2%到9%收益率网格上的精确价格
# ==========================================
y_grid = np.linspace(0.02, 0.09, 200)
P_exact_grid = np.array([calc_exact_price(y, cf_arr, t_arr) for y in y_grid])

# ==========================================
# 2. 5.3%附近基于久期的近似
# ==========================================
# 计算 y0 处的精确价格与久期
P0 = calc_exact_price(y0, cf_arr, t_arr)
MacD0 = calc_mac_duration(y0, cf_arr, t_arr)
ModD0 = MacD0 / (1 + y0)  # 修正久期

# 基于久期的线性近似： P_approx = P0 * (1 - ModD0 * Δy)
P_approx_grid = P0 * (1 - ModD0 * (y_grid - y0))

# ==========================================
# 3. +100bp的精确价格与久期法估计的相对变化
# ==========================================
y_up = y0 + delta_y
price_at_up100bp = calc_exact_price(y_up, cf_arr, t_arr)
dur_approx_change_up100bp = -ModD0 * delta_y

# ==========================================
# 4. 绘图与保存
# ==========================================
plt.rcParams['font.sans-serif'] = ['Arial', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

fig, ax = plt.subplots(figsize=(10, 6.5))

# 绘制精确价格曲线
line1, = ax.plot(y_grid * 100, P_exact_grid, color='blue', lw=2.5, label='Exact Price-Yield Curve')

# 绘制久期近似曲线
line2, = ax.plot(y_grid * 100, P_approx_grid, color='red', ls='--', lw=2, label='Duration Approximation (Tangent at 5.3%)')

# 标记初始点
ax.scatter([y0 * 100], [P0], color='black', s=60, zorder=5, label=f'Current State (y={y0*100:.1f}%, P={P0:.2f})')

# 标记收益率上升100bp的点
ax.scatter([y_up * 100], [price_at_up100bp], color='darkgreen', s=60, zorder=5, label=f'Up 100bp Exact (y={y_up*100:.1f}%, P={price_at_up100bp:.2f})')

# 绘制变动幅度双向箭头 (可调变动幅度的视觉体现)
ax.annotate('', xy=(y_up * 100, P0), xytext=(y0 * 100, P0),
            arrowprops=dict(arrowstyle='<->', color='green', lw=2))
ax.text((y0 * 100 + y_up * 100) / 2, P0 + 0.5, f'Δy = {delta_y*100:.0f}bp', color='green', ha='center', fontsize=11, fontweight='bold')

# 添加久期法估计的虚线
approx_price_up100bp = P0 * (1 + dur_approx_change_up100bp)
ax.scatter([y_up * 100], [approx_price_up100bp], color='orange', s=60, zorder=5, marker='x')
ax.plot([y_up * 100, y_up * 100], [price_at_up100bp, approx_price_up100bp], color='gray', ls=':', lw=1.5, label='Approx Error at +100bp')

# 图例分开 (分成两个图例框分别置于左上和右侧)
legend1 = ax.legend(handles=[line1], loc='upper left', frameon=True, framealpha=0.9)
ax.add_artist(legend1) # 单独添加第一个图例
legend2 = ax.legend(handles=[line2, line1.lines[0]], loc='center right', frameon=True, framealpha=0.9) # 逻辑补充
# 更彻底的图例分开：精确线和近似线分开
legend1 = ax.legend(handles=[line1], loc='upper left', title='Exact Curve', frameon=True, framealpha=0.9)
ax.add_artist(legend1)
legend2 = ax.legend(handles=[line2], loc='center right', title='Approximation', frameon=True, framealpha=0.9)

# 坐标轴与标题
ax.set_xlabel('Yield to Maturity (YTM, %)', fontsize=12)
ax.set_ylabel('Bond Price', fontsize=12)
ax.set_title('Bond Pricing: Exact Price vs. Duration Approximation', fontsize=14, fontweight='bold')
ax.grid(True, ls=':', alpha=0.6)

# 保存图形
figure_path = 'bond_price_yield_duration.png'
fig.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close()

# ==========================================
# 输出契约组装
# ==========================================
result = {
    'price_at_up100bp': price_at_up100bp,
    'dur_approx_change_up100bp': dur_approx_change_up100bp,
    'figure_path': figure_path
}

# 打印结果供课堂投屏检查
print(f"--- 债券定价与久期分析结果 ---")
print(f"初始 YTM: {y0*100:.1f}%  |  初始精确价格: {P0:.4f}")
print(f"修正久期: {ModD0:.4f}  |  麦考利久期: {MacD0:.4f}")
print(f"收益率 +100bp 后精确价格: {result['price_at_up100bp']:.4f}")
print(f"久期法估计的相对变化: {result['dur_approx_change_up100bp']:.4f} (即 {result['dur_approx_change_up100bp']*100:.2f}%)")
print(f"图形已保存至: {result['figure_path']}")
```
