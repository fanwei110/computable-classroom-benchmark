```python
import numpy as np

# 定义已知参数
beta = 1.42                   # 股票的 Beta 值
market_return_monthly = -0.058  # 上个月市场收益率（跌了 5.8%）
risk_free_annual = 0.047      # 无风险利率（年化 4.7%）

# 将年化无风险利率转换为月度无风险利率
# 按照标准金融计算惯例（APR），将年化利率简单除以 12 得到月度利率
risk_free_monthly = risk_free_annual / 12

# 根据 CAPM 公式计算股票的预期月度收益率
# CAPM 公式: E(R_i) = R_f + beta * (E(R_m) - R_f)
capm_return_monthly = risk_free_monthly + beta * (market_return_monthly - risk_free_monthly)

# 按照输出契约，将结果存入字典
result = {
    'capm_return_monthly': capm_return_monthly
}

# 打印验证（可选，方便查看结果）
print(f"月度无风险利率: {risk_free_monthly:.6f}")
print(f"按 CAPM 计算的月度预期收益率: {capm_return_monthly:.6f} (即 {capm_return_monthly*100:.4f}%)")
```
