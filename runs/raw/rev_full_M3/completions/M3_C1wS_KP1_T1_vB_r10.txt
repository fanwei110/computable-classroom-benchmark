```python
import numpy as np

# ==========================================
# 1. 由波动率与相关系数构造协方差矩阵
# ==========================================
vol = np.array([0.187, 0.243, 0.312])

# 根据题意构造相关系数矩阵 (资产1, 资产2, 资产3)
# rho_12 = 0.21, rho_13 = -0.13, rho_23 = 0.37
corr = np.array([
    [1.0,   0.21, -0.13],
    [0.21,  1.0,   0.37],
    [-0.13, 0.37,  1.0 ]
])

# 协方差矩阵公式: Σ = diag(σ) @ Corr @ diag(σ)
cov_matrix = np.diag(vol) @ corr @ np.diag(vol)

# ==========================================
# 2. 求最小方差权重 (允许做空，满仓约束 w1+w2+w3=1)
# ==========================================
# 闭式解: w_mvp = (Σ^-1 @ 1) / (1^T @ Σ^-1 @ 1)
ones = np.ones(3)
cov_inv = np.linalg.inv(cov_matrix)

# 计算权重向量
w_mvp = (cov_inv @ ones) / (ones @ cov_inv @ ones)

# ==========================================
# 3. 计算组合波动率
# ==========================================
# 组合方差 w'Σw，波动率取其平方根
port_var = w_mvp.T @ cov_matrix @ w_mvp
port_vol = np.sqrt(port_var)

# ==========================================
# 4. 按要求键名填充 result
# ==========================================
result = {
    'mvp_weights': w_mvp.tolist(),
    'mvp_vol_annual': port_vol
}

# ----- 课堂投屏展示用 (验证结果) -----
print("="*40)
print("马科维茨最小方差组合 (MVP) 计算结果")
print("="*40)
print(f"协方差矩阵:\n{cov_matrix}\n")
print(f"最小方差组合权重:\n  资产1: {w_mvp[0]:.4f}\n  资产2: {w_mvp[1]:.4f}\n  资产3: {w_mvp[2]:.4f}")
print(f"权重之和: {np.sum(w_mvp):.4f} (满足满仓约束)")
print(f"最小方差组合年化波动率: {port_vol:.4%}")
print("="*40)
```
