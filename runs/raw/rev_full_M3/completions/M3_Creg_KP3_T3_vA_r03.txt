```python
import numpy as np

# 债券基本参数
face_value = 100.0          # 面值
coupon_rate = 0.046         # 票息率 4.6%
y_initial = 0.053           # 初始收益率 5.3%
n_years = 7                 # 期限 7 年
yield_change_bp = 80        # 收益率上升的基点
yield_change = yield_change_bp / 10000.0  # 转换为绝对数值 0.008

# 构建现金流（假设按年付息，这也是此类题目的标准默认假设）
coupon = face_value * coupon_rate
cash_flows = np.full(n_years, coupon)
cash_flows[-1] += face_value  # 最后一期还本付息

# 计算初始价格
periods = np.arange(1, n_years + 1)
discount_factors_initial = 1 / (1 + y_initial) ** periods
price_initial = np.sum(cash_flows * discount_factors_initial)

# 计算收益率上升 80bps 后的新价格
y_new = y_initial + yield_change
discount_factors_new = 1 / (1 + y_new) ** periods
price_new = np.sum(cash_flows * discount_factors_new)

# 计算价格下跌的百分比（以小数形式表示，如 -0.048 代表下跌 4.8%）
price_drop_pct = (price_new - price_initial) / price_initial

# 按照输出契约存入字典
result = {
    'price_drop_pct': price_drop_pct
}

# 打印结果以便观察
print(result)
```
