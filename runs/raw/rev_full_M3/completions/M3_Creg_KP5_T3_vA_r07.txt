```python
import numpy as np
from scipy.stats import norm

# 给定参数
position = 2_700_000      # 头寸（元）
vol_annual = 0.24         # 年化波动率
confidence_level = 0.95   # 置信水平
trading_days = 252        # 一年中的交易日数（金融市场标准假设）

# 1. 将年化波动率转换为日波动率
# 假设收益率服从独立同分布，日波动率 = 年化波动率 / sqrt(交易日数)
vol_daily = vol_annual / np.sqrt(trading_days)

# 2. 计算95%置信水平下的正态分布分位数 (Z-score)
z_score = norm.ppf(confidence_level)

# 3. 计算95%一日 VaR
# VaR = 头寸 * Z-score * 日波动率
var_95_1d = position * z_score * vol_daily

# 按照输出契约存入字典
result = {
    'var_95_1d': var_95_1d
}

# 打印结果以供验证
print(f"95% 一日 VaR: {result['var_95_1d']:.2f} 元")
```
