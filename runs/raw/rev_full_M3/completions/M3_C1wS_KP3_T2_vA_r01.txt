```python
import numpy as np
import matplotlib.pyplot as plt

# ================= 参数设置 =================
F = 100                  # 面值
c_rate = 0.046           # 票息率
T = 7                    # 期限(年)
y0 = 0.053               # 初始收益率
C = F * c_rate           # 票息

# 可调收益率变动幅度 (100个基点 = 0.01，可在此处修改)
delta_y = 0.01           

# 收益率绘图区间
y_min, y_max = 0.02, 0.09
# ============================================

def bond_price(y, C, F, T):
    """
    计算债券精确价格
    支持标量与向量输入
    """
    t = np.arange(1, T + 1)
    y_arr = np.atleast_1d(y)
    
    # 计算票息现值
    pv_coupons = np.sum(C / (1 + y_arr[:, np.newaxis])**t, axis=1)
    # 计算面值现值
    pv_face = F / (1 + y_arr)**T
    
    prices = pv_coupons + pv_face
    # 若输入为标量，则返回标量
    return prices if len(prices) > 1 else prices.item()

def mac_duration(y, C, F, T):
    """
    计算麦考利久期
    """
    t = np.arange(1, T + 1)
    P = bond_price(y, C, F, T)
    
    pv_coupons = C / (1 + y)**t
    pv_face = F / (1 + y)**T
    
    # 加权时间求和
    weighted_time = np.sum(t * pv_coupons) + T * pv_face
    return weighted_time / P

# 1. 在 2% 到 9% 的收益率网格上为精确曲线定价
y_grid = np.linspace(y_min, y_max, 500)
P_grid = bond_price(y_grid, C, F, T)

# 2. 在 5.3% 附近叠加基于久期的近似
P0 = bond_price(y0, C, F, T)
mac_d = mac_duration(y0, C, F, T)
mod_d = mac_d / (1 + y0)  # 修正久期

# 久期一阶近似：P_approx(y) = P0 * (1 - ModD * (y - y0))
P_approx_grid = P0 * (1 - mod_d * (y_grid - y0))

# 3. 报告 +100bp 的精确价格与久期法估计的相对变化
y_up = y0 + delta_y
price_at_up100bp = bond_price(y_up, C, F, T)
dur_approx_change_up100bp = -mod_d * delta_y  # 相对价格变化 ΔP/P

# 4. 绘图与保存
plt.figure(figsize=(10, 6))
plt.plot(y_grid * 100, P_grid, label='Exact Price', color='blue', linewidth=2)
plt.plot(y_grid * 100, P_approx_grid, label='Duration Approximation', color='red', linestyle='--', linewidth=2)

# 标记初始点
plt.scatter([y0 * 100], [P0], color='blue', zorder=5)
plt.text(y0 * 100 - 0.2, P0 + 1.5, f'P0 = {P0:.2f}', color='blue')

# 标记上升delta_y后的精确点与近似点
P_approx_up = P0 * (1 + dur_approx_change_up100bp)
plt.scatter([y_up * 100], [price_at_up100bp], color='green', zorder=5, label=f'Exact at +{delta_y*100:.0f}bp')
plt.scatter([y_up * 100], [P_approx_up], color='orange', marker='x', s=100, zorder=5, label=f'Duration Est. at +{delta_y*100:.0f}bp')

# 连线展示误差
plt.vlines(y_up * 100, price_at_up100bp, P_approx_up, color='gray', linestyle=':', linewidth=1.5)

plt.title('Bond Price vs Yield and Duration Approximation')
plt.xlabel('Yield (%)')
plt.ylabel('Price')
plt.legend()
plt.grid(True, linestyle='--', alpha=0.7)

fig_path = 'bond_price_duration_curve.png'
plt.savefig(fig_path, dpi=150, bbox_inches='tight')
plt.close()

# 输出契约
result = {
    'price_at_up100bp': price_at_up100bp,
    'dur_approx_change_up100bp': dur_approx_change_up100bp,
    'figure_path': fig_path
}
```
