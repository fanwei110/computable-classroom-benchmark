```python
import numpy as np
import matplotlib.pyplot as plt

# ==================== 1. 定义资产参数 ====================
E1 = 0.071       # 资产1期望年收益
E2 = 0.124       # 资产2期望年收益
S1 = 0.163       # 资产1年化波动率
S2 = 0.289       # 资产2年化波动率

rhos = [0.15, 0.45, 0.75]  # 给定的三种相关系数
target_return = 0.10       # 目标期望收益

# ==================== 2. 计算与绘图 ====================
# 设定权重范围（允许卖空，权重可超出[0,1]）
w1_arr = np.linspace(-0.5, 1.5, 1000)
w2_arr = 1 - w1_arr

# 投资组合期望收益
port_returns = w1_arr * E1 + w2_arr * E2

# 初始化结果变量
mvp_vol_at_rho45 = None
frontier_vol_at_target = None

plt.figure(figsize=(10, 7))

for rho in rhos:
    # 协方差
    cov = rho * S1 * S2
    
    # 投资组合方差与波动率
    port_var = (w1_arr * S1)**2 + (w2_arr * S2)**2 + 2 * w1_arr * w2_arr * cov
    port_vol = np.sqrt(port_var)
    
    # 绘制前沿曲线
    plt.plot(port_vol, port_returns, label=f'$\\rho$ = {rho}')
    
    # 计算最小方差组合 (MVP)
    w1_mvp = (S2**2 - cov) / (S1**2 + S2**2 - 2 * cov)
    w2_mvp = 1 - w1_mvp
    
    ret_mvp = w1_mvp * E1 + w2_mvp * E2
    var_mvp = (w1_mvp * S1)**2 + (w2_mvp * S2)**2 + 2 * w1_mvp * w2_mvp * cov
    vol_mvp = np.sqrt(var_mvp)
    
    # 标出最小方差组合
    plt.scatter(vol_mvp, ret_mvp, marker='*', s=150, zorder=5)
    plt.annotate(f'MVP ($\\rho$={rho})\nVol={vol_mvp:.2%}', 
                 xy=(vol_mvp, ret_mvp), 
                 xytext=(10, -20), 
                 textcoords='offset points',
                 arrowprops=dict(arrowstyle='->', color='black', lw=1.5),
                 fontsize=9,
                 bbox=dict(boxstyle='round,pad=0.3', fc='white', alpha=0.8))

    # 针对相关系数为 0.45 时的特定计算
    if rho == 0.45:
        mvp_vol_at_rho45 = vol_mvp
        
        # 计算目标期望收益下的权重与最小波动率
        # E_target = w1 * E1 + (1 - w1) * E2  =>  w1 = (E_target - E2) / (E1 - E2)
        w1_target = (target_return - E2) / (E1 - E2)
        w2_target = 1 - w1_target
        
        var_target = (w1_target * S1)**2 + (w2_target * S2)**2 + 2 * w1_target * w2_target * cov
        frontier_vol_at_target = np.sqrt(var_target)

# ==================== 3. 图表修饰与保存 ====================
plt.title('Mean-Variance Frontiers for Different Correlations', fontsize=14)
plt.xlabel('Volatility (Annualized)', fontsize=12)
plt.ylabel('Expected Return (Annualized)', fontsize=12)
plt.legend(fontsize=11)
plt.grid(True, linestyle='--', alpha=0.7)
plt.xlim(0, max(S2, port_vol.max()) * 1.1)
plt.ylim(min(E1, E2) - 0.02, max(E1, E2) + 0.02)

# 保存图像
figure_path = 'mean_variance_frontier.png'
plt.savefig(figure_path, dpi=300, bbox_inches='tight')
plt.close()

# ==================== 4. 封装输出结果 ====================
result = {
    'mvp_vol_at_rho45': mvp_vol_at_rho45,
    'frontier_vol_at_target': frontier_vol_at_target,
    'figure_path': figure_path
}

# 打印报告（可选，便于查看）
print(f"相关系数为 0.45 时：")
print(f"最小方差组合的年化波动率: {result['mvp_vol_at_rho45']:.4%}")
print(f"目标期望收益 10% 下可达到的最小年化波动率: {result['frontier_vol_at_target']:.4%}")
```
