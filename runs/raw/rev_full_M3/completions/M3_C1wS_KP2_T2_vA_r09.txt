```python
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.ticker import FuncFormatter

# ================= 参数化设置（教师可在此调参） =================
rf = 0.023       # 无风险利率 (2.3%)
rm = 0.094       # 市场期望收益 (9.4%)
# ============================================================

# 1. 计算 SML 斜率与 beta=1.27 对应的期望收益
sml_slope = rm - rf
beta_target = 1.27
er_at_beta_127 = rf + beta_target * sml_slope

# 2. 准备股票数据并计算各自的 Alpha
stocks = {
    'X': {'beta': 0.62, 'er': 0.081},
    'Y': {'beta': 1.18, 'er': 0.131},
    'Z': {'beta': 1.51, 'er': 0.099},
}

for name, data in stocks.items():
    # Alpha = 实际期望收益 - SML上的理论期望收益
    sml_er = rf + data['beta'] * sml_slope
    data['alpha'] = data['er'] - sml_er

# 3. 绘图
# 设置中文字体（兼容常见操作系统）
plt.rcParams['font.sans-serif'] = ['SimHei', 'Microsoft YaHei', 'Arial Unicode MS', 'sans-serif']
plt.rcParams['axes.unicode_minus'] = False

fig, ax = plt.subplots(figsize=(10, 6.5))

# 绘制 SML 线 (Beta 从 0 到 2)
betas = np.linspace(0, 2, 100)
sml_ers = rf + betas * sml_slope
ax.plot(betas, sml_ers, label=f'SML (斜率 = {sml_slope:.2%})', color='royalblue', linewidth=2.5)

# 标注无风险利率点与市场组合点
ax.scatter([0, 1], [rf, rm], color='black', s=60, zorder=5)
ax.annotate(f'无风险利率 r_f = {rf:.1%}', xy=(0, rf), xytext=(0.1, rf - 0.012),
            arrowprops=dict(arrowstyle='->', color='black'), fontsize=10)
ax.annotate(f'市场组合 M (E[R_m] = {rm:.1%})', xy=(1, rm), xytext=(1.1, rm - 0.012),
            arrowprops=dict(arrowstyle='->', color='black'), fontsize=10)

# 绘制并标注三只股票
colors = {'X': 'green', 'Y': 'crimson', 'Z': 'darkorange'}
for name, data in stocks.items():
    ax.scatter(data['beta'], data['er'], color=colors[name], s=80, zorder=5)
    # Alpha为正向上标，为负向下标，防止文字覆盖线
    offset_y = 18 if data['alpha'] >= 0 else -28
    ax.annotate(f"{name}\n(β={data['beta']:.2f}, α={data['alpha']:.2%})",
                xy=(data['beta'], data['er']),
                xytext=(15, offset_y),
                textcoords='offset points',
                fontsize=10, color=colors[name], fontweight='bold',
                arrowprops=dict(arrowstyle='->', color=colors[name], lw=1.5))

# 格式美化
ax.set_title('证券市场线 (SML) 与 Alpha 定位', fontsize=14, fontweight='bold')
ax.set_xlabel('Beta (β)', fontsize=12)
ax.set_ylabel('期望收益率 E[R]', fontsize=12)
ax.set_xlim(0, 2)
ax.set_ylim(-0.02, 0.18)
ax.xaxis.set_ticks(np.arange(0, 2.1, 0.5))
ax.yaxis.set_major_formatter(FuncFormatter(lambda y, _: f'{y:.1%}'))
ax.grid(True, linestyle='--', alpha=0.6)
ax.legend(fontsize=11, loc='upper left')

# 保存图形
fig_path = 'sml_plot.png'
plt.savefig(fig_path, dpi=150, bbox_inches='tight')
plt.close()

# 4. 将结果存入 result 字典
result = {
    'sml_slope': sml_slope,
    'er_at_beta_127': er_at_beta_127,
    'figure_path': fig_path
}

# （以下为控制台输出，方便课堂投屏展示）
print("="*50)
print("《证券投资学》课堂展示：CAPM 与证券市场线")
print("="*50)
print(f"1. SML 斜率 (市场风险溢价): {sml_slope:.4f} ({sml_slope:.2%})")
print(f"2. Beta=1.27 对应的期望收益: {er_at_beta_127:.4f} ({er_at_beta_127:.2%})")
print("-" * 50)
print("3. 各股票 Alpha 偏离情况:")
for name, data in stocks.items():
    print(f"   股票 {name}: β={data['beta']:.2f}, 实际E[R]={data['er']:.2%}, Alpha={data['alpha']:.2%}")
print("-" * 50)
print(f"4. 图形已保存至: {fig_path}")
print(f"5. 结果字典: {result}")
```

**代码设计说明：**

1. **参数化设计**：`rf` 和 `rm` 放在代码最顶部。上课时修改这两个变量，重新运行即可一键更新所有计算（斜率、期望收益、Alpha值）以及图形上的文字标注，非常适合实时投屏调参演示。
2. **核心计算逻辑**：SML 斜率即为市场风险溢价 $E[R_m] - r_f$。代入 $\beta=1.27$ 计算得出对应的期望收益约为 11.32%。
3. **Alpha 标注**：在绘制 X、Y、Z 三个点时，不仅标出了原始坐标，还自动计算了偏离 SML 的 Alpha 值。根据 Alpha 的正负号，标注文本智能地向上或向下偏移，避免遮挡证券市场线。
4. **可视化规范**：X 轴限制在 0~2，Y 轴百分比格式化，并绘制了网格线与关键点（无风险利率点与市场组合点），图表直观且专业。
5. **输出契约**：严格按要求将 `sml_slope`（浮点数）、`er_at_beta_127`（浮点数）与 `figure_path`（字符串路径）存入并返回 `result` 字典。
