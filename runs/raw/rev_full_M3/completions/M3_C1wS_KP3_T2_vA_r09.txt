```python
import numpy as np
import matplotlib.pyplot as plt

# ================= 参数设置 =================
face = 100                  # 面值
coupon_rate = 0.046         # 票息率 4.6%
T = 7                       # 期限 7年
y0 = 0.053                  # 初始收益率 5.3%
delta_y = 0.01              # 可调的收益率变动幅度 (用于展示和近似计算)
C = face * coupon_rate      # 年票息额

# 假设每年付息1次，现金流发生在第1至第7年年末
t_arr = np.arange(1, T + 1)[:, None]  # shape (7, 1)，便于广播计算

# ================= 1. 精确曲线定价 =================
# 生成 2% 到 9% 的收益率网格
y_grid = np.linspace(0.02, 0.09, 1000) # shape (1000,)

# 精确价格：P = sum(C / (1+y)^t) + F / (1+y)^T
# 利用 numpy 广播机制，t_arr 为 (7,1)，y_grid 为 (1000,)，相乘后为 (7, 1000)
prices_exact = np.sum(C / (1 + y_grid)**t_arr, axis=0) + face / (1 + y_grid)**T

# ================= 2. 久期计算与近似曲线 =================
# 计算初始收益率 y0 下的精确价格
P0 = float(np.sum(C / (1 + y0)**t_arr) + face / (1 + y0)**T)

# 计算麦考利久期 (Macaulay Duration) 和 修正久期 (Modified Duration)
mac_dur = float(np.sum(t_arr * C / (1 + y0)**t_arr) / P0 + (T * face / (1 + y0)**T) / P0)
mod_dur = mac_dur / (1 + y0)

# 基于久期的近似价格曲线: P_approx = P0 * (1 - D_mod * (y - y0))
prices_approx = P0 * (1 - mod_dur * (y_grid - y0))

# ================= 3. 绘图 =================
# 设置字体以防中文乱码，兼容常见操作系统
plt.rcParams['font.sans-serif'] = ['SimHei', 'Arial Unicode MS', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

fig, ax = plt.subplots(figsize=(10, 6))

# 绘制精确价格曲线
ax.plot(y_grid * 100, prices_exact, label='Exact Price (精确现金流贴现)', color='blue', linewidth=2)
# 绘制久期近似曲线
ax.plot(y_grid * 100, prices_approx, label='Duration Approximation (久期一阶近似)', color='red', linestyle='--', linewidth=2)

# 标出初始定价点
ax.scatter([y0 * 100], [P0], color='black', zorder=5, label=f'Base Point (y={y0*100:.1f}%)')

# 绘制可调的收益率变动幅度区域 (展示 delta_y 的调整范围)
ax.axvspan((y0 - delta_y)*100, (y0 + delta_y)*100, color='gray', alpha=0.15, 
           label=f'Adjustable Δy Range (±{delta_y*100:.0f}bp)')

ax.set_xlabel('Yield to Maturity (收益率 %)')
ax.set_ylabel('Bond Price (债券价格)')
ax.set_title('债券价格-收益率曲线与久期近似\nBond Price-Yield Curve & Duration Approximation')
ax.legend(loc='upper right')
ax.grid(True, linestyle=':', alpha=0.7)

# 保存图形
fig_path = 'bond_price_yield_curve.png'
plt.savefig(fig_path, dpi=150, bbox_inches='tight')
plt.close()

# ================= 4. 报告 +100bp 的指标 =================
# 收益率上升 100个基点 (1%) 后的精确价格
y_up_100bp = y0 + 0.01
price_up_100bp = float(np.sum(C / (1 + y_up_100bp)**t_arr) + face / (1 + y_up_100bp)**T)

# 用久期法估计的相对价格变化: ΔP/P ≈ -D_mod * Δy
dur_approx_rel_change = -mod_dur * 0.01

# ================= 5. 封装输出 =================
result = {
    'price_at_up100bp': round(price_up_100bp, 4),
    'dur_approx_change_up100bp': round(dur_approx_rel_change, 6),
    'figure_path': fig_path
}

# 控制台打印以供教师投屏参考
if __name__ == '__main__':
    print("===== 计算结果 =====")
    print(f"初始收益率 ({y0*100:.1f}%) 下的债券精确价格: {P0:.4f}")
    print(f"修正久期 (Modified Duration): {mod_dur:.4f}")
    print(f"收益率上升100bp后的精确价格: {result['price_at_up100bp']}")
    print(f"久期法估计的相对价格变化: {result['dur_approx_change_up100bp']:.4%}")
    print(f"图表已保存至: {result['figure_path']}")
```
