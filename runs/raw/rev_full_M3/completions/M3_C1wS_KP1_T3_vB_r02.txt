```python
import numpy as np

# ================= 1. 确定权重与资产对应方式 =================
# 按照“A B 六四开”的常规表述习惯，设定资产A权重为60%，资产B权重为40%
w_A = 0.6
w_B = 0.4
w = np.array([w_A, w_B])

# ================= 2. 提取已知参数并构造协方差矩阵 =================
# 题目给定的年化波动率
vol_A = 0.184
vol_B = 0.297

# 计算方差
var_A = vol_A ** 2
var_B = vol_B ** 2

# 两种场景下的相关系数
rho_before = 0.3
rho_after = 0.8

# 计算协方差 cov(A, B) = rho * vol_A * vol_B
cov_before = rho_before * vol_A * vol_B
cov_after = rho_after * vol_A * vol_B

# 构造协方差矩阵 Sigma (2x2)
# Sigma = [[var_A, cov], [cov, var_B]]
Sigma_before = np.array([
    [var_A, cov_before],
    [cov_before, var_B]
])

Sigma_after = np.array([
    [var_A, cov_after],
    [cov_after, var_B]
])

# ================= 3. 计算两个组合波动率 =================
# 组合方差 formula: w' Sigma w
# 组合波动率 formula: sqrt(w' Sigma w)
var_portfolio_before = w.T @ Sigma_before @ w
var_portfolio_after  = w.T @ Sigma_after  @ w

vol_before_annual = np.sqrt(var_portfolio_before)
vol_after_annual  = np.sqrt(var_portfolio_after)

# ================= 4. 填充 result =================
result = {
    'vol_before_annual': vol_before_annual,
    'vol_after_annual': vol_after_annual
}

# ================= 课堂展示打印 =================
print("="*45)
print(" 马科维茨均值-方差组合理论 - 课堂实时计算")
print("="*45)
print(f"【输入假设】")
print(f"资产A权重: {w_A:.0%}，资产B权重: {w_B:.0%}")
print(f"资产A波动率: {vol_A:.1%}，资产B波动率: {vol_B:.1%}")
print("-" * 45)
print(f"【场景1：相关系数 = {rho_before}】")
print(f"协方差矩阵:\n{Sigma_before}")
print(f"组合波动率: {vol_before_annual:.4%}")
print("-" * 45)
print(f"【场景2：相关系数 = {rho_after}】")
print(f"协方差矩阵:\n{Sigma_after}")
print(f"组合波动率: {vol_after_annual:.4%}")
print("-" * 45)
print(f"【结论】相关性从0.3飙升至0.8，组合波动率上升约 {(vol_after_annual - vol_before_annual)*100:.2f}%")
print(f"\n最终输出字典 result: {result}")
```
