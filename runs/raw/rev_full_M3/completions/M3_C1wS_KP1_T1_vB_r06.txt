```python
import numpy as np

# ================= 1. 构造协方差矩阵 =================
# 波动率 (年化)
vols = np.array([0.187, 0.243, 0.312])

# 相关系数
rho_12 = 0.21
rho_13 = -0.13
rho_23 = 0.37
corr_matrix = np.array([
    [1.0,  rho_12, rho_13],
    [rho_12, 1.0,  rho_23],
    [rho_13, rho_23, 1.0 ]
])

# 协方差矩阵 Σ = D @ Corr @ D，其中 D 为波动率对角阵
D = np.diag(vols)
cov_matrix = D @ corr_matrix @ D

# ================= 2. 求解最小方差组合权重 =================
# 满仓约束下（允许做空），最小方差组合权重的闭式解为：
# w_mvp = Σ^(-1) @ 1 / (1^T @ Σ^(-1) @ 1)
ones = np.ones(3)
inv_cov = np.linalg.inv(cov_matrix)

# 计算 Σ^(-1) @ 1
inv_cov_ones = inv_cov @ ones

# 归一化使权重之和为1
w_mvp = inv_cov_ones / (ones @ inv_cov_ones)

# ================= 3. 计算组合波动率 =================
# 组合方差为 w'Σw，波动率为其平方根
port_var = w_mvp @ cov_matrix @ w_mvp
port_vol = np.sqrt(port_var)

# ================= 4. 填充结果字典 =================
result = {
    'mvp_weights': w_mvp.tolist(),
    'mvp_vol_annual': port_vol
}

# (可选) 课堂投屏展示辅助打印
print("--- 马科维茨最小方差组合计算结果 ---")
print(f"资产1权重: {w_mvp[0]:.4f} ({w_mvp[0]*100:.2f}%)")
print(f"资产2权重: {w_mvp[1]:.4f} ({w_mvp[1]*100:.2f}%)")
print(f"资产3权重: {w_mvp[2]:.4f} ({w_mvp[2]*100:.2f}%)")
print(f"权重之和: {np.sum(w_mvp):.4f}")
print(f"组合年化波动率: {port_vol:.4f} ({port_vol*100:.2f}%)")
print("-------------------------------------")
```
