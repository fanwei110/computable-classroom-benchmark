```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

def run_var_analysis():
    # -------------------- 参数配置 --------------------
    # 置信度参数化，方便后续调整（如0.95, 0.99等）
    confidence_level = 0.95
    position_value = 1_000_000  # 头寸：100万人民币
    
    # 设置中文字体，防止图表中文乱码，适配不同操作系统
    plt.rcParams['font.sans-serif'] = ['SimHei', 'Arial Unicode MS', 'DejaVu Sans']
    plt.rcParams['axes.unicode_minus'] = False  # 正常显示负号
    
    # -------------------- 步骤1：读取数据并构造损益 --------------------
    data_path = 'data/market_snapshot_v1.csv'
    df = pd.read_csv(data_path)
    
    # 提取基金日收益率序列（假设为小数形式，如0.02代表2%）
    daily_returns = df['fund']
    
    # 计算日损益（人民币）
    daily_pnl = daily_returns * position_value
    
    # -------------------- 步骤2：计算历史VaR --------------------
    alpha = 1 - confidence_level
    # 计算损益的经验分位数。VaR是损失的分位数，即损益分布的左侧alpha分位数
    var_point = np.percentile(daily_pnl, alpha * 100)
    
    # VaR通常以正数形式报告（代表可能损失的金额）
    var_amount = -var_point
    
    # -------------------- 步骤3：绘制直方图与VaR线 --------------------
    fig, ax = plt.subplots(figsize=(10, 6))
    
    # 绘制损益直方图
    ax.hist(daily_pnl, bins=50, edgecolor='black', alpha=0.7, color='skyblue', label='日损益频数')
    
    # 绘制VaR竖线及标注
    ax.axvline(var_point, color='red', linestyle='--', linewidth=2, 
               label=f'VaR ({confidence_level*100:.0f}%): {var_point:,.2f} RMB')
               
    # 在线附近添加文本标注，确保视觉清晰
    ax.annotate(f'VaR = {var_point:,.2f} RMB\n(损失 {var_amount:,.2f} RMB)', 
                xy=(var_point, ax.get_ylim()[1] * 0.8), 
                xytext=(var_point + daily_pnl.std()*0.5, ax.get_ylim()[1] * 0.9),
                arrowprops=dict(facecolor='black', shrink=0.05, width=1, headwidth=6),
                fontsize=12, bbox=dict(boxstyle="round,pad=0.3", fc="white", ec="red", alpha=0.8))
    
    ax.set_title(f'基金日损益分布与在险价值\n[头寸: {position_value/1e6:.0f}百万 | 置信度: {confidence_level*100:.0f}%]', fontsize=14)
    ax.set_xlabel('日损益 (RMB)', fontsize=12)
    ax.set_ylabel('频数', fontsize=12)
    ax.legend(loc='upper left', fontsize=11)
    ax.grid(axis='y', linestyle='--', alpha=0.7)
    
    # -------------------- 步骤4：保存图形并填充结果 --------------------
    figure_path = 'hist_var_plot.png'
    plt.savefig(figure_path, dpi=150, bbox_inches='tight')
    plt.close()
    
    # 将计算结果存入规定的字典
    result = {
        'hist_var_95_1d': var_amount,   # 95%一日VaR数值（正数表示损失敞口）
        'figure_path': figure_path
    }
    
    return result

if __name__ == '__main__':
    # 执行分析并输出结果
    result = run_var_analysis()
    print(f"95%一日历史VaR (人民币): {result['hist_var_95_1d']:,.2f}")
    print(f"图形已保存至: {result['figure_path']}")
```
