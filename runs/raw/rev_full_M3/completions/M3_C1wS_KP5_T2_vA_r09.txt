```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

def main():
    # ---------------- 参数设置 ----------------
    position = 1_000_000          # 头寸：100万元人民币
    confidence_level = 0.95       # 置信水平（可调参数，例如改为 0.99 即可计算99% VaR）
    
    # ---------------- 1. 读取数据，构造损益 ----------------
    # 读取快照 CSV 文件
    df = pd.read_csv('data/market_snapshot_v1.csv')
    
    # 计算基金净值的日收益率，并删除首行NaN
    daily_returns = df['fund'].pct_change().dropna()
    
    # 构造头寸的日损益（人民币）
    pnl = position * daily_returns
    
    # ---------------- 2. 计算历史法 VaR ----------------
    # 置信水平为 alpha 时，对应的分位数为 1 - alpha（左尾）
    quantile_level = 1 - confidence_level
    
    # 使用经验分布计算分位数（numpy默认使用线性插值）
    var_point = np.percentile(pnl, quantile_level * 100)
    
    # VaR 通常以正数表示损失金额，因此对分位数取负
    hist_var_1d = -var_point
    
    # ---------------- 3. 画直方图并加 VaR 线 ----------------
    fig, ax = plt.subplots(figsize=(10, 6))
    
    # 绘制日损益直方图
    ax.hist(pnl, bins=50, color='lightsteelblue', edgecolor='black', alpha=0.8)
    
    # 绘制 VaR 竖线（在原始损益坐标的对应分位点处画线）
    ax.axvline(var_point, color='red', linestyle='--', linewidth=2,
               label=f'{confidence_level*100:.0f}% 1-day VaR = {hist_var_1d:,.2f} RMB')
    
    # 添加标注和图例
    ax.set_title('Histogram of Daily P&L with Historical VaR', fontsize=14)
    ax.set_xlabel('Daily P&L (RMB)', fontsize=12)
    ax.set_ylabel('Frequency', fontsize=12)
    ax.legend(fontsize=12)
    ax.grid(True, linestyle='--', alpha=0.6)
    
    # ---------------- 4. 保存图形并填充 result ----------------
    figure_path = 'hist_var_plot.png'
    plt.savefig(figure_path, dpi=150, bbox_inches='tight')
    plt.close()
    
    # 封装结果字典
    result = {
        'hist_var_95_1d': hist_var_1d,
        'figure_path': figure_path
    }
    
    # 在控制台输出 VaR 的数值
    print(f"在 {confidence_level*100:.0f}% 置信水平下，1日历史法VaR为：{hist_var_1d:,.2f} 元人民币")

if __name__ == '__main__':
    main()
```
