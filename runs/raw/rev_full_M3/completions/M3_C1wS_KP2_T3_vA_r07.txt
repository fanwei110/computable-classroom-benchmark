```python
import numpy as np

# ================= 输入参数 =================
beta = 1.42                 # 股票的 Beta 值
market_return_monthly = -0.058  # 上个月市场收益率 (-5.8%)
risk_free_annual = 0.047    # 年化无风险利率 (4.7%)

# ================= 步骤1：年无风险利率折算为月利率 =================
# 假设：采用复利折算方式将年利率转换为月度利率，这在学术上更为严谨
# 公式: R_f_monthly = (1 + R_f_annual)^(1/12) - 1
risk_free_monthly = (1 + risk_free_annual) ** (1 / 12) - 1

# ================= 步骤2：在月度层面套用 CAPM =================
# CAPM 公式: E(R_i) = R_f + Beta * (E(R_m) - R_f)
# 在此场景下，使用已知的上月市场收益率代替期望市场收益率
capm_return_monthly = risk_free_monthly + beta * (market_return_monthly - risk_free_monthly)

# ================= 步骤3：填充 result =================
result = {
    'capm_return_monthly': capm_return_monthly
}

# 输出结果以便课堂投屏展示
print(f"年化无风险利率折算为月利率: {risk_free_monthly:.6%}")
print(f"按CAPM计算的该股票上月应有收益: {capm_return_monthly:.6%}")
print(f"result字典: {result}")
```
