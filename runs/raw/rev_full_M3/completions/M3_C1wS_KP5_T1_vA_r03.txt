```python
import numpy as np
from scipy import stats

# === 输入参数 ===
position = 1850000  # 头寸：185万元人民币
annual_vol = 0.218  # 年化波动率：21.8%

# === 假设处理 ===
# 金融市场参数法计算中，通常假设一年有252个交易日，使用平方根法则进行波动率期限转换
trading_days_per_year = 252

# === 步骤1：将年化波动率换算到一日期限 ===
daily_vol = annual_vol / np.sqrt(trading_days_per_year)

# === 步骤2：对两个置信水平套用正态分位数 ===
# 获取95%和99%置信水平下的标准正态分布左尾逆累积概率（即分位数）
z_95 = stats.norm.ppf(0.95)
z_99 = stats.norm.ppf(0.99)

# === 步骤3：计算VaR，缩放到十日期限，并以人民币报告 ===
# 95%置信水平的一日VaR
var_95_1d = position * daily_vol * z_95

# 99%置信水平的十日VaR（使用平方根法则将日波动率缩放至10日）
var_99_10d = position * daily_vol * np.sqrt(10) * z_99

# === 步骤4：填充 result ===
result = {
    'var_95_1d': var_95_1d,
    'var_99_10d': var_99_10d
}

# 课堂投屏展示输出
print(f"假设一年包含 {trading_days_per_year} 个交易日：")
print(f"日波动率: {daily_vol:.6f}")
print(f"95%置信水平的一日VaR: {result['var_95_1d']:,.2f} 人民币")
print(f"99%置信水平的十日VaR: {result['var_99_10d']:,.2f} 人民币")
```
