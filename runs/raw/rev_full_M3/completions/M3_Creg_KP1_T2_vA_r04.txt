```python
import numpy as np
import matplotlib.pyplot as plt

# ==================== 资产参数设定 ====================
mu1 = 0.071      # 资产1期望年收益
mu2 = 0.124      # 资产2期望年收益
sig1 = 0.163      # 资产1年化波动率
sig2 = 0.289      # 资产2年化波动率
rhos = [0.15, 0.45, 0.75]  # 相关系数

# ==================== 计算特定指标 ====================
rho_45 = 0.45
cov12_45 = rho_45 * sig1 * sig2

# 1. 计算相关系数为 0.45 时的最小方差组合 (MVP)
w1_mvp_45 = (sig2**2 - cov12_45) / (sig1**2 + sig2**2 - 2 * cov12_45)
w2_mvp_45 = 1 - w1_mvp_45
mvp_vol_45 = np.sqrt(w1_mvp_45**2 * sig1**2 + w2_mvp_45**2 * sig2**2 + 2 * w1_mvp_45 * w2_mvp_45 * cov12_45)

# 2. 计算目标期望收益 10% 下的最小波动率 (rho=0.45)
target_mu = 0.10
w1_target = (target_mu - mu2) / (mu1 - mu2)
w2_target = 1 - w1_target
frontier_vol_target = np.sqrt(w1_target**2 * sig1**2 + w2_target**2 * sig2**2 + 2 * w1_target * w2_target * cov12_45)

# ==================== 绘制均值-方差前沿 ====================
plt.figure(figsize=(10, 7))
colors = ['blue', 'green', 'red']

for i, rho in enumerate(rhos):
    cov12 = rho * sig1 * sig2
    
    # 由于允许卖空，生成从 -0.5 到 1.5 的权重，覆盖合理的收益范围
    w1_vals = np.linspace(-0.5, 1.5, 500)
    w2_vals = 1 - w1_vals
    mu_p = w1_vals * mu1 + w2_vals * mu2
    sig_p = np.sqrt(w1_vals**2 * sig1**2 + w2_vals**2 * sig2**2 + 2 * w1_vals * w2_vals * cov12)
    
    # 画出前沿曲线
    plt.plot(sig_p, mu_p, label=f'ρ = {rho}', color=colors[i])
    
    # 计算当前 rho 下的最小方差组合
    w1_mvp = (sig2**2 - cov12) / (sig1**2 + sig2**2 - 2 * cov12)
    w2_mvp = 1 - w1_mvp
    mu_mvp = w1_mvp * mu1 + w2_mvp * mu2
    sig_mvp = np.sqrt(w1_mvp**2 * sig1**2 + w2_mvp**2 * sig2**2 + 2 * w1_mvp * w2_mvp * cov12)
    
    # 标出最小方差组合点
    plt.scatter(sig_mvp, mu_mvp, color=colors[i], zorder=5)
    # 避免文本重叠，手动添加轻微偏移量
    plt.annotate(f'MVP (ρ={rho})', 
                 xy=(sig_mvp, mu_mvp), 
                 xytext=(sig_mvp + 0.015, mu_mvp - 0.005 - 0.003 * i),
                 arrowprops=dict(facecolor='black', shrink=0.05, width=1, headwidth=5),
                 fontsize=9)

# 图表修饰
plt.xlabel('Volatility (Standard Deviation)')
plt.ylabel('Expected Return')
plt.title('Mean-Variance Frontiers for Different Correlations')
plt.legend()
plt.grid(True, linestyle='--', alpha=0.6)
plt.tight_layout()

# 保存图像
fig_path = 'mean_variance_frontier.png'
plt.savefig(fig_path, dpi=150)
plt.close()

# ==================== 存入结果字典 ====================
result = {
    'mvp_vol_at_rho45': mvp_vol_45,
    'frontier_vol_at_target': frontier_vol_target,
    'figure_path': fig_path
}

# 打印结果供验证
print(result)
```
