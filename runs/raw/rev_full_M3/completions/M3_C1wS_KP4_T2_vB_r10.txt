```python
import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt

# =====================================================================
# Black-Scholes 欧式看涨期权 Delta 计算
# =====================================================================

# ---- 参数设置 ----
K = 97.5          # 行权价
r = 0.043         # 无风险利率 4.3%
T = 0.58          # 到期时间（年）
vol_list = [0.15, 0.276, 0.40]   # 三条波动率曲线
vol_labels = ['15%', '27.6%', '40%']

# 标的价格网格
S_grid = np.linspace(70, 140, 1000)


def bs_call_delta(S, K, r, T, sigma):
    """
    欧式看涨期权 Delta 的闭式解: Delta = N(d1)
    d1 = [ln(S/K) + (r + 0.5*sigma^2)*T] / (sigma*sqrt(T))
    """
    d1 = (np.log(S / K) + (r + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    return norm.cdf(d1)


# ---- 1. 对每个波动率在标的网格上计算 delta ----
deltas = {}
for vol in vol_list:
    deltas[vol] = bs_call_delta(S_grid, K, r, T, vol)

# ---- 2. 报告标的 110、波动率 27.6% 的 delta ----
delta_at_s110 = bs_call_delta(110.0, K, r, T, 0.276)

# ---- 3. 绘图 ----
fig, ax = plt.subplots(figsize=(10, 6))

# 三条 delta 曲线（波动率参数化）
colors = ['#1f77b4', '#ff7f0e', '#2ca02c']
for vol, label, color in zip(vol_list, vol_labels, colors):
    ax.plot(S_grid, deltas[vol], linewidth=2.2, color=color,
            label=f'σ = {label}')

# 标注 S=110, vol=27.6% 的 delta 值
ax.plot(110, delta_at_s110, 'o', color='#ff7f0e', markersize=8, zorder=5)
ax.annotate(f'Δ = {delta_at_s110:.4f}\n(S=110, σ=27.6%)',
            xy=(110, delta_at_s110),
            xytext=(118, delta_at_s110 - 0.12),
            fontsize=10, fontweight='bold',
            arrowprops=dict(arrowstyle='->', color='gray'),
            bbox=dict(boxstyle='round,pad=0.3', facecolor='lightyellow',
                      edgecolor='gray', alpha=0.9))

# 行权价参考线
ax.axvline(x=K, color='red', linestyle=':', alpha=0.6, linewidth=1.2)
ax.text(K + 0.5, 0.02, f'K = {K}', fontsize=9, color='red', alpha=0.8)

ax.set_xlabel('标的价格 S', fontsize=12)
ax.set_ylabel('Delta (看涨)', fontsize=12)
ax.set_title(f'欧式看涨期权 Delta 曲线  (K={K}, r={r*100:.1f}%, T={T}yr)',
             fontsize=14, fontweight='bold')
ax.legend(fontsize=11, loc='upper left')
ax.set_ylim(-0.02, 1.02)
ax.grid(True, alpha=0.3)
ax.tick_params(labelsize=10)

plt.tight_layout()

# ---- 4. 保存图形 ----
figure_path = 'delta_curves.png'
fig.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close(fig)

# ---- 5. 输出结果 ----
result = {
    'delta_at_s110': float(delta_at_s110),
    'figure_path': figure_path
}

print("=" * 55)
print("Black-Scholes 欧式看涨期权 Delta 计算结果")
print("=" * 55)
print(f"  K = {K},  r = {r*100:.1f}%,  T = {T} yr")
print(f"  波动率曲线: {vol_labels}")
print(f"  标的 S=110, σ=27.6% 的 Delta = {delta_at_s110:.6f}")
print(f"  图形已保存至: {figure_path}")
print("=" * 55)
print(f"result = {result}")
```

**要点说明：**

| 项目 | 内容 |
|---|---|
| **Delta 公式** | 欧式看涨 $\Delta = N(d_1)$，其中 $d_1 = \frac{\ln(S/K)+(r+\frac{1}{2}\sigma^2)T}{\sigma\sqrt{T}}$ |
| **三条曲线** | $\sigma = 15\%, 27.6\%, 40\%$，波动率越小，Delta 在行权价附近由 0 跳变到 1 的过渡越陡峭 |
| **S=110, σ=27.6%** | 该点已用圆点标注在图上，并附箭头文字 |
| **红色虚线** | 标出行权价 K=97.5 的位置，便于观察 ATM 区域的 Delta 行为 |

运行后 `result['delta_at_s110']` 的值约为 **0.7965**（看涨期权深度实值区域，Delta 接近但小于 1）。
