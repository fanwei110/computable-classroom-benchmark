```python
import numpy as np
from scipy.stats import norm

# ==================== 参数设定 ====================
portfolio_value = 1_850_000       # 头寸价值 (元)
annual_vol = 0.218                # 年化收益波动率
trading_days_per_year = 252       # 每年交易日数（业界标准参数）

# ==================== 波动率期限转换 ====================
# 使用平方根法则进行时间尺度转换
daily_vol = annual_vol / np.sqrt(trading_days_per_year)
vol_10d = daily_vol * np.sqrt(10)

# ==================== 计算分位数 ====================
# 在 delta-normal (参数法) 模型下，假设收益率服从正态分布
z_95 = norm.ppf(0.95)
z_99 = norm.ppf(0.99)

# ==================== 计算 VaR ====================
# VaR = 头寸价值 × 分位数 × 对应期限的波动率
var_95_1d = portfolio_value * z_95 * daily_vol
var_99_10d = portfolio_value * z_99 * vol_10d

# ==================== 输出结果 ====================
result = {
    'var_95_1d': var_95_1d,
    'var_99_10d': var_99_10d
}

# 打印结果以供验证
print(f"95% 一日 VaR: {result['var_95_1d']:,.2f} 元")
print(f"99% 十日 VaR: {result['var_99_10d']:,.2f} 元")
```
