```python
import numpy as np
import matplotlib.pyplot as plt

def main():
    # ================= 1. 参数设定 =================
    mu1, mu2 = 0.071, 0.124       # 两国/两资产预期收益率
    sigma1, sigma2 = 0.163, 0.289 # 波动率
    var1, var2 = sigma1**2, sigma2**2
    rhos = [0.15, 0.45, 0.75]     # 相关系数
    target_ret = 0.10             # 目标收益率

    # 权重扫描范围：允许卖空以展示完整的双曲线
    w1_arr = np.linspace(-0.5, 1.5, 1000)
    
    # 结果字典
    result = {}
    
    # 设置绘图风格与防止中文乱码（使用英文标签确保跨环境兼容）
    plt.figure(figsize=(10, 7))
    
    # ================= 2. 计算与绘图 =================
    for rho in rhos:
        # 构造协方差矩阵
        cov = rho * sigma1 * sigma2
        Sigma = np.array([[var1, cov], 
                          [cov, var2]])
        
        # 组合权重矩阵 W，shape为 (2, N)
        W = np.vstack([w1_arr, 1 - w1_arr])
        
        # 利用矩阵运算 w'Σw 和 w'μ 计算前沿
        mu_vec = np.array([mu1, mu2])
        port_ret = W.T @ mu_vec
        port_var = np.sum(W * (Sigma @ W), axis=0)
        port_vol = np.sqrt(port_var)
        
        # 绘制前沿曲线 (转为百分比以增强可读性)
        plt.plot(port_vol * 100, port_ret * 100, label=f'ρ = {rho}', lw=2)
        
        # ---- 解析法求最小方差组合 (MVP) ----
        # 对 w'Σw 求导令其等于0，得 w_mvp = (Σ^-1 * 1) / (1' * Σ^-1 * 1)
        # 此处两资产下有简化的解析解
        w1_mvp = (var2 - cov) / (var1 + var2 - 2 * cov)
        w_mvp = np.array([w1_mvp, 1 - w1_mvp])
        
        mvp_ret = w_mvp @ mu_vec
        mvp_var = w_mvp @ Sigma @ w_mvp
        mvp_vol = np.sqrt(mvp_var)
        
        # 在图上标出最小方差点
        plt.scatter(mvp_vol * 100, mvp_ret * 100, s=80, zorder=5, edgecolors='black')
        
        # ---- 针对 rho=0.45 计算特定指标 ----
        if np.isclose(rho, 0.45):
            # 存入最小方差组合波动率
            result['mvp_vol_at_rho45'] = mvp_vol
            
            # 计算目标收益 10% 下的最小波动率
            # 满仓约束下，两资产权重完全由目标收益决定
            w1_target = (target_ret - mu2) / (mu1 - mu2)
            w_target = np.array([w1_target, 1 - w1_target])
            target_var = w_target @ Sigma @ w_target
            target_vol = np.sqrt(target_var)
            
            # 存入目标收益最小波动率
            result['frontier_vol_at_target'] = target_vol
            
            # 在图上额外标出 rho=0.45 的目标收益点，方便课堂讲解
            plt.scatter(target_vol * 100, target_ret * 100, 
                        marker='*', color='red', s=300, zorder=6, 
                        label=f'Target 10% (ρ=0.45)\nVol = {target_vol*100:.2f}%')
            
            # 在 MVP 点旁添加文字注释
            plt.annotate(f'MVP Vol = {mvp_vol*100:.2f}%', 
                         xy=(mvp_vol * 100, mvp_ret * 100), 
                         xytext=(mvp_vol * 100 + 1.5, mvp_ret * 100 - 0.8),
                         arrowprops=dict(facecolor='black', width=0.5, shrink=0.05))

    # ================= 3. 图形修饰与保存 =================
    plt.title('Markowitz Mean-Variance Efficient Frontier (Two Assets)', fontsize=14)
    plt.xlabel('Volatility (%)', fontsize=12)
    plt.ylabel('Expected Return (%)', fontsize=12)
    plt.legend(fontsize=11)
    plt.grid(True, linestyle='--', alpha=0.7)
    
    # 保存图形
    fig_path = 'efficient_frontier.png'
    plt.savefig(fig_path, dpi=300, bbox_inches='tight')
    result['figure_path'] = fig_path
    
    # ================= 4. 打印结果验证 =================
    print("计算结果字典：")
    for k, v in result.items():
        if 'vol' in k:
            print(f"{k}: {v:.6f} (即 {v*100:.4f}%)")
        else:
            print(f"{k}: {v}")

if __name__ == '__main__':
    main()
```
