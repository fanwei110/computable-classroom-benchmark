```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# ==========================================
# 1. 生成模拟快照 CSV (确保脚本自包含且可复现)
# ==========================================
# 假设：由于未提供真实CSV，我们生成一个包含 'date' 和 'fund' 列的模拟数据
# 设置随机种子保证输出确定可复现
np.random.seed(42)
dates = pd.date_range(start='2023-01-01', periods=300, freq='B')
# 模拟基金净值：初始净值1.0，日均收益率0.05%，日均波动率1.5%
daily_returns = np.random.normal(loc=0.0005, scale=0.015, size=300)
nav = 1.0 * np.cumprod(1 + daily_returns)
df_mock = pd.DataFrame({'date': dates, 'fund': nav})
csv_path = 'snapshot.csv'
df_mock.to_csv(csv_path, index=False)

# ==========================================
# 2. 读取快照 CSV；计入无风险利率
# ==========================================
df = pd.read_csv(csv_path, parse_dates=['date'])
df.set_index('date', inplace=True)

# 假设：一年252个交易日，无风险利率年化2.1%
rf_annual = 0.021
trading_days = 252
rf_daily = rf_annual / trading_days

# 计算日收益率
df['daily_return'] = df['fund'].pct_change()

# 计算日超额收益率（扣除日度无风险利率）
df['excess_return'] = df['daily_return'] - rf_daily

# ==========================================
# 3. 计算 60 日滚动年化夏普，窗口可调
# ==========================================
# 窗口大小可调参数
window_size = 60

# 计算滚动均值和滚动标准差
rolling_mean_excess = df['excess_return'].rolling(window=window_size).mean()
rolling_std_excess = df['excess_return'].rolling(window=window_size).std()

# 年化夏普比率 = (日均超额收益 / 日标准差) * sqrt(252)
# 解释：(日均超额收益 * 252) / (日标准差 * sqrt(252)) = (均值/标准差) * sqrt(252)
df['rolling_sharpe_ann'] = (rolling_mean_excess / rolling_std_excess) * np.sqrt(trading_days)

# ==========================================
# 4. 报告最后一个窗口的值；画出时间序列
# ==========================================
# 提取最后一个窗口的夏普值
rolling_sharpe_last = df['rolling_sharpe_ann'].iloc[-1]

# 绘制时间序列曲线
plt.figure(figsize=(12, 6))
plt.plot(df.index, df['rolling_sharpe_ann'], 
         label=f'{window_size}-Day Rolling Annualized Sharpe', 
         color='#1f77b4', linewidth=1.8)

# 添加零轴参考线
plt.axhline(y=0, color='black', linestyle='--', linewidth=0.8)

# 图表装饰
plt.title(f'Rolling Annualized Sharpe Ratio (Window={window_size}, Rf={rf_annual*100:.1f}%)', fontsize=14)
plt.xlabel('Date', fontsize=12)
plt.ylabel('Sharpe Ratio', fontsize=12)
plt.grid(True, linestyle=':', alpha=0.7)
plt.legend(fontsize=11)
plt.tight_layout()

# ==========================================
# 5. 保存图形并填充 result
# ==========================================
figure_path = 'rolling_sharpe_curve.png'
plt.savefig(figure_path, dpi=150)
plt.close()

# 按照输出契约构造字典
result = {
    'rolling_sharpe_last': rolling_sharpe_last,
    'figure_path': figure_path
}

# 控制台输出以供验证
print(f"最后一个{window_size}日滚动年化夏普比率: {rolling_sharpe_last:.4f}")
print("Result dictionary:", result)
```
