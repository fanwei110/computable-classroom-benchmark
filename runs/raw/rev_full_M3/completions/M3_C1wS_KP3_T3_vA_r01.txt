```python
import numpy as np

# ==================== 债券参数设定 ====================
face_value = 100         # 面值
coupon_rate = 0.046      # 票息率 4.6%
ytm = 0.053              # 当前收益率 5.3%
maturity = 7             # 期限 7年
delta_y = 0.008          # 收益率上升 80 个基点 (0.80%)

# 假设：债券按年付息，现金流采用复利贴现（符合《证券投资学》常规惯例）

# ==================== 现金流构建 ====================
# 1到7期现金流，前6期仅为票息，第7期为票息+面值
cash_flows = np.full(maturity, face_value * coupon_rate)
cash_flows[-1] += face_value

# 时期数列
periods = np.arange(1, maturity + 1)

# ==================== 1. 计算当前收益率下的利率敏感性 ====================
# 贴现因子
discount_factors = (1 + ytm) ** -periods

# 当前债券价格
price = np.sum(cash_flows * discount_factors)

# Macaulay 久期
mac_duration = np.sum(periods * cash_flows * discount_factors) / price

# 修正久期
mod_duration = mac_duration / (1 + ytm)

# 凸性
convexity = np.sum(periods * (periods + 1) * cash_flows / (1 + ytm)**(periods + 2)) / price

# ==================== 2. 估算收益率上升 80 个基点的价格影响 ====================
# 价格变动百分比泰勒展开近似公式: ΔP/P ≈ -修正久期 * Δy + 0.5 * 凸性 * (Δy)^2
pct_change = -mod_duration * delta_y + 0.5 * convexity * (delta_y ** 2)

# 跌幅取价格变动百分比近似值的绝对值（以小数形式表示，如 0.045 代表跌 4.5%）
price_drop_pct = abs(pct_change)

# ==================== 3. 把跌幅存入 result ====================
result = {
    'price_drop_pct': price_drop_pct
}

# 以下为课堂投屏辅助展示信息，非核心赋值逻辑
print(f"当前债券价格: {price:.4f} 元")
print(f"Macaulay 久期: {mac_duration:.4f} 年")
print(f"修正久期: {mod_duration:.4f}")
print(f"凸性: {convexity:.4f}")
print(f"收益率上升 80bps 估算跌幅: {price_drop_pct:.4%}")
print(f"\n最终输出字典 result: {result}")
```
