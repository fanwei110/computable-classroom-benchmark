import numpy as np

def calculate_bond_price_drop():
    # 债券基本参数
    face_value = 100.0          # 面值
    coupon_rate = 0.046          # 票息率 4.6%
    ytm_initial = 0.053         # 初始收益率 5.3%
    maturity = 7                # 期限 7 年
    delta_y = 0.0080             # 收益率上升 80 个基点 (0.80%)
    
    # 假设按年付息（此类简化题目通常默认为年付息频率）
    periods = np.arange(1, maturity + 1)
    
    # 生成现金流序列：前 6 年每年 4.6，第 7 年 104.6 (票息+面值)
    coupon_payment = face_value * coupon_rate
    cash_flows = np.full(maturity, coupon_payment)
    cash_flows[-1] += face_value
    
    # 计算初始价格 P0
    discount_factors_initial = (1 + ytm_initial) ** -periods
    price_initial = np.sum(cash_flows * discount_factors_initial)
    
    # 计算收益率上升后的新价格 P1
    ytm_new = ytm_initial + delta_y
    discount_factors_new = (1 + ytm_new) ** -periods
    price_new = np.sum(cash_flows * discount_factors_new)
    
    # 计算价格下跌的百分比 (用正数表示跌幅，符合“跌多少”的语境)
    # 变量名包含 pct，按量化惯例存储为小数形式（如 0.04 表示跌 4%）
    price_drop_pct = (price_initial - price_new) / price_initial
    
    # 按照输出契约存入字典
    result = {'price_drop_pct': price_drop_pct}
    
    return result

if __name__ == '__main__':
    result = calculate_bond_price_drop()
    print(f"价格大概下跌百分比: {result['price_drop_pct']:.4%}")
    print(f"result = {result}")
