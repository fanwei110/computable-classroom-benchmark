```python
import numpy as np
from scipy.stats import norm

# ===================== 已知条件 =====================
portfolio_value = 1_850_000  # 仓位：185万人民币
annual_vol = 0.218           # 年化波动率：21.8%

# ===================== 假设处理 =====================
# 假设一年有252个交易日（金融市场参数法VaR计算的标准假设）
# 假设短期收益均值为0（delta-normal法计算短期VaR的惯例）
trading_days_per_year = 252

# ===================== 步骤1：年化波动率换算到一日期限 =====================
# 波动率随时间的平方根缩放：sigma_daily = sigma_annual / sqrt(T)
daily_vol = annual_vol / np.sqrt(trading_days_per_year)

# ===================== 步骤2：套用正态分位数 =====================
# 使用scipy计算正态分布的单尾分位数（Z值）
z_95 = norm.ppf(0.95)
z_99 = norm.ppf(0.99)

# ===================== 步骤3：计算并缩放VaR，以人民币报告 =====================
# 参数法（Delta-Normal）VaR 公式：VaR = 仓位价值 × 波动率 × Z值

# (1) 95% 1天 VaR
var_95_1d = portfolio_value * daily_vol * z_95

# (2) 99% 10天 VaR
# 将一日期限波动率缩放到十日期限：sigma_10d = sigma_1d * sqrt(10)
vol_10d = daily_vol * np.sqrt(10)
var_99_10d = portfolio_value * vol_10d * z_99

# ===================== 步骤4：填充 result =====================
result = {
    'var_95_1d': var_95_1d,
    'var_99_10d': var_99_10d
}

# -------------------- 投屏展示 --------------------
print("="*50)
print("《证券投资学》- 在险价值 计算")
print("="*50)
print(f"仓位价值:     {portfolio_value:>14,.2f} 人民币")
print(f"年化波动率:   {annual_vol*100:>14.1f} %")
print(f"日均波动率:   {daily_vol*100:>14.4f} %")
print("-"*50)
print(f"95% 1天 VaR:  {result['var_95_1d']:>14,.2f} 人民币")
print(f"99% 10天 VaR: {result['var_99_10d']:>14,.2f} 人民币")
print("="*50)
```
