```python
import numpy as np

# 1. 定义输入参数
# 三只风险资产的年化波动率
vols = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr_matrix = np.array([
    [1.0, 0.21, -0.13],
    [0.21, 1.0, 0.37],
    [-0.13, 0.37, 1.0]
])

# 2. 构建协方差矩阵
# 协方差矩阵 = 对角波动率矩阵 @ 相关系数矩阵 @ 对角波动率矩阵
cov_matrix = np.outer(vols, vols) * corr_matrix

# 3. 计算全局最小方差组合 (MVP)
# 全1向量
ones = np.ones(3)

# 求协方差矩阵的逆矩阵
cov_inv = np.linalg.inv(cov_matrix)

# 计算MVP权重：w = (Sigma^-1 * 1) / (1^T * Sigma^-1 * 1)
mvp_numerator = cov_inv @ ones
mvp_denominator = ones @ cov_inv @ ones
mvp_weights = mvp_numerator / mvp_denominator

# 计算MVP的年化波动率：sigma = sqrt(1 / (1^T * Sigma^-1 * 1))
mvp_var = 1.0 / mvp_denominator
mvp_vol_annual = np.sqrt(mvp_var)

# 4. 按照输出契约存储结果
result = {
    'mvp_weights': mvp_weights.tolist(),
    'mvp_vol_annual': float(mvp_vol_annual)
}

# 打印结果以供验证
print(result)
```
