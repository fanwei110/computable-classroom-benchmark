```python
import numpy as np
from scipy.stats import norm

# ==========================================
# 1. 基础参数与假设设定
# ==========================================
position = 1850000.0       # 仓位：185万人民币
annual_vol = 0.218         # 年化波动率：21.8%

# 假设：一年有252个交易日（金融市场标准假设）
trading_days_per_year = 252

# ==========================================
# 2. 年化波动率换算到一日期限 (Step 1)
# ==========================================
# 根据参数法(Delta-Normal)的平方根法则，日波动率 = 年化波动率 / sqrt(252)
daily_vol = annual_vol / np.sqrt(trading_days_per_year)

# ==========================================
# 3. 套用正态分位数并计算95% 1天期VaR (Step 2 & 3)
# ==========================================
# 95%置信水平对应的正态分布分位数
z_95 = norm.ppf(0.95)
# 1天期限，时间乘数 sqrt(1) = 1
var_95_1d = position * z_95 * daily_vol * np.sqrt(1)

# ==========================================
# 4. 缩放到10天期限并计算99% 10天期VaR (Step 3)
# ==========================================
# 99%置信水平对应的正态分布分位数
z_99 = norm.ppf(0.99)
# 10天期限，时间乘数 sqrt(10)
var_99_10d = position * z_99 * daily_vol * np.sqrt(10)

# ==========================================
# 5. 存入结果字典 (Step 4)
# ==========================================
# 按照金融惯例，VaR通常以正数报告损失金额，保留两位小数（人民币单位：元）
result = {
    'var_95_1d': round(var_95_1d, 2),
    'var_99_10d': round(var_99_10d, 2)
}

# 控制台输出，便于课堂投屏展示
print("="*50)
print("《证券投资学》—— 参数法 在险价值计算")
print("="*50)
print(f"初始仓位:     {position/10000:.2f} 万人民币")
print(f"年化波动率:   {annual_vol*100:.1f}%")
print(f"日化波动率:   {daily_vol*100:.4f}% (假设一年252个交易日)")
print("-"*50)
print(f"95% 1天期 VaR: {result['var_95_1d']:>10.2f} 元人民币")
print(f"99% 10天期 VaR: {result['var_99_10d']:>10.2f} 元人民币")
print("="*50)
```
