```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.optimize import minimize

# 定义风险资产参数
mu = np.array([0.071, 0.124])  # 期望年收益
sigma = np.array([0.163, 0.289])  # 年化波动率
correlations = [0.15, 0.45, 0.75]  # 相关系数列表

# 计算协方差矩阵
def get_cov_matrix(sigma1, sigma2, rho):
    cov = rho * sigma1 * sigma2
    return np.array([[sigma1**2, cov], [cov, sigma2**2]])

# 计算组合的期望收益和波动率
def portfolio_stats(weights, mu, cov):
    port_mu = weights @ mu
    port_vol = np.sqrt(weights @ cov @ weights.T)
    return port_mu, port_vol

# 寻找最小方差组合
def find_min_variance_portfolio(mu, cov):
    n = len(mu)
    def objective(weights):
        return weights @ cov @ weights.T
    constraints = ({'type': 'eq', 'fun': lambda w: np.sum(w) - 1})
    bounds = tuple((None, None) for _ in range(n))  # 允许卖空
    result = minimize(objective, x0=np.ones(n)/n, bounds=bounds, constraints=constraints)
    return result.x

# 寻找给定期望收益下的最小波动率组合
def find_portfolio_for_target_return(target_return, mu, cov):
    n = len(mu)
    def objective(weights):
        return weights @ cov @ weights.T
    constraints = (
        {'type': 'eq', 'fun': lambda w: np.sum(w) - 1},
        {'type': 'eq', 'fun': lambda w: w @ mu - target_return}
    )
    bounds = tuple((None, None) for _ in range(n))  # 允许卖空
    result = minimize(objective, x0=np.ones(n)/n, bounds=bounds, constraints=constraints)
    return result.x

# 生成均值-方差前沿
def generate_efficient_frontier(mu, cov, n_points=100):
    min_var_weights = find_min_variance_portfolio(mu, cov)
    min_var_mu, min_var_vol = portfolio_stats(min_var_weights, mu, cov)

    target_returns = np.linspace(min_var_mu, max(mu), n_points)
    frontier_vols = []

    for target in target_returns:
        weights = find_portfolio_for_target_return(target, mu, cov)
        _, vol = portfolio_stats(weights, mu, cov)
        frontier_vols.append(vol)

    return target_returns, frontier_vols, min_var_mu, min_var_vol

# 主计算过程
result = {}

# 计算rho=0.45时的特定结果
rho_45 = 0.45
cov_45 = get_cov_matrix(sigma[0], sigma[1], rho_45)
mvp_weights_45 = find_min_variance_portfolio(mu, cov_45)
mvp_mu_45, mvp_vol_45 = portfolio_stats(mvp_weights_45, mu, cov_45)
result['mvp_vol_at_rho45'] = mvp_vol_45

target_return = 0.10
target_weights = find_portfolio_for_target_return(target_return, mu, cov_45)
_, target_vol = portfolio_stats(target_weights, mu, cov_45)
result['frontier_vol_at_target'] = target_vol

# 绘制图形
plt.figure(figsize=(10, 6))

for rho in correlations:
    cov = get_cov_matrix(sigma[0], sigma[1], rho)
    target_returns, frontier_vols, min_var_mu, min_var_vol = generate_efficient_frontier(mu, cov)

    plt.plot(frontier_vols, target_returns, label=f'ρ={rho}')
    plt.scatter(min_var_vol, min_var_mu, color='red', zorder=5)
    plt.text(min_var_vol, min_var_mu, f'MVP (ρ={rho})', fontsize=9,
             verticalalignment='bottom', horizontalalignment='right')

plt.xlabel('Annualized Volatility')
plt.ylabel('Expected Annual Return')
plt.title('Mean-Variance Efficient Frontier for Different Correlations')
plt.legend()
plt.grid(True)

# 保存图形
figure_path = 'efficient_frontier.png'
plt.savefig(figure_path)
result['figure_path'] = figure_path

# 输出结果
print(result)
```

这个脚本会：
1. 计算并绘制三种相关系数（0.15, 0.45, 0.75）下的均值-方差前沿曲线
2. 在每条曲线上标出最小方差组合点（MVP）
3. 计算相关系数为0.45时的最小方差组合波动率
4. 计算相关系数为0.45时目标期望收益10%下的最小波动率
5. 将所有结果存入result字典，并保存图形到文件

输出的result字典包含三个键值对：
- 'mvp_vol_at_rho45': 相关系数0.45时最小方差组合的年化波动率
- 'frontier_vol_at_target': 相关系数0.45时目标期望收益10%下的最小年化波动率
- 'figure_path': 保存的图形文件路径
