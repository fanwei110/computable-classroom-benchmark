以下是自包含的Python脚本，满足所有要求：

```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import newton
import os

def calculate_bond_price(yield_to_maturity, face_value, coupon_rate, years_to_maturity, periods_per_year=2):
    """
    计算债券精确价格
    """
    coupon_payment = face_value * coupon_rate / periods_per_year
    total_periods = years_to_maturity * periods_per_year
    discount_rate = yield_to_maturity / periods_per_year

    # 现金流折现
    price = sum([coupon_payment / (1 + discount_rate)**t for t in range(1, total_periods + 1)])
    price += face_value / (1 + discount_rate)**total_periods

    return price

def calculate_duration(yield_to_maturity, face_value, coupon_rate, years_to_maturity, periods_per_year=2):
    """
    计算麦考利久期
    """
    coupon_payment = face_value * coupon_rate / periods_per_year
    total_periods = years_to_maturity * periods_per_year
    discount_rate = yield_to_maturity / periods_per_year
    bond_price = calculate_bond_price(yield_to_maturity, face_value, coupon_rate, years_to_maturity, periods_per_year)

    # 计算久期
    duration = sum([t * coupon_payment / (1 + discount_rate)**t for t in range(1, total_periods + 1)])
    duration += total_periods * face_value / (1 + discount_rate)**total_periods
    duration /= bond_price
    duration /= periods_per_year  # 转换为年化久期

    return duration

def calculate_modified_duration(yield_to_maturity, face_value, coupon_rate, years_to_maturity, periods_per_year=2):
    """
    计算修正久期
    """
    macaulay_duration = calculate_duration(yield_to_maturity, face_value, coupon_rate, years_to_maturity, periods_per_year)
    modified_duration = macaulay_duration / (1 + yield_to_maturity / periods_per_year)
    return modified_duration

def main():
    # 债券参数
    face_value = 100
    coupon_rate = 0.046
    years_to_maturity = 7
    current_yield = 0.053
    yield_change_bp = 100  # 基点变动幅度（可调）

    # 计算当前价格
    current_price = calculate_bond_price(current_yield, face_value, coupon_rate, years_to_maturity)

    # 计算久期
    modified_duration = calculate_modified_duration(current_yield, face_value, coupon_rate, years_to_maturity)

    # 收益率范围
    yields = np.linspace(0.02, 0.09, 100)
    prices = np.array([calculate_bond_price(y, face_value, coupon_rate, years_to_maturity) for y in yields])

    # 久期近似线
    delta_y = 0.0001  # 小幅变动用于近似
    approx_prices = current_price * (1 - modified_duration * (yields - current_yield))

    # 计算收益率上升100bp后的精确价格
    yield_up100bp = current_yield + yield_change_bp / 10000
    price_at_up100bp = calculate_bond_price(yield_up100bp, face_value, coupon_rate, years_to_maturity)

    # 久期法估计的相对价格变化
    dur_approx_change_up100bp = -modified_duration * (yield_change_bp / 10000)

    # 绘图
    plt.figure(figsize=(10, 6))
    plt.plot(yields * 100, prices, label='精确价格-收益率曲线', color='blue')
    plt.plot(yields * 100, approx_prices, '--', label=f'久期近似 (Δ={yield_change_bp}bp)', color='red')
    plt.scatter(current_yield * 100, current_price, color='green', zorder=5, label='当前点')
    plt.xlabel('收益率 (%)')
    plt.ylabel('债券价格')
    plt.title(f'债券价格-收益率曲线 (票息={coupon_rate*100}%, 期限={years_to_maturity}年)')
    plt.legend()
    plt.grid(True)

    # 保存图片
    figure_path = 'bond_price_yield_curve.png'
    plt.savefig(figure_path)
    plt.close()

    # 准备结果
    result = {
        'price_at_up100bp': price_at_up100bp,
        'dur_approx_change_up100bp': dur_approx_change_up100bp,
        'figure_path': os.path.abspath(figure_path)
    }

    return result

# 运行并输出结果
if __name__ == "__main__":
    result = main()
    print(result)
```

这个脚本会：

1. 计算并绘制从2%到9%收益率的精确价格-收益率曲线
2. 在当前收益率(5.3%)附近叠加基于修正久期的线性近似
3. 计算收益率上升100个基点后的精确价格
4. 计算久期法估计的相对价格变化
5. 将所有结果存入`result`字典，并保存图片到当前目录

输出的`result`字典包含：
- `price_at_up100bp`: 收益率上升100bp后的精确价格
- `dur_approx_change_up100bp`: 久期法估计的相对价格变化
- `figure_path`: 保存的图片文件路径

注意：脚本中收益率变动幅度(100基点)可以通过修改`yield_change_bp`变量来调整。
