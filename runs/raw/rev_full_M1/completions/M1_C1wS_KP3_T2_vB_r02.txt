```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import newton

# 参数设定
face_value = 100
coupon_rate = 0.046
years_to_maturity = 7
ytm_current = 0.053
yield_grid = np.linspace(0.02, 0.09, 100)  # 2%到9%的收益率网格

# 计算债券精确价格
def bond_price(yield_to_maturity, face_value, coupon_rate, years_to_maturity):
    coupon = face_value * coupon_rate
    periods = years_to_maturity
    cash_flows = np.array([coupon] * periods + [face_value + coupon])
    discount_factors = (1 + yield_to_maturity) ** np.arange(1, periods + 2)
    price = np.sum(cash_flows / discount_factors)
    return price

# 计算久期（Macaulay Duration）
def macaulay_duration(yield_to_maturity, face_value, coupon_rate, years_to_maturity):
    coupon = face_value * coupon_rate
    periods = years_to_maturity
    cash_flows = np.array([coupon] * periods + [face_value + coupon])
    discount_factors = (1 + yield_to_maturity) ** np.arange(1, periods + 2)
    discounted_cash_flows = cash_flows / discount_factors
    price = np.sum(discounted_cash_flows)
    weights = discounted_cash_flows / price
    time_periods = np.arange(1, periods + 2)
    duration = np.sum(weights * time_periods)
    return duration

# 计算修正久期
def modified_duration(yield_to_maturity, face_value, coupon_rate, years_to_maturity):
    mac_dur = macaulay_duration(yield_to_maturity, face_value, coupon_rate, years_to_maturity)
    mod_dur = mac_dur / (1 + yield_to_maturity)
    return mod_dur

# 计算凸性
def convexity(yield_to_maturity, face_value, coupon_rate, years_to_maturity):
    coupon = face_value * coupon_rate
    periods = years_to_maturity
    cash_flows = np.array([coupon] * periods + [face_value + coupon])
    discount_factors = (1 + yield_to_maturity) ** np.arange(1, periods + 2)
    discounted_cash_flows = cash_flows / discount_factors
    price = np.sum(discounted_cash_flows)
    time_periods = np.arange(1, periods + 2)
    weighted_time = time_periods * (time_periods + 1) / (1 + yield_to_maturity)**2
    conv = np.sum(discounted_cash_flows * weighted_time) / price
    return conv

# 计算精确价格曲线
prices_exact = np.array([bond_price(y, face_value, coupon_rate, years_to_maturity) for y in yield_grid])

# 在当前YTM附近计算久期和凸性
mod_dur = modified_duration(ytm_current, face_value, coupon_rate, years_to_maturity)
conv = convexity(ytm_current, face_value, coupon_rate, years_to_maturity)

# 久期近似价格曲线
prices_approx = bond_price(ytm_current, face_value, coupon_rate, years_to_maturity) * (
    1 - mod_dur * (yield_grid - ytm_current) + 0.5 * conv * (yield_grid - ytm_current)**2
)

# 计算+100bp后的精确价格和久期法估计的相对变化
ytm_up100bp = ytm_current + 0.01
price_up100bp_exact = bond_price(ytm_up100bp, face_value, coupon_rate, years_to_maturity)
price_current = bond_price(ytm_current, face_value, coupon_rate, years_to_maturity)
dur_approx_change_up100bp = -mod_dur * 0.01 * 100  # 久期法估计的百分比变化

# 绘图
plt.figure(figsize=(10, 6))
plt.plot(yield_grid * 100, prices_exact, label='Exact Price', color='blue')
plt.plot(yield_grid * 100, prices_approx, label='Duration + Convexity Approximation', linestyle='--', color='red')
plt.scatter(ytm_current * 100, bond_price(ytm_current, face_value, coupon_rate, years_to_maturity),
            color='green', zorder=5, label='Current YTM (5.3%)')
plt.xlabel('Yield to Maturity (%)')
plt.ylabel('Bond Price')
plt.title('Bond Price vs Yield to Maturity')
plt.legend()
plt.grid(True)

# 保存图形
figure_path = 'bond_price_yield_curve.png'
plt.savefig(figure_path)
plt.close()

# 填充result字典
result = {
    'price_at_up100bp': price_up100bp_exact,
    'dur_approx_change_up100bp': dur_approx_change_up100bp,
    'figure_path': figure_path
}

# 输出结果以供验证
print(result)
```

这个脚本会：
1. 在2%到9%的收益率网格上计算债券的精确价格
2. 在5.3%附近叠加基于久期和凸性的近似价格曲线
3. 计算收益率增加100bp后的精确价格和久期法估计的相对变化
4. 保存价格-收益率曲线图并将结果存入result字典

输出的result字典包含：
- 'price_at_up100bp': 收益率增加100bp后的精确价格
- 'dur_approx_change_up100bp': 久期法估计的价格相对变化百分比
- 'figure_path': 保存的图形文件路径
