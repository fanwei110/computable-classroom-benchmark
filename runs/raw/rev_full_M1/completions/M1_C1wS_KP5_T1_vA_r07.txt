```python
import numpy as np
from scipy.stats import norm

# 输入参数
position = 1_850_000          # 头寸（人民币元）
annual_volatility = 0.218     # 年化波动率（21.8%）

# 1. 年化波动率换算到一日波动率
# 假设一年有 252 个交易日
daily_volatility = annual_volatility / np.sqrt(252)

# 2. 正态分位数
z_95 = norm.ppf(0.95)         # 95% 置信水平的 z 值
z_99 = norm.ppf(0.99)         # 99% 置信水平的 z 值

# 3. 计算 VaR
# 一日 VaR（95% 置信水平）
var_95_1d = position * daily_volatility * z_95

# 十日 VaR（99% 置信水平）
# 波动率按平方根时间法则缩放
ten_day_volatility = daily_volatility * np.sqrt(10)
var_99_10d = position * ten_day_volatility * z_99

# 4. 结果存入字典
result = {
    'var_95_1d': round(var_95_1d, 2),    # 保留两位小数
    'var_99_10d': round(var_99_10d, 2)
}

# 输出 result 以便课堂展示
print(result)
```
