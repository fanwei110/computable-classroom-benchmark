```python
import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt
import os

def black_scholes_delta(S, K, T, r, sigma, option_type='call'):
    """
    计算 Black-Scholes 模型下的期权 Delta
    :param S: 标的资产价格
    :param K: 行权价
    :param T: 剩余期限（年）
    :param r: 无风险利率
    :param sigma: 波动率
    :param option_type: 期权类型（'call' 或 'put'）
    :return: Delta 值
    """
    d1 = (np.log(S / K) + (r + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    if option_type == 'call':
        delta = norm.cdf(d1)
    else:
        delta = norm.cdf(d1) - 1
    return delta

def plot_delta_curves(K, T, r, sigmas, S_range=(70, 140), num_points=100):
    """
    绘制不同波动率下的 Delta 曲线
    :param K: 行权价
    :param T: 剩余期限（年）
    :param r: 无风险利率
    :param sigmas: 波动率列表
    :param S_range: 标的价格范围
    :param num_points: 价格点数
    :return: 保存的图片路径
    """
    S_values = np.linspace(S_range[0], S_range[1], num_points)
    plt.figure(figsize=(10, 6))

    for sigma in sigmas:
        delta_values = [black_scholes_delta(S, K, T, r, sigma) for S in S_values]
        plt.plot(S_values, delta_values, label=f'σ = {sigma*100:.1f}%')

    plt.title('European Call Option Delta vs Underlying Price')
    plt.xlabel('Underlying Price')
    plt.ylabel('Delta')
    plt.legend()
    plt.grid(True)

    # 保存图片
    figure_path = 'delta_curves.png'
    plt.savefig(figure_path)
    plt.close()

    return figure_path

# 参数设置
strike_price = 97.5
time_to_maturity = 0.58
risk_free_rate = 0.043
volatilities = [0.15, 0.276, 0.40]
underlying_price_for_delta = 110
volatility_for_delta = 0.276

# 计算指定点的 Delta
delta_at_s110 = black_scholes_delta(
    S=underlying_price_for_delta,
    K=strike_price,
    T=time_to_maturity,
    r=risk_free_rate,
    sigma=volatility_for_delta
)

# 绘制 Delta 曲线
figure_path = plot_delta_curves(
    K=strike_price,
    T=time_to_maturity,
    r=risk_free_rate,
    sigmas=volatilities
)

# 准备结果字典
result = {
    'delta_at_s110': delta_at_s110,
    'figure_path': os.path.abspath(figure_path)
}

# 输出结果（供验证）
print(result)
```

这个脚本会：
1. 计算在标的价格110、波动率27.6%时的Delta值
2. 绘制标的价格从70到140时，波动率为15%、27.6%、40%三种情况下的Delta曲线
3. 将图片保存为'delta_curves.png'文件
4. 将所有结果存入名为`result`的字典中，包含：
   - 'delta_at_s110': 在S=110,σ=27.6%时的Delta值
   - 'figure_path': 保存的图片文件的绝对路径
