```python
import numpy as np
from scipy.optimize import newton

# 债券参数
face_value = 100          # 面值
annual_coupon_rate = 0.046  # 年票息率 4.6%
ytm_initial = 0.053       # 初始到期收益率 5.3%
years_to_maturity = 7     # 剩余期限 7 年
coupon_frequency = 2      # 每年付息次数（半年付息）
yield_change_bp = 80      # 收益率变动 80 个基点

# 计算每期票息
coupon_per_period = (face_value * annual_coupon_rate) / coupon_frequency

# 定义债券定价函数（现金流贴现）
def bond_price(ytm, face_value, coupon, periods, freq):
    ytm_period = ytm / freq
    cash_flows = np.full(periods, coupon)
    cash_flows[-1] += face_value  # 最后一期加上面值
    periods_arr = np.arange(1, periods + 1)
    price = np.sum(cash_flows / ((1 + ytm_period) ** periods_arr))
    return price

# 计算初始价格
periods_total = years_to_maturity * coupon_frequency
initial_price = bond_price(ytm_initial, face_value, coupon_per_period, periods_total, coupon_frequency)

# 计算久期（Macaulay Duration）
def macaulay_duration(ytm, face_value, coupon, periods, freq):
    ytm_period = ytm / freq
    cash_flows = np.full(periods, coupon)
    cash_flows[-1] += face_value
    periods_arr = np.arange(1, periods + 1)
    pv_cash_flows = cash_flows / ((1 + ytm_period) ** periods_arr)
    weighted_cash_flows = pv_cash_flows * periods_arr
    duration = np.sum(weighted_cash_flows) / np.sum(pv_cash_flows)
    return duration / freq  # 转换为年化久期

duration = macaulay_duration(ytm_initial, face_value, coupon_per_period, periods_total, coupon_frequency)

# 计算修正久期
modified_duration = duration / (1 + ytm_initial / coupon_frequency)

# 计算凸性
def convexity(ytm, face_value, coupon, periods, freq):
    ytm_period = ytm / freq
    cash_flows = np.full(periods, coupon)
    cash_flows[-1] += face_value
    periods_arr = np.arange(1, periods + 1)
    pv_cash_flows = cash_flows / ((1 + ytm_period) ** periods_arr)
    weighted_cash_flows = pv_cash_flows * periods_arr * (periods_arr + 1)
    conv = np.sum(weighted_cash_flows) / (np.sum(pv_cash_flows) * (1 + ytm_period)**2)
    return conv / (freq ** 2)  # 转换为年化凸性

conv = convexity(ytm_initial, face_value, coupon_per_period, periods_total, coupon_frequency)

# 收益率变动（80 个基点 = 0.0080）
delta_y = yield_change_bp / 10000

# 使用久期和凸性估算价格变动百分比
price_drop_pct_approx = -modified_duration * delta_y + 0.5 * conv * (delta_y ** 2)

# 精确计算新价格
new_ytm = ytm_initial + delta_y
new_price = bond_price(new_ytm, face_value, coupon_per_period, periods_total, coupon_frequency)
price_drop_pct_exact = (new_price - initial_price) / initial_price * 100

# 结果存储
result = {
    'price_drop_pct': price_drop_pct_exact  # 使用精确计算的跌幅
}

print(result)
```
