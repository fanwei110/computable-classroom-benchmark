```python
import numpy as np
from scipy.optimize import newton

# 债券参数
face_value = 100.0          # 面值
coupon_rate = 0.046         # 票息率 4.6%
ytm_current = 0.053         # 当前收益率 5.3%
maturity_years = 7          # 期限 7 年
settlement_freq = 2         # 每年付息次数（半年付息）
yield_increase_bps = 80     # 收益率上升 80 个基点

# 计算每期票息
coupon_per_period = face_value * coupon_rate / settlement_freq

# 计算总期数
total_periods = maturity_years * settlement_freq

# 定义债券定价函数：现金流贴现之和
def bond_price(yield_per_period, face_value, coupon, periods):
    """
    计算债券价格
    yield_per_period: 每期收益率（年化收益率 / settlement_freq）
    face_value: 面值
    coupon: 每期票息
    periods: 总期数
    """
    cash_flows = np.full(periods, coupon)
    cash_flows[-1] += face_value  # 最后一期加上面值
    discount_factors = (1 + yield_per_period) ** np.arange(1, periods + 1)
    return np.sum(cash_flows / discount_factors)

# 当前每期收益率
ytm_per_period_current = ytm_current / settlement_freq

# 计算当前价格
price_current = bond_price(ytm_per_period_current, face_value, coupon_per_period, total_periods)

# 计算久期（Macaulay Duration）和修正久期
def macaulay_duration(yield_per_period, face_value, coupon, periods):
    """
    计算 Macaulay 久期
    """
    cash_flows = np.full(periods, coupon)
    cash_flows[-1] += face_value
    discount_factors = (1 + yield_per_period) ** np.arange(1, periods + 1)
    pv_cash_flows = cash_flows / discount_factors
    weighted_times = np.arange(1, periods + 1) * pv_cash_flows
    return np.sum(weighted_times) / np.sum(pv_cash_flows)

def modified_duration(macaulay_dur, yield_per_period):
    """
    计算修正久期
    """
    return macaulay_dur / (1 + yield_per_period)

macaulay_dur = macaulay_duration(ytm_per_period_current, face_value, coupon_per_period, total_periods)
mod_dur = modified_duration(macaulay_dur, ytm_per_period_current)

# 计算凸性
def convexity(yield_per_period, face_value, coupon, periods):
    """
    计算凸性
    """
    cash_flows = np.full(periods, coupon)
    cash_flows[-1] += face_value
    discount_factors = (1 + yield_per_period) ** np.arange(1, periods + 1)
    pv_cash_flows = cash_flows / discount_factors
    weighted_times = np.arange(1, periods + 1) * (np.arange(1, periods + 1) + 1) * pv_cash_flows
    return np.sum(weighted_times) / (np.sum(pv_cash_flows) * (1 + yield_per_period)**2)

conv = convexity(ytm_per_period_current, face_value, coupon_per_period, total_periods)

# 收益率变化（绝对值）
delta_yield = yield_increase_bps / 10000  # 80 基点 = 0.0080
delta_yield_per_period = delta_yield / settlement_freq

# 使用久期和凸性估算价格变化
price_drop_estimate = -mod_dur * delta_yield + 0.5 * conv * (delta_yield)**2
price_drop_pct_estimate = price_drop_estimate * 100

# 精确计算新价格（验证）
ytm_new = ytm_current + delta_yield
ytm_per_period_new = ytm_new / settlement_freq
price_new = bond_price(ytm_per_period_new, face_value, coupon_per_period, total_periods)
price_drop_pct_exact = ((price_current - price_new) / price_current) * 100

# 结果存储
result = {
    'price_drop_pct': price_drop_pct_exact  # 使用精确计算的跌幅百分比
}

# 输出结果（供调试参考）
print(f"当前价格: {price_current:.4f}")
print(f"新价格: {price_new:.4f}")
print(f"精确跌幅: {price_drop_pct_exact:.4f}%")
print(f"久期-凸性估算跌幅: {price_drop_pct_estimate:.4f}%")
print("结果已存入 result 字典，键名为 'price_drop_pct'")
```
