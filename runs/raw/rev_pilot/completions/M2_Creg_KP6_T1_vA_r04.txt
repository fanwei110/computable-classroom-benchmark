```python
import numpy as np
import pandas as pd
import sys

# ============================================================
# Part 1: 读取课程数据快照 "fund" 列的日收益，计算年化夏普比率
# ============================================================
# 假设数据文件名为 fund_data.csv，其中包含 'fund' 列（日收益率，如 0.001 表示 0.1%）
try:
    data = pd.read_csv('fund_data.csv')
    daily_returns = data['fund'].values
except Exception as e:
    print("错误：无法读取 'fund_data.csv'。请确保文件存在且包含 'fund' 列。")
    sys.exit(1)

# 无风险利率（年化）
rf_annual = 0.021

# 日化无风险利率（简单折算，与业内惯例一致）
rf_daily = rf_annual / 252

# 超额日收益
excess_daily = daily_returns - rf_daily

# 年化夏普比率：√252 × 日均超额收益 / 日收益标准差（样本标准差，ddof=1）
sharpe_annual = np.sqrt(252) * np.mean(excess_daily) / np.std(daily_returns, ddof=1)

# ============================================================
# Part 2: Brinson 归因分析 —— 配置、选择与交互效应
# ============================================================
# 组合权重与行业收益
wP = np.array([0.45, 0.35, 0.20])
rP = np.array([0.083, 0.021, -0.014])

# 基准权重与行业收益
wB = np.array([0.40, 0.40, 0.20])
rB = np.array([0.067, 0.034, -0.009])

# 基准总收益
R_B = np.sum(wB * rB)

# 配置效应 (Allocation Effect): Σ (wP - wB) * (rB - R_B)
allocation_effect = np.sum((wP - wB) * (rB - R_B))

# 选择效应 (Selection Effect): Σ wB * (rP - rB)
selection_effect = np.sum(wB * (rP - rB))

# 交互效应 (Interaction Effect): Σ (wP - wB) * (rP - rB)
interaction_effect = np.sum((wP - wB) * (rP - rB))

# 校验总和是否等于超额收益（仅用于调试，不影响输出）
# excess_return = np.sum(wP * rP) - R_B
# print(f"Excess: {excess_return:.6f}, Sum: {allocation_effect + selection_effect + interaction_effect:.6f}")

# ============================================================
# 输出结果
# ============================================================
result = {
    'sharpe_annual': sharpe_annual,
    'allocation_effect': allocation_effect,
    'selection_effect': selection_effect,
    'interaction_effect': interaction_effect
}

print(result)
```
