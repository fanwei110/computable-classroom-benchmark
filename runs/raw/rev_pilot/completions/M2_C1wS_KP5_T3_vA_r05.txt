```python
import numpy as np
from scipy.stats import norm

# 给定参数
annual_vol = 0.24          # 年化波动率
position = 2_700_000        # 头寸金额（元）
confidence = 0.95           # 置信水平
trading_days = 252          # 假设一年252个交易日

# 1. 年化波动率换算为一日波动率
daily_vol = annual_vol / np.sqrt(trading_days)

# 2. 95% 正态分位数（标准正态分布的5%分位点）
z_score = norm.ppf(1 - confidence)  # 对应 -1.64485...

# 3. 计算 VaR（损失为正数）
var_95_1d = abs(position * daily_vol * z_score)

# 结果存入字典
result = {
    'var_95_1d': var_95_1d
}

# 输出结果
print(result)
```
