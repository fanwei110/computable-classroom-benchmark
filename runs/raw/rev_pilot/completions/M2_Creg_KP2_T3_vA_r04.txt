```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy import optimize

# ------------------------------
# 1. 给定参数
# ------------------------------
beta = 1.42
market_return_monthly = -0.058          # 市场下跌 5.8%
risk_free_annual = 0.047                # 年化无风险利率 4.7%

# ------------------------------
# 2. 将年化无风险利率转换为月利率（简单折算）
# ------------------------------
risk_free_monthly = risk_free_annual / 12.0

# ------------------------------
# 3. 利用 CAPM 计算预期月度收益
#    E(R) = Rf + β * (Rm - Rf)
# ------------------------------
capm_return_monthly = (
    risk_free_monthly
    + beta * (market_return_monthly - risk_free_monthly)
)

# ------------------------------
# 4. 按要求输出到字典 result
# ------------------------------
result = {
    'capm_return_monthly': capm_return_monthly
}

# 打印以验证（脚本运行时输出确定可复现）
print(result)
```
