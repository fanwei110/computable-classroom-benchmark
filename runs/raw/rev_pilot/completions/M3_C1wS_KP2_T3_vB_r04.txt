```python
import numpy as np

# ==========================================
# 《证券投资学》课堂实时编程演示
# 概念：CAPM 与证券市场线
# ==========================================

# --- 1. 输入参数 ---
beta = 1.42                     # 股票的 Beta 系数
market_return_monthly = -0.058  # 大盘上月收益率（跌 5.8%，计为 -0.058）
rf_annual = 0.047               # 年化无风险利率 4.7%

# --- 2. 把年无风险利率折算为月利率 ---
# 假设采用复利折算方式（行业内更严谨的折算方法）：
# (1 + rf_annual) = (1 + rf_monthly)^12
# 推导得：rf_monthly = (1 + rf_annual)^(1/12) - 1
rf_monthly = (1 + rf_annual) ** (1/12) - 1

# --- 3. 在月度层面套用 CAPM ---
# CAPM 公式：E(Ri) = Rf + β * [E(Rm) - Rf]
# 这里直接使用大盘上月的实际收益率作为 E(Rm) 的实现值代入计算
market_risk_premium_monthly = market_return_monthly - rf_monthly
capm_return_monthly = rf_monthly + beta * market_risk_premium_monthly

# --- 4. 填充 result 字典 ---
result = {
    'capm_return_monthly': capm_return_monthly
}

# --- 5. 课堂演示输出 ---
print("="*50)
print("CAPM 证券市场线：月度期望收益计算")
print("="*50)
print(f"已知条件：")
print(f"  Beta (β)            = {beta}")
print(f"  大盘月收益 (Rm)     = {market_return_monthly:.4f} ({market_return_monthly*100:.2f}%)")
print(f"  年化无风险利率 (Rf) = {rf_annual:.4f} ({rf_annual*100:.2f}%)")
print("-" * 50)
print(f"计算步骤：")
print(f"  1. 月度无风险利率折算 = (1 + {rf_annual})^(1/12) - 1 = {rf_monthly:.6f}")
print(f"  2. 月度市场风险溢价   = {market_return_monthly:.6f} - {rf_monthly:.6f} = {market_risk_premium_monthly:.6f}")
print(f"  3. CAPM 月度期望收益  = {rf_monthly:.6f} + {beta} × {market_risk_premium_monthly:.6f} = {capm_return_monthly:.6f}")
print("="*50)
print(f"结论：按 CAPM，该股票上月理论收益率应为 {capm_return_monthly*100:.2f}%")
print(f"\n最终输出字典 result: {result}")
```
