```python
import numpy as np
import pandas as pd

# 债券参数
face_value = 100.0          # 面值
coupon_rate = 0.046         # 票息率（年化）
years = 7                   # 剩余期限（年）
ytm = 0.053                 # 到期收益率（年化）
coupon = face_value * coupon_rate  # 每年票息

# 现金流时间点（年）
t = np.arange(1, years + 1)
# 各期现金流：前n-1年为票息，最后一年为票息+本金
cash_flows = np.full(years, coupon)
cash_flows[-1] += face_value

# 贴现因子
discount_factors = (1 + ytm) ** t
# 各期现金流现值
pv_cash_flows = cash_flows / discount_factors

# 价格
price = np.sum(pv_cash_flows)

# 权重（各期现值占总价格比重）
weights = pv_cash_flows / price

# 麦考利久期（加权平均时间，以年为单位）
macaulay_duration = np.sum(t * weights)

# 修正久期（考虑付息频率为1时：修正久期 = 麦考利久期 / (1 + ytm)）
modified_duration = macaulay_duration / (1 + ytm)

# 凸性：公式 convexity = (1 / price) * sum( t * (t+1) * CF_t / (1+ytm)^(t+2) )
convexity = np.sum(t * (t + 1) * cash_flows / (1 + ytm) ** (t + 2)) / price

# 将结果存入字典
result = {
    'price': price,
    'macaulay_duration_years': macaulay_duration,
    'modified_duration_years': modified_duration,
    'convexity': convexity
}

# 打印结果以便查看
print("债券计算结果：")
for key, value in result.items():
    print(f"{key}: {value:.6f}")

# result 字典即为最终输出，供后续使用
```
