```python
import numpy as np
from scipy.stats import norm

# ==========================================
# 1. 参数与假设设定
# ==========================================
annual_vol = 0.24          # 年化波动率 24%
position = 2_700_000       # 头寸金额 270万元
confidence_level = 0.95    # 95% 置信水平
trading_days = 252         # 假设：一年按252个交易日计算（中国A股市场通行假设）

# ==========================================
# 2. 把年化波动率换算到一日期限
# ==========================================
# Delta-Normal法下，波动率随时间的平方根缩放：sigma_daily = sigma_annual / sqrt(T)
daily_vol = annual_vol / np.sqrt(trading_days)

# ==========================================
# 3. 套用 95% 正态分位数
# ==========================================
# 获取标准正态分布下 95% 对应的分位数（即 Z 值）
z_score_95 = norm.ppf(confidence_level)

# ==========================================
# 4. 计算 95% 一日 VaR
# ==========================================
# VaR = 头寸绝对金额 × Z值 × 日波动率
var_95_1d = position * z_score_95 * daily_vol

# ==========================================
# 5. 按契约封装输出
# ==========================================
result = {
    'var_95_1d': var_95_1d
}

# --- 课堂投屏展示辅助信息（便于师生核对中间步骤） ---
print(f"【计算步骤拆解】")
print(f"1. 日波动率 = {annual_vol:.2%} / √{trading_days} = {daily_vol:.6%}")
print(f"2. 95%正态分位数(Z值) = {z_score_95:.4f}")
print(f"3. 95%一日VaR = {position:,.0f} × {z_score_95:.4f} × {daily_vol:.6%} = {var_95_1d:,.2f} 元")
print(f"\n【最终输出结果 result】\n{result}")
```
