```python
import numpy as np
from scipy import stats

# --- 给定参数 ---
annual_vol = 0.24        # 年化波动率 24%
position = 2700000      # 头寸金额 2,700,000 元
confidence_level = 0.95  # 95% 置信度
trading_days = 252       # 假设一年有 252 个交易日

# --- 计算步骤 ---
# 1. 将年化波动率转换为日波动率
# 日波动率 = 年化波动率 / sqrt(交易日天数)
daily_vol = annual_vol / np.sqrt(trading_days)

# 2. 计算头寸的日标准差（1日基础风险）
daily_std = position * daily_vol

# 3. 获取正态分布下 95% 置信度对应的分位数 (Z-score)
z_score = stats.norm.ppf(confidence_level)

# 4. 计算 95% 一日 VaR
var_95_1d = daily_std * z_score

# --- 按照输出契约构建结果字典 ---
result = {
    'var_95_1d': round(var_95_1d, 2)
}

# 打印结果以供验证
print(f"95% 一日 VaR: {result['var_95_1d']} 元")
```
