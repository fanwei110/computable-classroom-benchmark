```python
import numpy as np
from scipy.stats import norm

# 给定参数
annual_vol = 0.24          # 年化波动率
position = 2_700_000       # 头寸金额
confidence = 0.95          # 置信水平
trading_days = 252         # 一年交易日数

# 1. 年化波动率换算为日波动率
daily_vol = annual_vol / np.sqrt(trading_days)

# 2. 95%置信水平对应的正态分位数（左尾5%分位点，取绝对值表示损失大小）
z_score = abs(norm.ppf(1 - confidence))  # norm.ppf(0.05) = -1.64485...

# 3. 计算 95% 一日 VaR
var_95_1d = position * daily_vol * z_score

# 存储结果
result = {
    'var_95_1d': var_95_1d
}

# 输出结果
if __name__ == "__main__":
    print(result)
```
