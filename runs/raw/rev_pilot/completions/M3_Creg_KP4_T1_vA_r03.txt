```python
import numpy as np
from scipy.stats import norm

# --- 输入参数 ---
S = 103.7       # 标的资产现价
K = 97.5        # 行权价
sigma = 0.276   # 隐含波动率（年化）
r = 0.043       # 无风险利率（年化）
T = 0.58        # 剩余期限（年）

# --- Black-Scholes-Merton 模型计算 ---
# 计算d1和d2
d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
d2 = d1 - sigma * np.sqrt(T)

# 计算欧式看涨期权价格
# C = S * N(d1) - K * exp(-rT) * N(d2)
call_price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)

# 计算Delta
# 对于无分红股票的看涨期权，Delta = N(d1)
call_delta = norm.cdf(d1)

# 计算Vega
# Vega = S * sqrt(T) * N'(d1)，表示波动率变化1单位（如100%）时期权价格的变化量
# 注：实务中常按波动率变化1%（0.01）来报价Vega，若需1%的Vega，可乘以0.01
call_vega = S * np.sqrt(T) * norm.pdf(d1)

# --- 输出契约 ---
result = {
    'call_price': call_price,
    'call_delta': call_delta,
    'call_vega': call_vega
}

# 打印结果以供验证
if __name__ == "__main__":
    print(f"Call Price: {result['call_price']:.4f}")
    print(f"Call Delta: {result['call_delta']:.4f}")
    print(f"Call Vega:  {result['call_vega']:.4f}")
```
