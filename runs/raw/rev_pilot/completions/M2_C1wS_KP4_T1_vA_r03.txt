import numpy as np
from scipy.stats import norm

def black_scholes_call(S, K, T, r, sigma, q=0.0):
    """
    计算欧式看涨期权的 Black-Scholes 价格、Delta 和 Vega。
    
    参数:
    S : 标的现价
    K : 行权价
    T : 剩余到期时间（年）
    r : 无风险利率（连续复利）
    sigma : 隐含波动率（小数形式，例如 0.276 表示 27.6%）
    q : 连续股息率，默认为 0
    
    返回:
    price : 期权价格
    delta : Delta
    vega  : Vega（价格对波动率的偏导数，即波动率变动 1 单位时的价格变动）
    """
    # 1. 计算 d1 与 d2
    d1 = (np.log(S / K) + (r - q + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    
    # 2. 计算期权价格与 delta
    call_price = S * np.exp(-q * T) * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
    call_delta = np.exp(-q * T) * norm.cdf(d1)
    
    # 3. 计算 vega（注意：这里给出的是标准导数，波动率每变动 1（即 100%）时价格的变化）
    #    实务中常说的“vega”指波动率变动 1 个百分点（0.01）时的价格变化，即需要将该值乘以 0.01
    call_vega = S * np.exp(-q * T) * norm.pdf(d1) * np.sqrt(T)
    
    return call_price, call_delta, call_vega

if __name__ == "__main__":
    # 给定参数
    S = 103.7        # 标的现价
    K = 97.5         # 行权价
    sigma = 0.276    # 隐含波动率 27.6%
    r = 0.043        # 无风险利率 4.3%
    T = 0.58         # 剩余到期时间 0.58 年
    q = 0.0          # 不分红
    
    # 计算
    price, delta, vega = black_scholes_call(S, K, T, r, sigma, q)
    
    # 按输出契约构建字典，键名严格匹配
    result = {
        'call_price': price,
        'call_delta': delta,
        'call_vega': vega
    }
    
    # 打印结果，便于课堂投屏查看
    for key, value in result.items():
        print(f"{key}: {value:.6f}")
