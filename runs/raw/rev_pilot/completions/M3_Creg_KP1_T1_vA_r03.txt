```python
import numpy as np

# ==================== 输入参数 ====================
# 三只风险资产的年化波动率
vols = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corrs = np.array([
    [1.0,  0.21, -0.13],
    [0.21, 1.0,   0.37],
    [-0.13, 0.37, 1.0 ]
])

# ==================== 协方差矩阵构建 ====================
# cov(i,j) = corr(i,j) * vol_i * vol_j
cov_matrix = np.outer(vols, vols) * corrs

# ==================== 全局最小方差组合 (MVP) 计算 ====================
# 全局最小方差组合权重公式: w_mvp = (Sigma^-1 * 1) / (1^T * Sigma^-1 * 1)
ones = np.ones(len(vols))
inv_cov_matrix = np.linalg.inv(cov_matrix)

# 计算公式中的分子和分母
numerator = inv_cov_matrix @ ones
denominator = ones.T @ inv_cov_matrix @ ones

# 计算 MVP 权重
mvp_weights = numerator / denominator

# 计算 MVP 方差与年化波动率
# MVP方差公式: sigma_mvp^2 = 1 / (1^T * Sigma^-1 * 1)
mvp_var = 1.0 / denominator
mvp_vol_annual = np.sqrt(mvp_var)

# ==================== 结果输出 ====================
result = {
    'mvp_weights': mvp_weights.tolist(),
    'mvp_vol_annual': mvp_vol_annual
}

# 打印结果以便验证
print(result)
```
