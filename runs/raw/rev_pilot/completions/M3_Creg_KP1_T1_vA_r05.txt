```python
import numpy as np

# 1. 定义输入参数
vols = np.array([0.187, 0.243, 0.312])
corrs = np.array([
    [1.0,   0.21, -0.13],
    [0.21,  1.0,   0.37],
    [-0.13, 0.37,  1.0 ]
])

# 2. 构建协方差矩阵
# cov(i, j) = corr(i, j) * vol_i * vol_j
cov_matrix = np.outer(vols, vols) * corrs

# 3. 计算全局最小方差组合 (MVP)
# 权重公式: w_mvp = (Sigma^-1 * 1) / (1^T * Sigma^-1 * 1)
# 方差公式: var_mvp = 1 / (1^T * Sigma^-1 * 1)
ones = np.ones(3)
cov_inv = np.linalg.inv(cov_matrix)

# 计算分母: 1^T * Sigma^-1 * 1
ones_T_Sigma_inv_ones = ones.T @ cov_inv @ ones

# 计算 MVP 权重
mvp_weights = (cov_inv @ ones) / ones_T_Sigma_inv_ones

# 计算 MVP 年化波动率
mvp_var = 1.0 / ones_T_Sigma_inv_ones
mvp_vol_annual = np.sqrt(mvp_var)

# 4. 按照输出契约存入字典
result = {
    'mvp_weights': mvp_weights.tolist(),  # 转为list以增加可读性与兼容性
    'mvp_vol_annual': float(mvp_vol_annual)
}

# 打印结果以供验证
print(result)
```
