```python
import numpy as np
import pandas as pd

# ================= 假设处理说明 =================
# 1. 年化夏普比率计算中，假设每年有252个交易日，日无风险利率采用 r_annual / 252 计算。
# 2. 日收益率标准差采用样本标准差（ddof=1）。
# 3. “在基金收益中计入无风险利率”理解为计算超额收益：R_excess = R_fund - R_f_daily。
# 4. 为确保脚本完全自包含与可复现，若当前目录下无快照CSV，将自动生成符合正态分布的模拟数据。
# =================================================

# --- 第一步：读取快照 CSV 并处理基金收益 ---
try:
    # 尝试读取课堂提供的快照数据
    df = pd.read_csv('snapshot.csv')
    daily_returns = df['fund'].dropna().values
except FileNotFoundError:
    # 容错：若无CSV文件则生成模拟数据，确保脚本独立可运行
    print("未找到 snapshot.csv，已启用内部模拟数据以保证程序运行...")
    np.random.seed(42)
    # 模拟日均收益约0.03%，年化约7.5%；日波动约1.2%，年化约19%
    daily_returns = np.random.normal(0.0003, 0.012, 500)

# --- 第二步：计算年化夏普比率 ---
rf_annual = 0.021
trading_days = 252
rf_daily = rf_annual / trading_days

# 在基金收益中扣除无风险利率，得到超额收益
excess_daily_returns = daily_returns - rf_daily

# 计算日夏普比率并年化
mean_excess_daily = np.mean(excess_daily_returns)
std_excess_daily = np.std(excess_daily_returns, ddof=1) # 使用样本标准差

sharpe_daily = mean_excess_daily / std_excess_daily
sharpe_annual = sharpe_daily * np.sqrt(trading_days)

# --- 第三步：业绩归因（Brimson模型） ---
# 组合权重与收益
W_p = np.array([0.45, 0.35, 0.20])
R_p = np.array([0.083, 0.021, -0.014])

# 基准权重与收益
W_b = np.array([0.40, 0.40, 0.20])
R_b = np.array([0.067, 0.034, -0.009])

# 配置效应 (Allocation Effect)
allocation_effect = np.sum((W_p - W_b) * R_b)

# 选择效应 (Selection Effect)
selection_effect = np.sum(W_b * (R_p - R_b))

# 交互效应 (Interaction Effect)
interaction_effect = np.sum((W_p - W_b) * (R_p - R_b))

# --- 第四步：填充 result ---
result = {
    'sharpe_annual': sharpe_annual,
    'allocation_effect': allocation_effect,
    'selection_effect': selection_effect,
    'interaction_effect': interaction_effect
}

# 课堂投屏展示输出
if __name__ == '__main__':
    print("="*40)
    print("《证券投资学》课堂实时计算结果")
    print("="*40)
    print(f"年化夏普比率: {result['sharpe_annual']:.4f}")
    print(f"配置效应:     {result['allocation_effect']:.4f} ({result['allocation_effect']*100:.2f}%)")
    print(f"选择效应:     {result['selection_effect']:.4f} ({result['selection_effect']*100:.2f}%)")
    print(f"交互效应:     {result['interaction_effect']:.4f} ({result['interaction_effect']*100:.2f}%)")
    print("="*40)
```
