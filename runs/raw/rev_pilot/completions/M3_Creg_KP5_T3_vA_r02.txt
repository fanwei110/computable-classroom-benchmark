```python
import numpy as np
from scipy.stats import norm

# 已知参数
annual_vol = 0.24         # 年化波动率 24%
position = 2_700_000      # 头寸金额 2,700,000 元
trading_days = 252        # 一年的标准交易日天数（金融行业标准假设）

# 1. 将年化波动率转换为一日波动率
# 公式: sigma_daily = sigma_annual / sqrt(T)
daily_vol = annual_vol / np.sqrt(trading_days)

# 2. 计算 95% 置信水平对应的正态分布分位数 (Z-score)
confidence_level = 0.95
z_score = norm.ppf(confidence_level)

# 3. 计算 95% 一日 VaR
# 假设一日预期收益为0（这是计算短期 VaR 的标准做法）
# 公式: VaR = Position * Z * sigma_daily
var_95_1d = position * daily_vol * z_score

# 4. 按照输出契约存入字典
result = {
    'var_95_1d': var_95_1d
}

print(result)
```
