```python
import numpy as np

# 给定年化波动率
vols = np.array([0.187, 0.243, 0.312])

# 给定相关系数矩阵
corr = np.array([
    [1.0, 0.21, -0.13],
    [0.21, 1.0, 0.37],
    [-0.13, 0.37, 1.0]
])

# 计算协方差矩阵: Sigma = diag(vol) @ Corr @ diag(vol)
cov_matrix = np.outer(vols, vols) * corr

# 全1向量
ones = np.ones(3)

# 计算协方差矩阵的逆
cov_inv = np.linalg.inv(cov_matrix)

# 计算全局最小方差组合 (MVP) 的权重
# 公式: w_mvp = (Sigma^-1 * 1) / (1^T * Sigma^-1 * 1)
ones_T_cov_inv = ones.T @ cov_inv
denominator = ones_T_cov_inv @ ones
mvp_weights = ones_T_cov_inv / denominator

# 计算全局最小方差组合的年化波动率
# 公式: var_mvp = 1 / (1^T * Sigma^-1 * 1)
mvp_var = 1.0 / denominator
mvp_vol = np.sqrt(mvp_var)

# 按照输出契约存入字典
result = {
    'mvp_weights': mvp_weights.tolist(),
    'mvp_vol_annual': float(mvp_vol)
}

if __name__ == '__main__':
    # 打印结果以供验证
    print("全局最小方差组合结果：")
    print(f"权重: {result['mvp_weights']}")
    print(f"年化波动率: {result['mvp_vol_annual']:.4%}")
```
