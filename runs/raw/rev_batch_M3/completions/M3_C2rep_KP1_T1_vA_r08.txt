```python
import numpy as np

# ==========================================
# 1. 输入参数
# ==========================================
# 三只风险资产的年化波动率
vols = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr = np.array([
    [1.0,   0.21, -0.13],
    [0.21,  1.0,   0.37],
    [-0.13, 0.37,  1.0 ]
])

# ==========================================
# 2. 构造协方差矩阵
# ==========================================
# 协方差矩阵 Σ = diag(σ) @ Corr @ diag(σ)
# 等价于对应元素相乘：Σ_ij = σ_i * σ_j * ρ_ij
cov_matrix = np.outer(vols, vols) * corr

# ==========================================
# 3. 求解全局最小方差组合 (MVP) 权重
# ==========================================
# 允许卖空且资金全部投出（满仓约束 Σw = 1），MVP具有闭式解：
# w_mvp = Σ^(-1) * 1 / (1' * Σ^(-1) * 1)
ones = np.ones(3)
cov_inv = np.linalg.inv(cov_matrix)

# 计算未归一化的权重
w_raw = cov_inv @ ones

# 归一化使权重和严格为1
mvp_weights = w_raw / np.sum(w_raw)

# ==========================================
# 4. 计算组合年化波动率
# ==========================================
# 组合方差 = w'Σw，波动率 = sqrt(w'Σw)
mvp_var = mvp_weights.T @ cov_matrix @ mvp_weights
mvp_vol_annual = np.sqrt(mvp_var)

# ==========================================
# 5. 按要求键名填充结果字典
# ==========================================
result = {
    'mvp_weights': mvp_weights.tolist(),
    'mvp_vol_annual': float(mvp_vol_annual)
}

# 课堂演示打印（可选，方便投屏查看）
if __name__ == "__main__":
    print("全局最小方差组合（MVP）计算结果：")
    print(f"权重: {result['mvp_weights']}")
    print(f"年化波动率: {result['mvp_vol_annual']:.4%}")
```
