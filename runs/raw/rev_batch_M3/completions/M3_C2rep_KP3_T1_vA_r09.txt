```python
import numpy as np

# ====================
// 债券基本参数设置
# ====================
face_value = 100.0          # 面值
coupon_rate = 0.046         # 票息率 4.6%
years_to_maturity = 7       # 期限 7 年
ytm = 0.053                 # 到期收益率 5.3%
frequency = 1               // 题目未指明付息频率，假设为每年付息1次（常规设定）

# ====================
// 1. 计算现金流与价格
# ====================
# 期数序列
periods = np.arange(1, years_to_maturity * frequency + 1)

# 每期票息
coupon_payment = face_value * coupon_rate / frequency

# 各期现金流（到期一次还本，最后一期加上面值）
cash_flows = np.full(len(periods), coupon_payment)
cash_flows[-1] += face_value

# 每期贴现率
discount_rate_per_period = ytm / frequency

# 贴现因子与现金流现值
discount_factors = (1 + discount_rate_per_period) ** (-periods)
pv_cash_flows = cash_flows * discount_factors

# 债券价格
price = np.sum(pv_cash_flows)

# ====================
// 2. 计算麦考利久期与修正久期
# ====================
# 麦考利久期（期数）
macaulay_duration_periods = np.sum(periods * pv_cash_flows) / price

# 麦考利久期（年化）
macaulay_duration_years = macaulay_duration_periods / frequency

// 修正久期（年化）
modified_duration_years = macaulay_duration_years / (1 + discount_rate_per_period)

# ====================
// 3. 计算凸性
# ====================
# 凸性公式：1/P * Σ [ t(t+1) * CF_t / (1+y)^t ] / (1+y)^2
# 注意：t 为期数，y 为期利率
convexity_periods = np.sum(periods * (periods + 1) * pv_cash_flows) / (price * (1 + discount_rate_per_period)**2)

// 年化凸性（需除以付息频率的平方）
convexity = convexity_periods / (frequency ** 2)

# ====================
// 4. 存入结果字典
# ====================
result = {
    'price': price,
    'macaulay_duration_years': macaulay_duration_years,
    'modified_duration_years': modified_duration_years,
    'convexity': convexity
}

// 课堂投屏打印展示
print("--- 《证券投资学》课堂计算结果 ---")
print(f"债券参数: 面值={face_value}, 票息率={coupon_rate*100}%, 期限={years_to_maturity}年, YTM={ytm*100}%")
print(f"价格 (Price)              : {result['price']:.6f}")
print(f"麦考利久期 (Macaulay Dur) : {result['macaulay_duration_years']:.6f} 年")
print(f"修正久期 (Modified Dur)   : {result['modified_duration_years']:.6f} 年")
print(f"凸性 (Convexity)          : {result['convexity']:.6f}")
```
