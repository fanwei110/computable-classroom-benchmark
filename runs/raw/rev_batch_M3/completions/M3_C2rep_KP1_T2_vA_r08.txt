```python
import numpy as np
import matplotlib.pyplot as plt

# ==================== 1. 参数设定 ====================
mu = np.array([0.071, 0.124])        # 期望年收益
sigma = np.array([0.163, 0.289])     # 年化波动率
rhos = [0.15, 0.45, 0.75]           # 给定的相关系数
target_ret = 0.10                    # 目标期望收益

# 结果字典
result = {}

# ==================== 2. 计算 rho=0.45 时的指定指标 ====================
rho_45 = 0.45
cov_12_45 = rho_45 * sigma[0] * sigma[1]
cov_matrix_45 = np.array([
    [sigma[0]**2, cov_12_45],
    [cov_12_45, sigma[1]**2]
])

# 最小方差组合权重 (满仓约束 w1 + w2 = 1)
den_mvp = sigma[0]**2 + sigma[1]**2 - 2 * rho_45 * sigma[0] * sigma[1]
w1_mvp_45 = (sigma[1]**2 - rho_45 * sigma[0] * sigma[1]) / den_mvp
w_mvp_45 = np.array([w1_mvp_45, 1 - w1_mvp_45])

# 计算最小方差组合的年化波动率
mvp_vol_45 = np.sqrt(w_mvp_45 @ cov_matrix_45 @ w_mvp_45)
result['mvp_vol_at_rho45'] = float(mvp_vol_45)

# 目标期望收益 10% 下的最优权重
# 0.10 = w1 * 0.071 + (1 - w1) * 0.124
w1_target = (target_ret - mu[1]) / (mu[0] - mu[1])
w_target = np.array([w1_target, 1 - w1_target])

# 计算目标收益下的最小年化波动率
frontier_vol_at_target = np.sqrt(w_target @ cov_matrix_45 @ w_target)
result['frontier_vol_at_target'] = float(frontier_vol_at_target)

# ==================== 3. 绘制不同相关系数下的均值-方差前沿 ====================
plt.figure(figsize=(10, 7))

# 权重扫描范围 (允许卖空，故范围放宽)
w1_range = np.linspace(-0.5, 1.5, 500)
w2_range = 1 - w1_range

for rho in rhos:
    # 构造协方差矩阵
    cov_12 = rho * sigma[0] * sigma[1]
    cov_matrix = np.array([
        [sigma[0]**2, cov_12],
        [cov_12, sigma[1]**2]
    ])
    
    # 扫描计算前沿收益与波动率
    rets = w1_range * mu[0] + w2_range * mu[1]
    vols = np.sqrt(w1_range**2 * sigma[0]**2 + w2_range**2 * sigma[1]**2 
                   + 2 * w1_range * w2_range * cov_12)
    
    # 计算当前相关系数下的最小方差组合(MVP)
    den = sigma[0]**2 + sigma[1]**2 - 2 * rho * sigma[0] * sigma[1]
    w1_mvp = (sigma[1]**2 - rho * sigma[0] * sigma[1]) / den
    w_mvp = np.array([w1_mvp, 1 - w1_mvp])
    mvp_ret = w_mvp @ mu
    mvp_vol = np.sqrt(w_mvp @ cov_matrix @ w_mvp)
    
    # 绘制前沿曲线
    plt.plot(vols, rets, label=f'Frontier (ρ = {rho:.2f})', linewidth=2)
    
    # 标出最小方差组合
    plt.scatter(mvp_vol, mvp_ret, marker='D', s=80, zorder=5, 
               label=f'MVP (ρ = {rho:.2f})')

# ==================== 4. 图表格式设置与保存 ====================
plt.title('Mean-Variance Frontiers for Two Risky Assets', fontsize=14)
plt.xlabel('Annualized Volatility', fontsize=12)
plt.ylabel('Expected Annual Return', fontsize=12)
plt.legend(loc='upper left', fontsize=10)
plt.grid(True, linestyle='--', alpha=0.7)
plt.xlim(0, 0.50)
plt.ylim(0.04, 0.16)

# 保存图形
figure_path = 'markowitz_frontier.png'
plt.savefig(figure_path, dpi=300, bbox_inches='tight')
result['figure_path'] = figure_path

# 打印结果以供检查
print(result)
```
