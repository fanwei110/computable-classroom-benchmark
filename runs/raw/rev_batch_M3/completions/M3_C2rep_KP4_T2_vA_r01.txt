```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import norm

# ================= 基本参数设定 =================
K = 97.5          # 行权价
r = 0.043         # 无风险利率 (年化)
T = 0.58          # 剩余期限 (年)
q = 0.0           # 标的资产股息率 (题目指明不分红)

S_min = 70        # 标的价格网格下界
S_max = 140       # 标的价格网格上界

# 波动率参数化 (可调参数)
sigmas = [0.15, 0.276, 0.40]

# ================= 函数定义 =================
def bs_call_delta(S, K, T, r, sigma, q=0.0):
    """
    计算欧式看涨期权的 Black-Scholes Delta
    参数:
    S : 标的资产价格 (可以是标量或 numpy 数组)
    K : 行权价
    T : 到期时间 (年)
    r : 无风险利率
    sigma : 波动率
    q : 股息率
    """
    d1 = (np.log(S / K) + (r - q + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    delta = np.exp(-q * T) * norm.cdf(d1)
    return delta

# ================= 任务1: 计算Delta =================
# 生成标的价格网格
S_grid = np.linspace(S_min, S_max, 500)

# 计算不同波动率下的 Delta
deltas_dict = {}
for sigma in sigmas:
    deltas_dict[sigma] = bs_call_delta(S_grid, K, T, r, sigma, q)

# ================= 任务3: 报告特定值 =================
S_target = 110
sigma_target = 0.276
delta_at_s110 = bs_call_delta(S_target, K, T, r, sigma_target, q)

# ================= 任务2: 绘图 =================
plt.figure(figsize=(10, 6))

# 颜色映射
colors = ['blue', 'orange', 'green']

for sigma, color in zip(sigmas, colors):
    plt.plot(S_grid, deltas_dict[sigma], 
             label=f'σ = {sigma*100:.1f}%', 
             color=color, 
             linewidth=2)

plt.title('European Call Option Delta vs Underlying Price', fontsize=14)
plt.xlabel('Underlying Price (S)', fontsize=12)
plt.ylabel('Delta', fontsize=12)
plt.axhline(0, color='black', linewidth=0.5, linestyle='--')
plt.axhline(1, color='black', linewidth=0.5, linestyle='--')
plt.legend(fontsize=12)
plt.grid(True, linestyle='--', alpha=0.6)

# ================= 任务4: 保存图形并填充 result =================
figure_path = 'bs_delta_curves.png'
plt.savefig(figure_path, dpi=300, bbox_inches='tight')
plt.close()

# 结果字典
result = {
    'delta_at_s110': delta_at_s110,
    'figure_path': figure_path
}

# 打印验证 (投屏可见)
print(f"标的110、波动率27.6%时的Delta: {result['delta_at_s110']:.6f}")
print(f"图形已保存至: {result['figure_path']}")
```
