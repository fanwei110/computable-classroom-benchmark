```python
import numpy as np
from scipy.stats import norm

# 已知参数
S = 103.7      # 标的资产现价
K = 97.5       # 行权价
sigma = 0.276  # 波动率 27.6%
r = 0.043      # 无风险利率 4.3%
T = 0.58       # 剩余期限（年）

# 步骤1：计算 Black-Scholes 模型下的 d1 与 vega
# d1 的闭式解
d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))

# vega 的闭式解：Vega = S * sqrt(T) * N'(d1)
# 其中 N'(d1) 为标准正态分布的概率密度函数 pdf
vega = S * np.sqrt(T) * norm.pdf(d1)

# 隐含波动率上升一个百分点，即 d_sigma = 0.01
# 期权价格变化近似为 vega * d_sigma
d_sigma = 0.01
price_change = vega * d_sigma

# 步骤2：将结果存入字典
result = {
    'price_change': price_change
}

# 输出结果（便于教师投屏展示与核对）
print(f"Vega: {vega:.4f}")
print(f"Price Change: {price_change:.4f}")
print(f"Result Dictionary: {result}")
```
