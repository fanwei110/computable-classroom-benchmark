```python
import numpy as np
from scipy.stats import norm

# ==================== 输入参数与假设 ====================
position_value = 1_850_000      # 头寸价值（人民币）
annual_volatility = 0.218       # 年化收益波动率

# 假设：一年包含252个交易日，使用平方根法则进行波动率的期限缩放
trading_days_per_year = 252

# ==================== 步骤1：年化波动率换算到一日期限 ====================
# 根据平方根法则：sigma_daily = sigma_annual / sqrt(T)
sigma_daily = annual_volatility / np.sqrt(trading_days_per_year)

# ==================== 步骤2 & 3：套用正态分位数并计算VaR ====================
# Delta-normal模型假设损益服从正态分布。
# 使用scipy.stats.norm.ppf计算左尾分位数，取绝对值作为损失乘数。
z_95 = abs(norm.ppf(0.05))  # 95%置信水平下的正态分布分位数（约1.64485）
z_99 = abs(norm.ppf(0.01))  # 99%置信水平下的正态分布分位数（约2.32635）

# (i) 计算 95% 一日 VaR
# VaR = 头寸价值 × 分位数 × 期限波动率
var_95_1d = position_value * z_95 * sigma_daily

# (ii) 计算 99% 十日 VaR
# 期限缩放：10日波动率 = 1日波动率 × sqrt(10)
sigma_10d = sigma_daily * np.sqrt(10)
var_99_10d = position_value * z_99 * sigma_10d

# ==================== 步骤4：填充结果 ====================
result = {
    'var_95_1d': var_95_1d,
    'var_99_10d': var_99_10d
}

# 课堂投屏展示用（可选，便于学生直观理解）
print(f"头寸价值: {position_value:,.2f} 元")
print(f"年化波动率: {annual_volatility * 100:.1f}%")
print(f"日波动率: {sigma_daily:.6f}")
print("-" * 40)
print(f"95% 1日 VaR: {result['var_95_1d']:,.2f} 元")
print(f"99% 10日 VaR: {result['var_99_10d']:,.2f} 元")
```
