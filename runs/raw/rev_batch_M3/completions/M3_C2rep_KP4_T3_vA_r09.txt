```python
import numpy as np
from scipy.stats import norm

# ==========================================
# 已知参数
# ==========================================
S = 103.7       # 标的资产现价
K = 97.5        # 行权价
sigma = 0.276   # 隐含波动率 (27.6%)
r = 0.043       # 无风险利率 (4.3%)
T = 0.58        # 剩余到期时间 (年)
# 假设处理：题目未指明股息率，按欧式期权常规假设默认股息率 q = 0

# ==========================================
# Black-Scholes 欧式看涨期权定价与希腊字母闭式解
# ==========================================
def bs_call_price(S, K, T, r, sigma):
    """计算欧式看涨期权价格"""
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
    return price

def bs_vega(S, K, T, r, sigma):
    """计算期权 Vega（波动率每变动1个单位带来的价格变化）"""
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    vega = S * norm.pdf(d1) * np.sqrt(T)
    return vega

# ==========================================
# 1. 推算期权价格对这一个百分点波动率变化的响应
# ==========================================
# 计算原波动率下的期权价格
price_original = bs_call_price(S, K, T, r, sigma)

# 隐含波动率上升一个百分点（绝对量增加 0.01）
sigma_new = sigma + 0.01

# 计算新波动率下的期权价格
price_new = bs_call_price(S, K, T, r, sigma_new)

# 精确的期权价格变化量
price_change_exact = price_new - price_original

# 补充：使用希腊字母 Vega 进行线性近似推算（通常课堂也会对比此方法）
vega = bs_vega(S, K, T, r, sigma)
price_change_vega_approx = vega * 0.01

# ==========================================
# 2. 把价格变化存入 result
# ==========================================
# 采用精确计算的价格变化存入字典，以确保结果绝对准确
result = {
    'price_change': price_change_exact
}

# ==========================================
# 课堂投屏输出（便于教师展示与对比）
# ==========================================
print(f"原隐含波动率下的期权价格: {price_original:.4f}")
print(f"波动率上升1%后的期权价格: {price_new:.4f}")
print(f"期权价格精确变化量: {price_change_exact:.4f}")
print(f"使用 Vega 近似推算的变化量: {price_change_vega_approx:.4f}")
print("-" * 40)
print(f"最终输出结果: {result}")
```
