```python
import numpy as np

# ==================== 债券参数设定 ====================
F = 100          # 面值
c = 0.046        # 票息率 (4.6%)
y = 0.053        # 当前收益率 (5.3%)
T = 7            # 期限（年）
dy = 0.008       # 收益率变动幅度（上升80个基点 = 0.8% = 0.008）

# 假设：每年付息1次（题目未指明付息频率，采用最标准的年付息假设）

# ==================== 1. 计算现金流与当前价格 ====================
t = np.arange(1, T + 1)
CF = np.full(T, F * c)
CF[-1] += F  # 最后一期包含本金

# 贴现现金流
PV = CF / (1 + y)**t
P = np.sum(PV)

# ==================== 2. 计算利率敏感性（久期与凸性） ====================
# 麦考利久期
MacD = np.sum(t * PV) / P

# 修正久期
ModD = MacD / (1 + y)

# 凸性 (基于泰勒展开标准定义，分母包含 (1+y)^2)
Conv = np.sum(t * (t + 1) * CF / (1 + y)**(t + 2)) / P

# ==================== 3. 估算收益率上升80个基点的价格影响 ====================
# 价格变动百分比 ≈ -修正久期 × Δy + 0.5 × 凸性 × (Δy)^2
price_drop_pct = -ModD * dy + 0.5 * Conv * dy**2

# ==================== 输出结果 ====================
result = {'price_drop_pct': price_drop_pct}

# 课堂投屏辅助打印（便于学生观察中间计算过程）
print(f"当前债券价格 P = {P:.4f}")
print(f"麦考利久期 MacD = {MacD:.4f} 年")
print(f"修正久期 ModD = {ModD:.4f}")
print(f"凸性 Conv = {Conv:.4f}")
print(f"预计价格变动百分比 = {price_drop_pct:.4%} (负号代表下跌)")
print(f"result字典内容: {result}")
```
