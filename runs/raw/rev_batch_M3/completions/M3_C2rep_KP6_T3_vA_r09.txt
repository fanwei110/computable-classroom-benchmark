```python
import pandas as pd
import numpy as np

# ==========================================
# 1. 读取数据
# ==========================================
# 读取课程数据快照
df = pd.read_csv('data/market_snapshot_v1.csv')

# 提取 fund 列
fund = df['fund']

# ==========================================
# 2. 数据预处理与收益率计算
# ==========================================
# 启发式判断：fund列是净值(NAV)还是收益率(Return)
# 若包含负数，则一定是收益率；若全为正数，均值/标准差 > 1 通常为净值（均值在1.x，标准差在0.x），否则为收益率
if (fund < 0).any():
    returns = fund
elif fund.mean() / fund.std() > 1:
    returns = fund.pct_change().dropna()  # 净值序列转化为简单收益率
else:
    returns = fund

# ==========================================
# 3. 频率假设与年化参数设定
# ==========================================
# 若样本数 > 300，假设为日度数据（一年252个交易日）；否则假设为月度数据（一年12个月）
n_obs = len(returns)
if n_obs > 300:
    N = 252  # 日度数据年化因子
else:
    N = 12   # 月度数据年化因子

# 年化无风险利率设定
Rf_annual = 0.021

# 计算每期无风险利率（与收益率频率匹配）
Rf_period = Rf_annual / N

# ==========================================
# 4. 计算年化夏普比率
# ==========================================
# 计算每期收益率的均值和标准差
mean_return = returns.mean()
std_return = returns.std()  # 默认 ddof=1，计算样本标准差，符合统计学与金融惯例

# 每期夏普比率 = (每期平均收益 - 每期无风险利率) / 每期收益标准差
sharpe_period = (mean_return - Rf_period) / std_return

# 年化夏普比率 = 每期夏普比率 * sqrt(每年期数)
sharpe_annual = sharpe_period * np.sqrt(N)

# ==========================================
# 5. 存储结果
# ==========================================
result = {'sharpe_annual': sharpe_annual}

# 课堂演示打印
print(f"数据频率假设: 每年 {N} 期")
print(f"年化夏普比率: {sharpe_annual:.4f}")
print(f"结果字典: {result}")
```
