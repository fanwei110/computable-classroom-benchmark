```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# ==========================================
# 1. 配置与假设
# ==========================================
# 假设：
# - 一年按 252 个交易日计算，用于日度收益年化（均值乘252，标准差乘sqrt(252)）
# - 日度无风险利率采用单利折算：rf_daily = rf_annual / 252
# - 滚动标准差采用样本标准差 (ddof=1)，与 pandas 默认行为一致

DATA_PATH = 'data/market_snapshot_v1.csv'
RF_ANNUAL = 0.021
WINDOW = 60           # 窗口长度，可在此调整
TRADING_DAYS = 252    # 年化交易日因子

RF_DAILY = RF_ANNUAL / TRADING_DAYS

# ==========================================
# 2. 读取与预处理数据
# ==========================================
df = pd.read_csv(DATA_PATH)

# 确保 'fund' 列为数值型，并剔除可能的非数值与缺失值
df['fund'] = pd.to_numeric(df['fund'], errors='coerce')
df.dropna(subset=['fund'], inplace=True)

# 若数据中包含日期列，尝试将其设为索引以便绘图更直观
date_cols = [c for c in df.columns if 'date' in c.lower() or 'time' in c.lower()]
if date_cols:
    df.set_index(date_cols[0], inplace=True)

fund_returns = df['fund']

# ==========================================
# 3. 计算 60 日滚动年化夏普比率
# ==========================================
# 计算日超额收益
excess_returns = fund_returns - RF_DAILY

# 计算滚动均值与滚动标准差
rolling_mean = excess_returns.rolling(window=WINDOW).mean()
rolling_std = excess_returns.rolling(window=WINDOW).std()

# 年化夏普比率 = (日均超额收益 / 日超额收益标准差) * sqrt(252)
# 等价于：(日均超额收益 * 252) / (日超额收益标准差 * sqrt(252))
rolling_sharpe = (rolling_mean / rolling_std) * np.sqrt(TRADING_DAYS)

# 报告最后一个窗口的夏普值
rolling_sharpe_last = float(rolling_sharpe.iloc[-1])
print(f"最后一个 {WINDOW} 日窗口的年化夏普比率为: {rolling_sharpe_last:.4f}")

# ==========================================
# 4. 绘图与保存
# ==========================================
plt.figure(figsize=(12, 6))
plt.plot(rolling_sharpe.index, rolling_sharpe.values, 
         label=f'{WINDOW}-Day Rolling Annualized Sharpe', 
         color='darkblue', linewidth=1.2)

plt.title(f'{WINDOW}-Day Rolling Annualized Sharpe Ratio (Fund)', fontsize=14)
plt.xlabel('Date / Index', fontsize=12)
plt.ylabel('Sharpe Ratio', fontsize=12)
plt.axhline(0, color='black', linewidth=0.8, linestyle='--')
plt.grid(True, linestyle=':', alpha=0.6)
plt.legend(fontsize=11)
plt.tight_layout()

FIGURE_PATH = 'rolling_sharpe_ratio.png'
plt.savefig(FIGURE_PATH, dpi=150)
plt.close()

# ==========================================
# 5. 填充 result 字典
# ==========================================
result = {
    'rolling_sharpe_last': rolling_sharpe_last,
    'figure_path': FIGURE_PATH
}
```
