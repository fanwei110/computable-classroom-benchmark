```python
import numpy as np
from scipy.stats import norm

# ==========================================
# 1. 参数设定与假设
# ==========================================
# 题目给定参数
annual_vol = 0.24          # 年化波动率 24%
position_value = 2700000   # 头寸金额 2,700,000 元
confidence_level = 0.95    # 95% 置信水平

# 假设：一年按 252 个交易日计算（金融市场标准假设）
trading_days_per_year = 252

# ==========================================
# 2. 把年化波动率换算到一日期限
# ==========================================
# 根据平方根法则：日波动率 = 年化波动率 / sqrt(252)
daily_vol = annual_vol / np.sqrt(trading_days_per_year)

# ==========================================
# 3. 套用 95% 正态分位数
# ==========================================
# 参数法VaR计算：VaR = 头寸金额 × 日波动率 × 对应置信水平的分位数
# 95% 的左尾分位数绝对值（即 norm.ppf(0.95)）
z_score_95 = norm.ppf(confidence_level)

# ==========================================
# 4. 计算 95% 一日 VaR
# ==========================================
var_95_1d = position_value * daily_vol * z_score_95

# ==========================================
# 5. 按输出契约存入 result 字典
# ==========================================
result = {
    'var_95_1d': var_95_1d
}

# 课堂投屏展示计算逻辑与结果（可选打印）
print(f"年化波动率: {annual_vol:.2%}")
print(f"日波动率: {daily_vol:.4%}")
print(f"95% 正态分位数 (Z值): {z_score_95:.4f}")
print(f"头寸金额: {position_value:,.2f} 元")
print(f"-" * 40)
print(f"95% 一日 VaR: {result['var_95_1d']:,.2f} 元")
```
