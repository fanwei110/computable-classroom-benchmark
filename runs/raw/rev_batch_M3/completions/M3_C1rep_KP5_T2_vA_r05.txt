```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os

# 设置中文字体，防止图表中文乱码
plt.rcParams['font.sans-serif'] = ['SimHei', 'Arial Unicode MS']
plt.rcParams['axes.unicode_minus'] = False

# 1. 读取数据
file_path = 'data/market_snapshot_v1.csv'
df = pd.read_csv(file_path)

# 2. 计算日收益率 (假设fund列为基金净值或价格序列，计算其百分比变化即为日收益)
# 若fund列已经是日收益率，pct_change()依然适用但逻辑会变，此处按常规净值/价格序列处理
daily_returns = df['fund'].pct_change().dropna()

# 3. 设定参数
position = 1000000  # 头寸 100万元
confidence_level = 0.95  # 置信水平（可调）
alpha = 1 - confidence_level

# 4. 计算日损益 (PnL)
pnl = position * daily_returns

# 5. 计算历史法VaR
# 找到损益分布的下 alpha 分位数，VaR取其绝对值（表示可能的最大损失金额）
var_pnl = np.percentile(pnl, alpha * 100)
hist_var_95_1d = -var_pnl

# 6. 绘制直方图及VaR线
plt.figure(figsize=(10, 6))
plt.hist(pnl, bins=50, alpha=0.75, edgecolor='black', color='steelblue')
plt.axvline(var_pnl, color='red', linestyle='dashed', linewidth=2, 
            label=f'{confidence_level*100:.0f}% 1日历史VaR = {hist_var_95_1d:,.2f} 元')

plt.title('日损益分布直方图及历史VaR')
plt.xlabel('日损益 (元)')
plt.ylabel('频数')
plt.legend()
plt.grid(axis='y', alpha=0.5)

# 7. 保存图表
figure_path = 'pnl_var_histogram.png'
plt.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close()

# 8. 按照输出契约构建结果字典
result = {
    'hist_var_95_1d': hist_var_95_1d,
    'figure_path': figure_path
}

# 打印VaR数值
print(f"在{confidence_level*100}%置信水平下，1日历史VaR为: {hist_var_95_1d:,.2f} 元")
```
