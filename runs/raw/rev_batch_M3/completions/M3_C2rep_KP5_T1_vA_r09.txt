```python
import numpy as np
from scipy.stats import norm

# =====================
# 已知参数与假设
# =====================
V = 1_850_000          # 头寸价值（人民币元）
sigma_y = 0.218        # 年化收益波动率
trading_days = 252     # 假设一年有252个交易日，用于年化到日频的换算

# =====================
# 1. 年化波动率换算到一日期限
# 在 delta-normal 模型下，假设收益率独立同分布，使用平方根法则缩放时间
# sigma_1d = sigma_y / sqrt(T)
# =====================
sigma_1d = sigma_y / np.sqrt(trading_days)

# =====================
# 2. 对两个置信水平套用正态分位数
# 95% 对应左侧损失分位数，即 norm.ppf(0.95)
# 99% 对应左侧损失分位数，即 norm.ppf(0.99)
# =====================
z_95 = norm.ppf(0.95)
z_99 = norm.ppf(0.99)

# =====================
# 3. 计算 VaR 并缩放到十日期限
# VaR = 头寸价值 * 分位数 * 对应期限的波动率
# 十日波动率 sigma_10d = sigma_1d * sqrt(10)
# =====================
# (i) 95% 一日 VaR
var_95_1d = V * z_95 * sigma_1d

# (ii) 99% 十日 VaR
sigma_10d = sigma_1d * np.sqrt(10)
var_99_10d = V * z_99 * sigma_10d

# =====================
# 4. 填充 result 字典
# =====================
result = {
    'var_95_1d': var_95_1d,
    'var_99_10d': var_99_10d
}

# 课堂投屏展示用输出
print(f"--- 在险价值计算结果 ---")
print(f"一日波动率: {sigma_1d:.6f}")
print(f"十日波动率: {sigma_10d:.6f}")
print(f"95% 一日 VaR: {result['var_95_1d']:>12.2f} 元")
print(f"99% 十日 VaR: {result['var_99_10d']:>12.2f} 元")
```
