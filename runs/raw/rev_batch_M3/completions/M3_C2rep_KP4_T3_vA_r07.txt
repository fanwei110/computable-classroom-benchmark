```python
import numpy as np
from scipy.stats import norm

# ================= 参数设定 =================
S = 103.7       # 标的资产现价
K = 97.5        # 行权价
sigma = 0.276   # 当前波动率 (27.6%)
r = 0.043       # 无风险利率 (4.3%)
T = 0.58        # 剩余期限 (年)
delta_sigma = 0.01  # 隐含波动率上升一个百分点 (1% = 0.01)

# ================= 核心函数 =================
def bs_call_price(S, K, sigma, r, T):
    """Black-Scholes 欧式看涨期权定价闭式解"""
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
    return price

def bs_vega(S, K, sigma, r, T):
    """Black-Scholes 欧式看涨期权 Vega 闭式解 (对波动率的一阶偏导)"""
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    vega = S * norm.pdf(d1) * np.sqrt(T)
    return vega

# ================= 计算步骤 =================
# 1. 推算期权价格对这一个百分点波动率变化的响应
# 方式A：基于全价重估的精确响应（推荐，完全反映非线性特征）
price_original = bs_call_price(S, K, sigma, r, T)
price_new = bs_call_price(S, K, sigma + delta_sigma, r, T)
price_change_exact = price_new - price_original

# 方式B：基于希腊字母 Vega 的线性近似响应（作为概念印证）
vega = bs_vega(S, K, sigma, r, T)
price_change_approx = vega * delta_sigma

# 2. 把价格变化存入 result
# 采用全价重估的精确差值，以确保数值严格准确
result = {
    'price_change': price_change_exact
}

# ================= 课堂展示输出 =================
print("=" * 55)
print("《证券投资学》课堂演示：波动率变化对期权价格的响应")
print("=" * 55)
print(f"标的现价: {S}, 行权价: {K}, 波动率: {sigma*100:.1f}%, 利率: {r*100:.1f}%, 期限: {T}年")
print("-" * 55)
print(f"原波动率下的期权价格: {price_original:.4f}")
print(f"波动率上升1%后的期权价格: {price_new:.4f}")
print(f"【精确】期权价格变化: {price_change_exact:.4f}")
print("-" * 55)
print(f"期权 Vega 值 (每1%波动率): {vega:.4f}")
print(f"【Vega近似】期权价格变化: {price_change_approx:.4f}")
print(f"非线性(Vomma)导致的误差: {price_change_exact - price_change_approx:.6f}")
print("-" * 55)
print(f"最终输出变量 result: {result}")
```
