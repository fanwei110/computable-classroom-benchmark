import numpy as np

def calculate_bond_metrics():
    # 基础参数设定
    face_value = 100.0          # 面值
    coupon_rate = 0.046         # 票息率 4.6%
    years_to_maturity = 7       # 期限 7 年
    yield_to_maturity = 0.053   # 到期收益率 5.3%
    
    # 假设按年付息
    coupon_payment = face_value * coupon_rate
    
    # 构建现金流时间点与对应的现金流
    # t = [1, 2, 3, 4, 5, 6, 7]
    # cf = [4.6, 4.6, 4.6, 4.6, 4.6, 4.6, 104.6]
    t = np.arange(1, years_to_maturity + 1)
    cash_flows = np.full(years_to_maturity, coupon_payment)
    cash_flows[-1] += face_value  # 最后一期还本付息
    
    # 贴现因子
    discount_factors = (1 + yield_to_maturity) ** t
    
    # 1. 计算价格：现金流贴现之和
    pv_cash_flows = cash_flows / discount_factors
    price = np.sum(pv_cash_flows)
    
    # 2. 计算麦考利久期与修正久期
    # 麦考利久期 = sum(t * CF_t / (1+y)^t) / P
    macaulay_duration = np.sum(t * pv_cash_flows) / price
    # 修正久期 = 麦考利久期 / (1+y)
    modified_duration = macaulay_duration / (1 + yield_to_maturity)
    
    # 3. 计算凸性
    # 凸性 = sum(t*(t+1) * CF_t / (1+y)^(t+2)) / P
    # 等价于 sum(t*(t+1) * CF_t / (1+y)^t) / (P * (1+y)^2)
    convexity = np.sum(t * (t + 1) * pv_cash_flows) / (price * (1 + yield_to_maturity) ** 2)
    
    # 4. 填充 result
    result = {
        'price': price,
        'macaulay_duration_years': macaulay_duration,
        'modified_duration_years': modified_duration,
        'convexity': convexity
    }
    
    return result

if __name__ == '__main__':
    result = calculate_bond_metrics()
    
    # 格式化输出，便于课堂投屏展示
    print("="==== 按年付息假设下的债券定价与久期/凸性计算结果 ====================")
    for key, value in result.items():
        print(f"{key:.<30} {value:.6f}")
    print("===========================================================================")
