```python
import numpy as np
import matplotlib.pyplot as plt
import matplotlib

# 设置中文字体和负号显示
matplotlib.rcParams['font.sans-serif'] = ['SimHei']
matplotlib.rcParams['axes.unicode_minus'] = False

# 债券基本参数
F = 100                # 面值
coupon_rate = 0.046    # 票息率 4.6%
C = F * coupon_rate    # 票息额
n = 7                  # 期限 7年
y0 = 0.053             # 当前YTM 5.3%

# 变动幅度可调参数（此处设定为100bp，可修改该值以调整变动幅度）
delta_y = 0.01

# 精确价格计算函数
def bond_price(y, C, F, n):
    if y == 0:
        return C * n + F
    return C * (1 - (1+y)**-n) / y + F * (1+y)**-n

# 修正久期计算函数
def modified_duration(y, C, F, n, P):
    cf_times = np.arange(1, n+1)
    cf = np.full(n, C)
    cf[-1] += F
    pv_cf = cf / (1+y)**cf_times
    mac_dur = np.sum(cf_times * pv_cf) / P
    return mac_dur / (1+y)

# 计算当前YTM下的精确价格和修正久期
P0 = bond_price(y0, C, F, n)
ModD = modified_duration(y0, C, F, n, P0)

# 计算收益率上升100bp后的精确价格
y_up = y0 + delta_y
price_at_up100bp = bond_price(y_up, C, F, n)

# 计算久期法估算的相对变化
dur_approx_change_up100bp = -ModD * delta_y

# 绘图：收益率范围从2%到9%
y_range = np.linspace(0.02, 0.09, 500)
p_exact = np.array([bond_price(y, C, F, n) for y in y_range])
p_dur_approx = P0 * (1 - ModD * (y_range - y0))

plt.figure(figsize=(10, 6))
plt.plot(y_range*100, p_exact, label='精确价格', color='blue')
plt.plot(y_range*100, p_dur_approx, label='久期近似', color='red', linestyle='--')
plt.axvline(y0*100, color='gray', linestyle=':', label=f'当前YTM = {y0*100:.1f}%')

# 标注变动点
plt.scatter([y_up*100], [price_at_up100bp], color='blue', zorder=5, label=f'+100bp精确价格 = {price_at_up100bp:.4f}')
plt.scatter([y_up*100], [P0*(1+dur_approx_change_up100bp)], color='red', zorder=5, label=f'+100bp久期近似价格 = {P0*(1+dur_approx_change_up100bp):.4f}')

plt.title('价格-收益率曲线与久期近似对比')
plt.xlabel('收益率 (%)')
plt.ylabel('价格')
plt.legend()
plt.grid(True)

fig_path = 'price_yield_curve.png'
plt.savefig(fig_path)

# 将所有要求的输出存入名为 result 的字典
result = {
    'price_at_up100bp': price_at_up100bp,
    'dur_approx_change_up100bp': dur_approx_change_up100bp,
    'figure_path': fig_path
}

print(result)
```
