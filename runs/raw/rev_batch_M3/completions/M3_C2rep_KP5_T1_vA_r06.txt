```python
import numpy as np
from scipy.stats import norm

# ==========================================
# 已知条件
# ==========================================
position_value = 1_850_000  # 头寸价值（元）
annual_vol = 0.218         # 年化收益波动率

# ==========================================
# 假设处理
# ==========================================
# 证券投资学惯例：一年按252个交易日计算，使用平方根法则进行期限缩放
trading_days_per_year = 252

# ==========================================
# 步骤1：年化波动率换算到一日期限
# ==========================================
daily_vol = annual_vol / np.sqrt(trading_days_per_year)

# ==========================================
# 步骤2：套用正态分位数 (Delta-Normal 模型核心)
# ==========================================
z_95 = norm.ppf(0.95)
z_99 = norm.ppf(0.99)

# ==========================================
# 步骤3：计算VaR并缩放期限，以人民币报告
# ==========================================
# (i) 95% 一日 VaR
var_95_1d = position_value * z_95 * daily_vol

# (ii) 99% 十日 VaR (运用平方根法则将一日波动率缩放至十日)
var_99_10d = position_value * z_99 * daily_vol * np.sqrt(10)

# ==========================================
# 步骤4：填充 result 字典
# ==========================================
result = {
    'var_95_1d': round(var_95_1d, 2),
    'var_99_10d': round(var_99_10d, 2)
}

# 课堂投屏展示辅助信息
if __name__ == "__main__":
    print(f"--- Delta-Normal VaR 计算结果 ---")
    print(f"假设: 一年 {trading_days_per_year} 个交易日")
    print(f"日波动率: {daily_vol:.6%}")
    print(f"Z(95%): {z_95:.4f} | Z(99%): {z_99:.4f}")
    print(f"(i) 95% 一日 VaR: ¥{result['var_95_1d']:,.2f}")
    print(f"(ii) 99% 十日 VaR: ¥{result['var_99_10d']:,.2f}")
    print(f"--------------------------------")
```
