```python
import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt

def bs_call_delta(S, K, T, r, sigma):
    """
    计算无分红欧式看涨期权的 Black-Scholes Delta 闭式解。
    
    参数:
    S : 标的资产价格 (可以是标量或数组)
    K : 行权价
    T : 剩余期限 (年)
    r : 无风险利率 (年化)
    sigma : 波动率 (年化，可调参数)
    """
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    delta = norm.cdf(d1)
    return delta

# ==================== 题目给定参数 ====================
K = 97.5          # 行权价
r = 0.043         # 无风险利率 4.3%
T = 0.58          # 剩余期限 0.58 年
S_grid = np.linspace(70, 140, 500)  # 标的价格从 70 到 140 的网格

# 波动率参数化设定 (15%, 27.6%, 40%)
volatilities = [0.15, 0.276, 0.40]

# ==================== 步骤 1: 计算每个波动率下的 delta ====================
delta_curves = {}
for sigma in volatilities:
    delta_curves[sigma] = bs_call_delta(S_grid, K, T, r, sigma)

# ==================== 步骤 2: 绘制三条带标注曲线 ====================
plt.figure(figsize=(10, 6))
for sigma in volatilities:
    # 将小数转换为百分数显示在图例中
    plt.plot(S_grid, delta_curves[sigma], label=f'$\\sigma$ = {sigma*100:.1f}%')

plt.title('European Call Option Delta (K=97.5, r=4.3%, T=0.58yr)', fontsize=14)
plt.xlabel('Underlying Price (S)', fontsize=12)
plt.ylabel('Delta', fontsize=12)
plt.ylim(-0.05, 1.05)  # Delta 取值在 (0, 1) 之间，设置合理的纵轴范围
plt.legend(fontsize=11)
plt.grid(True, linestyle='--', alpha=0.7)

# 保存图形
figure_path = 'bs_call_delta_curve.png'
plt.savefig(figure_path, dpi=300, bbox_inches='tight')
plt.close()

# ==================== 步骤 3: 报告标的 110、波动率 27.6% 的 delta ====================
S_target = 110
sigma_target = 0.276
delta_at_s110 = bs_call_delta(S_target, K, T, r, sigma_target)

# ==================== 步骤 4: 填充 result 字典 ====================
result = {
    'delta_at_s110': delta_at_s110,
    'figure_path': figure_path
}

# 以下为控制台输出验证，方便教师投屏时确认结果
print(f"标的 110、波动率 27.6% 时的 Delta: {result['delta_at_s110']:.6f}")
print(f"图形已保存至: {result['figure_path']}")
```
