```python
import numpy as np

# ==================== 已知条件 ====================
beta = 1.42                     # 股票的 Beta 系数
market_return_monthly = -0.058  # 上个月市场收益率 (跌 5.8%)
risk_free_rate_annual = 0.047   # 年无风险利率 (4.7%)

# ==================== 步骤 1：折算年无风险利率为月利率 ====================
# 假设：给定的年无风险利率为有效年利率(EAR)，采用复利折算方式计算月度有效利率。
# 这在时间价值意义上是严谨的，若假设为APR则直接除以12，此处采用复利折算。
risk_free_rate_monthly = (1 + risk_free_rate_annual) ** (1 / 12) - 1

# ==================== 步骤 2：在月度层面套用 CAPM ====================
# CAPM 公式: E(R_i) = R_f + beta * (E(R_m) - R_f)
# 在本例中，将上个月市场的实际收益率作为市场期望收益率的代理进行计算。
capm_return_monthly = risk_free_rate_monthly + beta * (market_return_monthly - risk_free_rate_monthly)

# ==================== 步骤 3：填充 result ====================
result = {
    'capm_return_monthly': capm_return_monthly
}

# ==================== 课堂投屏展示输出 ====================
print(f"年无风险利率折算为月利率: {risk_free_rate_monthly:.6%}")
print(f"按 CAPM 计算该股票上个月本该收益率: {capm_return_monthly:.6%}")
print("\n最终结果字典 result:")
print(result)
```
