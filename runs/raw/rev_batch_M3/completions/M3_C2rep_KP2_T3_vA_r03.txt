```python
import numpy as np

# ==================== 输入参数 ====================
beta = 1.42                      # 股票的 Beta 值
market_return_monthly = -0.058   # 上个月市场收益率 (-5.8%)
risk_free_rate_annual = 0.047    # 年化无风险利率 (4.7%)

# ==================== 步骤 1：年无风险利率折算为月利率 ====================
# 假设：采用简单算术平均折算，即月无风险利率 = 年无风险利率 / 12
# （注：在常规《证券投资学》教学与月度CAPM实务中，此为最标准且简便的假设）
risk_free_rate_monthly = risk_free_rate_annual / 12

# ==================== 步骤 2：在月度层面套用 CAPM ====================
# CAPM 公式: E(R_i) = R_f + β * (E(R_m) - R_f)
# 其中：
# E(R_i) 为股票预期收益率 (capm_return_monthly)
# R_f 为月度无风险利率 (risk_free_rate_monthly)
# E(R_m) 为市场预期收益率 (market_return_monthly)
capm_return_monthly = risk_free_rate_monthly + beta * (market_return_monthly - risk_free_rate_monthly)

# ==================== 步骤 3：填充 result ====================
result = {
    'capm_return_monthly': capm_return_monthly
}

# 课堂投屏展示辅助打印
if __name__ == "__main__":
    print(f"年化无风险利率: {risk_free_rate_annual:.2%}")
    print(f"月度无风险利率: {risk_free_rate_monthly:.4%}")
    print(f"市场月度收益率: {market_return_monthly:.2%}")
    print(f"股票 Beta: {beta}")
    print("-" * 30)
    print(f"按 CAPM 计算的该股票上个月本该收益: {capm_return_monthly:.4%}")
    print("-" * 30)
    print(f"输出契约字典 result: {result}")
```!
