```python
import numpy as np
from scipy.stats import norm

# ==================== 已知参数 ====================
position_value = 2_700_000  # 头寸金额（元）
annual_volatility = 0.24    # 年化波动率（24%）
confidence_level = 0.95     # 置信水平

# ==================== 假设处理 ====================
# 假设：一年中的交易日数。金融界计算日度VaR最常用的参数法假设为252个交易日
trading_days_per_year = 252

# ==================== 步骤1：年化波动率换算到一日期限 ====================
# 根据波动率的平方根法则：日波动率 = 年化波动率 / sqrt(一年交易日数)
daily_volatility = annual_volatility / np.sqrt(trading_days_per_year)

# ==================== 步骤2：套用95%正态分位数 ====================
# 使用参数法，计算标准正态分布下，95%置信水平对应的分位数（Z值）
z_score_95 = norm.ppf(confidence_level)

# ==================== 步骤3：计算95%一日VaR ====================
# 参数法VaR公式：VaR = 头寸金额 × 一日波动率 × 正态分位数
var_95_1d = position_value * daily_volatility * z_score_95

# ==================== 按契约存入结果 ====================
result = {
    'var_95_1d': var_95_1d
}

# ==================== 课堂投屏辅助打印 ====================
print("="*40)
print("《证券投资学》- 在险价值计算演示")
print("="*40)
print(f"1. 头寸金额:       {position_value:,.2f} 元")
print(f"2. 年化波动率:     {annual_volatility:.2%}")
print(f"3. 假设一年交易日: {trading_days_per_year} 天")
print(f"4. 一日波动率:     {daily_volatility:.6%}")
print(f"5. 95%正态分位数:  {z_score_95:.4f}")
print("-"*40)
print(f"计算得出 95% 一日 VaR: {var_95_1d:,.2f} 元")
print("="*40)
print(f"result字典内容: {result}")
```
