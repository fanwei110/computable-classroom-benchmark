```python
import numpy as np

# ==================== 债券参数设定 ====================
# 题目给定参数
face_value = 100.0          # 面值
coupon_rate = 0.046         # 票息率 4.6%
n_years = 7                 # 期限 7 年
ytm = 0.053                 # 到期收益率 5.3%

# 假设处理：未指明付息频率，默认按年付息
# ==================== 构建现金流 ====================
# 时间轴 t = 1, 2, ..., 7
times = np.arange(1, n_years + 1)

# 1~6年现金流为票息，第7年现金流为票息+面值
cash_flows = np.full(n_years, face_value * coupon_rate)
cash_flows[-1] += face_value

# ==================== 1. 计算价格 ====================
# 贴现因子 DF = 1 / (1+y)^t
discount_factors = (1 + ytm) ** times
pv_cash_flows = cash_flows / discount_factors
price = np.sum(pv_cash_flows)

# ==================== 2. 计算久期 ====================
# 麦考利久期 MacD = Σ [ t * PV(CF_t) ] / P
macaulay_duration = np.sum(times * pv_cash_flows) / price

# 修正久期 ModD = MacD / (1+y)
modified_duration = macaulay_duration / (1 + ytm)

# ==================== 3. 计算凸性 ====================
# 凸性定义需匹配泰勒展开式：ΔP/P ≈ -ModD * Δy + 0.5 * Convexity * (Δy)^2
# 凸性 Conv = (1/P) * Σ [ t(t+1) * CF_t / (1+y)^(t+2) ]
convexity = np.sum(times * (times + 1) * cash_flows / (1 + ytm)**(times + 2)) / price

# ==================== 4. 填充 result ====================
result = {
    'price': round(price, 4),
    'macaulay_duration_years': round(macaulay_duration, 4),
    'modified_duration_years': round(modified_duration, 4),
    'convexity': round(convexity, 4)
}

# 课堂投屏展示结果
print(result)
```
