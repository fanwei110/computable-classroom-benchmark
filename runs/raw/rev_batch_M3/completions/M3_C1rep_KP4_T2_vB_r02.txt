```python
import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt

# 参数设置
K = 97.5
r = 0.043
T = 0.58
S_range = np.linspace(70, 140, 500)

# vol可调：在此列表中修改或添加波动率即可
vols = [0.15, 0.276, 0.40]

# 定义计算看涨期权Delta的函数
def call_delta(S, K, r, T, sigma):
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    return norm.cdf(d1)

# 绘图
plt.figure(figsize=(10, 6))
for vol in vols:
    deltas = call_delta(S_range, K, r, T, vol)
    plt.plot(S_range, deltas, label=f'Vol = {vol*100}%')

# 添加行权价参考线
plt.axvline(x=K, color='grey', linestyle='--', alpha=0.7, label=f'Strike = {K}')

plt.title(f'Call Option Delta (K={K}, r={r*100}%, T={T})')
plt.xlabel('Underlying Price (S)')
plt.ylabel('Delta')
plt.legend()
plt.grid(True, linestyle='--', alpha=0.6)

# 保存图片
figure_path = 'delta_curves.png'
plt.savefig(figure_path, dpi=300, bbox_inches='tight')
plt.close()

# 计算标的110，vol 27.6%的delta
delta_at_s110 = call_delta(110, K, r, T, 0.276)

# 按照契约输出
result = {
    'delta_at_s110': delta_at_s110,
    'figure_path': figure_path
}

print(result)
```
