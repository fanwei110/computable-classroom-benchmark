```python
import numpy as np

# ==================== 1. 输入参数 ====================
# 年化波动率
vols = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr = np.array([
    [1.00,  0.21, -0.13],
    [0.21,  1.00,  0.37],
    [-0.13, 0.37,  1.00]
])

# ==================== 2. 构造协方差矩阵 ====================
# 协方差矩阵 Σ = diag(σ) @ Corr @ diag(σ)
D = np.diag(vols)
cov_matrix = D @ corr @ D

# ==================== 3. 求解全局最小方差组合(MVP) ====================
# 满仓约束下，MVP的闭式解为：w_mvp = (Σ^-1 * 1) / (1' * Σ^-1 * 1)
# 为提高数值稳定性，使用 np.linalg.solve 求解 Σ * w_unnorm = 1
ones = np.ones(len(vols))

# 求解未归一化的权重
w_unnorm = np.linalg.solve(cov_matrix, ones)

# 归一化使权重和为1
mvp_weights = w_unnorm / np.sum(w_unnorm)

# ==================== 4. 计算组合年化波动率 ====================
# 组合方差 = w'Σw，组合年化波动率 = sqrt(w'Σw)
mvp_var = mvp_weights.T @ cov_matrix @ mvp_weights
mvp_vol_annual = np.sqrt(mvp_var)

# ==================== 5. 按契约输出结果 ====================
result = {
    'mvp_weights': mvp_weights.tolist(),
    'mvp_vol_annual': float(mvp_vol_annual)
}

# (可选) 课堂投屏展示打印
print("="*50)
print("马科维茨均值-方差组合理论 - 全局最小方差组合(MVP)")
print("="*50)
print(f"协方差矩阵:\n{cov_matrix}\n")
print(f"MVP权重: {result['mvp_weights']}")
print(f"MVP年化波动率: {result['mvp_vol_annual']:.4%}")
print("="*50)
```
