```python
import numpy as np

# ==========================================
# 《证券投资学》课堂实时编程：债券久期与凸性
# ==========================================

# 债券参数
face_value = 100       # 面值
coupon_rate = 0.046    # 票息率 4.6%
ytm = 0.053            # 收益率 5.3%
maturity = 7           # 期限 7年
delta_y = 0.008        # 收益率上升 80 个基点 (0.80%)

# 假设处理：未指明付息频率，默认按年付息（年度复利）

# 1. 计算现金流与当前价格
times = np.arange(1, maturity + 1)
cash_flows = np.full(maturity, face_value * coupon_rate)
cash_flows[-1] += face_value  # 最后一期包含本金

# 贴现因子与现值
discount_factors = (1 + ytm) ** times
pv_cash_flows = cash_flows / discount_factors
current_price = np.sum(pv_cash_flows)

# 2. 计算利率敏感性指标：久期与凸性
# 麦考利久期 (Macaulay Duration)
mac_duration = np.sum(times * pv_cash_flows) / current_price

# 修正久期 (Modified Duration)
mod_duration = mac_duration / (1 + ytm)

# 凸性 (Convexity)
convexity = np.sum(times * (times + 1) * pv_cash_flows) / (current_price * (1 + ytm) ** 2)

# 3. 估算收益率上升 80 个基点后的价格影响（使用久期与凸性）
# 价格变动百分比 ≈ -修正久期 × Δy + 0.5 × 凸性 × (Δy)^2
# 收益率上升，价格下跌，该百分比数值为负（表示跌幅）
price_drop_pct = -mod_duration * delta_y + 0.5 * convexity * (delta_y ** 2)

# 输出中间过程以便课堂展示
print(f"当前债券价格: {current_price:.4f}")
print(f"麦考利久期: {mac_duration:.4f} 年")
print(f"修正久期: {mod_duration:.4f}")
print(f"凸性: {convexity:.4f}")
print(f"收益率上升80bp后，预估价格变动百分比: {price_drop_pct:.4%}")

# 将跌幅存入指定字典
result = {
    'price_drop_pct': price_drop_pct
}
```
