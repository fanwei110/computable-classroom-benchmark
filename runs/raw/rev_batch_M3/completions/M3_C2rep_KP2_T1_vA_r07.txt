```python
import numpy as np

# ==================== 输入参数 ====================
rf = 0.023          # 无风险利率 2.3%
rm = 0.094          # 市场期望收益 9.4%
beta_x = 0.62       # 股票 X 的 beta
beta_y = 1.18       # 股票 Y 的 beta
beta_z = 1.51       # 股票 Z 的 beta
actual_ry = 0.131   # 股票 Y 的实际收益 13.1%

# ==================== CAPM 计算 ====================
# 市场风险溢价
market_premium = rm - rf

# 1. 套用 CAPM 公式：E[Ri] = rf + beta * (E[Rm] - rf)
er_x = rf + beta_x * market_premium
er_y = rf + beta_y * market_premium
er_z = rf + beta_z * market_premium

# 2. 计算 Y 的 alpha：alpha = 实际收益 - CAPM 期望收益
alpha_y = actual_ry - er_y

# ==================== 结果汇总 ====================
result = {
    'er_x': er_x,
    'er_y': er_y,
    'er_z': er_z,
    'alpha_y': alpha_y
}

# 为了方便课堂投屏展示，打印结果（保留6位小数以体现计算过程）
print("CAPM 计算结果：")
print(f"股票 X 期望收益 (er_x): {er_x:.6f} ({er_x*100:.4f}%)")
print(f"股票 Y 期望收益 (er_y): {er_y:.6f} ({er_y*100:.4f}%)")
print(f"股票 Z 期望收益 (er_z): {er_z:.6f} ({er_z*100:.4f}%)")
print(f"股票 Y 的 Alpha (alpha_y): {alpha_y:.6f} ({alpha_y*100:.4f}%)")
print("\nresult 字典内容：", result)
```
