```python
import numpy as np
from scipy.stats import norm

# ==========================================
// 初始参数
// ==========================================
annual_vol = 0.24          # 年化波动率 24%
position = 2_700_000       # 头寸金额 2,700,000 元
confidence_level = 0.95    # 置信水平 95%

// 假设：一年有252个交易日（金融市场常见假设）
trading_days_per_year = 252

// ==========================================
// 1. 将年化波动率换算到一日期限
// ==========================================
// 根据平方根法则，日波动率 = 年化波动率 / sqrt(交易日数)
daily_vol = annual_vol / np.sqrt(trading_days_per_year)

// ==========================================
// 2. 套用 95% 正态分位数（参数法 delta-normal）
// ==========================================
// 获取标准正态分布下 95% 的分位数 (单尾)
z_score = norm.ppf(confidence_level)

// ==========================================
// 3. 计算 95% 一日 VaR
// ==========================================
// VaR = 头寸金额 * 分位数 * 日波动率
var_95_1d = position * z_score * daily_vol

// ==========================================
// 输出契约：存入 result 字典
// ==========================================
result = {
    'var_95_1d': var_95_1d
}

// 打印结果供课堂投屏展示
print(f"--- 在险价值计算过程 ---")
print(f"年化波动率: {annual_vol:.2%}")
print(f"假设年交易日数: {trading_days_per_year}")
print(f"日波动率: {daily_vol:.6f}")
print(f"95% 正态分位数: {z_score:.4f}")
print(f"头寸金额: {position:,.2f} 元")
print(f"------------------------")
print(f"95% 一日 VaR: {var_95_1d:,.2f} 元")
print(f"\n最终输出字典 result: {result}")
```
