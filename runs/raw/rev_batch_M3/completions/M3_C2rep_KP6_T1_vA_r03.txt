```python
import numpy as np
import pandas as pd

# ==================== 第一部分：计算年化夏普比率 ====================

# 1. 读取快照 CSV
df = pd.read_csv('data/market_snapshot_v1.csv')
daily_returns = df['fund'].dropna()  # 剔除可能存在的缺失值保证计算健壮

# 假设处理：一年按252个交易日计算，日化无风险利率采用单利折算
trading_days = 252
rf_annual = 0.021
rf_daily = rf_annual / trading_days

# 在基金收益中计入无风险利率，计算日超额收益
excess_returns = daily_returns - rf_daily

# 2. 计算年化夏普比率
mean_excess = np.mean(excess_returns)
std_excess = np.std(excess_returns, ddof=1)  # 使用ddof=1计算样本标准差

# 年化处理：夏普比率年化 = 日夏普比率 * sqrt(交易日)
sharpe_daily = mean_excess / std_excess
sharpe_annual = sharpe_daily * np.sqrt(trading_days)


# ==================== 第二部分：计算业绩归因效应 ====================

# 组合与基准数据
w_p = np.array([0.45, 0.35, 0.20])      # 组合权重
r_p = np.array([0.083, 0.021, -0.014])  # 组合行业收益

w_b = np.array([0.40, 0.40, 0.20])      # 基准权重
r_b = np.array([0.067, 0.034, -0.009])  # 基准行业收益

# 假设处理：采用BHB(Brinson-Hood-Beebower)模型进行业绩归因分解
# 权重差 = 组合权重 - 基准权重
weight_diff = w_p - w_b
# 收益差 = 组合收益 - 基准收益
return_diff = r_p - r_b

# 3. 计算三大效应
# 配置效应 (Allocation Effect): 衡量因偏离基准权重而产生的收益影响
allocation_effect = np.sum(weight_diff * r_b)

# 选择效应 (Selection Effect): 衡量因超越基准收益而产生的收益影响
selection_effect = np.sum(w_b * return_diff)

# 交互效应 (Interaction Effect): 衡量权重偏离与收益偏离共同作用的影响
interaction_effect = np.sum(weight_diff * return_diff)


# ==================== 第四部分：填充 result ====================

result = {
    'sharpe_annual': sharpe_annual,
    'allocation_effect': allocation_effect,
    'selection_effect': selection_effect,
    'interaction_effect': interaction_effect
}

# 课堂投屏展示辅助（打印输出验证）
if __name__ == '__main__':
    print("=== 风险调整后业绩与归因分析结果 ===")
    for key, value in result.items():
        print(f"{key}: {value:.6f}")
```
