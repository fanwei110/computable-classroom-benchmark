```python
import math

# 已知参数
annual_vol = 0.24
position = 2700000
z_score_95 = 1.644853 # 95%置信水平对应的正态分布Z值
trading_days = 252 # 假设一年有252个交易日

# 计算日波动率
daily_vol = annual_vol / math.sqrt(trading_days)

# 计算95%一日VaR
var_95_1d = position * z_score_95 * daily_vol

# 按照输出契约存入字典
result = {'var_95_1d': round(var_95_1d, 2)}
```
