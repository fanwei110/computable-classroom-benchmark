以下是满足您要求的Python代码实现。代码使用了`numpy`和`matplotlib`进行计算和绘图，使用`scipy.stats.norm`计算正态分布累积概率以得出看涨期权的Delta值。

```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import norm
import os

# 参数设置
K = 97.5          # 行权价
r = 0.043         # 无风险利率 4.3%
T = 0.58          # 到期时间
S_range = np.linspace(70, 140, 700) # 标的价格范围 70 到 140

# 可调波动率列表，包含15%, 27.6%, 40%三条
vols = [0.15, 0.276, 0.40] 

# 计算欧式看涨期权 Delta 的函数
def call_delta(S, K, r, T, vol):
    d1 = (np.log(S / K) + (r + 0.5 * vol**2) * T) / (vol * np.sqrt(T))
    return norm.cdf(d1)

# 计算 标的110，vol 27.6% 的delta值
delta_s110 = call_delta(110, K, r, T, 0.276)

# 绘制Delta曲线
plt.figure(figsize=(10, 6))
for vol in vols:
    deltas = call_delta(S_range, K, r, T, vol)
    plt.plot(S_range, deltas, label=f'Vol = {vol*100}%')

# 图表格式及图例设置
plt.title('Call Option Delta vs Spot Price')
plt.xlabel('Spot Price (S)')
plt.ylabel('Delta')
plt.axvline(x=K, color='grey', linestyle='--', alpha=0.7, label=f'Strike = {K}')
plt.legend()
plt.grid(True)

# 保存图表
figure_path = 'delta_curve.png'
plt.savefig(figure_path)
plt.close()

# 按照输出契约，将结果存入名为 result 的字典
result = {
    'delta_at_s110': float(delta_s110),
    'figure_path': os.path.abspath(figure_path)
}

# 打印结果字典（方便查看）
print(result)
```
