```python
import math

# 给定的输入数据
sigma_A = 0.184
sigma_B = 0.297
w_A = 0.6
w_B = 0.4
rho_before = 0.3
rho_after = 0.8

# 计算变化前的组合方差和波动率
var_before = (w_A**2 * sigma_A**2 +
              w_B**2 * sigma_B**2 +
              2 * w_A * w_B * rho_before * sigma_A * sigma_B)
vol_before_annual = math.sqrt(var_before)

# 计算变化后的组合方差和波动率
var_after = (w_A**2 * sigma_A**2 +
             w_B**2 * sigma_B**2 +
             2 * w_A * w_B * rho_after * sigma_A * sigma_B)
vol_after_annual = math.sqrt(var_after)

# 将结果存入字典
result = {
    'vol_before_annual': vol_before_annual,
    'vol_after_annual': vol_after_annual
}

# 打印结果以供查看
print(result)
```
