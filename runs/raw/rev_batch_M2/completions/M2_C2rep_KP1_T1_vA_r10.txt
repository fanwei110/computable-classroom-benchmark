```python
import numpy as np

def main():
    # 1. 给定的年化波动率与相关系数矩阵
    sigma = np.array([0.187, 0.243, 0.312])  # 年化波动率
    corr = np.array([
        [1.0,  0.21, -0.13],
        [0.21, 1.0,   0.37],
        [-0.13, 0.37,  1.0 ]
    ])  # 相关系数矩阵

    # 2. 构造协方差矩阵：Σ_ij = σ_i * σ_j * ρ_ij
    cov = np.outer(sigma, sigma) * corr

    # 3. 计算全局最小方差组合权重（允许卖空，满仓即权重之和为1）
    #    闭式解：w = Σ^{-1} 1 / (1' Σ^{-1} 1)
    ones = np.ones(len(sigma))
    inv_cov = np.linalg.inv(cov)          # 协方差阵的逆
    numerator = inv_cov @ ones            # Σ^{-1} 1
    denominator = ones @ numerator        # 1' Σ^{-1} 1
    w_mvp = numerator / denominator       # 最小方差组合权重

    # 4. 组合年化波动率 = sqrt(w' Σ w)
    var_mvp = w_mvp @ cov @ w_mvp
    vol_mvp = np.sqrt(var_mvp)

    # 5. 按要求组装结果字典
    result = {
        'mvp_weights': w_mvp.tolist(),    # 转为列表，顺序对应三只资产
        'mvp_vol_annual': vol_mvp
    }

    # 课堂投屏展示（可选打印）
    print("全局最小方差组合（允许卖空，全投资）")
    print(f"权重：{result['mvp_weights']}")
    print(f"年化波动率：{result['mvp_vol_annual']:.4f}")

    return result

# 自包含脚本入口
if __name__ == "__main__":
    result = main()
```
