```python
import numpy as np

# ==================================================
# 基本设定（沿用题设，默认每年付息一次的年复利）
# ==================================================
FACE = 100.0               # 面值
COUPON_RATE = 0.046        # 票息率（年化）
YTM = 0.053                # 到期收益率（年化）
MATURITY = 7               # 剩余年限（年）
FREQ = 1                   # 付息频率（每年1次，与复利频率一致）

# --------------------------------------------------
# 1. 生成现金流与时间点
# --------------------------------------------------
times = np.arange(1, MATURITY + 1, dtype=float)          # 现金流发生时间（年）
coupons = np.full(MATURITY, COUPON_RATE * FACE)          # 各期票息
coupons[-1] += FACE                                      # 最后一期归还本金

# --------------------------------------------------
# 2. 计算债券价格（全价）
# --------------------------------------------------
disc_factors = (1 + YTM) ** (-times)                     # 折现因子
pv_cfs = coupons * disc_factors                         # 各期现金流现值
price = np.sum(pv_cfs)                                  # 债券价格

# --------------------------------------------------
# 3. 计算麦考利久期、修正久期与凸性
# --------------------------------------------------
mac_dur = np.sum(times * pv_cfs) / price                # 麦考利久期
mod_dur = mac_dur / (1 + YTM)                           # 修正久期

# 凸性（年复利离散形式）
convexity = np.sum(times * (times + 1) * pv_cfs) / (price * (1 + YTM) ** 2)

# --------------------------------------------------
# 4. 估算收益率上升80个基点的价格影响
# --------------------------------------------------
delta_y = 0.008                                         # 80个基点 = 0.8%
delta_p_pct = -mod_dur * delta_y + 0.5 * convexity * delta_y ** 2  # 价格变动比例（负数为下跌）

# 价格跌幅（以百分比表示，正数）
price_drop_pct = -delta_p_pct * 100.0

# --------------------------------------------------
# 5. 存入结果字典并展示
# --------------------------------------------------
result = {'price_drop_pct': price_drop_pct}

# 课堂展示用输出
print(f"债券价格                 : {price:.4f}")
print(f"麦考利久期               : {mac_dur:.4f} 年")
print(f"修正久期                 : {mod_dur:.4f}")
print(f"凸性                     : {convexity:.4f}")
print(f"收益率上升 80 bp 时 ΔP/P : {delta_p_pct * 100:.4f}%")
print(f"预计价格跌幅 (price_drop_pct) : {price_drop_pct:.4f}%")
print()
print("结果字典 (result) :", result)
```
