```python
# 计算CAPM期望收益（月度）
beta = 1.42
market_return_monthly = -0.058  # 大盘月收益率 -5.8%
rf_annual = 0.047  # 无风险年化利率 4.7%
rf_monthly = rf_annual / 12  # 简单月化

# CAPM: E(R) = rf + beta * (rm - rf)
capm_return_monthly = rf_monthly + beta * (market_return_monthly - rf_monthly)

result = {'capm_return_monthly': capm_return_monthly}
print(result)
```
