```python
import numpy as np
import pandas as pd
from scipy import stats  # 虽然没有直接用到，但按要求导入
import matplotlib.pyplot as plt  # 导入但不一定使用

# ============================================================
# 第一部分：计算年化夏普比率
# ============================================================

# 读取数据
data = pd.read_csv("data/market_snapshot_v1.csv")
fund_returns = data["fund"].values  # 日收益序列，假设为小数形式

# 参数设定
risk_free_annual = 0.021       # 年化无风险利率
trading_days = 252             # 年交易日数

# 日无风险利率（简单折算）
rf_daily = risk_free_annual / trading_days

# 每日超额收益
excess_returns = fund_returns - rf_daily

# 样本均值和样本标准差（自由度 ddof=1）
mean_excess_daily = np.mean(excess_returns)
std_excess_daily = np.std(excess_returns, ddof=1)

# 年化均值和年化标准差
mean_excess_annual = mean_excess_daily * trading_days
std_excess_annual = std_excess_daily * np.sqrt(trading_days)

# 年化夏普比率
sharpe_annual = mean_excess_annual / std_excess_annual

# ============================================================
# 第二部分：业绩归因 —— 配置效应、选择效应与交互效应
# ============================================================

# 组合与基准数据
weights_p = np.array([0.45, 0.35, 0.20])   # 组合权重
returns_p = np.array([0.083, 0.021, -0.014])  # 组合行业收益
weights_b = np.array([0.40, 0.40, 0.20])   # 基准权重
returns_b = np.array([0.067, 0.034, -0.009])  # 基准行业收益

# 基准总收益
R_b = np.sum(weights_b * returns_b)

# 配置效应 (Allocation effect)
allocation_effect = np.sum((weights_p - weights_b) * (returns_b - R_b))

# 选择效应 (Selection effect)
selection_effect = np.sum(weights_b * (returns_p - returns_b))

# 交互效应 (Interaction effect)
interaction_effect = np.sum((weights_p - weights_b) * (returns_p - returns_b))

# 主动收益（用于验证，不强制输出）
R_p = np.sum(weights_p * returns_p)
active_return = R_p - R_b
# 验证拆分是否正确（由于浮点舍入，允许极小误差）
assert np.abs(active_return - (allocation_effect + selection_effect + interaction_effect)) < 1e-12

# ============================================================
# 输出字典
# ============================================================
result = {
    'sharpe_annual': sharpe_annual,
    'allocation_effect': allocation_effect,
    'selection_effect': selection_effect,
    'interaction_effect': interaction_effect
}

# 直接打印以在课堂展示
print(result)
```
