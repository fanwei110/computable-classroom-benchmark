```python
import numpy as np
import matplotlib.pyplot as plt

# 给定数据
r1, r2 = 0.071, 0.124
sigma1, sigma2 = 0.163, 0.289
rhos = [0.15, 0.45, 0.75]
target_return = 0.10

# 计算函数
def portfolio_stats(w1, r1, r2, s1, s2, rho):
    w2 = 1 - w1
    ret = w1 * r1 + w2 * r2
    vol = np.sqrt(w1**2 * s1**2 + w2**2 * s2**2 + 2*w1*w2*rho*s1*s2)
    return ret, vol

def mvp(r1, r2, s1, s2, rho):
    cov = rho * s1 * s2
    w1 = (s2**2 - cov) / (s1**2 + s2**2 - 2*cov)
    ret, vol = portfolio_stats(w1, r1, r2, s1, s2, rho)
    return w1, ret, vol

# 计算所需指标
_, _, mvp_vol_rho45 = mvp(r1, r2, sigma1, sigma2, 0.45)

# 目标收益10%对应的权重及在rho=0.45时的波动率
w1_target = (target_return - r2) / (r1 - r2)
_, frontier_vol_target = portfolio_stats(w1_target, r1, r2, sigma1, sigma2, 0.45)

# 绘图
fig, ax = plt.subplots(figsize=(10, 6))
colors = ['blue', 'green', 'red']
for rho, c in zip(rhos, colors):
    w1_range = np.linspace(-0.5, 1.5, 500)
    rets, vols = portfolio_stats(w1_range, r1, r2, sigma1, sigma2, rho)
    ax.plot(vols, rets, label=f'ρ = {rho}', color=c)
    # 最小方差点
    w1_mvp, ret_mvp, vol_mvp = mvp(r1, r2, sigma1, sigma2, rho)
    ax.scatter(vol_mvp, ret_mvp, color=c, marker='o', zorder=5)
    ax.annotate(f'MVP\n({vol_mvp:.3f}, {ret_mvp:.3f})', 
                (vol_mvp, ret_mvp), textcoords="offset points", xytext=(10,-10), fontsize=8, color=c)

# 标注目标收益点（仅rho=0.45作为示例）
ax.scatter(frontier_vol_target, target_return, color='green', marker='*', s=150, zorder=6)
ax.annotate(f'Target 10%\n(σ={frontier_vol_target:.3f})', 
            (frontier_vol_target, target_return), textcoords="offset points", xytext=(10,10), fontsize=9, color='green')

ax.set_xlabel('Volatility (σ)')
ax.set_ylabel('Expected Return')
ax.set_title('Efficient Frontier for Two Assets')
ax.legend()
ax.grid(True, linestyle='--', alpha=0.6)

# 保存图片
figure_path = 'efficient_frontier.png'
plt.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close()

# 结果字典
result = {
    'mvp_vol_at_rho45': round(mvp_vol_rho45, 6),
    'frontier_vol_at_target': round(frontier_vol_target, 6),
    'figure_path': figure_path
}

print(result)
```

**计算结果：**
- `mvp_vol_at_rho45`（ρ=0.45 时最小方差组合的波动率）: **0.161693**（约 16.17%）
- `frontier_vol_at_target`（目标收益 10% 时的最小波动率）: **0.202383**（约 20.24%）
- 图像已保存至 `efficient_frontier.png`。
