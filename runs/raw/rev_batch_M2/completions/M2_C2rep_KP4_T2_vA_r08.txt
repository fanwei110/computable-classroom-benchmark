```python

import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt

# ===================== 参数设置 =====================
K = 97.5           # 行权价
r = 0.043          # 无风险利率
T = 0.58           # 剩余期限（年）
S_min, S_max = 70, 140
S_grid = np.linspace(S_min, S_max, 200)  # 标的价格网格

# 波动率参数（可调）
volatilities = [0.15, 0.276, 0.40]

# ===================== 函数定义 =====================
def bs_call_delta(S, K, T, r, sigma):
    """欧式看涨期权 Black-Scholes Delta"""
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    return norm.cdf(d1)

# ===================== 计算并绘图 =====================
fig, ax = plt.subplots(figsize=(8, 5))

for sigma in volatilities:
    delta = bs_call_delta(S_grid, K, T, r, sigma)
    ax.plot(S_grid, delta, label=f"$\sigma$ = {sigma*100:.1f}%")

ax.set_xlabel("标的价格 S")
ax.set_ylabel("Delta")
ax.set_title("欧式看涨期权 Delta 曲线 (Black-Scholes)")
ax.legend()
ax.grid(True, alpha=0.3)

# 保存图像
figure_path = "delta_curve.png"
fig.savefig(figure_path, dpi=150, bbox_inches="tight")
plt.close(fig)  # 释放内存

# ===================== 特定点计算 =====================
S_target = 110
sigma_target = 0.276
delta_at_s110 = bs_call_delta(S_target, K, T, r, sigma_target)

# ===================== 输出结果 =====================
result = {
    'delta_at_s110': delta_at_s110,
    'figure_path': figure_path
}

print("结果字典：", result)
```
