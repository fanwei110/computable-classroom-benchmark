```python
import numpy as np

def main():
    # 债券参数
    face_value = 100.0
    coupon_rate = 0.046
    years = 7
    ytm = 0.053

    # 现金流时间点（年）
    t = np.arange(1, years + 1, dtype=float)

    # 各期票息现金流
    coupon = face_value * coupon_rate
    cashflows = np.full(years, coupon)
    # 到期还本
    cashflows[-1] += face_value

    # 贴现因子
    discount_factors = (1 + ytm) ** (-t)
    pv = cashflows * discount_factors

    # 价格
    price = np.sum(pv)

    # 麦考利久期
    macaulay_duration = np.sum(t * pv) / price

    # 修正久期
    modified_duration = macaulay_duration / (1 + ytm)

    # 凸性
    # 凸性 = Σ [t*(t+1)*PV] / [P * (1+y)^2]
    convexity = np.sum(t * (t + 1) * pv) / (price * (1 + ytm)**2)

    # 输出字典
    result = {
        'price': price,
        'macaulay_duration_years': macaulay_duration,
        'modified_duration_years': modified_duration,
        'convexity': convexity
    }

    # 打印结果，便于课堂投屏查看
    for key, value in result.items():
        print(f"{key}: {value:.6f}")

    return result

if __name__ == "__main__":
    result = main()
```
