```python
import numpy as np
from scipy.stats import norm

# ---------- 给定参数 ----------
position_value = 1_850_000          # 头寸价值（人民币）
annual_vol = 0.218                  # 年化波动率
trading_days = 252                  # 假设年交易日数

# ---------- 波动率转换 ----------
daily_vol = annual_vol / np.sqrt(trading_days)         # 一日波动率
ten_day_vol = daily_vol * np.sqrt(10)                  # 十日波动率

# ---------- 正态分位数（左尾） ----------
z_95 = abs(norm.ppf(0.05))   # 95% 置信水平对应分位数
z_99 = abs(norm.ppf(0.01))   # 99% 置信水平对应分位数

# ---------- VaR 计算 ----------
var_95_1d = position_value * daily_vol * z_95
var_99_10d = position_value * ten_day_vol * z_99

# ---------- 输出字典 ----------
result = {
    'var_95_1d': round(var_95_1d, 2),
    'var_99_10d': round(var_99_10d, 2)
}

# 课堂投屏可直接打印
print(result)
```
