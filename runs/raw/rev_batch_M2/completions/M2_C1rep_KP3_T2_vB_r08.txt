以下是满足要求的Python代码，它计算精确价格、久期近似，绘制价格收益率曲线，并将所有结果存入字典 `result`。

```python
import numpy as np
import matplotlib.pyplot as plt

# 债券参数
face_value = 100
coupon_rate = 0.046
n = 7
ytm_initial = 0.053

# 收益率变动参数（可调）
shift_bp = 100          # 基点数
delta_y = shift_bp / 10000  # 转换为小数 (100bp = 0.01)

# 收益率区间（可调）
y_min = 0.02
y_max = 0.09
yields = np.linspace(y_min, y_max, 500)

# ---------- 债券定价函数 ----------
def bond_price(y, face, coupon, periods):
    """年付息债券价格"""
    cf = np.full(periods, coupon * face)
    cf[-1] += face
    t = np.arange(1, periods + 1)
    pv = cf / (1 + y) ** t
    return pv.sum()

def macaulay_duration(y, face, coupon, periods):
    """麦考利久期"""
    price = bond_price(y, face, coupon, periods)
    cf = np.full(periods, coupon * face)
    cf[-1] += face
    t = np.arange(1, periods + 1)
    weighted_pv = t * cf / (1 + y) ** t
    return weighted_pv.sum() / price

# ---------- 初始时点的计算 ----------
P0 = bond_price(ytm_initial, face_value, coupon_rate, n)
D_mac = macaulay_duration(ytm_initial, face_value, coupon_rate, n)
D_mod = D_mac / (1 + ytm_initial)

# 精确价格曲线 & 久期近似直线
exact_prices = np.array([bond_price(y, face_value, coupon_rate, n) for y in yields])
approx_prices = P0 * (1 - D_mod * (yields - ytm_initial))

# 收益率 +100bp 时的精确价格与久期近似相对变化
ytm_up = ytm_initial + delta_y
P_up = bond_price(ytm_up, face_value, coupon_rate, n)
dur_approx_change = -D_mod * delta_y      # 久期法估算的相对变化

# ---------- 画图 ----------
plt.figure(figsize=(10, 6))
plt.plot(yields * 100, exact_prices, label='精确价格', linewidth=2)
plt.plot(yields * 100, approx_prices, '--', label='久期近似', linewidth=2)
plt.axvline(ytm_initial * 100, color='gray', linestyle=':', alpha=0.7)
plt.axhline(P0, color='gray', linestyle=':', alpha=0.7)
plt.xlabel('到期收益率 (%)')
plt.ylabel('价格')
plt.title('债券价格收益率曲线与久期近似')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()

# 保存图形
figure_path = 'price_yield_curve.png'
plt.savefig(figure_path)
plt.close()

# ---------- 输出结果 ----------
result = {
    'price_at_up100bp': round(P_up, 6),               # 收益率上升100bp后的精确价格
    'dur_approx_change_up100bp': round(dur_approx_change, 6),  # 久期法估算的相对变化
    'figure_path': figure_path
}

print(result)
```
