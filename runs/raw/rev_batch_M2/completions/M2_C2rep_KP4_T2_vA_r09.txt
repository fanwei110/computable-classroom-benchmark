```python
import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt
import pandas as pd  # 仅用于满足依赖要求，实际未使用

# ==================== 参数定义 ====================
K = 97.5            # 行权价
r = 0.043           # 无风险利率
T = 0.58            # 剩余时间（年）
S_min, S_max = 70, 140  # 标的价格范围
S_grid = np.linspace(S_min, S_max, 500)   # 标的网格

# 可调参数：波动率列表
volatilities = [0.15, 0.276, 0.40]

# ==================== 函数定义 ====================
def bs_call_delta(S, K, r, T, sigma):
    """计算欧式看涨期权的 Black-Scholes Delta"""
    if T <= 0:
        # 到期时，delta 为 1 if S>K else 0 (实际为Heaviside)
        return (S > K).astype(float)
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    return norm.cdf(d1)

# ==================== 计算与绘图 ====================
plt.figure(figsize=(10, 6))

for sigma in volatilities:
    delta = bs_call_delta(S_grid, K, r, T, sigma)
    label_str = f"$\\sigma = {sigma*100:.1f}\\%$"
    plt.plot(S_grid, delta, linewidth=2, label=label_str)

# 特殊点：S=110, sigma=27.6%
S_target = 110.0
sigma_target = 0.276
delta_target = bs_call_delta(S_target, K, r, T, sigma_target)
# 在图上标注
plt.scatter(S_target, delta_target, color='red', s=60, zorder=5,
            label=f"$S={S_target}$, $\\sigma={sigma_target*100:.1f}\\%$: $\\Delta={delta_target:.4f}$")

plt.xlabel("标的价格 $S$", fontsize=12)
plt.ylabel("Delta $\\Delta$", fontsize=12)
plt.title("欧式看涨期权 Delta 曲线 (Black-Scholes)", fontsize=14)
plt.legend(fontsize=10)
plt.grid(alpha=0.3)
plt.tight_layout()

# 保存图片
figure_path = "delta_curve.png"
plt.savefig(figure_path, dpi=150)
plt.close()  # 释放内存

# ==================== 输出结果 ====================
result = {
    'delta_at_s110': delta_target,
    'figure_path': figure_path
}

# 打印结果供查看
print("result =", result)
```
