```python
# 定义函数计算历史法VaR
def calculate_historical_var(returns, position_value=1000000, confidence_level=0.95):
    """
    计算历史法VaR
    
    参数:
    returns: 收益率序列
    position_value: 头寸价值
    confidence_level: 置信水平
    """
    # 计算分位数（左尾）
    var_percentile = returns.quantile(1 - confidence_level)
    
    # 计算VaR金额
    var_amount = position_value * abs(var_percentile)
    
    return var_percentile, var_amount

# 计算95%置信水平的历史VaR
confidence_level = 0.95
var_percentile_95, var_amount_95 = calculate_historical_var(returns, 1000000, confidence_level)

print(f"95%置信水平历史法VaR:")
print(f"收益率VaR: {var_percentile_95:.4%}")
print(f"金额VaR: ¥{var_amount_95:,.2f}")
```

```python
# 可调置信水平的VaR计算函数
def adjustable_var_analysis(returns, position_value=1000000, confidence_level=0.95):
    """
    可调置信水平的VaR分析
    """
    var_percentile, var_amount = calculate_historical_var(returns, position_value, confidence_level)
    
    # 创建图表
    daily_pnl = returns * position_value
    
    plt.figure(figsize=(12, 6))
    
    # 绘制直方图
    plt.hist(daily_pnl, bins=50, alpha=0.7, color='skyblue', 
             edgecolor='black', linewidth=0.5)
    
    # VaR线
    var_line = -var_amount
    plt.axvline(x=var_line, color='red', linestyle='--', linewidth=2, 
                label=f'{confidence_level:.0%} Historical VaR: ¥{var_amount:,.2f}')
    
    plt.text(var_line, plt.ylim()[1]*0.9, f'VaR ({confidence_level:.0%}): ¥{var_amount:,.2f}', 
             color='red', fontweight='bold', ha='right')
    
    plt.xlabel('Daily Profit/Loss (¥)', fontsize=12)
    plt.ylabel('Frequency', fontsize=12)
    plt.title(f'Daily Profit/Loss Distribution with Historical VaR\n(Position: ¥{position_value:,}, Confidence Level: {confidence_level:.0%})', 
              fontsize=14, fontweight='bold')
    plt.legend(loc='upper left')
    plt.grid(True, alpha=0.3)
    
    # 保存图表
    figure_path = f'historical_var_{confidence_level:.0%}_chart.png'
    plt.savefig(figure_path, dpi=300, bbox_inches='tight')
    plt.show()
    
    return var_percentile, var_amount, figure_path

# 测试不同置信水平
print("测试不同置信水平的VaR:")
for cl in [0.90, 0.95, 0.99]:
    var_pct, var_amt, _ = adjustable_var_analysis(returns, 1000000, cl)
    print(f"置信水平 {cl:.0%}: 收益率VaR = {var_pct:.4%}, 金额VaR = ¥{var_amt:,.2f}")
```
