import numpy as np

def compute_mvp():
    """
    计算三资产全局最小方差组合（允许卖空，满仓约束）。
    
    返回:
        result: dict, 包含 'mvp_weights' (组合权重列表) 和 'mvp_vol_annual' (年化组合波动率浮点数)
    """
    # 1. 输入年化波动率与相关系数矩阵
    sigmas = np.array([0.187, 0.243, 0.312])  # 年化波动率
    corr = np.array([
        [1.00,  0.21, -0.13],
        [0.21,  1.00,  0.37],
        [-0.13, 0.37,  1.00]
    ])

    # 2. 构造协方差矩阵 Σ[i,j] = σ_i * σ_j * ρ_ij
    Sigma = np.outer(sigmas, sigmas) * corr

    # 3. 全局最小方差组合闭式解（权重和=1）
    n = len(sigmas)
    ones = np.ones(n)
    inv_Sigma = np.linalg.inv(Sigma)
    # w = (Σ^{-1} * 1) / (1^T * Σ^{-1} * 1)
    w = inv_Sigma @ ones / (ones.T @ inv_Sigma @ ones)

    # 4. 组合年化波动率 = sqrt(w^T Σ w)
    mvp_var = w.T @ Sigma @ w
    mvp_vol = np.sqrt(mvp_var)

    # 5. 整理输出
    result = {
        'mvp_weights': w.tolist(),            # 转为列表便于观察
        'mvp_vol_annual': float(mvp_vol)      # 确保为Python原生float
    }
    return result

# 主执行：生成字典 result
result = compute_mvp()

# 当脚本直接运行时，打印结果以方便课堂查看
if __name__ == '__main__':
    print("全局最小方差组合权重：", result['mvp_weights'])
    print("年化组合波动率：", f"{result['mvp_vol_annual']:.4%}")
