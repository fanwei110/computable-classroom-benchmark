```python
import numpy as np

# ========================
# 给定参数
# ========================
beta = 1.42                      # 股票 beta
market_return_monthly = -0.058   # 上个月市场收益率（-5.8%）
rf_annual = 0.047                # 无风险年利率（4.7%）

# ========================
# 1. 将年无风险利率折算为月利率
# 假设采用简单收益率折算（年利率 / 12），这也是短期情形下的常见教学处理。
# ========================
rf_monthly = rf_annual / 12

# ========================
# 2. 在月度层面应用 CAPM
# E(R_i) = R_f + beta * (R_m - R_f)
# ========================
capm_return_monthly = rf_monthly + beta * (market_return_monthly - rf_monthly)

# ========================
# 3. 填充输出字典
# ========================
result = {
    'capm_return_monthly': capm_return_monthly
}

if __name__ == "__main__":
    print(f"月无风险利率: {rf_monthly:.6f} ({rf_monthly*100:.4f}%)")
    print(f"CAPM 预期月度收益率: {capm_return_monthly:.6f} ({capm_return_monthly*100:.4f}%)")
    print("\nresult =", result)
```
