```python
import numpy as np
from scipy.stats import norm

def black_scholes_call(S, K, T, r, sigma):
    """
    计算欧式看涨期权的 Black-Scholes 价格
    """
    if T <= 0:
        return max(S - K, 0.0)
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
    return price

# 给定参数
S = 103.7      # 现价
K = 97.5       # 行权价
sigma = 0.276  # 波动率 27.6%
r = 0.043      # 无风险利率 4.3%
T = 0.58       # 剩余期限 0.58 年

# 计算两个波动率下的期权价格
price_base = black_scholes_call(S, K, T, r, sigma)
price_shifted = black_scholes_call(S, K, T, r, sigma + 0.01)  # 上升一个百分点

# 价格变化
price_change = price_shifted - price_base

# 输出结果字典
result = {
    'price_change': price_change
}

print(result)
```
