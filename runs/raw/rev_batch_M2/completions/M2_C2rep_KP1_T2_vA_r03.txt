```python
import numpy as np
import matplotlib
matplotlib.use('Agg')   # 非交互式后端，确保在无图形界面环境下可保存图片
import matplotlib.pyplot as plt

def main():
    # ---------- 给定的参数 ----------
    mu = np.array([0.071, 0.124])          # 年化期望收益
    sigma = np.array([0.163, 0.289])       # 年化波动率（标准差）
    rhos = [0.15, 0.45, 0.75]             # 需考察的相关系数
    target_return = 0.10                   # 目标期望收益 10%

    var1, var2 = sigma[0]**2, sigma[1]**2  # 方差

    result = {}                            # 最终输出字典

    # ---------- 开始绘图 ----------
    plt.figure(figsize=(8, 6))

    for rho in rhos:
        # --- 构造协方差矩阵 ---
        cov12 = rho * sigma[0] * sigma[1]

        # --- 最小方差组合 (MVP) ---
        denom = var1 + var2 - 2 * cov12
        w1_mvp = (var2 - cov12) / denom
        w2_mvp = 1.0 - w1_mvp
        mu_mvp = w1_mvp * mu[0] + w2_mvp * mu[1]
        std_mvp = np.sqrt(w1_mvp**2 * var1 + w2_mvp**2 * var2 + 2 * w1_mvp * w2_mvp * cov12)

        # 若为所要求的相关系数 0.45，存储结果
        if abs(rho - 0.45) < 1e-12:
            result['mvp_vol_at_rho45'] = float(std_mvp)

            # --- 目标期望收益 10% 对应的组合 ---
            # 满仓 w1 + w2 = 1，期望收益唯一确定权重
            w1_target = (target_return - mu[1]) / (mu[0] - mu[1])
            w2_target = 1.0 - w1_target
            std_target = np.sqrt(w1_target**2 * var1 + w2_target**2 * var2 +
                                 2 * w1_target * w2_target * cov12)
            result['frontier_vol_at_target'] = float(std_target)

        # --- 扫描权重建模整个均值-方差前沿 ---
        w1_range = np.linspace(-2.0, 4.0, 6000)   # 范围足够覆盖所有曲线及卖空情况
        w2_range = 1.0 - w1_range
        mu_range = w1_range * mu[0] + w2_range * mu[1]
        std_range = np.sqrt(w1_range**2 * var1 + w2_range**2 * var2 +
                            2 * w1_range * w2_range * cov12)

        plt.plot(std_range, mu_range, label=rf'$\rho = {rho}$')

        # 标出最小方差组合（统一使用红点，便于全班观看）
        plt.scatter(std_mvp, mu_mvp, color='red', s=60, zorder=5)

    # ---------- 图形修饰 ----------
    # 为图例添加一个代表“最小方差组合”的红色散点条目
    plt.scatter([], [], color='red', s=60, label='Minimum Variance Portfolio')
    plt.xlabel('Volatility (standard deviation)')
    plt.ylabel('Expected Return')
    plt.title('Mean-Variance Frontiers')
    plt.legend()
    plt.grid(True)

    # 保存图像
    figure_filename = 'frontier.png'
    plt.savefig(figure_filename, dpi=150, bbox_inches='tight')
    plt.close()
    result['figure_path'] = figure_filename

    return result


if __name__ == '__main__':
    output = main()
    # 打印结果以便教师在控制台查看
    print("=== RESULTS ===")
    for k, v in output.items():
        if isinstance(v, float):
            print(f"{k}: {v:.6f} ({v*100:.2f}%)")
        else:
            print(f"{k}: {v}")
```
