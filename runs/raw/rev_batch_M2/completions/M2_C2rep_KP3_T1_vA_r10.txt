```python
import numpy as np

def main():
    # 债券参数
    face_value = 100.0
    coupon_rate = 0.046   # 4.6%
    maturity = 7          # 年
    ytm = 0.053           # 到期收益率 5.3%
    frequency = 1         # 每年付息一次

    # 现金流时间点
    t = np.arange(1, maturity + 1)
    # 每期票息
    coupon = face_value * coupon_rate
    # 现金流：最后一期加上本金
    cash_flows = np.full(maturity, coupon)
    cash_flows[-1] += face_value

    # 贴现因子
    discount_factors = (1 + ytm) ** (-t)
    pv = cash_flows * discount_factors

    # 价格
    price = np.sum(pv)

    # 麦考利久期
    macaulay_duration = np.sum(t * pv) / price

    # 修正久期
    modified_duration = macaulay_duration / (1 + ytm)

    # 凸性
    # 公式: convexity = 1/(P * (1+y)^2) * sum( t*(t+1) * CF_t / (1+y)^t )
    convexity = np.sum(t * (t + 1) * pv) / (price * (1 + ytm) ** 2)

    # 结果字典
    result = {
        'price': price,
        'macaulay_duration_years': macaulay_duration,
        'modified_duration_years': modified_duration,
        'convexity': convexity
    }

    # 打印结果，便于投屏展示
    for key, val in result.items():
        print(f"{key}: {val:.6f}")

    return result

if __name__ == "__main__":
    result = main()
```
