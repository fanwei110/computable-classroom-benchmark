```python
import numpy as np

# 债券参数
face_value = 100.0       # 面值
coupon_rate = 0.046      # 票息率（年化）
yield_rate = 0.053       # 当前收益率（年化）
maturity = 7             # 期限（年）
delta_y = 0.0080         # 收益率上升 80 个基点
coupon_payment = face_value * coupon_rate  # 每年票息

# 假设每年付息一次，付息时间点：1, 2, ..., 7
times = np.arange(1, maturity + 1, dtype=float)
cash_flows = np.full(maturity, coupon_payment)
cash_flows[-1] += face_value  # 最后一期包含本金

# ---------- 1. 当前价格及利率敏感性 ----------
discount_factors = (1 + yield_rate) ** (-times)
pv_cf = cash_flows * discount_factors
price = np.sum(pv_cf)

# 麦考利久期
weights = pv_cf / price
macaulay_duration = np.sum(weights * times)

# 修正久期
modified_duration = macaulay_duration / (1 + yield_rate)

# 凸性（年化）
convexity = np.sum(weights * times * (times + 1)) / (1 + yield_rate)**2

# ---------- 2. 估算收益率上升后的价格影响 ----------
# 基于久期和凸性的近似变化百分比 (ΔP/P 为小数)
approx_price_change_pct = (-modified_duration * delta_y +
                           0.5 * convexity * delta_y**2) * 100   # 转为百分比
# 注意：ΔP/P 为负时表示价格下跌，跌幅为正数
approx_price_drop_pct = -approx_price_change_pct  # 跌幅百分比

# 精确价格计算（新收益率）
new_yield = yield_rate + delta_y
new_discount_factors = (1 + new_yield) ** (-times)
new_price = np.sum(cash_flows * new_discount_factors)
exact_price_change_pct = (new_price - price) / price * 100  # 负值
exact_price_drop_pct = -exact_price_change_pct  # 正值，精确跌幅百分比

# ---------- 3. 存入结果 ----------
# 按题目要求，使用精确价格跌幅作为最终输出，确保确定且可复现
result = {
    'price_drop_pct': round(exact_price_drop_pct, 6)
}

# 输出关键中间结果供课堂查看
print(f"债券当前价格: {price:.4f}")
print(f"麦考利久期: {macaulay_duration:.4f} 年")
print(f"修正久期: {modified_duration:.4f}")
print(f"凸性: {convexity:.4f}")
print(f"近似价格跌幅（久期+凸性）: {approx_price_drop_pct:.4f}%")
print(f"精确价格跌幅（重算现值）: {exact_price_drop_pct:.4f}%")
print(f"\n最终结果字典: {result}")
```
