```python
import pandas as pd
import numpy as np

# ---------- 第一部分：年化夏普比率 ----------
# 读取数据快照
data = pd.read_csv('data/market_snapshot_v1.csv')
fund_returns = data['fund']

# 无风险利率设定
rf_annual = 0.021
trading_days = 252
rf_daily = rf_annual / trading_days

# 计算日超额收益
excess_daily = fund_returns - rf_daily

# 日平均超额收益与标准差
mean_excess_daily = excess_daily.mean()
std_excess_daily = excess_daily.std(ddof=1)   # 样本标准差

# 年化夏普比率
sharpe_annual = np.sqrt(trading_days) * mean_excess_daily / std_excess_daily

# ---------- 第二部分：业绩归因 ----------
# 组合与基准的权重、行业收益
w_p = np.array([0.45, 0.35, 0.20])
r_p = np.array([0.083, 0.021, -0.014])
w_b = np.array([0.40, 0.40, 0.20])
r_b = np.array([0.067, 0.034, -0.009])

# 基准总收益
R_b = np.sum(w_b * r_b)

# 配置效应 (Allocation effect)
allocation_effect = np.sum((w_p - w_b) * (r_b - R_b))

# 选择效应 (Selection effect)
selection_effect = np.sum(w_b * (r_p - r_b))

# 交互效应 (Interaction effect)
interaction_effect = np.sum((w_p - w_b) * (r_p - r_b))

# ---------- 结果输出 ----------
result = {
    'sharpe_annual': sharpe_annual,
    'allocation_effect': allocation_effect,
    'selection_effect': selection_effect,
    'interaction_effect': interaction_effect
}

print(result)
```
