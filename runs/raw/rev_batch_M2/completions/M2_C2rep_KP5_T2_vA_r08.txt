```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# ========================
# 可调参数
# ========================
DATA_PATH = 'data/market_snapshot_v1.csv'   # 数据文件路径
POSITION_VALUE = 1_000_000                  # 头寸规模（人民币元）
CONFIDENCE_LEVEL = 0.95                     # VaR 置信水平，可调
ALPHA = 1 - CONFIDENCE_LEVEL                # 损失分位数水平

# ========================
# 1. 读取数据，构造日损益
# ========================
df = pd.read_csv(DATA_PATH)
# 提取 fund 列的日收益率，并剔除缺失值
returns = df['fund'].dropna().values.astype(float)

# 智能识别收益率格式：若绝对值最大值大于1，假定为百分比形式，自动除以100
if np.max(np.abs(returns)) > 1:
    print("检测到收益率绝对值大于1，假定为百分比形式，已自动除以100。")
    returns = returns / 100.0

# 计算每日损益（人民币）
pnl = POSITION_VALUE * returns

# ========================
# 2. 计算 95% 历史 VaR
# ========================
# 损益分布的 alpha 分位数（例如 5% 分位数）
var_percentile = np.percentile(pnl, ALPHA * 100)
# 历史 VaR 通常表示为正数损失
hist_var = -var_percentile

# ========================
# 3. 绘制直方图与 VaR 线
# ========================
plt.figure(figsize=(10, 6))
n, bins, patches = plt.hist(pnl, bins=50, alpha=0.7,
                            color='skyblue', edgecolor='black',
                            label='Daily P&L')

# 在 VaR 处画竖线
plt.axvline(x=var_percentile, color='red', linestyle='--',
            linewidth=2, label=f'{CONFIDENCE_LEVEL*100:.0f}% VaR')

# 标注 VaR 数值
ymax = np.max(n) if len(n) > 0 else 0
plt.text(var_percentile, ymax * 0.9,
         f'VaR = ¥{hist_var:,.2f}',
         rotation=90, verticalalignment='top',
         color='red', fontweight='bold')

plt.xlabel('Daily P&L (¥)')
plt.ylabel('Frequency')
plt.title(f'Daily P&L Distribution and {CONFIDENCE_LEVEL*100:.0f}% 1‑day Historical VaR')
plt.legend()
plt.tight_layout()

# 保存图形
figure_path = 'var_histogram.png'
plt.savefig(figure_path, dpi=150)
plt.close()  # 释放图形内存

# ========================
# 4. 构造输出字典
# ========================
result = {
    'hist_var_95_1d': round(hist_var, 2),   # 95% 一日历史 VaR（人民币）
    'figure_path': figure_path              # 保存的图形文件路径
}

print("输出结果：")
print(result)
```
