```python
import numpy as np
from scipy.stats import norm

# 给定参数
S = 103.7      # 标的资产当前价格
K = 97.5       # 行权价
sigma = 0.276  # 隐含波动率（年化）
r = 0.043      # 无风险利率（连续复利）
T = 0.58       # 剩余期限（年）

# Black-Scholes 欧式看涨期权定价公式
# 首先计算 d1 和 d2
d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
d2 = d1 - sigma * np.sqrt(T)

# 看涨期权价格
call_price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)

# delta = dC/dS = N(d1)
call_delta = norm.cdf(d1)

# vega = dC/dsigma = S * sqrt(T) * n(d1)
# 其中 n(d1) 是标准正态在 d1 处的概率密度函数
call_vega = S * np.sqrt(T) * norm.pdf(d1)

# 按照要求存入字典
result = {
    'call_price': call_price,
    'call_delta': call_delta,
    'call_vega': call_vega
}

# 输出结果（保留适当精度以方便课堂演示）
print(f"看涨期权价格: {call_price:.6f}")
print(f"看涨期权 delta: {call_delta:.6f}")
print(f"看涨期权 vega:  {call_vega:.6f}")
```
