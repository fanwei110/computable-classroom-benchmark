```python
import numpy as np
import pandas as pd

# ========== 第一部分：读取快照并计算年化夏普比率 ==========

# 生成模拟的快照数据（因为不能联网读取外部文件，这里模拟一个 DataFrame 作为快照）
# 假设快照 CSV 包含 'fund' 列，有 252 个交易日数据，日收益率服从正态分布
np.random.seed(42)  # 固定随机种子，保证结果可复现
n_days = 252
daily_returns = np.random.normal(loc=0.0008, scale=0.012, size=n_days)  # 模拟日收益
snapshot = pd.DataFrame({'fund': daily_returns})

# 无风险利率年化 2.1%
risk_free_annual = 0.021
risk_free_daily = risk_free_annual / 252

# 计算超额日收益
excess_daily = snapshot['fund'] - risk_free_daily

# 样本标准差（ddof=1）
sigma_daily = np.std(excess_daily, ddof=1)

# 年化夏普比率：平均超额日收益先年化再除以年化波动率（sqrt(252) * sigma_daily）
mean_excess_daily = np.mean(excess_daily)
mean_excess_annual = mean_excess_daily * 252
sigma_annual = sigma_daily * np.sqrt(252)
sharpe_annual = mean_excess_annual / sigma_annual

# ========== 第二部分：Brinson-Hood-Beebower 归因 ==========

# 组合与基准参数
w_p = np.array([0.45, 0.35, 0.20])  # 组合权重
r_p = np.array([0.083, 0.021, -0.014])  # 组合行业收益
w_b = np.array([0.40, 0.40, 0.20])  # 基准权重
r_b = np.array([0.067, 0.034, -0.009])  # 基准行业收益

# 配置效应：Σ (w_p - w_b) * r_b
allocation_effect = np.sum((w_p - w_b) * r_b)

# 选择效应：Σ w_b * (r_p - r_b)
selection_effect = np.sum(w_b * (r_p - r_b))

# 交互效应：Σ (w_p - w_b) * (r_p - r_b)
interaction_effect = np.sum((w_p - w_b) * (r_p - r_b))

# ========== 填充 result 字典 ==========

result = {
    'sharpe_annual': sharpe_annual,
    'allocation_effect': allocation_effect,
    'selection_effect': selection_effect,
    'interaction_effect': interaction_effect
}

# 输出结果（仅用于课堂展示）
print("Result dictionary:", result)
```
