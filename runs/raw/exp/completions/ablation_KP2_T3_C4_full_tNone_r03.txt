```python
import numpy as np

# 给定参数
beta = 1.42
market_return_monthly = -0.058          # 上个月市场跌了5.8%
annual_risk_free_rate = 0.047           # 年无风险利率 4.7%

# 1. 把年无风险利率按简单除法折算为月利率
monthly_risk_free_rate = annual_risk_free_rate / 12

# 2. 在月度层面套用 CAPM：E(R_i) = R_f + beta * (R_m - R_f)
capm_return_monthly = monthly_risk_free_rate + beta * (market_return_monthly - monthly_risk_free_rate)

# 3. 填充 result 字典
result = {
    'capm_return_monthly': capm_return_monthly
}

# 打印结果便于课堂演示
print(f"月无风险利率: {monthly_risk_free_rate:.6f}")
print(f"CAPM 预期月度收益: {capm_return_monthly:.6f} (即 {capm_return_monthly*100:.2f}%)")
```
