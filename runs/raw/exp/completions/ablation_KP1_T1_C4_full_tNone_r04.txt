```python
import numpy as np
import pandas as pd

# 给定数据
vols = np.array([0.187, 0.243, 0.312])  # 年化波动率
corr_matrix = np.array([
    [1.0, 0.21, -0.13],
    [0.21, 1.0, 0.37],
    [-0.13, 0.37, 1.0]
])

# 1. 构造协方差矩阵
#    协方差 = 波动率_i * 波动率_j * 相关系数_ij
cov_matrix = np.outer(vols, vols) * corr_matrix

# 2. 求全局最小方差组合权重 (允许卖空且满仓)
#    闭式解：w = (Σ⁻¹ * 1) / (1' * Σ⁻¹ * 1)
n = len(vols)
ones = np.ones(n)
cov_inv = np.linalg.inv(cov_matrix)
mvp_weights = cov_inv @ ones / (ones @ cov_inv @ ones)

# 3. 组合年化波动率
mvp_var = mvp_weights @ cov_matrix @ mvp_weights
mvp_vol_annual = np.sqrt(mvp_var)

# 4. 输出结果字典
result = {
    'mvp_weights': mvp_weights,
    'mvp_vol_annual': mvp_vol_annual
}

# 打印供课堂查看
print("全局最小方差组合权重 (w1, w2, w3):")
print(np.round(mvp_weights, 6))
print(f"\n年化波动率: {mvp_vol_annual:.6f} ({mvp_vol_annual*100:.4f}%)")
```
