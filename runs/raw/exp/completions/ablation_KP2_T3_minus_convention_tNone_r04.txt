```python
import numpy as np

# 给定参数
beta = 1.42
market_return_monthly = -0.058  # 市场月收益率（小数）
annual_risk_free = 0.047        # 年无风险利率（小数）

# 步骤1：年无风险利率按简单除法折算为月利率
monthly_risk_free = annual_risk_free / 12

# 步骤2：在月度层面套用CAPM
# E(R_i) = R_f + beta * (R_m - R_f)
capm_return_monthly = monthly_risk_free + beta * (market_return_monthly - monthly_risk_free)

# 步骤3：填充 result 字典
result = {
    'capm_return_monthly': capm_return_monthly
}

# 输出结果（保留足够精度）
print(result)
```
