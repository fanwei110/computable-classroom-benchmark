```python
import numpy as np

# 步骤1：由波动率与相关系数构造协方差矩阵
vol = np.array([0.187, 0.243, 0.312])  # 年化波动率
corr = np.array([
    [1.0, 0.21, -0.13],
    [0.21, 1.0, 0.37],
    [-0.13, 0.37, 1.0]
])
# 协方差矩阵 = diag(vol) * corr * diag(vol)
cov = np.diag(vol) @ corr @ np.diag(vol)

# 步骤2：求最小方差权重（闭式解）
# 满仓约束下最小化 w' Σ w，无其他约束，允许卖空
# 解析解：w = (Σ^{-1} * 1) / (1' * Σ^{-1} * 1)
n = cov.shape[0]
one = np.ones(n)
inv_cov = np.linalg.inv(cov)
mvp_weights = inv_cov @ one / (one @ inv_cov @ one)

# 步骤3：组合波动率
mvp_var = mvp_weights @ cov @ mvp_weights
mvp_vol_annual = np.sqrt(mvp_var)

# 步骤4：按要求填充 result
result = {
    'mvp_weights': mvp_weights,
    'mvp_vol_annual': mvp_vol_annual
}

# 输出结果（教师投屏用）
print("全局最小方差组合权重：", mvp_weights)
print("年化波动率：", mvp_vol_annual)
```
