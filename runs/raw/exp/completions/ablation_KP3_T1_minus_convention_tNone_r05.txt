```python
import numpy as np

# 债券参数
face_value = 100.0
coupon_rate = 0.046
coupon = face_value * coupon_rate  # 每年票息 = 4.6
maturity = 7                      # 剩余年限
ytm = 0.053                      # 到期收益率

# 1. 计算价格：按年复利贴现
# 票息支付年份 t = 1,2,...,7
t = np.arange(1, maturity + 1)
# 每年贴现因子
discount = (1 + ytm) ** t
# 票息现金流现值
pv_coupons = np.sum(coupon / discount)
# 到期面值现值
pv_face = face_value / (1 + ytm) ** maturity
price = pv_coupons + pv_face

# 2. 麦考利久期 (年)
# 分子：t * CFt / (1+y)^t
macaulay_numerator = np.sum(t * coupon / discount) + maturity * face_value / (1 + ytm) ** maturity
macaulay_duration = macaulay_numerator / price

# 修正久期
modified_duration = macaulay_duration / (1 + ytm)

# 3. 凸性 (年平方)
# 公式：Σ[t(t+1)CF/(1+y)^(t+2)] / P
convexity_numerator = np.sum(t * (t + 1) * coupon / ((1 + ytm) ** (t + 2))) + \
                     maturity * (maturity + 1) * face_value / ((1 + ytm) ** (maturity + 2))
convexity = convexity_numerator / price

# 4. 填充 result 字典
result = {
    'price': round(price, 6),
    'macaulay_duration_years': round(macaulay_duration, 6),
    'modified_duration_years': round(modified_duration, 6),
    'convexity': round(convexity, 6)
}

# 输出结果（教师投屏用）
print("计算结果：")
for key, value in result.items():
    print(f"{key}: {value}")
```
