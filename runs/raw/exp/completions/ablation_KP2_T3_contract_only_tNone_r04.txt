```python
import numpy as np

# 给定参数
beta = 1.42                     # 股票 beta 系数
market_monthly_return = -0.058 # 市场月收益率（-5.8%）
risk_free_annual = 0.047       # 年化无风险利率 4.7%

# 将年化无风险利率转换为月无风险利率（按单利或复利？通常CAPM使用单利近似）
# 这里按常见做法：月无风险利率 = 年无风险利率 / 12
risk_free_monthly = risk_free_annual / 12

# CAPM 公式: E(R_i) = R_f + beta * (E(R_m) - R_f)
# 注意：这里R_f和R_m都使用同期限的月收益率
capm_return_monthly = risk_free_monthly + beta * (market_monthly_return - risk_free_monthly)

# 按照输出契约构造结果字典
result = {
    'capm_return_monthly': capm_return_monthly
}

# 输出结果（可复现）
print("CAPM 月期望收益率（以小数表示）:", result['capm_return_monthly'])
print("即大约", round(result['capm_return_monthly'] * 100, 4), "%")
```
