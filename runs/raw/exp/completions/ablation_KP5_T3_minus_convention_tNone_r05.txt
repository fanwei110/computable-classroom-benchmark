```python
import numpy as np
from scipy.stats import norm

# 参数
annual_vol = 0.24          # 年化波动率 24%
position = 2_700_000       # 头寸 2,700,000 元
confidence = 0.95          # 置信水平 95%
trading_days = 252         # 每年交易天数

# 1. 去年化到一日波动率
daily_vol = annual_vol / np.sqrt(trading_days)

# 2. 单尾 95% 正态分位数 (均值=0)
z_95 = norm.ppf(confidence)   # 因为要找左侧损失分位数，但通常 VaR 用绝对值表示

# 3. 计算 VaR (正值)
var_95_1d = z_95 * daily_vol * position

# 4. 存入字典
result = {
    'var_95_1d': var_95_1d
}

# 输出结果
print(result)
```
