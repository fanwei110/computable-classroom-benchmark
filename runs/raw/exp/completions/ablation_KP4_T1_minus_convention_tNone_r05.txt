```python
import numpy as np
from scipy.stats import norm

# 参数设定
S = 103.7        # 标的资产现价
K = 97.5         # 行权价
sigma = 0.276    # 隐含波动率 (27.6%)
r = 0.043        # 无风险利率 (4.3%)
T = 0.58         # 剩余期限 (年)

# 1. 计算 d1 与 d2 (连续复利)
d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
d2 = d1 - sigma * np.sqrt(T)

# 2. 期权价格与 delta
call_price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
call_delta = norm.cdf(d1)      # 看涨期权 delta

# 3. vega (每单位波动率变化引起的价格变化)
call_vega = S * norm.pdf(d1) * np.sqrt(T)   # 注意：vega 通常以“波动率变化 1%”计量，这里返回基础值

# 4. 填充 result 字典
result = {
    'call_price': round(call_price, 4),
    'call_delta': round(call_delta, 4),
    'call_vega':  round(call_vega, 6)   # vega 一般保留较多小数位，此处保留6位
}

# 打印结果供课堂展示
print("计算结果：")
print(f"期权价格 (call_price) = {result['call_price']}")
print(f"Delta (call_delta)     = {result['call_delta']}")
print(f"Vega  (call_vega)      = {result['call_vega']}")
```
