```python
import numpy as np

# 年化波动率
sigma = np.array([0.187, 0.243, 0.312])
# 相关系数矩阵
corr = np.array([
    [1.0, 0.21, -0.13],
    [0.21, 1.0, 0.37],
    [-0.13, 0.37, 1.0]
])

# 协方差矩阵
Sigma = np.outer(sigma, sigma) * corr

# 全局最小方差组合权重 (允许卖空，无约束，和为1)
# w = (Σ^{-1} * 1) / (1^T * Σ^{-1} * 1)
n = 3
ones = np.ones(n)
Sigma_inv = np.linalg.inv(Sigma)
w_mvp = Sigma_inv @ ones / (ones @ Sigma_inv @ ones)

# 年化波动率
vol_mvp = np.sqrt(w_mvp @ Sigma @ w_mvp)

# 输出字典
result = {
    'mvp_weights': w_mvp,             # ndarray of shape (3,)
    'mvp_vol_annual': vol_mvp         # float
}

# 可选：打印结果
print("全局最小方差组合权重 (风险资产1,2,3):", np.round(w_mvp, 6))
print("年化波动率:", round(vol_mvp, 6))
```
