```python
import numpy as np
import pandas as pd

# -------------------- 输入数据 --------------------
# 年化波动率（小数形式）
vol_1 = 0.187
vol_2 = 0.243
vol_3 = 0.312

# 相关系数矩阵
corr12 = 0.21
corr13 = -0.13
corr23 = 0.37

# -------------------- 1. 构造协方差矩阵 --------------------
# 协方差 = 波动率_i * 波动率_j * 相关系数_ij
cov_matrix = np.array([
    [vol_1**2,             vol_1 * vol_2 * corr12, vol_1 * vol_3 * corr13],
    [vol_1 * vol_2 * corr12, vol_2**2,             vol_2 * vol_3 * corr23],
    [vol_1 * vol_3 * corr13, vol_2 * vol_3 * corr23, vol_3**2]
])

# -------------------- 2. 求解全局最小方差组合权重（允许卖空，满仓） --------------------
# 闭式解：w = (Σ⁻¹ 𝟙) / (𝟙' Σ⁻¹ 𝟙)
ones = np.ones(3)
inv_cov = np.linalg.inv(cov_matrix)
mvp_weights = inv_cov @ ones / (ones @ inv_cov @ ones)

# -------------------- 3. 计算组合年化波动率 --------------------
mvp_variance = mvp_weights @ cov_matrix @ mvp_weights
mvp_vol_annual = np.sqrt(mvp_variance)

# -------------------- 4. 构建结果字典 --------------------
result = {
    'mvp_weights': mvp_weights,          # numpy array, 三个权重
    'mvp_vol_annual': mvp_vol_annual     # float, 年化波动率（小数）
}

# 打印检查（课堂展示用）
print("全局最小方差组合权重：")
print(f"资产1: {mvp_weights[0]:.6f}")
print(f"资产2: {mvp_weights[1]:.6f}")
print(f"资产3: {mvp_weights[2]:.6f}")
print(f"权重和: {np.sum(mvp_weights):.2e}")   # 应接近1
print(f"\n年化波动率: {mvp_vol_annual:.6f} ({mvp_vol_annual*100:.2f}%)")
```
