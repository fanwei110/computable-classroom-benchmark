```python
import numpy as np
import pandas as pd

# ========== 第一部分：夏普比率 ==========
# 模拟课程快照数据（共100个交易日）
np.random.seed(42)
dates = pd.date_range('2023-01-01', periods=100, freq='B')
# 生成随机日收益率，均值0.08%/天，标准差0.6%/天
daily_returns = np.random.normal(0.0008, 0.006, 100)
fund_data = pd.DataFrame({'fund': daily_returns}, index=dates)

# 无风险利率年化2.1% -> 日化
rfr_annual = 0.021
rfr_daily = rfr_annual / 252

# 计算超额日收益率
excess_daily = fund_data['fund'] - rfr_daily

# 年化夏普比率
annual_sharpe = np.sqrt(252) * excess_daily.mean() / excess_daily.std(ddof=1)

# ========== 第二部分：Brinson归因 ==========
w_p = np.array([0.45, 0.35, 0.20])  # 组合权重
r_p = np.array([0.083, 0.021, -0.014])  # 组合行业收益
w_b = np.array([0.40, 0.40, 0.20])  # 基准权重
r_b = np.array([0.067, 0.034, -0.009])  # 基准行业收益

# 配置效应: Σ(w_p - w_b) * r_b
allocation = np.sum((w_p - w_b) * r_b)

# 选择效应: Σ w_b * (r_p - r_b)
selection = np.sum(w_b * (r_p - r_b))

# 交互效应: Σ (w_p - w_b) * (r_p - r_b)
interaction = np.sum((w_p - w_b) * (r_p - r_b))

# ========== 输出 ==========
result = {
    'sharpe_annual': round(annual_sharpe, 6),
    'allocation_effect': round(allocation, 6),
    'selection_effect': round(selection, 6),
    'interaction_effect': round(interaction, 6)
}

print(result)
```
