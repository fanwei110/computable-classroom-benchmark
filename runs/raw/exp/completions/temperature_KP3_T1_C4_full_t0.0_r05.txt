```python
import numpy as np

# 债券参数
face_value = 100.0          # 面值
coupon_rate = 0.046         # 票息率 4.6%
maturity_years = 7          # 期限 7 年
yield_to_maturity = 0.053   # 到期收益率 5.3%

# 每年付息一次
coupon = face_value * coupon_rate  # 每年票息额

# 1. 计算价格：各期现金流贴现求和
# 现金流：第1~6年每年票息，第7年票息+面值
cash_flows = np.array([coupon] * (maturity_years - 1) + [coupon + face_value])
discount_factors = (1 + yield_to_maturity) ** np.arange(1, maturity_years + 1)
price = np.sum(cash_flows / discount_factors)

# 2. 计算麦考利久期
# 权重 = 每期现金流现值 / 价格
pv_cash_flows = cash_flows / discount_factors
weights = pv_cash_flows / price
time_periods = np.arange(1, maturity_years + 1)
macaulay_duration = np.sum(time_periods * weights)

# 修正久期
modified_duration = macaulay_duration / (1 + yield_to_maturity)

# 3. 计算凸性 (年平方)
# 凸性 = Σ[t(t+1) * CF_t / (1+y)^(t+2)] / P
convexity_numerator = np.sum(time_periods * (time_periods + 1) * cash_flows / (1 + yield_to_maturity) ** (time_periods + 2))
convexity = convexity_numerator / price

# 4. 填充 result 字典
result = {
    'price': round(price, 6),
    'macaulay_duration_years': round(macaulay_duration, 6),
    'modified_duration_years': round(modified_duration, 6),
    'convexity': round(convexity, 6)
}

# 输出结果（供课堂展示）
print("计算结果：")
for key, value in result.items():
    print(f"{key}: {value}")
```
