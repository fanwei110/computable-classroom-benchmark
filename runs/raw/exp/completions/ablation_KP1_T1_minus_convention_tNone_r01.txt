```python
import numpy as np

# 步骤1：由波动率与相关系数构造协方差矩阵
vols = np.array([0.187, 0.243, 0.312])   # 年化波动率小数
corr_matrix = np.array([
    [1.0,   0.21, -0.13],
    [0.21,  1.0,   0.37],
    [-0.13, 0.37,  1.0 ]
])
# 协方差矩阵 = diag(vols) * corr_matrix * diag(vols)
cov_matrix = np.diag(vols) @ corr_matrix @ np.diag(vols)

# 步骤2：求最小方差权重（闭式解，允许卖空，满仓约束）
# 对于满仓约束（w'1=1）且无其他约束，最小方差组合权重的闭式解：
# w = (Σ⁻¹ 1) / (1' Σ⁻¹ 1)
ones = np.ones(3)
cov_inv = np.linalg.inv(cov_matrix)
mvp_weights = cov_inv @ ones / (ones @ cov_inv @ ones)

# 步骤3：组合波动率（年化）
mvp_variance = mvp_weights @ cov_matrix @ mvp_weights
mvp_vol_annual = np.sqrt(mvp_variance)   # 小数形式

# 步骤4：按要求存入字典
result = {
    'mvp_weights': mvp_weights,
    'mvp_vol_annual': mvp_vol_annual
}

# 显示结果（方便课堂投屏查看）
print("全局最小方差组合权重 (mvp_weights):")
print(np.round(mvp_weights, 6))
print(f"\n年化波动率 (mvp_vol_annual): {mvp_vol_annual:.6f} ({mvp_vol_annual*100:.2f}%)")
```
