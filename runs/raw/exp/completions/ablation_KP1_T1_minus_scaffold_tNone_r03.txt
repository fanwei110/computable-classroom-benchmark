```python
import numpy as np
import pandas as pd
import scipy.optimize as sco

# ---------- 原始数据 ----------
# 年化波动率（小数）
sigma = np.array([0.187, 0.243, 0.312])

# 相关系数矩阵
corr = np.array([
    [1.0,   0.21, -0.13],
    [0.21,  1.0,   0.37],
    [-0.13, 0.37,  1.0 ]
])

# 协方差矩阵（年化）
cov = np.outer(sigma, sigma) * corr  # 等价于 sigma * sigma * corr

# ---------- 求解全局最小方差组合 ----------
# 目标函数：组合方差 w^T Σ w
def portfolio_variance(weights):
    return weights.T @ cov @ weights

# 约束：满仓（权重之和为1，允许卖空）
constraints = ({'type': 'eq', 'fun': lambda w: np.sum(w) - 1.0})

# 初始猜测：等权重
w0 = np.array([1/3, 1/3, 1/3])

# 边界：无限制（允许卖空），但scipy要求边界，设为[-5,5]足够宽松
bounds = [(-5.0, 5.0) for _ in range(3)]

# 优化求解
result_opt = sco.minimize(portfolio_variance, w0, method='SLSQP',
                          bounds=bounds, constraints=constraints)

# ---------- 提取结果 ----------
mvp_weights = result_opt.x
mvp_var = portfolio_variance(mvp_weights)
mvp_vol_annual = np.sqrt(mvp_var)   # 年化波动率（小数）

# ---------- 输出契约 ----------
result = {
    'mvp_weights': mvp_weights,      # 三个资产的权重数组
    'mvp_vol_annual': mvp_vol_annual # 小数，如0.14表示14%
}

# ---------- 打印清晰结果（供课堂展示）----------
print("=== 全局最小方差组合（MVP） ===")
print(f"资产1权重: {mvp_weights[0]:.6f}")
print(f"资产2权重: {mvp_weights[1]:.6f}")
print(f"资产3权重: {mvp_weights[2]:.6f}")
print(f"组合年化波动率: {mvp_vol_annual:.6f} ({mvp_vol_annual*100:.2f}%)")
print("\nresult 字典内容:", result)
```
