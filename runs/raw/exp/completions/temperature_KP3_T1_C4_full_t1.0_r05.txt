```python
import numpy as np

# 债券参数
face_value = 100.0          # 面值
coupon_rate = 0.046         # 票息率
years = 7                   # 期限
ytm = 0.053                 # 到期收益率（小数）

# 1. 现金流时间及金额
t = np.arange(1, years + 1)       # 1,2,...,7
coupon = coupon_rate * face_value  # 每年票息
cf = np.full(years, coupon)        # 前6年票息
cf[-1] = coupon + face_value       # 最后一年票息+面值

# 价格 = Σ CF_t / (1+y)^t
discount = (1 + ytm) ** t
price = np.sum(cf / discount)

# 2. 麦考利久期 = Σ [t * CF_t / (1+y)^t] / Price
macaulay_dur = np.sum(t * cf / discount) / price

# 修正久期 = 麦考利久期 / (1+y)
modified_dur = macaulay_dur / (1 + ytm)

# 3. 凸性 = Σ [t*(t+1) * CF_t / (1+y)^(t+2)] / Price
convexity = np.sum(t * (t + 1) * cf / ((1 + ytm) ** (t + 2))) / price

# 4. 输出结果字典
result = {
    'price': price,
    'macaulay_duration_years': macaulay_dur,
    'modified_duration_years': modified_dur,
    'convexity': convexity
}

# 打印结果（为课堂演示）
print("计算结果：")
for k, v in result.items():
    print(f"{k:35s} = {v:.10f}")
```
