```python
import numpy as np
from scipy.stats import norm

# 设定参数
annual_vol = 0.24          # 年化波动率 24%
position = 2_700_000       # 头寸 2,700,000 元
confidence = 0.95          # 置信水平 95%
trading_days = 252         # 每年交易日天数

# 步骤1：年化波动率去年化到一日波动率
daily_vol = annual_vol / np.sqrt(trading_days)

# 步骤2：单尾 95% 正态分位数（均值取零）
z = norm.ppf(confidence)   # 单尾 95% 对应分位数约 1.64485

# 步骤3：计算 VaR = 头寸 × 一日波动率 × 分位数（正损失金额）
var_95_1d = position * daily_vol * z

# 输出结果存入字典
result = {
    'var_95_1d': var_95_1d
}

print(result)
```
