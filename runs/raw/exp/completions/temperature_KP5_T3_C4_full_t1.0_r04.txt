```python
import numpy as np
from scipy.stats import norm

# 给定参数
annual_vol = 0.24          # 年化波动率 24%
position = 2_700_000       # 头寸 2,700,000 元
confidence_level = 0.95    # 置信水平 95%
trading_days = 252         # 每年交易日数

# 步骤1：去年化 —— 年化波动率 / sqrt(252) 得到一日波动率
daily_vol = annual_vol / np.sqrt(trading_days)

# 步骤2：单尾95%正态分位数，均值取零
z_score = norm.ppf(confidence_level)   # 约 1.64485

# 步骤3：计算 VaR = 头寸 * z * 一日波动率（均值取零）
var_95_1d = position * z_score * daily_vol

# 存入 result 字典
result = {
    'var_95_1d': var_95_1d
}

# 打印结果（仅为课堂展示）
print(f"年化波动率: {annual_vol:.2%}")
print(f"一日波动率: {daily_vol:.6f}")
print(f"95% 单尾分位数: {z_score:.4f}")
print(f"头寸: {position:,.0f} 元")
print(f"95% 一日 VaR: {var_95_1d:,.2f} 元")
```
