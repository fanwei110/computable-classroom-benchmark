```python
import numpy as np

# ---------- 债券参数 ----------
face_value = 100.0          # 面值
coupon_rate = 0.046         # 票息率
coupon = face_value * coupon_rate  # 每年票息额
years_to_maturity = 7       # 剩余期限
ytm = 0.053                 # 到期收益率（年复利）
n_periods = years_to_maturity  # 付息次数，每年一次

# ---------- 1. 现金流向量与时间向量 ----------
# 第1年至第6年支付票息，第7年支付票息+面值
cf = np.array([coupon] * (n_periods - 1) + [coupon + face_value])
t = np.arange(1, n_periods + 1, dtype=float)  # 时间点 1,...,7

# ---------- 2. 折现因子 ----------
disc = (1 + ytm) ** (-t)      # 1/(1+y)^t

# ---------- 3. 价格 ----------
price = np.sum(cf * disc)
# 保留任意精度，Python float 默认双精度

# ---------- 4. 麦考利久期 (Macaulay duration) ----------
# D_mac = Σ [ t * CF_t / (1+y)^t ] / Price
macaulay_duration = np.sum(t * cf * disc) / price

# ---------- 5. 修正久期 (Modified duration) ----------
modified_duration = macaulay_duration / (1 + ytm)

# ---------- 6. 凸性 (Convexity) ----------
# Convexity = Σ [ t(t+1) * CF_t / (1+y)^(t+2) ] / Price
# 先将分母指数写成 (1+y)^(t+2) = (1+y)^t * (1+y)^2
# 我们可以复用 disc (即 (1+y)^(-t)) 来高效计算
convexity = np.sum(t * (t + 1) * cf * disc / (1 + ytm)**2) / price

# ---------- 7. 输出字典 ----------
result = {
    'price': price,
    'macaulay_duration_years': macaulay_duration,
    'modified_duration_years': modified_duration,
    'convexity': convexity
}

# ---------- 显示结果（投屏用）----------
print("========== 债券定价与风险指标 ==========")
print(f"面值: {face_value}")
print(f"票息率: {coupon_rate*100:.2f}%")
print(f"期限: {years_to_maturity} 年")
print(f"到期收益率: {ytm*100:.2f}%")
print("---------------------------------------")
print(f"价格 (Price):                 {price:.6f}")
print(f"麦考利久期 (Macaulay D):     {macaulay_duration:.6f} 年")
print(f"修正久期 (Modified D):       {modified_duration:.6f} 年")
print(f"凸性 (Convexity):            {convexity:.6f} 年²")
print("=======================================")
```
