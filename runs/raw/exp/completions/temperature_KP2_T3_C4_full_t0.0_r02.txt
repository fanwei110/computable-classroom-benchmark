```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy import stats

# 给定参数
beta = 1.42
market_return_monthly = -0.058   # 上个月市场收益率（小数）
annual_rf = 0.047                # 年无风险利率（小数）

# 1. 年无风险利率折算为月利率（简单除法）
monthly_rf = annual_rf / 12.0

# 2. 月度 CAPM：E(ri) = rf + beta * (rm - rf)
capm_return_monthly = monthly_rf + beta * (market_return_monthly - monthly_rf)

# 3. 填充 result 字典
result = {
    'capm_return_monthly': capm_return_monthly
}

# 输出结果（教师投屏用）
print("月无风险利率:", monthly_rf)
print("CAPM 月度期望收益:", capm_return_monthly)
print("result 字典:", result)
```
