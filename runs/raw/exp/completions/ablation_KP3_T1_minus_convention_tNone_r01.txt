```python
import numpy as np

# 债券参数
face_value = 100.0          # 面值
coupon_rate = 0.046         # 票息率
coupon = face_value * coupon_rate  # 每年票息额 4.6
maturity_years = 7          # 期限（年）
ytm = 0.053                 # 到期收益率

# 1. 计算价格
# 生成未来现金流时间点（第1年到第7年）
t = np.arange(1, maturity_years + 1)  # [1,2,3,4,5,6,7]

# 现金流：前6年是票息，第7年是票息+面值
cash_flows = np.full(maturity_years, coupon)
cash_flows[-1] += face_value  # 最后一期加上面值

# 贴现因子
discount_factors = 1 / (1 + ytm) ** t

# 价格 = 现金流贴现之和
price = np.sum(cash_flows * discount_factors)

# 2. 计算麦考利久期（年）
# 分子：t * CF * 贴现因子，分母：价格
macaulay_duration = np.sum(t * cash_flows * discount_factors) / price

# 修正久期
modified_duration = macaulay_duration / (1 + ytm)

# 3. 计算凸性（年平方）
# 凸性公式 = Σ [ t(t+1) * CF / (1+y)^(t+2) ] / P
convexity = np.sum(t * (t + 1) * cash_flows / (1 + ytm) ** (t + 2)) / price

# 4. 构造结果字典
result = {
    'price': round(price, 6),
    'macaulay_duration_years': round(macaulay_duration, 6),
    'modified_duration_years': round(modified_duration, 6),
    'convexity': round(convexity, 6)
}

# 输出结果（教师投屏可见）
print("计算结果如下：")
print(result)

# 也可逐行显示更清晰
print(f"\n价格: {result['price']:.6f}")
print(f"麦考利久期（年）: {result['macaulay_duration_years']:.6f}")
print(f"修正久期（年）: {result['modified_duration_years']:.6f}")
print(f"凸性（年平方）: {result['convexity']:.6f}")
```
