face_value = 100.0          # 面值
coupon_rate = 0.046         # 票息率 4.6%
yield_rate = 0.053          # 当前收益率 5.3%
years = 7                   # 剩余年限 7 年
delta_y = 0.0080            # 收益率上升 80 个基点 (0.80%)

# ========================
# 2. 计算现金流与当前价格
# ========================
t = np.arange(1, years + 1)                     # 年份 1~7
coupon = coupon_rate * face_value               # 每年票息 4.6
cf = np.full(years, coupon)                     # 前6年票息
cf[-1] += face_value                            # 第7年加上本金：104.6

df = (1 + yield_rate) ** -t                     # 贴现因子
pv = cf * df                                    # 每期现金流现值
price = np.sum(pv)                              # 当前债券价格

# ========================
# 3. 利率敏感性：久期与凸性
# ========================
# 麦考利久期
mac_duration = np.sum(t * pv) / price
# 修正久期
mod_duration = mac_duration / (1 + yield_rate)
# 凸性（现金流加权）
convexity = np.sum(t * (t + 1) * pv / (1 + yield_rate)**2) / price

# ========================
# 4. 估计收益率上升 80 bp 时的价格变动
# ========================
# 价格变动率 (ΔP/P)，小数形式
delta_p_pct = -mod_duration * delta_y + 0.5 * convexity * (delta_y ** 2)
# 跌幅为正数百分比（即“跌了多少”）
price_drop_pct = -delta_p_pct * 100.0

# ========================

}

# ========================

print(f"当前债券价格：{price:.4f}")
print(f"麦考利久期：{mac_duration:.4f} 年")
