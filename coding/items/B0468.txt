import numpy as np

# ================= 债券参数假设 =================
# 题目未指明付息频率，默认假设为每年付息一次（Annual Coupon）
F = 100.0           # 债券面值
c_rate = 0.046      # 票息率 4.6%
y = 0.053           # 当前到期收益率 5.3%
T = 7               # 期限 7 年
delta_y = 0.008     # 收益率上升 80 个基点 (0.8%)

# ================= 1. 计算当前收益率下的利率敏感性 =================
# 生成各期现金流与对应时间
t = np.arange(1, T + 1)
CF = np.full(T, c_rate * F)
CF[-1] += F  # 最后一期加入本金

# 折现因子
df = (1 + y) ** t

# 当前债券价格 (现金流贴现之和)
P = np.sum(CF / df)

# 麦考利久期
D_mac = np.sum(t * CF / df) / P

# 修正久期
D_mod = D_mac / (1 + y)

# 凸性
# 公式: Conv = (1/P) * sum[ t(t+1) * CF_t / (1+y)^(t+2) ]
Conv = np.sum(t * (t + 1) * CF / ((1 + y) ** (t + 2))) / P

# 使用久期与凸性估算价格变动百分比:
# dP/P ≈ -D_mod * dy + 0.5 * Conv * dy^2
