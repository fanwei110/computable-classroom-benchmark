import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# ==========================================
# 参数配置 (可调)
# ==========================================
WINDOW_SIZE = 60            # 滚动窗口长度（交易日），可根据需要调整
ANNUAL_RF_PERCENT = 2.1     # 年化无风险利率 (%)
TRADING_DAYS_PER_YEAR = 252 # 每年交易日数假设

# ==========================================
# 1. 读取快照 CSV；计入无风险利率
# ==========================================
# 假设CSV第一列为日期/时间索引，自动解析
df = pd.read_csv('data/market_snapshot_v1.csv', index_col=0, parse_dates=True)

# 数据规模兼容性处理：
# 国内数据源(如Tushare)日收益常为百分比(如1.5代表1.5%)，而标准量化框架常为小数(如0.015)。
# 此处通过均值启发式判断，统一转换为小数以保障计算一致性。
if df['fund'].abs().mean() > 0.5:
    df['fund'] = df['fund'] / 100.0

# 无风险利率换算：采用精确的日度复利公式计算日无风险利率
annual_rf_decimal = ANNUAL_RF_PERCENT / 100.0
daily_rf_decimal = (1 + annual_rf_decimal) ** (1 / TRADING_DAYS_PER_YEAR) - 1

# ==========================================
# 2. 计算 60 日滚动年化夏普，窗口可调
# ==========================================
# 每日超额收益 = 当日收益 - 日无风险利率
daily_excess_returns = df['fund'] - daily_rf_decimal

# 计算滚动窗口内的日均超额收益
rolling_mean_excess = daily_excess_returns.rolling(window=WINDOW_SIZE).mean()

# 计算滚动窗口内的日收益标准差（总风险，ddof=1为样本标准差，pandas默认）
rolling_std = df['fund'].rolling(window=WINDOW_SIZE).std()

# 日度夏普比率 = 日均超额收益 / 日收益标准差
rolling_sharpe_daily = rolling_mean_excess / rolling_std

# 年化夏普比率 = 日度夏普比率 * sqrt(一年交易日数)
rolling_sharpe_annualized = rolling_sharpe_daily * np.sqrt(TRADING_DAYS_PER_YEAR)

# ==========================================
# 3. 报告最后一个窗口的值；画出时间序列
# ==========================================
# 提取最近一个完整窗口的夏普比率
rolling_sharpe_last = rolling_sharpe_annualized.iloc[-1]
print(f"最近一个 {WINDOW_SIZE} 日窗口的年化夏普比率: {rolling_sharpe_last:.4f}")

# 绘图
fig, ax = plt.subplots(figsize=(12, 6))
rolling_sharpe_annualized.plot(
    ax=ax, 
    color='tab:blue', 
    linewidth=1.5, 
    label=f'{WINDOW_SIZE}-Day Rolling Annualized Sharpe'
)

# 添加零线与图表装饰
ax.axhline(0, color='black', linewidth=0.8, linestyle='--')
ax.set_title(f'Rolling Annualized Sharpe Ratio (Window={WINDOW_SIZE}, Rf={ANNUAL_RF_PERCENT}%)', fontsize=14)
ax.set_xlabel('Date', fontsize=12)
ax.set_ylabel('Annualized Sharpe Ratio', fontsize=12)
ax.grid(True, linestyle='--', alpha=0.6)
ax.legend(fontsize=11)

# ==========================================
# 4. 保存图形并填充 result
# ==========================================
fig_path = 'rolling_sharpe_ratio.png'
plt.savefig(fig_path, dpi=150, bbox_inches='tight')
plt.close()

# 严格遵循输出契约
result = {
    'rolling_sharpe_last': rolling_sharpe_last,
    'figure_path': fig_path
}
