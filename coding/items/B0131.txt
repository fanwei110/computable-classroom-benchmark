import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# ================= 1. 准备模拟数据 =================
# 假设 df 已存在，此处生成包含 'fund' 列的模拟数据供运行测试
np.random.seed(42)
dates = pd.date_range(start='2023-01-01', periods=300)
# 模拟基金净值
fund_nav = 1.0 * (1 + np.random.normal(0.0005, 0.015, 300)).cumprod()
df = pd.DataFrame({'fund': fund_nav}, index=dates)

# ================= 2. 计算滚动夏普比率 =================
window = 60  # 滚动窗口大小，可在此处调节
rf_annual = 0.021  # 年化无风险利率 2.1%
rf_daily = rf_annual / 252  # 转换为日度无风险利率

# 假设 'fund' 列为净值，计算日收益率；若 'fund' 已是日收益率，请去掉 .pct_change()
returns = df['fund'].pct_change()

# 计算滚动均值与滚动标准差
rolling_mean = returns.rolling(window=window).mean()
rolling_std = returns.rolling(window=window).std()

# 计算年化滚动夏普比率
rolling_sharpe = ((rolling_mean - rf_daily) / rolling_std) * np.sqrt(252)

# 获取最后一个窗口的夏普比率值
rolling_sharpe_last = rolling_sharpe.iloc[-1]

# ================= 3. 绘图并保存 =================
plt.figure(figsize=(12, 6))
plt.plot(rolling_sharpe.index, rolling_sharpe, label=f'{window}-Day Rolling Sharpe', color='tab:blue')
plt.axhline(0, color='red', linestyle='--', linewidth=1, alpha=0.7)
plt.title(f'{window}-Day Rolling Sharpe Ratio (rf={rf_annual:.1%})', fontsize=14)
plt.xlabel('Date', fontsize=12)
plt.ylabel('Annualized Sharpe Ratio', fontsize=12)
plt.legend()
plt.grid(True, linestyle=':', alpha=0.6)

# 保存图片
figure_path = 'rolling_sharpe_plot.png'
plt.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close()

# ================= 4. 按照输出契约构造结果字典 =================
result = {
    'rolling_sharpe_last': rolling_sharpe_last,
    'figure_path': figure_path
}

# 打印确认
print(f"最后一个窗口的夏普比率: {result['rolling_sharpe_last']:.4f}")
print(f"图片已保存至: {result['figure_path']}")
