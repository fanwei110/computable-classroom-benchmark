import numpy as np
import matplotlib.pyplot as plt

# 设置matplotlib以正常显示中文和负号
plt.rcParams['font.sans-serif'] = ['Arial Unicode MS', 'SimHei', 'STSong']
plt.rcParams['axes.unicode_minus'] = False

# --- 债券参数 ---
F = 100          # 面值
coupon_rate = 0.046  # 票息率 4.6%
C = F * coupon_rate  # 票息额 4.6
N = 7            # 期限 7年
y0 = 0.053       # 当前收益率 5.3%

# --- 可调参数：收益率变动幅度 ---
# 这里的 delta_y 控制收益率变动的基点数，100个基点 = 0.01
delta_y = 0.01  

# --- 计算函数 ---
def bond_price(y):
    """计算债券精确价格"""
    return C * (1 - (1 + y)**(-N)) / y + F * (1 + y)**(-N)

# --- 核心计算 ---
# 当前价格
P0 = bond_price(y0)

# 计算麦考利久期和修正久期
t = np.arange(1, N + 1)
pv_cashflows = C / (1 + y0)**t
pv_face = F / (1 + y0)**N
mac_dur = np.sum(t * pv_cashflows) / P0 + N * pv_face / P0
mod_dur = mac_dur / (1 + y0)  # 修正久期

# 1. 收益率上升100个基点后的精确价格
y_up_100bp = y0 + delta_y
price_at_up100bp = bond_price(y_up_100bp)

# 2. 用久期估计的相对价格变化 (%)
dur_approx_change_up100bp = -mod_dur * delta_y

# --- 绘图数据准备 ---
y_range = np.linspace(0.02, 0.09, 700) # 收益率从 2% 到 9%
exact_prices = np.array([bond_price(y) for y in y_range])

# 一阶久期近似价格曲线（基于当前价格P0和修正久期在y0处的切线）
approx_prices = P0 * (1 - mod_dur * (y_range - y0))

# --- 绘图 ---
plt.figure(figsize=(10, 6))
plt.plot(y_range * 100, exact_prices, label='精确价格', linewidth=2, color='blue')
plt.plot(y_range * 100, approx_prices, label='久期近似', linewidth=2, linestyle='--', color='red')

# 标记当前点和上升100bp的点
plt.axvline(x=y0*100, color='gray', linestyle=':', alpha=0.7, label=f'当前收益率 ({y0*100:.1f}%)')
plt.scatter([y_up_100bp*100], [price_at_up100bp], color='green', s=60, zorder=5, label=f'上升100bp精确价格 ({price_at_up100bp:.2f})')

plt.title('债券价格随收益率变化曲线及久期近似')
plt.xlabel('收益率 (%)')
plt.ylabel('价格')
plt.legend()
plt.grid(True, linestyle=':', alpha=0.6)

# 保存图表
fig_path = 'bond_price_yield_curve.png'
plt.savefig(fig_path, dpi=150, bbox_inches='tight')
plt.close()

# --- 构建输出契约字典 ---
result = {
    'price_at_up100bp': round(price_at_up100bp, 4),
    'dur_approx_change_up100bp': round(dur_approx_change_up100bp, 4),
    'figure_path': fig_path
}

# 打印结果以供核对
print(result)
