rolling_std = excess_returns.rolling(window=window).std()

rolling_sharpe = (rolling_mean / rolling_std) * np.sqrt(252)

# 获取最后一个窗口的夏普比率值
rolling_sharpe_last = rolling_sharpe.iloc[-1]

# ================= 4. 绘图并保存 =================
plt.figure(figsize=(12, 6))
plt.plot(rolling_sharpe.index, rolling_sharpe, label=f'{window}-Day Rolling Sharpe Ratio', color='tab:blue')
plt.title(f'{window}-Day Rolling Sharpe Ratio (Risk-Free Rate = {rf*100}%)', fontsize=14)
plt.xlabel('Date', fontsize=12)
plt.ylabel('Sharpe Ratio', fontsize=12)
plt.axhline(0, color='red', linestyle='--', linewidth=1, alpha=0.7)
plt.legend()
plt.grid(True, alpha=0.3)

# 保存图片
figure_path = 'rolling_sharpe_plot.png'
plt.savefig(figure_path, dpi=150, bbox_inches='tight')
plt.close()

# ================= 5. 按照输出契约组装结果 =================
result = {
    'rolling_sharpe_last': rolling_sharpe_last,
    'figure_path': figure_path
}

# 打印结果供查看
print(f"最后一个窗口的滚动夏普值为: {rolling_sharpe_last:.4f}")
print(f"图表已保存至: {figure_path}")
print(result)
