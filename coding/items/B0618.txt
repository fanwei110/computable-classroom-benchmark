```python
import numpy as np

# 波动率
vols = np.array([0.187, 0.243, 0.312])
# 相关系数矩阵
corr = np.array([
    [1.0, 0.21, -0.13],
    [0.21, 1.0, 0.37],
    [-0.13, 0.37, 1.0]
])
# 协方差矩阵
cov = corr * np.outer(vols, vols)

# 最小方差组合权重
ones = np.ones(3)
cov_inv = np.linalg.inv(cov)
weights = cov_inv @ ones / (ones @ cov_inv @ ones)

# 组合波动率
port_vol = np.sqrt(weights @ cov @ weights)

result = {
    'mvp_weights': weights.tolist(),
    'mvp_vol_annual': port_vol
}
```
