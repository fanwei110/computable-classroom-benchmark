vol_before_annual, vol_after_annual

vol_before_annual: 提交值 0.1849  |  参考值 0.18485375841459106

--- 提取出的代码 ---
import numpy as np

# ==========================================
# 1. 设定资产参数与组合权重
# ==========================================
# 资产年化波动率
sigma_A = 0.184
sigma_B = 0.297

# 假设60/40组合中，资产A占比60%，资产B占比40%
# (符合常规书写顺序A:B = 60:40)
w_A = 0.6
w_B = 0.4
w = np.array([w_A, w_B])

# ==========================================
# 2. 构造两个相关系数下的协方差矩阵
# ==========================================
rho_before = 0.3
rho_after = 0.8

# 协方差 cov(A,B) = rho * sigma_A * sigma_B
cov_AB_before = rho_before * sigma_A * sigma_B
cov_AB_after = rho_after * sigma_A * sigma_B

# 构建协方差矩阵 Sigma
Sigma_before = np.array([
    [sigma_A**2, cov_AB_before],
    [cov_AB_before, sigma_B**2]
])

Sigma_after = np.array([
    [sigma_A**2, cov_AB_after],
    [cov_AB_after, sigma_B**2]
])

# ==========================================
# 3. 计算两个组合波动率
# ==========================================
# 组合方差 sigma_p^2 = w' Σ w
# 组合年化波动率 sigma_p = sqrt(w' Σ w)
var_before = w.T @ Sigma_before @ w
var_after = w.T @ Sigma_after @ w

vol_before_annual = np.sqrt(var_before)
vol_after_annual = np.sqrt(var_after)

# ==========================================
# 4. 填充 result 字典
# ==========================================
# 为符合金融数据习惯，保留4位小数（即精确到万分位）
result = {
    'vol_before_annual': round(vol_before_annual, 4),
    'vol_after_annual': round(vol_after_annual, 4)
}

# 课堂投屏展示结果
if __name__ == "__main__":
    print("马科维茨均值-方差组合理论 - 相关性变动影响计算")
    print(f"资产A波动率: {sigma_A:.1%}, 资产B波动率: {sigma_B:.1%}")
    print(f"组合权重: A={w_A:.0%}, B={w_B:.0%}")
    print("-" * 45)
    print(f"相关系数为 {rho_before} 时，组合年化波动率: {result['vol_before_annual']:.2%}")
    print(f"相关系数为 {rho_after} 时，组合年化波动率: {result['vol_after_annual']:.2%}")
    print("-" * 45)
    print(f"波动率变动: {result['vol_after_annual'] - result['vol_before_annual']:.2%} (随相关性上升而增加)")
