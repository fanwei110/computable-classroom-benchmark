from scipy.stats import norm
import json

# 输入参数

sigma = 0.276   # 隐含波动率（年化）

# 第一步：计算 d1 和 d2
d1 = (np.log(S / K) + (r + sigma**2 / 2) * T) / (sigma * np.sqrt(T))
d2 = d1 - sigma * np.sqrt(T)

print(f"d1 = {d1:.6f}")
print(f"d2 = {d2:.6f}")

# 第二步：计算期权价格和Delta
