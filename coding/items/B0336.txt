import numpy as np

# ==========================================
# 《证券投资学》课堂实时编程：马科维茨组合理论计算
# ==========================================

# 1. 确定 60/40 权重与两只资产的对应方式
# 按照惯例，将资产A的权重设为60%，资产B的权重设为40%
w = np.array([0.6, 0.4])

# 给定的年化波动率（标准差）
vol_A = 0.184
vol_B = 0.297

# 计算各资产的方差
var_A = vol_A ** 2
var_B = vol_B ** 2

# 2. 构造相关系数 0.3 与 0.8 两个协方差矩阵
# 协方差公式：Cov(A, B) = rho * vol_A * vol_B
rho_before = 0.3
cov_before = rho_before * vol_A * vol_B
Sigma_before = np.array([
    [var_A, cov_before],
    [cov_before, var_B]
])

rho_after = 0.8
cov_after = rho_after * vol_A * vol_B
Sigma_after = np.array([
    [var_A, cov_after],
    [cov_after, var_B]
])

# 3. 计算两个组合波动率
# 组合方差公式：sigma_p^2 = w' * Sigma * w
# 组合波动率：sigma_p = sqrt(sigma_p^2)

port_var_before = w.T @ Sigma_before @ w
port_vol_before = np.sqrt(port_var_before)

port_var_after = w.T @ Sigma_after @ w
port_vol_after = np.sqrt(port_var_after)

# 4. 填充 result 字典
# 为了符合金融数据展示惯例，将结果四舍五入保留4位小数（即百分比保留2位小数）
result = {
    'vol_before_annual': round(float(port_vol_before), 4),
    'vol_after_annual': round(float(port_vol_after), 4)
}

# 课堂投屏辅助输出（便于师生直观验证）
print("="*40)
print("马科维茨均值-方差组合理论：相关性影响分析")
print("="*40)
print(f"资产配置：资产A 60%，资产B 40%")
print(f"相关系数从 0.3 升至 0.8\n")
print(f"调整前组合年化波动率: {result['vol_before_annual']*100:.2f}%")
print(f"调整后组合年化波动率: {result['vol_after_annual']*100:.2f}%")
print(f"波动率变化幅度: {(result['vol_after_annual'] - result['vol_before_annual'])*100:.2f}%")
print("结论：随着相关系数上升，组合的分散化效应减弱，组合总波动率上升。")
print("="*40)
