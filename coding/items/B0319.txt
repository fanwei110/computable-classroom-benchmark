from scipy.stats import norm

# ================== 参数设定 ==================

sigma = 0.276   # 波动率 (27.6%)
r = 0.043       # 无风险利率 (4.3%)
T = 0.58        # 剩余期限（年）

# ================== 闭式解计算 ==================
# 计算 d1
d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))

# 隐含波动率上升一个百分点，即变化量 d_sigma = 0.01
d_sigma = 0.01

# 利用 Vega 推算（一阶近似）期权价格对这一百分点波动率变化的响应

# ================== 输出契约 ==================
